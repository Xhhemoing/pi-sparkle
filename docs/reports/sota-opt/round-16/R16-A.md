  events.push({
    id: nextId(),
    schemaVersion: 1,
    occurredAt: NOW_ISO as Event["occurredAt"],
    runId,
    type: "RUN_STARTED",
    actor: "system",
    payload: { title: "bench" }
  } as unknown as Event);
  if (opts.attached) {
    events.push({
      id: nextId(),
      schemaVersion: 1,
      occurredAt: NOW_ISO as Event["occurredAt"],
      runId,
      type: "RUN_ATTACHED",
      actor: "supervisor",
      payload: { episodeId: opts.episodeId ?? "ep_bench", runId, attachedAt: NOW_ISO }
    } as unknown as Event);
  }
  for (let i = 0; i < opts.filler; i += 1) {
    events.push({
      id: nextId(),
      schemaVersion: 1,
      occurredAt: NOW_ISO as Event["occurredAt"],
      runId,
      type: "CHILD_MESSAGE",
      actor: "child",
      payload: { taskId: `tsk_${i % 5}`, content: "..." }
    } as unknown as Event);
  }
  return events;
}

// Regex-miss filler for the S axis: no substring matches
// /pass|passed|verified|succeed/i (worst case: full-length scan).
function missFiller(length: number): string {
  const unit = "lorem ipsum dolor sit amet consetetur ";
  let out = "";
  while (out.length < length) out += unit;
  return out.slice(0, length);
}

function mkChild(
  kind: "PASSED" | "FAILED",
  opts?: {
    claimSuccessWithCheckGap?: boolean;
    constraintCount?: number;
    checkCount?: number;
    evidenceCount?: number;
    summaryLength?: number;
    taskId?: string;
    idSuffix?: string; // when set, evidence/artifact ids use this suffix core
  }
): {
  child: ChildShape;
  spec: SpecShape;
  contract: RequirementContract;
} {
  const summary =
    opts?.summaryLength !== undefined
      ? missFiller(opts.summaryLength)
      : opts?.claimSuccessWithCheckGap === true
        ? "all checks passed"
        : "tests ran";
  const constraintCount = opts?.constraintCount ?? 3;
  const checkCount = opts?.checkCount ?? 1;
  const evidenceCount = opts?.evidenceCount ?? 1;
  const core = opts?.idSuffix ?? "";
  const evidenceIds = Array.from(
    { length: evidenceCount },
    (_, i) => `evd_${core}${i}`
  );
  return {
    child: {
      taskId: opts?.taskId ?? "tsk_bench",
      outcome: kind === "PASSED" ? "SUCCESS" : "FAILURE",
      summary,
      evidenceIds,
      artifactIds: [`art_${core}1`],
      terminalResult: { verification: { kind, evidenceIds: [evidenceIds[0] ?? "evd_1"] } }
    } as unknown as ChildShape,
    spec: {
      role: "tester",
      acceptanceCriteria: Array.from({ length: checkCount }, (_, i) => ({
        id: `chk_${i}`,
        description: `check ${i} passes`
      }))
    } as unknown as SpecShape,
    contract: {
      constraints: Array.from({ length: constraintCount }, (_, i) => ({
        id: `con_${i}`,
        description: `constraint ${i} text`
      }))
    } as unknown as RequirementContract
  };
}

function genObservation(rng: () => number, trial: number): ChildObservation {
  const kind = pick(rng, ["PASSED", "FAILED"] as const);
  const evidenceIds =
    rng() < 0.75 ? [`evd_${trial}`, "evd_shared"] : rng() < 0.5 ? ["evd_shared"] : [];
  return {
    taskId: `tsk_${trial % 7}`,
    role: pick(rng, ["implementer", "tester", "scout", "worker"]),
    outcome: pick(rng, ["SUCCESS", "PARTIAL", "FAILURE", "CANCELLED", "TIMEOUT"] as const),
    summary: pick(rng, [
      "tests passed",
      "child completed the task",
      "",
      "wip",
      "verified output",
      "all checks succeed"
    ]),
    evidenceIds,
    artifactIds: rng() < 0.5 ? [`art_${trial}`] : [],
    verification: {
      kind,
      evidenceIds:
        rng() < 0.8 ? [`evd_v_${trial}`, ...(rng() < 0.3 ? ["evd_shared"] : [])] : []
    },
    requiredChecks: rng() < 0.5 ? (rng() < 0.5 ? ["test"] : ["chk_0", "chk_1"]) : [],
    constraints: Array.from({ length: Math.floor(rng() * 5) }, (_, i) => ({
      id: `c_${i}`,
      text: `constraint ${i}`,
      kind: "constraint" as const,
      mandatory: true as const
    }))
  };
}

/* ============================================================
 * F1: V-alias lemma over the REAL code (pins the axis-4b V-walk
 * inventory). assessment.evidenceRefs = insertion-ordered dedup
 * union; FAIL dims alias the SAME array; toolSituation evidenceIds
 * are content-equal but reference-distinct (S2-A-2 witnessed).
 * ============================================================ */

  let failDimsSeen = 0;

  for (let trial = 0; trial < 8000; trial += 1) {
    const observation = genObservation(rng, trial);
    const decision = assessChildObservation({ observation, episodeId: "ep_x", runId: "run_x" });
    if (!decision.apply) continue;

      if (!seen.has(id)) {
        seen.add(id);
        expected.push(id);
      }
    }
    for (const id of observation.verification?.evidenceIds ?? []) {
      if (!seen.has(id)) {
        seen.add(id);
        expected.push(id);
      }
    }
    const refs = decision.assessment.evidenceRefs;
    check(
      "F1 (evidenceRefs = insertion-ordered dedup union)",
      refs.length === expected.length && refs.every((id, i) => id === expected[i]),
      `trial ${trial}: ${JSON.stringify(refs)} vs ${JSON.stringify(expected)}`

    for (const dim of decision.assessment.dimensions) {
      if (dim.verdict === "FAIL") {
        failDimsSeen += 1;
        check(
          "F1 (FAIL dimension evidenceRefs aliases assessment.evidenceRefs)",
          dim.evidenceRefs === refs,
          `trial ${trial}: ${dim.id}`
        );
      }
    }
    // second evidenceRefsOf call lives inside prescoreInputFromObservation;
    // witness via a fresh call: content-equal, reference-distinct.

    const toolIds = pi.toolSituations[0]?.evidenceIds;
    if (toolIds !== undefined) {

        "F1 (toolSituation evidenceIds content-equal to refs)",

function fitLinear(
  p1: { x: number; t: number },
  p2: { x: number; t: number }
): { a: number; b: number } {
  const b = (p2.t - p1.t) / (p2.x - p1.x);
  return { a: p1.t - b * p1.x, b };
}

function crossingX(fit: { a: number; b: number }, perGateMsLine: number): number {
  if (fit.b <= 0) return Number.POSITIVE_INFINITY;
  return (perGateMsLine - fit.a) / fit.b;
}
