let failures = 0;
function check(name: string, ok: boolean, detail?: string): void {
  if (!ok) {
    failures += 1;
    console.error(`FAIL ${name}${detail === undefined ? "" : `: ${detail}`}`);
  }
}

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function pick<T>(rng: () => number, items: readonly T[]): T {
  return items[Math.floor(rng() * items.length)] as T;
}

function bench(fn: () => void, reps: number): number {
  fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}

async function benchAsync(fn: () => Promise<void>, reps: number): Promise<number> {
  await fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) await fn();
  return (performance.now() - t0) / reps;
}

/* Configured-state policies (R7-I lesson; k=1 is bit-identical to the
 * R8-D..R15-D generator; k>1 scales avoid/prefer counts for axis 4e). */
function configuredPolicy(variantTag: string, k = 1): string {
  const avoid = Array.from({ length: 40 * k }, (_, i) => ({
    modelId: i % 8 === 0 ? "cheap" : `m${i % 7}`,
    family: i % 3 === 0 ? pick(() => (i % 10) / 10, ["edit", "test", "review"]) : `fam${i % 12}`,
    reason: `deterministic FAIL pattern ${i} (${variantTag}): attributable replay regression observed across paired episodes with stable family-level attribution`
  }));
  const prefer = Array.from({ length: 40 * k }, (_, i) => ({
    family: i % 4 === 0 ? "test" : `fam${i % 12}`,
    modelId: i % 4 === 0 ? "cheap" : `m${(i + 1) % 7}`
  }));
  const assignments = Array.from({ length: 7 }, (_, i) => ({
    role: ["planner", "executor", "tester", "reviewer", "scout", "debugger", "worker"][i] as string,
    model: i % 2 === 0 ? "premium" : "cheap",
    family: ["edit", "test", "review"][i % 3] as string
  }));
  if (variantTag === "candidate") {
    avoid.push({ modelId: "cheap", family: "refactor", reason: "candidate-only: deterministic FAIL on refactor replay (new attribution)" });
  }
  return JSON.stringify({ primaryModelId: "premium", avoid, prefer, assignments });
}

// Regex-neutral filler for the objective axis: no analyze-task keyword hits
// change class as length grows (lorem ipsum has none of the risk/deploy
// vocabulary), so the O axis measures scan cost, not classification flips.
function neutralFiller(length: number): string {
  const unit = "lorem ipsum dolor sit amet consetetur sadipscing elitr sed diam nonumy ";
  let out = "";
  while (out.length < length) out += unit;
  return out.slice(0, length);
}

function buildRoutingRegistry(
  tag: string,

/* Configured-state registry (same construction as R8-D..R15-D). */
function buildConfiguredRegistry(
  tag: string,
  baselinePolicy = CONFIGURED_BASELINE_POLICY,
  candidatePolicy = CONFIGURED_CANDIDATE_POLICY
): {
  registry: ResourceRegistry;
  candidateId: CandidateId;
  baseline: ResourceVersion;
  identity: ResourceIdentity;
} {
  const built = buildRoutingRegistry(tag, baselinePolicy, candidatePolicy);
  const { registry } = built;
  const versions: { identity: ResourceIdentity; v1: ResourceVersion; v2?: ResourceVersion }[] = [];
  for (let i = 0; i < 8; i += 1) {
    const identity: ResourceIdentity = {
      kind: "prompt",
      name: `agent-prompt-${i}`,
      scope: { kind: "project", projectId: createProjectId(() => `${tag}proj`) }
    };
    const body = `# prompt ${i}\n${`Follow the review checklist, cite evidence for every claim, and keep the diff bounded to the stated scope (${i}).\n`.repeat(30 + i * 8)}`;
    const v1 = registry.registerBaseline({ identity, content: body, author: HUMAN });
    const candidate = registry.createCandidate({
      identity,
      content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    const entry: { identity: ResourceIdentity; v1: ResourceVersion; v2?: ResourceVersion } = { identity, v1 };
    if (i < 5) {
      const result = promoteWithRegistry(registry, {
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
      entry.v2 = result.newVersion as ResourceVersion;
    } else if (i === 5) {
      registry.beginPromotion({
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
    }
    versions.push(entry);
  }
  const first = versions[0] as { identity: ResourceIdentity; v1: ResourceVersion; v2?: ResourceVersion };
  registry.rollback({
    identity: first.identity,
    expectedCurrentVersionId: (first.v2 as ResourceVersion).versionId,
    targetVersionId: first.v1.versionId,
    reason: "degradation",
    evidence: ["ev:configured-degradation"],
    automatic: false
  });
  const second = versions[1] as { identity: ResourceIdentity; v1: ResourceVersion; v2?: ResourceVersion };
  registry.rollback({
    identity: second.identity,
    expectedCurrentVersionId: (second.v2 as ResourceVersion).versionId,
    targetVersionId: second.v1.versionId,
    reason: "user",
    evidence: ["ev:configured-user"],
    automatic: false
  });
  registry.retire((versions[2] as { v1: ResourceVersion }).v1.versionId);
  registry.retire((versions[3] as { v1: ResourceVersion }).v1.versionId);
  return built;
}

interface DatasetShape {
  readonly hashPad?: number; // episodeHash = "eh_" + "h"*pad + 6-digit index
  readonly objLen?: number; // objective = neutralFiller(objLen)
  readonly taskSuffixLen?: number; // taskId suffix total length (index-padded)
  readonly successRate?: number; // default 0.7
  readonly familyRate?: number; // default 0.5
  readonly poisonTaskIdAt?: number; // index whose taskId suffix is 65 chars (F2)
}

function episodeHashFor(i: number, shape?: DatasetShape): string {
  if (shape?.hashPad === undefined) return `eh_${i}`;
  return `eh_${"h".repeat(shape.hashPad)}${String(i).padStart(6, "0")}`;
}

function taskIdFor(i: number, shape?: DatasetShape): string {
  if (shape?.poisonTaskIdAt === i) {
    return `tsk_${"x".repeat(65)}`;
  }
  if (shape?.taskSuffixLen === undefined) return `tsk_e${i}`;
  const index = i.toString(36);
  const pad = Math.max(0, shape.taskSuffixLen - index.length);
  return `tsk_${"t".repeat(pad)}${index}`;
}

async function writeDataset(
  dir: string,
  episodes: number,
  rng: () => number,
  shape?: DatasetShape
): Promise<void> {
  await mkdir(dir, { recursive: true });
  const successRate = shape?.successRate ?? 0.7;
  const familyRate = shape?.familyRate ?? 0.5;
  const eps = Array.from({ length: episodes }, (_, i) => ({
    episodeHash: episodeHashFor(i, shape),
    taskId: taskIdFor(i, shape),
    role: pick(rng, ROLES),
    objective: shape?.objLen === undefined ? pick(rng, OBJECTIVES) : neutralFiller(shape.objLen),
    originalWorkspace: "/repos/alpha",
    ...(rng() < successRate ? { taskSuccess: rng() < 0.5 ? "PASS" : "FAIL" } : {}),
    ...(rng() < familyRate ? { taskFamily: pick(rng, ["edit", "test", "review"]) } : {})
  }));
  const manifest = { datasetId: "ds-r16d", environmentVersion: "env-1", episodes: eps };
  await writeFile(join(dir, "manifest.json"), `${JSON.stringify(manifest, null, 2)}\n`, "utf8");
}

const FORBIDDEN_POLICY_FIELDS = new Set([
  "permission",
  "credential",
  "secret",
  "token",
  "apikey",
  "password",
  "authorization"
]);
function assertNoForbiddenFieldsReplica(value: unknown): void {
  if (Array.isArray(value)) {
    for (const entry of value) {
      assertNoForbiddenFieldsReplica(entry);
    }
    return;
  }
  if (!isRecord(value)) {
    return;
  }
  for (const [key, child] of Object.entries(value)) {
    if (FORBIDDEN_POLICY_FIELDS.has(key.toLowerCase())) {
      throw new DomainValidationError(`routing-policy contains forbidden field ${key}`);
    }
    assertNoForbiddenFieldsReplica(child);
  }
}
function parseEpisodeReplica(value: unknown, index: number): EpisodeShape {
  if (!isRecord(value)) {
    throw new DomainValidationError(`dataset episodes[${index}] must be an object`);
  }
  if (typeof value.episodeHash !== "string" || value.episodeHash.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires episodeHash`);
  }
  if (typeof value.taskId !== "string") {
    throw new DomainValidationError(`dataset episodes[${index}] requires taskId`);
  }
  parseTaskId(value.taskId);
  if (!isAgentRole(value.role)) {
    throw new DomainValidationError(`dataset episodes[${index}] has invalid role`);
  }
  if (typeof value.objective !== "string" || value.objective.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires objective`);
  }
  if (typeof value.originalWorkspace !== "string" || value.originalWorkspace.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires originalWorkspace`);
  }
  let taskSuccess: "PASS" | "FAIL" | undefined;
  if (value.taskSuccess !== undefined) {
    if (value.taskSuccess !== "PASS" && value.taskSuccess !== "FAIL") {
      throw new DomainValidationError(
        `dataset episodes[${index}] taskSuccess must be PASS or FAIL when present`
      );
    }
    taskSuccess = value.taskSuccess;
  }
  const taskFamily =
    typeof value.taskFamily === "string" && value.taskFamily.trim() !== ""
      ? value.taskFamily
      : undefined;
  return {
    episodeHash: value.episodeHash,
    taskId: value.taskId,
    role: value.role,
    objective: value.objective,
    originalWorkspace: value.originalWorkspace,
    ...(taskFamily !== undefined ? { taskFamily } : {}),
    ...(taskSuccess !== undefined ? { taskSuccess } : {})
  };
}
function catalogCostReplica(catalog: ModelRouterConfig, modelId: string): number {

  if (model === undefined) {
    throw new DomainValidationError(`selected model ${modelId} is not in the catalog`);
  }
  return model.estimatedCostUsd;
}

interface ManifestShape {
  readonly datasetId: string;
  readonly environmentVersion: string;
  readonly episodes: readonly unknown[];

function parseManifestHead(parsed: unknown): ManifestShape {
  if (!isRecord(parsed)) {
    throw new DomainValidationError("dataset manifest must be an object");
  }
  if (typeof parsed.datasetId !== "string" || parsed.datasetId.trim() === "") {
    throw new DomainValidationError("dataset manifest datasetId is required");
  }
  if (typeof parsed.environmentVersion !== "string" || parsed.environmentVersion.trim() === "") {
    throw new DomainValidationError("dataset manifest environmentVersion is required");
  }
  if (!Array.isArray(parsed.episodes) || parsed.episodes.length === 0) {
    throw new DomainValidationError("dataset manifest episodes must be a non-empty array");
  }
  return { datasetId: parsed.datasetId, environmentVersion: parsed.environmentVersion, episodes: parsed.episodes };
}
/** Replica of the full evalRoutingPolicy pipeline from prepared file bytes. */
function replicaReport(
  registry: ResourceRegistry,
  candidateId: CandidateId,
  manifestRaw: string,
  datasetDir: string,
  outputRoot: string
): RoutingEvalReport {
  const candidate = registry.getCandidate(candidateId) as ImprovementCandidate;
  const candidateContent = registry.getContent(candidate.contentHash) as string;
  assertNoForbiddenFieldsReplica(JSON.parse(candidateContent));
  const candidatePolicy = parseLearnedRoutingPolicy(candidateContent);
  const parent = registry.getVersion(candidate.parentVersionId) as ResourceVersion;
  const baselineContent = registry.getContent(parent.contentHash) as string;
  assertNoForbiddenFieldsReplica(JSON.parse(baselineContent));
  const baselinePolicy = parseLearnedRoutingPolicy(baselineContent);
  const manifest = parseManifestHead(JSON.parse(manifestRaw) as unknown);
  const episodes = manifest.episodes.map((entry, index) => parseEpisodeReplica(entry, index));
  createIsolationGuard({
    readOnlyRoots: [...episodes.map((episode) => episode.originalWorkspace), datasetDir],
    outputRoot
  });
  const catalog = catalogFromPrimary({ primaryModelId: baselinePolicy.primaryModelId });
  const tasks = episodes.map((episode) => ({
    taskId: parseTaskId(episode.taskId),
    role: episode.role,
    objective: episode.objective
  }));
  const baseline = assignTasks({ catalog, tasks, learned: baselinePolicy });

    candidateHash: candidate.contentHash,
    environmentVersion: manifest.environmentVersion,
    evaluatorVersion: ROUTING_EVALUATOR_VERSION
  });

    stableStringify({

      cache: {
        candidateHash: candidate.contentHash,
        environmentVersion: manifest.environmentVersion,

  }, reps);

    manifestKB: manifestRaw.length / 1024
  };
}

function fitLeastSquares(
  points: readonly { readonly x: number; readonly y: number }[]
): { slope: number; intercept: number } {
  const n = points.length;
  let sx = 0;
  let sy = 0;
  let sxx = 0;
  let sxy = 0;
  for (const p of points) {
    sx += p.x;
    sy += p.y;
    sxx += p.x * p.x;
    sxy += p.x * p.y;
  }
  const slope = (n * sxy - sx * sy) / (n * sxx - sx * sx);
  const intercept = (sy - slope * sx) / n;
  return { slope, intercept };
}
function crossing(fit: { slope: number; intercept: number }, lineMs: number): number {
  if (fit.slope <= 0) return Number.POSITIVE_INFINITY;
  return (lineMs - fit.intercept) / fit.slope;
}
