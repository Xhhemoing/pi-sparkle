## §3 新轴 SYSCENSUS：内核边界 syscall 普查（首创；无候选）

**动机与新颖性**：R1-G…R15-G 全部测量都在 JS/fs-API 边界之上（bench
计时、原型计数、A/A 标定、引擎代际）；「I/O 动词已契约饱和」的裁决
始终依赖 store 层 API 面清点。本轮首次沉到内核动词层直接验证：若 JS
层之下存在任何隐藏可删 I/O（重复 open、多余 stat、冗余读、未见 sync
变体），只有 ptrace 全量清点能看见。前十五轮无一轮做过（非 R14-G
引擎态/规模计时轴、非 R15-G A/A 计时轴的换名）。

**方法（边际 K-缩放）**：驱动脚本（附录 B，全文 + sha256）执行 K 次
规范 start→COMPLETED 命令循环——夹具与归档矩阵 C1 格逐字节同形

器）。`strace -f -c` 包全进程（tsx 父进程 + node 全线程），取
K=8 / 48 / 88 三尺度，边际 = (count(K₂)−count(K₁))/40。启动/模块加载

/**
 * R16-G kernel-boundary syscall census driver (NOT committed; archived in
 * the report appendix). Performs K canonical start->COMPLETED command
 * cycles (byte-identical fixture shape to the archived R12-G matrix C1
 * cell: 3-node chain, supplied childResults, fixed clock, sequential ids).
 * Run under `strace -f -c` at two K scales; the marginal syscall count per
 * run is (count(K2) - count(K1)) / (K2 - K1), which cancels process
 * startup, module loading, and JIT noise.
 *
 * Usage: npx tsx tmp-r16g-census.ts <K>
 */
import { mkdtempSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";

import { validateConfidenceScore, type FlowEdge, type FlowNode, type Flowchart } from "./src/domain/flowchart.js";
import { createTaskId } from "./src/domain/ids.js";
import { parseIsoTimestamp } from "./src/domain/timestamp.js";
import { createModelRouter, type ModelRouter } from "./src/supervisor/model-router.js";
import { createFilePauseController } from "./src/run/pause-controller.js";
import { startFlowchartRun } from "./src/run/flowchart-run.js";
import type { ChildNodeResult } from "./src/supervisor/flowchart-supervisor.js";

const K = Number(process.argv[2] ?? "8");

function sequenceGenerator(): () => string {
  let n = 0;
  return () => `00000000-0000-4000-8000-${String(n++).padStart(12, "0")}`;
}

const routerConfig = {
  policyVersion: "router-v1",
  models: [
    { id: "cheap", version: "cheap-v1", roles: ["actor", "critic"] as const, maxComplexity: "MEDIUM" as const, estimatedCostUsd: 0.1, estimatedDurationMs: 1_000 },
    { id: "premium", version: "premium-v1", roles: ["actor", "critic", "judge", "router"] as const, maxComplexity: "HIGH" as const, estimatedCostUsd: 0.5, estimatedDurationMs: 4_000 }
  ]
};
function router(): ModelRouter { return createModelRouter(routerConfig); }

function mnode(id: string): FlowNode {
  return {
    id,
    taskId: createTaskId(() => id),
    role: "actor",
    objective: `Do ${id}`,
    modelPolicy: { allowedModels: ["cheap", "premium"] },
    confidenceThreshold: validateConfidenceScore(0.7),
    approvalRequired: false
  };
}
const sEdge = (from: string, to: string): FlowEdge => ({ from, to, condition: { type: "success", expected: true } });
function chainFlowchart(): Flowchart {
  return {
    id: "census-chain",
    nodes: [mnode("a"), mnode("b"), mnode("c")],
    edges: [sEdge("a", "b"), sEdge("b", "c")]
  };
}
function fakeResult(confidence: number, evidence: string): ChildNodeResult {
  return {
    outcome: "SUCCESS",
    confidence: validateConfidenceScore(confidence),
    evidenceIds: [evidence],
    facts: [{ key: "coverage", value: "green", confidence: validateConfidenceScore(confidence) }]
  };
}

const stateRoot = mkdtempSync(join(tmpdir(), "r16g-census-state-"));
const projectRoot = mkdtempSync(join(tmpdir(), "r16g-census-proj-"));
try {
  const now = () => parseIsoTimestamp("2026-08-24T09:00:00.000Z");
  const generateId = sequenceGenerator();
  let completed = 0;
  for (let i = 0; i < K; i += 1) {
    const deps = {
      stateRoot,
      router: router(),
      now,
      generateId,
      pause: createFilePauseController(stateRoot, now)
    };
    const out = await startFlowchartRun(deps, {
      projectRoot,
      flowchart: chainFlowchart(),
      objective: "census chain",
      childResults: { a: fakeResult(0.9, "evd_a"), b: fakeResult(0.9, "evd_b"), c: fakeResult(0.9, "evd_c") }
    });
    if (out.status === "COMPLETED") completed += 1;
  }
  console.log(`K=${K} completed=${completed}`);
} finally {
  rmSync(stateRoot, { recursive: true, force: true });
  rmSync(projectRoot, { recursive: true, force: true });
}
