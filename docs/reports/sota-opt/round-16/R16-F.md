保存为 `tmp-r16f-axes.ts` 后 `npx tsx tmp-r16f-axes.ts` 运行。这不是候选裁决仿真（本轮无候选进入测量门），是 §1.3 三条新轴的收尾测量。

```ts
/**
 * R16-F new-axis measurements (temporary; NOT candidate adjudications and NOT
 * committed — no winner this round). Three axes no earlier F round measured
 * directly, mirroring the Round-16 sibling practice (R16-A/D measured scale
 * crossing points; R16-B added an in-process sampling-profiler budget):
 *
 *   axis 1  CPU sampling-profile attribution of the full-experiment anchor.
 *           Fifteen F rounds priced the anchor by component timing (validate
 *           ~83%, scan pools, copy pools, rng pool) but never asked a
 *           profiler "where do the samples actually land?". This runs the
 *           production anchor under an in-process `node:inspector` sampling
 *           session (same method as R16-B §2d) and aggregates self/inclusive
 *           time per function. If the ledger arithmetic is right, the profile
 *           must reproduce it through an independent instrument — and any
 *           hidden site the component-timing method cannot see (iterator
 *           machinery, spread/copy helpers, GC) shows up as a named line.
 *
 *   axis 2  A-scaling crossing points for the closed quadratic pools. The
 *           reopen conditions on S8-F-1 / S9-F-1 (assign-site membership),
 *           S9-F-2 (assignments.some witness), S1-F-8 (outcomes dedup + cost
 *           re-accumulation) and the S2-F-1 rng-replay pool are all keyed on
 *           "A grows >= 1 order of magnitude", but no round measured at what
 *           A each pool actually crosses an absolute landing-eligible size.
 *           This measures the anchor and each pool at A ∈ {500, 1000, 2000,
 *           4000} (P = 2A, anchor shape preserved), fits the quadratic
 *           coefficient on the two largest cells, and reports the crossing
 *           A* at 10 ms and 30 ms per pool — turning the qualitative reopen
 *           conditions into measured thresholds.
 *
 *   axis 3  string-encoding cell (one-byte vs two-byte population entries),
 *           completing the fixture-representation family: R14-F measured
 *           provenance (interned vs de-interned), R15-F measured length
 *           (L 11→96); both used ASCII (one-byte) strings, and the
 *           production hash source (src/domain/hash.ts hash32) emits ASCII
 *           hex — so two-byte is upward stress only. A single char above
 *           U+00FF forces V8's two-byte representation, doubling the byte
 *           width behind every content compare and first-time hash. This
 *           cell re-prices the anchor at L=11 in all four
 *           (encoding × provenance) corners and checks the decision
 *           sequences stay identical (the rng path is content-independent).
 *
 * Uses the production modules directly. Run: npx tsx tmp-r16f-axes.ts
 */

import { Session } from "node:inspector/promises";
import { createCandidateId, createResourceVersionId } from "./src/domain/ids.js";
import { validateExperimentPlan, type ExperimentPlan } from "./src/experiments/plan.js";
import { createSeededRng } from "./src/experiments/replay.js";
import { createShadowRunner, type ExperimentOutcome, type ShadowState } from "./src/experiments/shadow.js";

function timeMs(fn: () => void, rounds = 5): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

/** Population entries of exact length 11, same shape family as the historical
 *  fixture (`ep_` + zero-padded base36 index — high common prefix). A prefix
 *  containing a char above U+00FF forces the two-byte representation. */
function makePopulation(size: number, prefix = "ep_"): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`${prefix}${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

function makeShadowPlan(population: readonly string[], maxAssignments: number): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: "exp_r16f_bench",
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r16fbase"),
    candidateId: createCandidateId(() => "r16fcand"),
    population,
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments, maxWallClockMs: 1e12 },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

function fullExperiment(plan: ExperimentPlan, hashes: readonly string[], A: number): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (let k = 0; k < A; k++) {
    const h = hashes[k] as string;
    state = runner.assign(state, h, 0);
    const outcome: ExperimentOutcome = {
      episodeHash: h,
      utility: 0.5,
      costUsd: 0.01,
      guardrailBreached: false,
    };
    state = runner.recordOutcome(state, outcome, 0);
  }
  if (state.assignments.length !== A) throw new Error("unexpected assignment count");
  return state;
}

/* ------------------------------------------------------------------ */
/* axis 1 — sampling-profile attribution of the anchor (P=2000/A=1000) */
/* ------------------------------------------------------------------ */
const P0 = 2000;
const A0 = 1000;
const anchorPopulation = makePopulation(P0);
const anchorPlan = makeShadowPlan(anchorPopulation, A0);
validateExperimentPlan(anchorPlan);
const anchorHashes = anchorPopulation.slice(0, A0);

// warm through the same JIT state before profiling
fullExperiment(anchorPlan, anchorHashes, A0);
fullExperiment(anchorPlan, anchorHashes, A0);

interface RawProfileNode {
  readonly id: number;
  readonly callFrame: { readonly functionName: string; readonly url: string };
  readonly hitCount?: number;
  readonly children?: readonly number[];
}

const session = new Session();
session.connect();
await session.post("Profiler.enable");
await session.post("Profiler.setSamplingInterval", { interval: 100 });
await session.post("Profiler.start");
const PROF_RUNS = 15;
const tp0 = process.hrtime.bigint();
for (let r = 0; r < PROF_RUNS; r++) {
  fullExperiment(anchorPlan, anchorHashes, A0);
}
const tp1 = process.hrtime.bigint();
const { profile } = await session.post("Profiler.stop");
await session.post("Profiler.disable");
session.disconnect();
const profWallMs = Number(tp1 - tp0) / 1e6;

const nodes = profile.nodes as readonly RawProfileNode[];
const byId = new Map<number, RawProfileNode>();
for (const node of nodes) byId.set(node.id, node);
const inclusive = new Map<number, number>();
function inclusiveHits(id: number): number {
  const cached = inclusive.get(id);
  if (cached !== undefined) return cached;
  const node = byId.get(id) as RawProfileNode;
  let total = node.hitCount ?? 0;
  for (const child of node.children ?? []) total += inclusiveHits(child);
  inclusive.set(id, total);
  return total;
}
let totalHits = 0;
for (const node of nodes) totalHits += node.hitCount ?? 0;
const selfByName = new Map<string, number>();
const inclByName = new Map<string, number>();
for (const node of nodes) {
  const name = node.callFrame.functionName === "" ? "(anonymous)" : node.callFrame.functionName;
  selfByName.set(name, (selfByName.get(name) ?? 0) + (node.hitCount ?? 0));
  // none of the slice functions are recursive on the anchor path, so summing
  // per-node inclusive hits per name does not double count here
  inclByName.set(name, (inclByName.get(name) ?? 0) + inclusiveHits(node.id));
}
const msPerHit = profWallMs / totalHits;
const perExperiment = (hits: number): string => ((hits * msPerHit) / PROF_RUNS).toFixed(2);
const pct = (hits: number): string => ((hits / totalHits) * 100).toFixed(1);
const topSelf = [...selfByName.entries()].sort((a, b) => b[1] - a[1]).slice(0, 12);
process.stdout.write(
  `axis 1 (profile): ${PROF_RUNS} anchor runs wall=${profWallMs.toFixed(1)}ms samples=${totalHits} (${(profWallMs / PROF_RUNS).toFixed(2)}ms/run under profiler)\n`
);
for (const [name, hits] of topSelf) {
  process.stdout.write(
    `axis 1 self: ${name.padEnd(28)} ${pct(hits).padStart(5)}% = ${perExperiment(hits)}ms/experiment\n`
  );
}
for (const name of [
  "validateExperimentPlan",
  "assertUniqueNonEmpty",
  "restoreShadowState",
  "recordExperimentOutcome",
  "shadowDecisionAt",
  "requirePopulationEpisode",
  "assign",
  "recordOutcome",
]) {
  const incl = inclByName.get(name) ?? 0;
  process.stdout.write(
    `axis 1 incl: ${name.padEnd(28)} ${pct(incl).padStart(5)}% = ${perExperiment(incl)}ms/experiment\n`
  );
}

/* ------------------------------------------------------------------ */
/* axis 2 — A-scaling crossing points for the closed quadratic pools   */
/* ------------------------------------------------------------------ */
const CELLS = [500, 1000, 2000, 4000] as const;
interface CellResult {
  readonly A: number;
  readonly anchor: number;
  readonly scan: number;
  readonly cost: number;
  readonly rng: number;
  readonly includes: number;
  readonly validate2A: number;
}
const cells: CellResult[] = [];
for (const A of CELLS) {
  const P = 2 * A;
  const population = makePopulation(P);
  const plan = makeShadowPlan(population, A);
  validateExperimentPlan(plan);
  const hashes = population.slice(0, A);
  const rounds = A <= 1000 ? 5 : A <= 2000 ? 3 : 2;

  const anchor = timeMs(() => {
    fullExperiment(plan, hashes, A);
  }, rounds);

  // one O(A²/2) linear string-compare scan pool — the shared shape of the
  // three per-call scans (requireUniqueAssignment miss at assign,
  // assignments.some hit-at-end and outcomes dedup miss at recordOutcome)
  let scanSink = 0;
  const scan = timeMs(() => {
    for (let k = 1; k <= A; k++) {
      const probe = hashes[k - 1] as string;
      for (let i = 0; i < k; i++) {
        if (hashes[i] === probe) scanSink += 1;
      }
    }
  }, rounds);

  // cost re-accumulation pool (accumulatedCostUsd shape): Σ k float adds
  const state = fullExperiment(plan, hashes, A);
  const outcomes = state.outcomes;
  let costSink = 0;
  const cost = timeMs(() => {
    for (let k = 1; k <= A; k++) {
      let total = 0;
      for (let i = 0; i < k; i++) {
        const item = outcomes[i] as ExperimentOutcome;
        if (item.missing === true) continue;
        total += item.costUsd;
      }
      costSink += total;
    }
  }, rounds);

  // rng replay pool (shadowDecisionAt shape): Σ (k+1) mulberry32 steps
  let rngSink = 0;
  const rng = timeMs(() => {
    for (let k = 0; k < A; k++) {
      const draw = createSeededRng(42);
      let value = 0;
      for (let i = 0; i <= k; i++) value = draw();
      rngSink += value;
    }
  }, rounds);

  // assign-site includes pool: hit at index k (anchor order)
  let incSink = 0;
  const includes = timeMs(() => {
    for (let k = 0; k < A; k++) {
      if (population.includes(hashes[k] as string)) incSink += 1;
    }
  }, rounds);

  // validate pool: 2A fail-closed re-validations (assign + recordOutcome)
  const validate2A = timeMs(() => {
    for (let r = 0; r < 2 * A; r++) validateExperimentPlan(plan);
  }, A <= 1000 ? 3 : 2);

  cells.push({ A, anchor, scan, cost, rng, includes, validate2A });
  process.stdout.write(
    `axis 2 A=${String(A).padStart(4)} (P=${P}): anchor=${anchor.toFixed(2)}ms | scan-pool=${scan.toFixed(2)}ms ` +
      `cost-pool=${cost.toFixed(2)}ms rng-pool=${rng.toFixed(2)}ms includes=${includes.toFixed(2)}ms ` +
      `validate×2A=${validate2A.toFixed(2)}ms\n`
  );
  void scanSink;
  void costSink;
  void rngSink;
  void incSink;
}
// quadratic fit on the two largest cells: pool(A) = c·A² → A*(target) = sqrt(target/c)
const last = cells[cells.length - 1] as CellResult;
const prev = cells[cells.length - 2] as CellResult;
function crossing(name: string, at: (cell: CellResult) => number): void {
  const c = (at(last) / (last.A * last.A) + at(prev) / (prev.A * prev.A)) / 2;
  const a10 = Math.sqrt(10 / c);
  const a30 = Math.sqrt(30 / c);
  process.stdout.write(
    `axis 2 crossing ${name.padEnd(12)}: c=${(c * 1e6).toFixed(3)}ns/A² → A*(10ms)≈${Math.round(a10)} A*(30ms)≈${Math.round(a30)}\n`
  );
}
crossing("scan-pool", (cell) => cell.scan);
crossing("cost-pool", (cell) => cell.cost);
crossing("rng-pool", (cell) => cell.rng);
crossing("includes", (cell) => cell.includes);
crossing("anchor", (cell) => cell.anchor);

/* ------------------------------------------------------------------ */
/* axis 3 — encoding cell: one-byte vs two-byte × interned/de-interned */
/* ------------------------------------------------------------------ */
const ENCODINGS = [
  ["one-byte", "ep_"],
  ["two-byte", "\u0113p_"],
] as const;
let referenceDecisions: string | undefined;
for (const [label, prefix] of ENCODINGS) {
  const population = makePopulation(P0, prefix);
  const plan = makeShadowPlan(population, A0);
  validateExperimentPlan(plan);
  const interned = population.slice(0, A0);
  const deinterned = JSON.parse(JSON.stringify(interned)) as string[];
  for (let i = 0; i < A0; i++) {
    if (deinterned[i] !== interned[i]) throw new Error("round-trip mismatch");
  }

  // determinism: decision sequence is content-independent, and interned vs
  // de-interned callers land field-identical states
  const sInt = fullExperiment(plan, interned, A0);
  const sDeint = fullExperiment(plan, deinterned, A0);
  const decisions = sInt.assignments.map((a) => a.shadowDecision).join(",");
  if (referenceDecisions === undefined) {
    referenceDecisions = decisions;
  } else if (decisions !== referenceDecisions) {
    throw new Error(`decision sequence diverged for ${label}`);
  }
  if (
    sInt.halted !== sDeint.halted ||
    sInt.haltReason !== sDeint.haltReason ||
    sInt.guardrailBreaches !== sDeint.guardrailBreaches ||
    sInt.assignments.length !== sDeint.assignments.length ||
    sInt.outcomes.length !== sDeint.outcomes.length
  ) {
    throw new Error(`state divergence for ${label}`);
  }
  for (let i = 0; i < A0; i++) {
    const a = sInt.assignments[i];
    const b = sDeint.assignments[i];
    if (a?.episodeHash !== b?.episodeHash || a?.shadowDecision !== b?.shadowDecision) {
      throw new Error(`assignment divergence for ${label} index=${i}`);
    }
  }

  const tInt = timeMs(() => {
    fullExperiment(plan, interned, A0);
  }, 5);
  const tDeint = timeMs(() => {
    fullExperiment(plan, deinterned, A0);
  }, 5);
  const REP = 2000;
  let alnSink = 0;
  const tAln = timeMs(() => {
    for (let r = 0; r < REP; r++) {
      for (let i = 0; i < A0; i++) {
        if (deinterned[i] === population[i]) alnSink += 1;
      }
    }
  }, 5);
  const tValidate = timeMs(() => {
    for (let r = 0; r < REP; r++) validateExperimentPlan(plan);
  }, 3);
  process.stdout.write(
    `axis 3 ${label}: anchor interned=${tInt.toFixed(2)}ms de-interned=${tDeint.toFixed(2)}ms ` +
      `provenance-delta=${(tDeint - tInt).toFixed(2)}ms | de-interned aligned-compare x${REP}=${tAln.toFixed(2)}ms | ` +
      `validate x${REP}=${tValidate.toFixed(2)}ms\n`
  );
  void alnSink;
}
process.stdout.write(
  "axis 3 determinism: decision sequences identical across encodings; provenance variants field-identical\n"
);
```
