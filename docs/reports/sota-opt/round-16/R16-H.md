let failures = 0;
let checks = 0;
function check(name: string, ok: boolean, detail?: string): void {
  checks += 1;
  if (!ok) {
    failures += 1;
    process.stderr.write(`FAIL ${name}${detail === undefined ? "" : `: ${detail}`}\n`);
  }
}
function log(line: string): void {
  process.stdout.write(`${line}\n`);
}
function median(xs: number[]): number {
  const sorted = [...xs].sort((a, b) => a - b);
  return sorted[Math.floor(sorted.length / 2)] ?? Number.NaN;
}
function bench(fn: () => void, reps: number): number {
  fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}
async function benchAsync(fn: () => Promise<void>, reps: number): Promise<number> {
  await fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) await fn();
  return (performance.now() - t0) / reps;
}
/** Time-boxed variant for cells whose per-call cost spans 5 µs..seconds. */
async function benchAdaptive(fn: () => Promise<void>, budgetMs: number): Promise<{ ms: number; reps: number }> {
  await fn();
  let t0 = performance.now();
  await fn();
  const est = performance.now() - t0;
  const reps = Math.max(1, Math.min(3000, Math.floor(budgetMs / Math.max(est, 0.0005))));
  t0 = performance.now();
  for (let i = 0; i < reps; i += 1) await fn();
  return { ms: (performance.now() - t0) / reps, reps };
}

 * Hot tier, three states (R8-H..R15-H anchor shapes). Measured FIRST so
 * anchor conditions stay comparable (benchAsync single warm call + 3000
 * timed reps, no extra chain executions before).

async function measureChain(label: string, objective: string, habits: HeuristicHabits | undefined): Promise<void> {
  const args = { objective, ...(habits !== undefined ? { habits } : {}) };
  const chain = await benchAsync(async () => {
    await extractHeuristicContract(args);
  }, 3000);
  const prod = await extractHeuristicContract(args);
  const real = prod.contract;
  const tasks: CoverageTaskRef[] = Array.from({ length: 5 }, (_, i) => ({
    id: `tsk_0000000${i}` as TaskId,
    acceptanceCriteria: real.acceptanceCriteria.map((criterion) => ({ id: criterion.id }))
  }));
  const options: CoverageStartOptions | undefined =
    real.questions.length > 0
      ? { resolvedQuestionIds: real.questions.map((question) => question.id) }
      : undefined;
  const gate = bench(() => assertCoverageAllowsStart(real, tasks, options), 50000);
  const precedence = bench(() => void applyPrecedence(real, "user-first"), 100000);
  const total = chain + gate + precedence;
  log(
    `CEILING ${label}: extractHeuristicContract=${(chain * 1e6).toFixed(0)}ns + run-start gate(Q=${real.questions.length}${options !== undefined ? ", all resolved" : ", no options"})=${(gate * 1e6).toFixed(0)}ns + applyPrecedence=${(precedence * 1e6).toFixed(0)}ns = ${(total * 1e6).toFixed(0)}ns once-per-run production total`
  );
}

/* ================================================================
 * NEW AXIS part 1: string-substrate fidelity (mirrors R16-B's corpus
 * string-substrate axis). Fresh substrate string per iteration — exactly
 * like production, where each run's objective is a freshly materialized
 * string. Construction-only loops are measured too so the substrate delta
 * can be attributed.
 * ================================================================ */
{
  const jsonPayload = JSON.stringify({ objective: OBJ });
  const mid = Math.floor(OBJ.length / 2);
  const padParent = `<<<<PAD>>>>${OBJ}<<<<PAD>>>>`;
  const CJK_SUFFIX = "（登录重试）";

  const makeJson = (): string => (JSON.parse(jsonPayload) as { objective: string }).objective;
  const makeCons = (): string => OBJ.slice(0, mid) + OBJ.slice(mid);
  const makeSliced = (): string => padParent.substring(11, 11 + OBJ.length);
  const makeTwoByte = (): string => OBJ + CJK_SUFFIX;

  // Deterministic equivalence: ASCII substrates must yield byte-identical
  // objectives and shape-identical contracts vs the flat-literal control.
  const flatProd = await extractHeuristicContract({ objective: OBJ });
  const shapeOf = (c: RequirementContract): string =>
    [
      c.acceptanceCriteria.map((a) => a.id).join("|"),
      c.deliverables.map((d) => d.id).join("|"),
      c.questions.map((q) => q.id).join("|"),
      c.constraints.map((k) => k.id).join("|")
    ].join(" // ");
  for (const [name, make] of [
    ["JSON", makeJson],
    ["CONS", makeCons],
    ["SLICED", makeSliced]
  ] as const) {
    const s = make();
    check(`substrate ${name} produces identical bytes`, s === OBJ && s.length === OBJ.length);
    const prod = await extractHeuristicContract({ objective: make() });
    check(
      `substrate ${name} contract identical shape + confidence`,
      shapeOf(prod.contract) === shapeOf(flatProd.contract) &&
        prod.confidence === flatProd.confidence &&
        prod.contract.objective === flatProd.contract.objective
    );
  }
  {
    const prod2 = await extractHeuristicContract({ objective: makeTwoByte() });
    check(
      "substrate TWOBYTE same shape (C=2, Q=0, conf 0.86, session.ts target found)",
      prod2.contract.acceptanceCriteria.length === 2 &&
        prod2.contract.questions.length === 0 &&
        prod2.confidence === 0.86 &&
        prod2.contract.deliverables.some((d) => d.description === "src/auth/session.ts")
    );
  }

  // Paired position-alternating windows vs the flat-literal control —
  // drift-robust (a sequential protocol here would alias JIT tiering drift
  // into the substrate delta; the R15-H A/A floor for CHAIN-level duels is
  // ±0.82-2.25µs/window, so the median over 6 windows is the adjudicated
  // reading and only magnitude-class conclusions are drawn).
  const cells: [string, () => string][] = [
    ["JSON.parse product", makeJson],
    ["ConsString (a+b)", makeCons],
    ["SlicedString (substring)", makeSliced],
    ["Two-byte (CJK suffix)", makeTwoByte]
  ];
  const chainOn = (make: () => string) => async (): Promise<void> => {
    await extractHeuristicContract({ objective: make() });
  };
  const flatArm = async (): Promise<void> => {
    await extractHeuristicContract({ objective: OBJ });
  };
  for (const [name, make] of cells) {
    const makeOnly = bench(() => void make(), 200000);
    const cellArm = chainOn(make);
    // warm both arms identically
    for (let i = 0; i < 300; i += 1) {
      await flatArm();
      await cellArm();
    }
    const deltas: number[] = [];
    const flats: number[] = [];
    for (let w = 0; w < 6; w += 1) {
      let tFlat: number;
      let tCell: number;
      if (w % 2 === 0) {
        tFlat = await benchAsync(flatArm, 1000);
        tCell = await benchAsync(cellArm, 1000);
      } else {
        tCell = await benchAsync(cellArm, 1000);
        tFlat = await benchAsync(flatArm, 1000);
      }
      flats.push(tFlat);
      deltas.push((tCell - tFlat - makeOnly) * 1e6);
    }
    const med = median(deltas);
    const flatMed = median(flats) * 1e6;
    log(
      `SUBSTRATE ${name}: median paired delta=${med.toFixed(0)}ns/run (windows [${deltas.map((d) => d.toFixed(0)).join(", ")}]; flat-arm median=${flatMed.toFixed(0)}ns; make-only=${(makeOnly * 1e6).toFixed(1)}ns)`
    );
    check(`substrate cell finite (${name})`, deltas.every(Number.isFinite));
    check(
      `substrate ${name} no cliff (|median delta| < 1× flat chain)`,
      Math.abs(med) < flatMed
    );
  }
}

/* ================================================================
 * NEW AXIS part 2: adversarial shape × length scan (payload-shape axis;
 * extends R13-H Face R, which scanned LENGTH on a benign corpus only).
 * Measures extractHeuristicContract alone. Adaptive reps; a shape whose
 * per-call cost exceeds 1200ms stops scanning larger lengths (the growth
 * law is already pinned by then).
 * ================================================================ */
{
  const PREFIX = "fix the retry bug now "; // 5 words, has verb, length>=12 → never vague
  const BENIGN_UNIT = "fix the login retry bug in src/auth/session.ts and keep tests green ";
  const mkShape = (shape: string, L: number): string => {
    const body = Math.max(0, L - PREFIX.length);
    switch (shape) {
      case "BENIGN":
        return (PREFIX + BENIGN_UNIT.repeat(Math.ceil(body / BENIGN_UNIT.length))).slice(0, L);
      case "WORDRUN":
        return PREFIX + "a".repeat(body);
      case "DOTRUN":
        return (PREFIX + "a.".repeat(Math.ceil(body / 2))).slice(0, L);
      case "SLASHRUN":
        return (PREFIX + "a/".repeat(Math.ceil(body / 2))).slice(0, L);
      default:
        throw new Error(`unknown shape ${shape}`);
    }
  };
  const lengths = [64, 512, 4096, 32768];
  for (const shape of ["BENIGN", "WORDRUN", "DOTRUN", "SLASHRUN"]) {
    const points: { L: number; ms: number }[] = [];
    for (const L of lengths) {
      const objective = mkShape(shape, L);
      check(`shape ${shape} L=${L} exact length`, objective.length === L);
      const { ms, reps } = await benchAdaptive(async () => {
        await extractHeuristicContract({ objective });
      }, 1500);
      points.push({ L, ms });
      log(`SHAPE ${shape} L=${L}: ${ms >= 1 ? `${ms.toFixed(2)}ms` : `${(ms * 1e6).toFixed(0)}ns`}/call (reps=${reps})`);
      check(`shape cell finite (${shape} L=${L})`, Number.isFinite(ms) && ms > 0);
      if (ms > 1200) break; // growth law pinned; don't burn minutes on the next decade
    }
    if (points.length >= 2) {
      const p1 = points[points.length - 2]!;
      const p2 = points[points.length - 1]!;
      const k = Math.log(p2.ms / p1.ms) / Math.log(p2.L / p1.L);
      const crossing = p2.ms >= 10 ? p2.L : Math.round(p2.L * Math.pow(10 / p2.ms, 1 / k));
      log(
        `SHAPE ${shape} fit: tail exponent k=${k.toFixed(2)} → measured-curve 10ms crossing ≈ ${crossing.toLocaleString("en-US")} chars${p2.ms >= 10 ? " (measured directly at/below largest cell)" : ""}`
      );
    }
  }

  // Component attribution: pin the measured k=2 growth on the PATH_RE leg
  // (namedTargets) — the other regex legs (isVague / shouldScout) must stay
  // linear on the same pathological payloads.
  {
    const { namedTargets, isVague, shouldScout } = await import("/workspace/src/requirement/heuristic.js");
    for (const shape of ["WORDRUN", "DOTRUN"]) {
      const s512 = mkShape(shape, 512);
      const s4096 = mkShape(shape, 4096);
      const nt512 = bench(() => void namedTargets(s512), 2000);
      const nt4096 = bench(() => void namedTargets(s4096), 50);
      const iv4096 = bench(() => void isVague(s4096), 2000);
      const sc4096 = bench(() => void shouldScout(s4096), 2000);
      const kNt = Math.log(nt4096 / nt512) / Math.log(8);
      log(
        `ATTRIB ${shape}: namedTargets 512→4096 = ${(nt512 * 1e6).toFixed(0)}→${(nt4096 * 1e6).toFixed(0)}ns (k=${kNt.toFixed(2)}); isVague(4096)=${(iv4096 * 1e6).toFixed(0)}ns shouldScout(4096)=${(sc4096 * 1e6).toFixed(0)}ns — quadratic lives in PATH_RE, not the linear legs`
      );
      check(`attribution: ${shape} namedTargets superlinear (k>1.6)`, kNt > 1.6);
      check(
        `attribution: ${shape} linear legs stay far below namedTargets at 4096`,
        iv4096 < nt4096 / 10 && sc4096 < nt4096 / 10
      );
    }
  }

  // REALPASTE: the realistic worst case a user can actually paste — a 4KB
  // stack trace after "fix this crash:".
  {
    const frame = "    at Object.handler (/srv/app/node_modules/express/lib/router/layer.js:95:5)\n";
    let paste = "fix this crash: ";
    while (paste.length < 4096) paste += frame;
    paste = paste.slice(0, 4096);
    const { ms, reps } = await benchAdaptive(async () => {
      await extractHeuristicContract({ objective: paste });
    }, 1500);
    log(`SHAPE REALPASTE (4KB stack trace): ${ms >= 1 ? `${ms.toFixed(2)}ms` : `${(ms * 1e6).toFixed(0)}ns`}/call (reps=${reps})`);
    check("REALPASTE finite and sub-ms (realistic worst case stays µs-class)", Number.isFinite(ms) && ms > 0 && ms < 1);
    const prod = await extractHeuristicContract({ objective: paste });
    check(
      "REALPASTE deterministic shape (not vague, conf 0.86, path deliverables, q-tests emitted)",
      prod.confidence === 0.86 &&
        prod.contract.deliverables.length > 1 &&
        prod.contract.questions.some((q) => q.id === "q-tests")
    );
  }
}
