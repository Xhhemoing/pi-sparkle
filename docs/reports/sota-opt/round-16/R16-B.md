## 2e. 新测量格 B：语料字符串基底保真（rope vs 生产 JSON 平坦串）

2. **R15-B「单窗符号会说谎」的跨运行现场复演**：M=10 批级差额 run1/run2

   税改用同种子二次生成，两臂互不污染。后续轮次若需复测基底，必须

/* ============================================================
 * §0d NEW measurement cell A: CPU-profile cost-attribution decomposition.
 * V8 sampling profiler (node:inspector, in-process session, 100us interval)
 * attached around a window of production replay batches. For every slice
 * function we report the measured self share and inclusive (subtree) share
 * of on-CPU time. share x ceiling = a hard budget for any candidate confined
 * to that site. Regex/irregexp time attributes to the calling JS frame's
 * self time (noted in the report). (program)/(garbage collector) rows keep
 * VM overhead visible and cross-check R14-B's GC budget cell.
 * ============================================================ */
interface ProfileNode {
  readonly id: number;
  readonly callFrame: { readonly functionName: string; readonly url: string; readonly lineNumber: number };
  readonly hitCount?: number;
  readonly children?: readonly number[];
}
interface Profile {
  readonly nodes: readonly ProfileNode[];
  readonly startTime: number;
  readonly endTime: number;
  readonly samples?: readonly number[];
  readonly timeDeltas?: readonly number[];
}
function post<T>(session: Session, method: string, params?: object): Promise<T> {
  return new Promise((resolve, reject) => {
    (session.post as (m: string, p: object | undefined, cb: (err: Error | null, res: unknown) => void) => void)(
      method, params, (err, res) => (err ? reject(err) : resolve(res as T))
    );
  });
}
async function profileCell(
  label: string,
  catalog: ModelRouterConfig,
  learned: LearnedRoutingPolicy,
  batches: number
): Promise<void> {
  const replayTasks = genReplayTasks(mulberry32(0xb44b01), 2000);
  const run = (): void => {
    assignTasks({ catalog, tasks: replayTasks, learned });
  };
  // Steady-state warm-up (same discipline as every prior pass).
  for (let i = 0; i < 10; i += 1) run();
  const session = new Session();
  session.connect();
  await post(session, "Profiler.enable");
  await post(session, "Profiler.setSamplingInterval", { interval: 100 });
  await post(session, "Profiler.start");
  const t0 = performance.now();
  for (let i = 0; i < batches; i += 1) run();
  const windowMs = performance.now() - t0;
  const { profile } = await post<{ profile: Profile }>(session, "Profiler.stop");
  session.disconnect();

  const byId = new Map<number, ProfileNode>();
  for (const node of profile.nodes) byId.set(node.id, node);
  // Subtree (inclusive) hit counts via iterative post-order.
  const inclusive = new Map<number, number>();
  const computeInclusive = (rootId: number): number => {
    const stack: [number, boolean][] = [[rootId, false]];
    while (stack.length > 0) {
      const [id, expanded] = stack.pop()!;
      const node = byId.get(id)!;
      if (!expanded) {
        stack.push([id, true]);
        for (const child of node.children ?? []) stack.push([child, false]);
      } else {
        let total = node.hitCount ?? 0;
        for (const child of node.children ?? []) total += inclusive.get(child) ?? 0;
        inclusive.set(id, total);
      }
    }
    return inclusive.get(rootId)!;
  };
  for (const node of profile.nodes) if (!inclusive.has(node.id)) computeInclusive(node.id);
  let totalHits = 0;
  for (const node of profile.nodes) totalHits += node.hitCount ?? 0;

  const selfByName = new Map<string, number>();
  const inclByName = new Map<string, number>();
  const nameOf = (node: ProfileNode): string => {
    const fn = node.callFrame.functionName === "" ? "(anonymous)" : node.callFrame.functionName;
    const url = node.callFrame.url;
    const src = url.includes("/src/") ? url.slice(url.indexOf("/src/") + 1) : url === "" ? "" : "(vm)";
    return src === "" ? fn : `${fn} @${src}:${node.callFrame.lineNumber + 1}`;
  };
  for (const node of profile.nodes) {
    const key = nameOf(node);
    selfByName.set(key, (selfByName.get(key) ?? 0) + (node.hitCount ?? 0));
    inclByName.set(key, (inclByName.get(key) ?? 0) + (inclusive.get(node.id) ?? 0));
  }
  const usPerHit = ((profile.endTime - profile.startTime) || windowMs * 1e3) / Math.max(1, totalHits);
  const rows = [...selfByName.entries()]
    .map(([key, self]) => ({ key, self, incl: inclByName.get(key) ?? 0 }))
    .sort((a, b) => b.self - a.self);
  console.log(
    `profile-cell [${label}]: window=${windowMs.toFixed(0)}ms/${batches} batches | samples=${totalHits} (~${usPerHit.toFixed(0)}us/sample)`
  );
  const topN = rows.slice(0, 24);
  for (const row of topN) {
    console.log(
      `  self=${((row.self / totalHits) * 100).toFixed(2).padStart(6)}% incl=${((row.incl / totalHits) * 100).toFixed(2).padStart(6)}%  ${row.key}`
    );
  }
  // Named slice-site budget lines (per-eval x2 budget = incl share x window/batches x 2).
  const SITES = [
    "assignTasks", "assignPlanned", "analyzeTask", "familyOf", "complexityOf", "route",
    "validateInput", "resolveRouteDefaults", "partitionLiveCandidates", "evaluateLiveCandidate",
    "evaluateCandidate", "toModelDescriptor", "buildDecision", "makeApprovalPlan",
    "oneHotDistribution", "coldStartRoutingScore", "selectLiveModel", "compareLiveCandidates",
    "planAssignmentPolicy", "pickPreferredModel", "applyLearnedRouting", "flowchartRoleForAgentRole",
    "createModelRouter", "validateConfig", "catalogModel", "(garbage collector)", "(program)"
  ];
  const perBatchMs = windowMs / batches;
  const budgetBits: string[] = [];
  for (const site of SITES) {
    let self = 0;
    let incl = 0;
    for (const [key, value] of selfByName) {
      if (key === site || key.startsWith(`${site} @`)) {
        self += value;
        incl += inclByName.get(key) ?? 0;
      }
    }
    if (self === 0 && incl === 0) continue;
    const inclShare = incl / totalHits;
    budgetBits.push(
      `${site}: self=${((self / totalHits) * 100).toFixed(2)}% incl=${(inclShare * 100).toFixed(2)}% budget=${(inclShare * perBatchMs * 2).toFixed(3)}ms/eval`
    );
  }
  console.log(`  site-budgets [${label}]: ${budgetBits.join(" | ")}`);
  if (label === "M=7 configured, learned") {
    const biggest = rows.filter((r) => r.key.includes("@src/"))[0];
    conclusions.push(
      `profile M=7 top-slice-site self=${biggest === undefined ? "n/a" : `${biggest.key.split(" @")[0]}:${((biggest.self / totalHits) * 100).toFixed(1)}%`} per-batch=${perBatchMs.toFixed(2)}ms (largest-site-budget-below-landing-line=true)`
    );
  }
}
{
  await profileCell("M=2 default, learned", catalogFromPrimary({ primaryModelId: "premium" }), learnedM2(), 400);
  await profileCell("M=7 configured, learned", configuredCatalog(), configuredLearned(), 400);
  await profileCell("M=10 stress, learned", tenModelCatalog(), tenModelLearned(), 300);
}

/**
 * R16-B substrate attestation probe v2 (native-predicate-free).
 * V8 12.4 exposes no cons-string predicate (%StringIsCons/%StringIsFlat are
 * absent; %FlattenString preserves identity), so cons-ness is attested by its
 * only observable: the one-time flattening tax on first regex touch.
 * Arm G is built fresh by the fifteen-pass corpus generator (ropes via +=)
 * and is NEVER serialized before the timed first sweep (JSON.stringify would
 * flatten it in place). Arm P is a JSON round-trip of an INDEPENDENT
 * generation with the same seed (identical values, fresh flat strings).
 * Pass 1 sweep timing difference = flattening tax (attests ropes existed).
 * Pass 2 sweep timing difference = steady-state residual (what the main
 * harness substrate cell measures).
 * Run: node --import tsx /tmp/r16b-substrate-probe.mts

import { performance } from "node:perf_hooks";

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function pick<T>(rng: () => number, items: readonly T[]): T {
  return items[Math.floor(rng() * items.length)] as T;
}
const ROLES = ["worker", "scout", "planner", "implementer", "reviewer", "tester", "debugger"] as const;
const OBJECTIVE_TEMPLATES = [
  "Survey the payment module", "Plan the checkout migration", "Run the unit tests",
  "Deploy payment credentials to production", "Implement retry logic for the ledger sync",
  "Review the audit log formatting nits", "Refactor and rename the tracking roller",
  "Fix the flaky spec for gate transitions", "Research and compare vector store options",
  "Investigate why the drop table migration failed in prod", "验证路由细节并补充测试", ""

  "deploy", "production", "credentials", "review", "tests", "coverage", "plan",

function genObjective(rng: () => number): string {
  if (rng() < 0.5) return pick(rng, OBJECTIVE_TEMPLATES);

}
interface Row { readonly role: string; readonly objective: string }
