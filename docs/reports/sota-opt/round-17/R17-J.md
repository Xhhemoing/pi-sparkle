```ts
function priceThrow(label: string, n: number, op: () => void, expectMessagePart: string, expectDomain: boolean): void {
  // Fidelity: assert the endpoint actually rejects with the pinned error once.
  let witnessed = false;
  try {
    op();
  } catch (error) {
    witnessed = true;
    if (expectDomain && !(error instanceof DomainValidationError)) {
      throw new Error(`${label}: expected DomainValidationError, got ${String(error)}`);
    }
    const message = error instanceof Error ? error.message : String(error);
    if (!message.includes(expectMessagePart)) {
      throw new Error(`${label}: message "${message}" missing "${expectMessagePart}"`);
    }
  }
  if (!witnessed) throw new Error(`${label}: endpoint did not reject`);
  // Warmup then timed reps, catch included (caller-experienced price).
  for (let i = 0; i < Math.min(n, 20000); i++) {
    try { op(); } catch { sink += 1; }
  }
  const t0 = hr();
  for (let i = 0; i < n; i++) {
    try { op(); } catch { sink += 1; }
  }
  const t1 = hr();
  console.log(`  ${label}: ${fmtNs(nsPer(t0, t1, n))}/rejection`);
}

}
```

```ts
/**
 * R17-J new-axis harness B: cold-process budget. J-slice counterpart of the
 * R17-A (once-per-run ICU collator ~5.0ms) and R17-F (+16.8-22.0ms first
 * experiment once-per-process) cold-budget cells: with each cell run in its
 * own fresh process, what does the FIRST operation of each production path
 * cost over its steady state (JIT warmup, libuv threadpool spin-up, first
 * mkdir/openat on a cold dentry cache)? Timing starts after module import,
 * so the figure isolates the per-path first-op premium, not the loader.
 * Note: the J slice has no localeCompare call sites, so the R17-A ICU
 * collator pathology is structurally absent here (verified by grep).
 * Usage: tsx tmp-r17j-cold.ts <save|jsonl-false|jsonl-true|cascade|index|plan>
 * Temporary script: not committed (no production winner -> report only).
 */
import { mkdtempSync, rmSync, mkdirSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join, dirname } from "node:path";

import {
  configurePreferencePersistence,
  recordPreference,
  resetPreferenceStore
} from "./src/preferences/store.js";
import { appendJsonlLine } from "./src/persist/jsonl.js";
import { cascadeFeedbackTombstones } from "./src/privacy/deletion.js";
import { feedbackLogPath, feedbackTombstonesPath } from "./src/feedback/store.js";
import { buildProjectContextIndex, type CodeMapSymbol } from "./src/context/index.js";
import { applyAnswers } from "./src/track/clarify.js";
import { planFromContract, acceptanceForRole } from "./src/track/plan.js";
import type { ProjectSnapshot } from "./src/domain/project.js";
import type { RequirementContract, DecisionQuestion } from "./src/domain/contract.js";
import type { EpisodeId } from "./src/domain/ids.js";

const hr = (): bigint => process.hrtime.bigint();
const usOf = (a: bigint, b: bigint): number => Number(b - a) / 1000;
const cell = process.argv[2] ?? "";
let idCounter = 0;
const gen = (): string => `bench${(idCounter++).toString(36)}`;

function report(label: string, firstUs: number, steadyUs: number, n: number): void {
  console.log(
    `${label}: first=${firstUs.toFixed(1)}us steady(mean of ${n})=${steadyUs.toFixed(1)}us premium=${(firstUs - steadyUs).toFixed(1)}us`
  );
}

function seedFeedback(stateRoot: string): void {
  const log = feedbackLogPath(stateRoot);
  mkdirSync(dirname(log), { recursive: true });
  const lines: string[] = [];
  for (let i = 0; i < 200; i++) {
    const hit = i % 10 === 0;
    lines.push(
      JSON.stringify({
        id: `fb-${i}`,
        episodeId: hit ? "eps_hit" : `eps_other${i % 7}`,
        kind: "human",
        rubricVersion: "r1",
        score: 50 + (i % 40),
        evidenceRefs: [],
        redacted: false,
        createdAt: "2026-08-24T00:00:00.000Z",
        body: `free text feedback payload number ${i} `.repeat(3),
        summary: `summary ${i}`
      })
    );
  }
  writeFileSync(log, lines.join("\n") + "\n", "utf8");
  writeFileSync(
    feedbackTombstonesPath(stateRoot),
    JSON.stringify(["fb-pre-a", "fb-pre-b", "fb-pre-c", "fb-pre-d", "fb-pre-e"], null, 2) + "\n",
    "utf8"
  );
}

async function main(): Promise<void> {
  if (cell === "save") {
    // First persisted preference write in the process: creates the adaptation
    // dir + snapshot file, JITs the applyObservation/rebuildViews/saveToDisk
    // chain. Steady: writes 2..201 on the same store.
    const dir = mkdtempSync(join(tmpdir(), "r17j-cold-pref-"));
    resetPreferenceStore();
    configurePreferencePersistence(join(dir, "sub", "prefs.json"));
    const write = (i: number): void => {
      recordPreference("project", `key-${i % 7}`, `pref-${i % 11}`, `value-w${i}`, `eps_w${i.toString(36)}` as EpisodeId, 1.0, false);
    };
    const t0 = hr();
    write(0);
    const t1 = hr();
    const n = 200;
    const t2 = hr();
    for (let i = 1; i <= n; i++) write(i);
    const t3 = hr();
    report("save-cold", usOf(t0, t1), usOf(t2, t3) / n, n);
    resetPreferenceStore();
    configurePreferencePersistence(undefined);
    rmSync(dir, { recursive: true, force: true });
  } else if (cell === "jsonl-false" || cell === "jsonl-true") {
    // First append in the process: first async fs op (threadpool spin-up +
    // eventfd machinery), dir + file creation. Steady on the same file.
    const fsync = cell === "jsonl-true";
    const dir = mkdtempSync(join(tmpdir(), "r17j-cold-jsonl-"));
    const file = join(dir, "sub", "log.jsonl");
    const line = JSON.stringify({ id: "evt_x", type: "BENCH", payload: { n: 42, text: "x".repeat(120) } });
    const t0 = hr();
    await appendJsonlLine(file, line, fsync);
    const t1 = hr();
    const n = fsync ? 100 : 500;
    const t2 = hr();
    for (let i = 0; i < n; i++) await appendJsonlLine(file, line, fsync);
    const t3 = hr();
    report(`${cell}-cold`, usOf(t0, t1), usOf(t2, t3) / n, n);
    rmSync(dir, { recursive: true, force: true });
  } else if (cell === "cascade") {
    // Seeding is fully synchronous, so cascade #1 is the first async fs op in
    // the process AND the first execution of the cascade chain (JIT cold).
    const stateRoot = mkdtempSync(join(tmpdir(), "r17j-cold-casc-"));
    seedFeedback(stateRoot);
    const t0 = hr();
    const first = await cascadeFeedbackTombstones(stateRoot, "eps_hit" as EpisodeId);
    const t1 = hr();
    if (first.length !== 20) throw new Error(`expected 20 cascaded, got ${first.length}`);
    const n = 20;
    let total = 0n;
    for (let i = 0; i < n; i++) {
      seedFeedback(stateRoot);
      const t2 = hr();
      const out = await cascadeFeedbackTombstones(stateRoot, "eps_hit" as EpisodeId);
      const t3 = hr();
      if (out.length !== 20) throw new Error(`expected 20 cascaded, got ${out.length}`);
      total += t3 - t2;
    }
    report("cascade-cold", usOf(t0, t1), Number(total) / 1000 / n, n);
    rmSync(stateRoot, { recursive: true, force: true });
  } else if (cell === "index") {
    // Pure CPU: first build vs steady isolates JIT/IC warmup for the
    // once-per-run index path.
    const facts: { key: string; value: string; confidence: string }[] = [];
    for (let i = 0; i < 12; i++) facts.push({ key: `architecture.layer${i}`, value: `layer ${i} does thing ${i}`, confidence: "HIGH" });
    for (let i = 0; i < 12; i++) facts.push({ key: `risk.area${i}`, value: `risk in area ${i}`, confidence: "MEDIUM" });
    for (let i = 0; i < 16; i++) facts.push({ key: `misc.fact${i}`, value: `misc value ${i}`, confidence: "LOW" });
    const codeMap: CodeMapSymbol[] = [];
    for (let i = 0; i < 30; i++) {
      codeMap.push({
        path: `src/mod${i % 8}/file${i}.ts`,
        symbol: `symbol${i}`,
        kind: (["class", "function", "interface", "type", "method", "module"] as const)[i % 6]!,
        public: i % 3 === 0,
        calls: [`callee${i % 5}`, `callee${(i + 2) % 5}`]
      });
    }
    const snapshot = {
      id: "prj_bench0001",
      rootPath: "/repo",
      discoveredAt: "2026-08-24T00:00:00.000Z",
      instructionFiles: [{ path: "/repo/AGENTS.md" }, { path: "/repo/docs/AGENTS.md" }, { path: "/repo/src/deep/AGENTS.md" }],
      manifests: [{ path: "/repo/package.json" }, { path: "/repo/pnpm-lock.yaml" }, { path: "/repo/tsconfig.json" }, { path: "/repo/sub/package.json" }],
      commands: [
        { name: "test", command: "pnpm test" },
        { name: "lint", command: "pnpm lint" },
        { name: "build", command: "pnpm build" }
      ],
      facts
    } as unknown as ProjectSnapshot;
    const options = {
      dirtyPaths: ["/repo/src/a.ts", "/repo/src/b.ts", "/repo/gen/out.js", "/repo/src/c.ts", "/repo/docs/x.md", "/repo/src/d.ts"],
      generatedPaths: ["/repo/gen", "/repo/dist", "/repo/.cache"],
      sourceHashes: { "/repo/package.json": "h1", "/repo/AGENTS.md": "h2", "architecture.layer0": "h3" },
      priorHashes: { "/repo/package.json": "h1", "/repo/AGENTS.md": "hOLD" },
      codeMap
    };
    let sink = 0;
    const t0 = hr();
    sink += buildProjectContextIndex(snapshot, options).facts.length;
    const t1 = hr();
    const n = 2000;
    const t2 = hr();
    for (let i = 0; i < n; i++) sink += buildProjectContextIndex(snapshot, options).facts.length;
    const t3 = hr();
    report("index-cold", usOf(t0, t1), usOf(t2, t3) / n, n);
    console.log(`[sink=${sink}]`);
  } else if (cell === "plan") {
    const questions: DecisionQuestion[] = [
      { id: "q-tests", question: "Include tests?", options: ["yes", "no"], default: "yes" },
      { id: "q-done", question: "What does done look like?", options: [], default: "shipped" },
      { id: "q-write", question: "Write code or investigate?", options: ["write", "investigate"], default: "write" }
    ];
    const answers = { "q-tests": "yes", "q-done": "feature shipped and verified", "q-write": "write the code" } as const;
    const contract: RequirementContract = {
      schemaVersion: 1,
      objective: "Implement the new export module and add tests covering redaction",
      deliverables: [],
      constraints: [{ id: "c-smallest", description: "smallest change", enforceable: false } as never],
      nonGoals: [],
      acceptanceCriteria: [
        { id: "ac-1", description: "Export module produces valid JSON", observableCheck: "run export" },
        { id: "ac-tests", description: "Unit tests pass", observableCheck: "pnpm test" },
        { id: "ac-2", description: "Redaction verified on fixtures", observableCheck: "fixture diff" }
      ],
      assumptions: [],
      questions,
      authority: [],
      sourceRefs: []
    };
    const habits = { preferReview: true } as const;
    let sink = 0;
    const once = (): void => {
      const applied = applyAnswers(questions, answers);
      const children = planFromContract({ contract, habits, answers: applied.resolved, generateId: gen });
      for (const child of children) sink += acceptanceForRole(child.role, contract).length;
    };
    const t0 = hr();
    once();
    const t1 = hr();
    const n = 20000;
    const t2 = hr();
    for (let i = 0; i < n; i++) once();
    const t3 = hr();
    report("plan-cold", usOf(t0, t1), usOf(t2, t3) / n, n);
    console.log(`[sink=${sink}]`);
  } else {
    throw new Error(`unknown cell ${cell}`);
  }
}

main().catch((error) => {
  console.error(error);
  process.exit(1);
});
```
