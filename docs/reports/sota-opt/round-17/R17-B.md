function mulberry32(seed: number): () => number {
  let a = seed >>> 0;

    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);

/* ============================================================
 * Faithful per-task replica of assignTasks' internal pipeline, used by cells
 * A (refusal census) and B (per-task timing). Byte-identical output to
 * production assignTasks is ASSERTED before any replica-derived number is
 * reported. The replica exists because assignPlanned is module-private; it
 * reproduces its exact composition through the public surface.
 * ============================================================ */
interface PerTaskPipeline {
  readonly runTask: (task: AssignableTask) => TaskAssignment;
}
function buildPipeline(
  catalog: ModelRouterConfig,
  learned: LearnedRoutingPolicy | undefined,
  prior: PublicPriorSnapshot | undefined
): PerTaskPipeline {
  const router = createModelRouter(catalog);
  const catalogIds = catalog.models.map((model) => model.id);
  const plan = planAssignmentPolicy(router.config.models, catalogIds);
  const limits: RoutingLimits = { remainingTimeMs: Number.MAX_SAFE_INTEGER };
  const runTask = (task: AssignableTask): TaskAssignment => {
    const analysis = analyzeTask(task.objective, task.role, {
      ...(task.contractRisk !== undefined ? { contractRisk: task.contractRisk } : {}),
      ...(task.contextTokens !== undefined ? { contextTokens: task.contextTokens } : {}),
      ...(task.outputTokens !== undefined ? { outputTokens: task.outputTokens } : {})
    });
    let allowedModels: readonly string[] = [...plan.allowedIds];
    let preferredModel = pickPreferredModel(plan, analysis, prior);
    if (learned !== undefined) {
      const applied = applyLearnedRouting(analysis.family, allowedModels, preferredModel, learned);
      allowedModels = [...applied.allowedModels];
      preferredModel = applied.preferredModel;
    }
    const decision = router.route({
      taskId: task.taskId,
      role: flowchartRoleForAgentRole(task.role),
      complexity: analysis.complexity,
      modelPolicy: { allowedModels, preferredModel },
      approvalRequired: analysis.highRisk,
      highRisk: analysis.highRisk,
      family: analysis.family,
      featureVersion: ASSIGN_FEATURE_VERSION,
      agentRole: task.role,
      requiredCapabilities: analysis.requiredCapabilities,
      ...(analysis.contextTokens !== undefined ? { contextNeeded: analysis.contextTokens } : {}),
      ...(analysis.outputTokens !== undefined ? { outputNeeded: analysis.outputTokens } : {}),
      limits
    });
    return { taskId: task.taskId, role: task.role, analysis, decision, allowedModels, preferredModel };
  };
  return { runTask };
}

{
  const replayTasks = genReplayTasks(mulberry32(0xb44b01), 2000);

 * concentration census. Per-task timing: min over 8 sweeps (min denoises GC
 * pauses and scheduler preemption); quantiles + class shares from those mins.
 * ============================================================ */
{

    ["M=2 default", catalogFromPrimary({ primaryModelId: "premium" }), learnedM2()],

    // Warm-up sweeps to steady state.
    for (let i = 0; i < 4; i += 1) for (const task of replayTasks) pipeline.runTask(task);
    const n = replayTasks.length;
    const minCostUs = new Float64Array(n).fill(Number.POSITIVE_INFINITY);
    for (let sweep = 0; sweep < 8; sweep += 1) {
      for (let i = 0; i < n; i += 1) {
        const task = replayTasks[i]!;
        const t0 = performance.now();
        pipeline.runTask(task);
        const cost = (performance.now() - t0) * 1e3;
        if (cost < minCostUs[i]!) minCostUs[i] = cost;
      }
    }
    const sorted = [...minCostUs].sort((a, b) => a - b);
    const q = (p: number): number => sorted[Math.min(n - 1, Math.floor(p * n))]!;
    const total = sorted.reduce((acc, v) => acc + v, 0);
    const maxUs = sorted[n - 1]!;
    console.log(
      `per-task distribution N=2000 [${label}, learned]: p50=${q(0.5).toFixed(2)}us p90=${q(0.9).toFixed(2)}us p99=${q(0.99).toFixed(2)}us p99.9=${q(0.999).toFixed(2)}us max=${maxUs.toFixed(2)}us | sum=${(total / 1e3).toFixed(2)}ms | tail-cap max*N=${((maxUs * n) / 1e3).toFixed(1)}ms(hypothetical)`
    );
    // Class concentration: share of total per-task cost by class.
    const byClass = new Map<string, number>();
    const addClass = (key: string, cost: number): void => {
      byClass.set(key, (byClass.get(key) ?? 0) + cost);
    };
    for (let i = 0; i < n; i += 1) {
      const task = replayTasks[i]!;
      const assignment = pipeline.runTask(task);
      const cost = minCostUs[i]!;
      addClass(`risk=${assignment.analysis.highRisk}`, cost);
      addClass(`family=${assignment.analysis.family}`, cost);
      const len = task.objective.length;
      addClass(`len=${len === 0 ? "0" : len <= 30 ? "1-30" : len <= 60 ? "31-60" : len <= 120 ? "61-120" : ">120"}`, cost);
    }
    const shares = [...byClass.entries()]
      .map(([key, cost]) => ({ key, sharePct: (cost / total) * 100, ms: cost / 1e3 }))
      .sort((a, b) => b.sharePct - a.sharePct);
    const top = shares.filter((s) => s.key.startsWith("family=") || s.key.startsWith("risk=")).slice(0, 6);
    console.log(
      `  class shares [${label}]: ${top.map((s) => `${s.key}:${s.sharePct.toFixed(1)}%(${s.ms.toFixed(2)}ms)`).join(" | ")}`
    );
    const lenShares = shares.filter((s) => s.key.startsWith("len="));
    console.log(
      `  length-bucket shares [${label}]: ${lenShares.map((s) => `${s.key}:${s.sharePct.toFixed(1)}%`).join(" | ")}`
    );
    if (label === "M=10 stress") {
      const biggestFamily = shares.find((s) => s.key.startsWith("family="))!;
      conclusions.push(
        `per-task tail M=10: p50=${q(0.5).toFixed(2)}us p99=${q(0.99).toFixed(2)}us max=${maxUs.toFixed(2)}us (max/p50=${(maxUs / q(0.5)).toFixed(1)}x); largest class ${biggestFamily.key}=${biggestFamily.sharePct.toFixed(1)}% => no class concentrates a pool beyond the sealed site budgets`
      );
    }
  }
}
