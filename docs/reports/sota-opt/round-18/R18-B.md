  "部署", "生产环境", "凭据", "评审", "测试", "覆盖率", "规划",
  "设计", "调研", "研究", "重构", "实现", "新增", "修复",
  "迁移", "验证", "\n", "该模块", "碎片", "、"
];
function genCjkObjective(rng: () => number): string {
  if (rng() < 0.5) return pick(rng, CJK_TEMPLATES);
  let text = "";
  const parts = 1 + Math.floor(rng() * 8);
  for (let i = 0; i < parts; i += 1) text += pick(rng, CJK_FRAGMENTS) + "、";
  return text;
}

}

    const produced = assignTasks({ catalog, tasks: replayTasks, ...(learned !== undefined ? { learned } : {}) });
    let budgetRows = 0;
    let deadlineRows = 0;
    let otherRows = 0;
    for (const item of produced) {
      for (const row of item.decision.rejections) {
        if (row.constraint === "budget") budgetRows += 1;
        else if (row.constraint === "deadline") deadlineRows += 1;
        else otherRows += 1;
      }
    }
    check(
      `limits census [${label}]: zero budget/deadline rejection rows under default limits`,
      budgetRows === 0 && deadlineRows === 0,
      `budget=${budgetRows} deadline=${deadlineRows}`
    );
    console.log(
      `limits census [${label}]: budget rows=${budgetRows} deadline rows=${deadlineRows} other rows=${otherRows}`
    );
  }
  // (b) Pressure catalog: m0..m7 expensive+slow+unapproved, m8/m9 cheap+fast+
  // approved — finite budget/deadline excludes m0..m7 on EVERY task while
  // every task still routes (m8/m9 clear high-risk too). This is the only
  // configuration shape that turns finite limits into steady-state refusal-
  // row formatting flow instead of an R17-B fail-fast abort.
  const pressureCatalog: ModelRouterConfig = {
    policyVersion: "sim-r18b-pressure",
    models: Array.from({ length: 10 }, (_, i) =>
      i < 8
        ? {
            id: `m${i}`,
            version: `m${i}-v1`,
            roles: ["actor", "critic"] as const,
            maxComplexity: "HIGH" as TaskComplexity,
            estimatedCostUsd: 0.3 + 0.05 * i,
            estimatedDurationMs: 8_000 + 500 * i,
            approvedForHighRisk: false
          }
        : {
            id: `m${i}`,
            version: `m${i}-v1`,
            roles: ["actor", "critic"] as const,
            maxComplexity: "HIGH" as TaskComplexity,
            estimatedCostUsd: i === 8 ? 0.05 : 0.06,
            estimatedDurationMs: i === 8 ? 800 : 900,
            approvedForHighRisk: true
          }
    )
  };
  const arms: readonly [string, { remainingCostUsd?: number; remainingTimeMs?: number }, number, number][] = [
    ["infinite limits", {}, 0, 0],
    ["budget=0.25", { remainingCostUsd: 0.25 }, 8, 0],
    ["deadline=1000ms", { remainingTimeMs: 1000 }, 0, 8],
    ["budget+deadline", { remainingCostUsd: 0.25, remainingTimeMs: 1000 }, 8, 8]
  ];
  let infCost = 0;
  for (const [label, limitArgs, expectBudget, expectDeadline] of arms) {
    const produced = assignTasks({ catalog: pressureCatalog, tasks: replayTasks, ...limitArgs });
    const pressured = expectBudget > 0 || expectDeadline > 0;
    let rowShapeOk = true;
    for (const item of produced) {
      let budgetRows = 0;
      let deadlineRows = 0;
      for (const row of item.decision.rejections) {
        if (row.constraint === "budget") budgetRows += 1;
        else if (row.constraint === "deadline") deadlineRows += 1;
      }
      if (budgetRows !== expectBudget || deadlineRows !== expectDeadline) rowShapeOk = false;
      // Under finite pressure only m8/m9 stay eligible. In the infinite
      // baseline arm preferPrimary tasks legitimately route to the preferred
      // most-expensive model m7 (plan.primaryPreferredId), so membership is
      // only asserted on the pressured arms.
      if (pressured && item.decision.model !== "m8" && item.decision.model !== "m9") rowShapeOk = false;
    }
    check(
      `limits pressure [${label}]: every task routes${pressured ? " to m8/m9" : ""} with ${expectBudget} budget + ${expectDeadline} deadline rows`,
      rowShapeOk
    );
    const cost = bench(() => assignTasks({ catalog: pressureCatalog, tasks: replayTasks, ...limitArgs }), 30);
    if (label === "infinite limits") infCost = cost;
    console.log(
      `limits pressure replay N=2000 [${label}]: ${(cost * 1e3).toFixed(1)}us per call (${(cost * 2).toFixed(2)}ms per eval x2)${label === "infinite limits" ? "" : ` | delta vs infinite=${((cost - infCost) * 2).toFixed(2)}ms per eval`}`
    );
    if (label === "budget+deadline") {
      conclusions.push(
        `limits-pressure hypothetical-N2000 budget+deadline (16 formatted rows/task): per-eval=${(cost * 2).toFixed(1)}ms delta=${((cost - infCost) * 2).toFixed(1)}ms (still-below-landing-line=${cost * 2 < 30})`
      );
    }
  }
  // (c) Unit per-row price + limits-shape probe on the default M=10 catalog.
  const pressureRouter = createModelRouter(pressureCatalog);
  const allIds = pressureCatalog.models.map((m) => m.id);
  const baseInput = {
    taskId: "tsk_limits" as TaskId,
    role: "actor" as const,
    complexity: "MEDIUM" as TaskComplexity,
    modelPolicy: { allowedModels: allIds },
    family: "edit",
    featureVersion: ASSIGN_FEATURE_VERSION,
    requiredCapabilities: ["tool-use"]
  };
  const infUnit = bench(() => {
    pressureRouter.route({ ...baseInput, limits: { remainingTimeMs: Number.MAX_SAFE_INTEGER } });
  }, 20000);
  const budgetUnit = bench(() => {
    pressureRouter.route({
      ...baseInput,
      limits: { remainingTimeMs: Number.MAX_SAFE_INTEGER, remainingCostUsd: 0.25 }
    });
  }, 20000);
  const bothUnit = bench(() => {
    pressureRouter.route({ ...baseInput, limits: { remainingTimeMs: 1000, remainingCostUsd: 0.25 } });
  }, 20000);
  console.log(
    `limits unit price (pressure catalog M=10): infinite=${(infUnit * 1e6).toFixed(0)}ns | budget(8 rows)=${(budgetUnit * 1e6).toFixed(0)}ns | budget+deadline(16 rows)=${(bothUnit * 1e6).toFixed(0)}ns per route | per-formatted-row≈${(((bothUnit - infUnit) / 16) * 1e6).toFixed(0)}ns`
  );
  const router10 = createModelRouter(tenModelCatalog());
  const ids10 = tenModelCatalog().models.map((m) => m.id);
  const shapeBase = { ...baseInput, modelPolicy: { allowedModels: ids10 } };
  const shapeTime = bench(() => {
    router10.route({ ...shapeBase, limits: { remainingTimeMs: Number.MAX_SAFE_INTEGER } });
  }, 20000);
  const shapeBoth = bench(() => {
    router10.route({
      ...shapeBase,
      limits: { remainingTimeMs: Number.MAX_SAFE_INTEGER, remainingCostUsd: 1e12 }
    });
  }, 20000);
  console.log(
    `limits shape probe (default M=10, zero refusal rows both arms): {time}=${(shapeTime * 1e6).toFixed(0)}ns vs {time,cost}=${(shapeBoth * 1e6).toFixed(0)}ns per route (delta=${((shapeBoth - shapeTime) * 1e6).toFixed(1)}ns)`
  );
  // (d) Real-scale configured face: N=30 with pressure limits.
  const cli = genTasks(mulberry32(0xb44b01), 30);
  const cliInf = bench(() => assignTasks({ catalog: pressureCatalog, tasks: cli }), 2000);
  const cliPressure = bench(
    () => assignTasks({ catalog: pressureCatalog, tasks: cli, remainingCostUsd: 0.25, remainingTimeMs: 1000 }),
    2000
  );
  console.log(
    `limits real-scale live face N=30 (pressure catalog): infinite=${(cliInf * 1e3).toFixed(1)}us | budget+deadline=${(cliPressure * 1e3).toFixed(1)}us per call (delta=${((cliPressure - cliInf) * 1e3).toFixed(1)}us)`
  );
  conclusions.push(
    `limits real-scale configured face N=30: delta=${((cliPressure - cliInf) * 1e3).toFixed(1)}us per batch; unit per-formatted-row≈${(((bothUnit - infUnit) / 16) * 1e6).toFixed(0)}ns`
  );
}

console.log(`\nCONCLUSIONS: ${conclusions.join(" | ")}`);
if (failures > 0) {
  console.error(`\n${failures} check(s) FAILED`);
  process.exit(1);
}
