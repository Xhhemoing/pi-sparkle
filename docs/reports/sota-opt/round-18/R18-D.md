function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/* 1c: promote WHOLE transaction (CLI pre-reads + lock(load + gate + op +
 * save)) on K fresh state dirs, with accounting closure. */
{
  const report = configuredEvalReport as NonNullable<typeof configuredEvalReport>;
  const K = 30;
  interface PromoteDir {
    readonly stateRoot: string;
    readonly contentFile: string;
    readonly reviewFile: string;
    readonly evalFile: string;
    readonly candidateId: CandidateId;
    readonly expected: ResourceVersionId;
    readonly newVersionId?: ResourceVersionId;
  }
  const dirs: PromoteDir[] = [];
  for (let i = 0; i < K; i += 1) {
    const stateRoot = join(workRoot, `state-tx-p${i}`);
    const { registry, candidateId, baseline } = buildConfiguredRegistry(`tx${i}`);
    const candidate = registry.getCandidate(candidateId) as ImprovementCandidate;
    await saveAdaptationRegistry(stateRoot, registry);
    const contentFile = join(stateRoot, "candidate-policy.json");
    const reviewFile = join(stateRoot, "review.json");
    const evalFile = join(stateRoot, "eval.json");
    await writeFile(contentFile, CONFIGURED_CANDIDATE_POLICY, "utf8");
    await writeFile(
      reviewFile,
      JSON.stringify(mkReview(candidateId, candidate.contentHash, `tx${i}`), null, 2),
      "utf8"
    );
    await writeFile(
      evalFile,
      JSON.stringify({ ...report, candidateId, contentHash: candidate.contentHash }, null, 2),
      "utf8"
    );
    dirs.push({ stateRoot, contentFile, reviewFile, evalFile, candidateId, expected: baseline.versionId });
  }

  // CLI pre-read glue in isolation (readFile x3 + JSON.parse x2 + slice-owned
  // parsePromotionReview + parseRoutingEvalReport), on dir 0.
  const first = dirs[0] as PromoteDir;
  const prereadUs = (await benchAsync(async () => {
    const content = await readFile(first.contentFile, "utf8");
    const review = parsePromotionReview(JSON.parse(await readFile(first.reviewFile, "utf8")) as unknown);
    const evalReport = parseRoutingEvalReport(JSON.parse(await readFile(first.evalFile, "utf8")) as unknown);
    if (content.length === 0 || review.candidateId !== first.candidateId || evalReport.candidateId !== first.candidateId) {
      throw new Error("preread fidelity");
    }
  }, 50)) * 1e3;

  // Whole transaction, one pass over K fresh dirs (promote mutates state).
  const newVersionIds: ResourceVersionId[] = [];
  const t0 = performance.now();
  for (const dir of dirs) {
    const content = await readFile(dir.contentFile, "utf8");
    const review = parsePromotionReview(JSON.parse(await readFile(dir.reviewFile, "utf8")) as unknown);
    const evalReport = parseRoutingEvalReport(JSON.parse(await readFile(dir.evalFile, "utf8")) as unknown);
    const result = await withAdaptationRegistryLock(dir.stateRoot, async () => {
      const registry = await loadAdaptationRegistry(dir.stateRoot);
      const candidate = registry.getCandidate(dir.candidateId);
      if (candidate === undefined) {
        throw new DomainValidationError(`unknown candidate: ${dir.candidateId}`);
      }
      if (candidate.identity.kind === "routing-policy" && evalReport === undefined) {
        throw new DomainValidationError("routing-policy promote requires --eval-file");
      }
      const input: PromoteInput = {
        candidateId: dir.candidateId,
        expectedCurrentVersionId: dir.expected,
        content,
        approvedBy: { kind: "human", identity: "cli-user" },
        review,
        changeNote: {
          scope: "cli-promote",
          evidence: ["explicit CLI --approve"],
          guardrails: ["proposal-first"],
          rollbackVersionId: dir.expected
        },
        explicitApproval: true,
        evalReport
      };
      const promotedResult = promoteWithRegistry(registry, input);
      if (promotedResult.ok && promotedResult.newVersion !== undefined) {
        await saveAdaptationRegistry(dir.stateRoot, registry);
      }
      return promotedResult;
    });
    check(`S1c promote tx ok (${dir.candidateId})`, result.ok === true && result.newVersion !== undefined);
    newVersionIds.push((result.newVersion as ResourceVersion).versionId);
  }
  const promoteTxMs = (performance.now() - t0) / K;

  // Fidelity: reload one dir from disk; pointer moved and ledger recorded.
  {
    const reloaded = await loadAdaptationRegistry(first.stateRoot);
    const cand = reloaded.getCandidate(first.candidateId) as ImprovementCandidate;
    const active = reloaded.getActiveVersion(cand.identity);
    check("S1c fidelity: active pointer moved on disk", active?.versionId === newVersionIds[0]);
    check(
      "S1c fidelity: promoted ledger entry persisted",
      reloaded.ledger().some((entry) => entry.kind === "promoted" && entry.candidateId === first.candidateId)
    );
    check("S1c fidelity: candidate approved", cand.status === "approved");
  }

  const partsMs = prereadUs / 1e3 + lockGlueUs / 1e3 + cfgLoadMs + promoteOpUs / 1e3 + cfgSaveMs;
  console.log(
    `S1c promote WHOLE transaction (K=${K} fresh dirs): ${promoteTxMs.toFixed(2)}ms per command | parts: preread=${prereadUs.toFixed(0)}us (I-slice reads + slice-owned parses) + lock=${lockGlueUs.toFixed(0)}us + load=${(cfgLoadMs * 1e3).toFixed(0)}us + op=${promoteOpUs.toFixed(0)}us + save(+fsync)=${(cfgSaveMs * 1e3).toFixed(0)}us = ${partsMs.toFixed(2)}ms | residual=${((promoteTxMs - partsMs) * 1e3).toFixed(0)}us (${(((promoteTxMs - partsMs) / promoteTxMs) * 100).toFixed(0)}% of whole)`
  );

  /* 1d: rollback WHOLE transaction on the same K dirs (now post-promote:
   * active = v2, roll back to baseline via the CLI shape). */
  const t1 = performance.now();
  for (const [i, dir] of dirs.entries()) {
    const result = await withAdaptationRegistryLock(dir.stateRoot, async () => {
      const registry = await loadAdaptationRegistry(dir.stateRoot);
      const expectedVersion = registry.getVersion(newVersionIds[i] as ResourceVersionId);
      const targetVersion = registry.getVersion(dir.expected);
      const identity = expectedVersion?.identity ?? targetVersion?.identity;
      if (identity === undefined) {
        throw new DomainValidationError(`unknown rollback version: ${String(newVersionIds[i])}`);
      }
      const rolledBack = rollbackActive(registry, {
        identity,
        expectedCurrentVersionId: newVersionIds[i] as ResourceVersionId,
        targetVersionId: dir.expected,
        reason: "user",
        automatic: false,
        evidence: ["cli-rollback:user"]
      });
      await saveAdaptationRegistry(dir.stateRoot, registry);
      return rolledBack;
    });
    check(`S1d rollback tx ok (dir ${i})`, result.ok === true && result.active.versionId === dir.expected);
  }
  const rollbackTxMs = (performance.now() - t1) / K;
  {
    const reloaded = await loadAdaptationRegistry(first.stateRoot);
    const cand = reloaded.getCandidate(first.candidateId) as ImprovementCandidate;
    const active = reloaded.getActiveVersion(cand.identity);
    check("S1d fidelity: pointer restored to baseline on disk", active?.versionId === first.expected);
    check(
      "S1d fidelity: rolled-back ledger entry persisted",
      reloaded.rollbackLedger().some((entry) => entry.kind === "rolled-back" && entry.toVersionId === first.expected)
    );
  }
  const rbPartsMs = lockGlueUs / 1e3 + cfgLoadMs + rollbackOpUs / 1e3 + cfgSaveMs;
  console.log(
    `S1d rollback WHOLE transaction (K=${K}): ${rollbackTxMs.toFixed(2)}ms per command | parts: lock=${lockGlueUs.toFixed(0)}us + load=${(cfgLoadMs * 1e3).toFixed(0)}us + op=${rollbackOpUs.toFixed(1)}us + save(+fsync)=${(cfgSaveMs * 1e3).toFixed(0)}us = ${rbPartsMs.toFixed(2)}ms | residual=${((rollbackTxMs - rbPartsMs) * 1e3).toFixed(0)}us (${(((rollbackTxMs - rbPartsMs) / rollbackTxMs) * 100).toFixed(0)}% of whole)`
  );
}

await rm(workRoot, { recursive: true, force: true });

}
