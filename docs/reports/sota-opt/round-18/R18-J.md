  mkdirSync(dirname(log), { recursive: true });
  const lines: string[] = [];
  for (let i = 0; i < 200; i++) {
    const hit = i % 10 === 0;
    lines.push(
      JSON.stringify({
        id: `fb-${i}`,
        episodeId: hit ? "eps_hit" : `eps_other${i % 7}`,
        kind: "human",
        rubricVersion: "r1",
        score: 50 + (i % 40),
        evidenceRefs: [],
        redacted: false,
        createdAt: "2026-08-24T00:00:00.000Z",
        body: `free text feedback payload number ${i} `.repeat(3),
        summary: `summary ${i}`
      })
    );
  }
  writeFileSync(log, lines.join("\n") + "\n", "utf8");
  writeFileSync(
    feedbackTombstonesPath(stateRoot),
    JSON.stringify(["fb-pre-a", "fb-pre-b", "fb-pre-c", "fb-pre-d", "fb-pre-e"], null, 2) + "\n",
    "utf8"
  );
}

// ------------------------------------------------------------------ B1
async function accountDeleteEpisode(): Promise<void> {
  console.log("[B1 deleteEpisodeRecords account: statExists x2 + rm x2 + cascade(MATCH) + glue]");
  const stateRoot = mkdtempSync(join(tmpdir(), "r18j-acct-del-"));
  const episodesDir = join(runtimeRoot(stateRoot), "episodes");
  mkdirSync(episodesDir, { recursive: true });
  const epFiles = [join(episodesDir, "eps_hit.jsonl"), join(episodesDir, "eps_hit.events.jsonl")];
  const seedEpisodeFiles = (): void => {
    for (const file of epFiles) writeFileSync(file, `${JSON.stringify({ t: "x" })}\n`, "utf8");
  };

  // e2e: per-iteration re-seed (untimed), timer around the command only.
  const iters = 30;
  const e2e: number[] = [];
  for (let run = 0; run < 3; run++) {
    seedFeedback(stateRoot);
    seedEpisodeFiles();
    await deleteEpisodeRecords(stateRoot, "eps_hit" as EpisodeId); // warm
    let total = 0n;
    for (let i = 0; i < iters; i++) {
      seedFeedback(stateRoot);
      seedEpisodeFiles();
      const t0 = hr();
      const out = await deleteEpisodeRecords(stateRoot, "eps_hit" as EpisodeId);
      const t1 = hr();
      if (out.cascadedFeedbackTombstones.length !== 20 || out.removedPaths.length !== 2) {
        throw new Error("unexpected delete outcome");
      }
      total += t1 - t0;
    }
    e2e.push(Number(total) / 1000 / iters);
  }

  // phase: statExists twin (stat in try/catch) on an existing file.
  seedEpisodeFiles();
  const statN = 3000;
  const statUs: number[] = [];
  for (let run = 0; run < 3; run++) {
    const t0 = hr();
    for (let i = 0; i < statN; i++) {
      try {
        await stat(epFiles[0]!);
        sink += 1;
      } catch {
        sink -= 1;
      }
    }
    const t1 = hr();
    statUs.push(usPer(t0, t1, statN));
  }

  // phase: rm(force) on an existing small file (re-created untimed).
  const rmN = 800;
  const rmUs: number[] = [];
  for (let run = 0; run < 3; run++) {
    let total = 0n;
    for (let i = 0; i < rmN; i++) {
      writeFileSync(epFiles[0]!, "x\n", "utf8");
      const t0 = hr();
      await rm(epFiles[0]!, { force: true });
      const t1 = hr();
      total += t1 - t0;
    }
    rmUs.push(Number(total) / 1000 / rmN);
  }

  // phase: cascade(MATCH) alone on the same fixture (untimed re-seed).
  const cascUs: number[] = [];
  for (let run = 0; run < 3; run++) {
    let total = 0n;
    for (let i = 0; i < iters; i++) {
      seedFeedback(stateRoot);
      const t0 = hr();
      const out = await cascadeFeedbackTombstones(stateRoot, "eps_hit" as EpisodeId);
      const t1 = hr();
      if (out.length !== 20) throw new Error("expected 20 cascaded");
      total += t1 - t0;
    }
    cascUs.push(Number(total) / 1000 / iters);
  }

  const mid = (xs: number[]): number => [...xs].sort((a, b) => a - b)[1]!;
  const phases = 2 * mid(statUs) + 2 * mid(rmUs) + mid(cascUs);
  console.log(`  e2e deleteEpisodeRecords: ${e2e.map(f).join(" / ")} us/cmd`);
  console.log(`  phase statExists(present): ${statUs.map(f).join(" / ")} us/call  x2`);
  console.log(`  phase rm(present,force):  ${rmUs.map(f).join(" / ")} us/call  x2`);
  console.log(`  phase cascade(MATCH):     ${cascUs.map(f).join(" / ")} us/call  x1`);
  console.log(
    `  account: sum(phases,median)=${f(phases)} us vs e2e(median)=${f(mid(e2e))} us -> glue residual ${f(mid(e2e) - phases)} us`
  );
  rmSync(stateRoot, { recursive: true, force: true });
}

async function accountAppendFeedback(): Promise<void> {
  console.log("[B2 appendFeedback account: redact + stringify + jsonl(false) + glue]");
  const stateRoot = mkdtempSync(join(tmpdir(), "r18j-acct-fb-"));
  const record: FeedbackRecord = {
    id: "fb-acct",
    episodeId: "eps_acct" as EpisodeId,
    kind: "human",
    rubricVersion: "r1",
    score: 80,
    evidenceRefs: [],
    redacted: false,
    createdAt: "2026-08-24T00:00:00.000Z" as FeedbackRecord["createdAt"],
    body: "clean feedback body with no secrets and a reasonable length ".repeat(2),
    summary: "clean summary"
  };
  // e2e appendFeedback on an existing log.
  const iters = 2000;
  const e2e: number[] = [];
  for (let run = 0; run < 3; run++) {
    for (let i = 0; i < 100; i++) await appendFeedback(stateRoot, record);
    const t0 = hr();
    for (let i = 0; i < iters; i++) await appendFeedback(stateRoot, record);
    const t1 = hr();
    e2e.push(usPer(t0, t1, iters));
  }

  // phase: redact under the production-policy replica (module-private in
  // feedback/store.ts; literal reproduced here).
  const prodPolicy = {
    redactPII: true,
    maxBodyChars: 400,
    forbiddenSubstrings: ["sk-", "api_key", "API_KEY", "BEGIN PRIVATE"]
  } as const;
  const redacted = redactFeedback(record, prodPolicy).feedback;
  const redactUs: number[] = [];
  for (let run = 0; run < 3; run++) {
    const N = 200000;
    for (let i = 0; i < 20000; i++) sink += redactFeedback(record, prodPolicy).feedback.redacted ? 1 : 0;
    const t0 = hr();
    for (let i = 0; i < N; i++) sink += redactFeedback(record, prodPolicy).feedback.redacted ? 1 : 0;
    const t1 = hr();
    redactUs.push(usPer(t0, t1, N));
  }

  // phase: stringify of the redacted record.
  const strUs: number[] = [];
  for (let run = 0; run < 3; run++) {
    const N = 200000;
    for (let i = 0; i < 20000; i++) sink += JSON.stringify(redacted).length;
    const t0 = hr();
    for (let i = 0; i < N; i++) sink += JSON.stringify(redacted).length;
    const t1 = hr();
    strUs.push(usPer(t0, t1, N));
  }

  // phase: raw jsonl append (fsync=false) of the same line.
  const line = JSON.stringify(redacted);
  const appendUs: number[] = [];
  {
    const dir = mkdtempSync(join(tmpdir(), "r18j-acct-jsonl-"));
    const file = join(dir, "log.jsonl");
    for (let run = 0; run < 3; run++) {
      for (let i = 0; i < 100; i++) await appendJsonlLine(file, line, false);
      const t0 = hr();
      for (let i = 0; i < iters; i++) await appendJsonlLine(file, line, false);
      const t1 = hr();
      appendUs.push(usPer(t0, t1, iters));
    }
    rmSync(dir, { recursive: true, force: true });
  }

  const mid = (xs: number[]): number => [...xs].sort((a, b) => a - b)[1]!;
  const phases = mid(redactUs) + mid(strUs) + mid(appendUs);
  console.log(`  e2e appendFeedback:        ${e2e.map(f).join(" / ")} us/append`);
  console.log(`  phase redact(prod policy): ${redactUs.map((x) => x.toFixed(3)).join(" / ")} us/call`);
  console.log(`  phase stringify(record):   ${strUs.map((x) => x.toFixed(3)).join(" / ")} us/call`);
  console.log(`  phase jsonl append(false): ${appendUs.map(f).join(" / ")} us/call`);
  console.log(
    `  account: sum(phases,median)=${f(phases)} us vs e2e(median)=${f(mid(e2e))} us -> glue residual ${f(mid(e2e) - phases)} us`
  );
  rmSync(stateRoot, { recursive: true, force: true });
}

async function accountRecordPreference(): Promise<void> {
  console.log("[B3 recordPreference(persisted) account: CPU twin + stringify + mkdir/write + glue]");
  const writeOnce = (i: number): void => {
    recordPreference(
      "project",
      `key-${i % 7}`,
      `pref-${i % 11}`,
      `value-w${i}`,
      `eps_w${i.toString(36)}` as EpisodeId,
      1.0,
      false
    );
  };
  const seed = (): void => {
    for (let i = 0; i < 50; i++) {
      recordPreference(
        (["user", "project", "role"] as const)[i % 3]!,
        `key-${i % 7}`,
        `pref-${i % 11}`,
        i % 2 === 0 ? `value-${i}` : i,
        `eps_seed${i.toString(36)}` as EpisodeId,
        1.0,
        i % 4 === 0
      );
    }
  };
  const N = 200;

  // e2e persisted (anchor protocol).
  const e2e: number[] = [];
  for (let run = 0; run < 3; run++) {
    const dir = mkdtempSync(join(tmpdir(), "r18j-acct-pref-"));
    resetPreferenceStore();
    configurePreferencePersistence(join(dir, "prefs.json"));
    seed();
    const t0 = hr();
    for (let i = 0; i < N; i++) writeOnce(i);
    const t1 = hr();
    e2e.push(usPer(t0, t1, N));
    resetPreferenceStore();
    configurePreferencePersistence(undefined);
    rmSync(dir, { recursive: true, force: true });
  }

  // phase: CPU twin (unpersisted, identical growth 50 -> 250).
  const cpu: number[] = [];
  for (let run = 0; run < 3; run++) {
    resetPreferenceStore();
    configurePreferencePersistence(undefined);
    seed();
    const t0 = hr();
    for (let i = 0; i < N; i++) writeOnce(i);
    const t1 = hr();
    cpu.push(usPer(t0, t1, N));
    resetPreferenceStore();
  }

  // phase: snapshot stringify with the same growth curve. Build observation
  // objects exactly the way recordPreference does (createId + nowIso), then
  // per write push one and stringify the whole snapshot.
  type Obs = Record<string, unknown>;
  const makeObs = (i: number, explicitEvery4: boolean): Obs => ({
    id: createId("EvidenceId"),
    scope: (["user", "project", "role"] as const)[i % 3]!,
    scopeKey: `key-${i % 7}`,
    key: `pref-${i % 11}`,
    value: i % 2 === 0 ? `value-${i}` : i,
    evidenceEpisodeId: `eps_seed${i.toString(36)}`,
    weight: 1.0,
    createdAt: nowIso(),
    explicit: explicitEvery4 ? i % 4 === 0 : false,
    recurrenceCount: 1
  });
  const strUs: number[] = [];
  const generated: string[] = [];
  for (let run = 0; run < 3; run++) {
    const arr: Obs[] = [];
    for (let i = 0; i < 50; i++) arr.push(makeObs(i, true));
    let total = 0n;
    for (let i = 0; i < N; i++) {
      arr.push(makeObs(i, false));
      const t0 = hr();
      const s = JSON.stringify({ observations: arr, tombstones: [] });
      const t1 = hr();
      total += t1 - t0;
      if (run === 0) generated.push(s);
      sink += s.length;
    }
    strUs.push(Number(total) / 1000 / N);
  }

  // phase: mkdir(self-heal) + writeFile of the same byte curve.
  const writeUs: number[] = [];
  for (let run = 0; run < 3; run++) {
    const dir = mkdtempSync(join(tmpdir(), "r18j-acct-wr-"));
    const file = join(dir, "prefs.json");
    const parent = dirname(file);
    let total = 0n;
    for (const s of generated) {
      const t0 = hr();
      mkdirSync(parent, { recursive: true });
      writeFileSync(file, s);
      const t1 = hr();
      total += t1 - t0;
    }
    writeUs.push(Number(total) / 1000 / generated.length);
  }

  const mid = (xs: number[]): number => [...xs].sort((a, b) => a - b)[1]!;
  const phases = mid(cpu) + mid(strUs) + mid(writeUs);
  console.log(`  e2e recordPreference(persisted): ${e2e.map(f).join(" / ")} us/write`);
  console.log(`  phase CPU twin (unpersisted):    ${cpu.map(f).join(" / ")} us/write`);
  console.log(`  phase snapshot stringify:        ${strUs.map(f).join(" / ")} us/write`);
  console.log(`  phase mkdir+writeFile:           ${writeUs.map(f).join(" / ")} us/write`);
  console.log(
    `  account: sum(phases,median)=${f(phases)} us vs e2e(median)=${f(mid(e2e))} us -> glue residual ${f(mid(e2e) - phases)} us`
  );
}

async function main(): Promise<void> {
  console.log(`node ${process.version}`);
  await accountDeleteEpisode();
  await accountAppendFeedback();
  await accountRecordPreference();
  console.log(`[sink=${sink}]`);
}

main().catch((error) => {
  console.error(error);
  process.exit(1);
});
