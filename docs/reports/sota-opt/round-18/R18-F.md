 * composition and reject endpoints, R18-B added fixture-population axes,
 * R18-D published a transaction-composition ledger with no hidden segments):
 *
 *   axis 1  alignment-fraction dose-response of the landed S7-F-1 unit.
 *           Seventeen rounds re-proved the four-order signs (prefix /
 *           half-aligned / scattered / reversed) but never published the
 *           interior of the curve: at what aligned-prefix fraction f does the
 *           saving over the frozen S6-F-1 form sit, and is there any interior
 *           crossover valley where the sign could dip negative? This sweeps
 *           f ∈ {0, 1/16, 1/8, 1/4, 1/2, 3/4, 7/8, 15/16, 1} at the anchor
 *           scale (P=2000, A=1000, 2000 restores/cell), comparing the landed
 *           production restore against an inline S6-F-1-era reference that
 *           calls the PRODUCTION validateExperimentPlan — so the delta
 *           isolates exactly the S7-F-1 alignment machinery (S7-F-2 is live
 *           on both sides). State equality is asserted per cell.
 *
 *   axis 2  fresh-plan warm-process tax. R17-F axis 2 measured the
 *           cold-process budget (+16.8–22.0 ms) but that number conflates
 *           function compilation, IC warmup, and V8 string-hash population.
 *           Production `evaluateProposalShadow` receives a FRESH plan per
 *           proposal, so in a warm process each proposal pays exactly the
 *           per-plan component once: the first validate on a fresh
 *           population fills every string's cached hash; the remaining
 *           2A+1 validates of that same experiment reuse them. This measures
 *           first-validate minus steady-validate and first-experiment minus
 *           second-experiment on 8 fresh plans in a warm process —
 *           decomposing the cold budget and pricing the true production
 *           per-proposal overhead over the anchor.
 *
 *   axis 3  post-halt operation pricing — the one config-state ×
 *           command-class cell never directly priced. A halted experiment
 *           still pays the full fail-closed restore (validate + alignment
 *           scan + haltReason trim probe) on every operation; assign then
 *           returns early (skipping membership includes, uniqueness scan,
 *           and the rng replay) and cancel is idempotent. Production traffic
 *           through the post-halt paths is zero (`evaluateProposalShadow`
 *           breaks on halt — re-verified this round), so these are
 *           informational floors, not pools.
 *

import { DomainValidationError } from "./src/domain/errors.js";

function timeMs(fn: () => void, rounds = 3): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

function onceMs(fn: () => void): number {
  const t0 = process.hrtime.bigint();
  fn();
  const t1 = process.hrtime.bigint();
  return Number(t1 - t0) / 1e6;
}

/** Historical fixture shape: `ep_` + zero-padded base36 index, L=11. */
function makePopulation(size: number, tag = ""): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${tag}${i.toString(36).padStart(8 - tag.length, "0")}`);
  }
  return population;
}

function makeShadowPlan(population: readonly string[], maxAssignments: number, id = "exp_r18f_bench"): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: id,
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r18fbase"),
    candidateId: createCandidateId(() => "r18fcand"),
    population,
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments, maxWallClockMs: 1e12 },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

const P = 2000;
const A = 1000;

function fullExperiment(plan: ExperimentPlan, hashes: readonly string[]): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (let k = 0; k < A; k++) {
    const h = hashes[k] as string;
    state = runner.assign(state, h, 0);
    const outcome: ExperimentOutcome = {
      episodeHash: h,
      utility: 0.5,
      costUsd: 0.01,
      guardrailBreached: false,
    };
    state = runner.recordOutcome(state, outcome, 0);
  }
  if (state.assignments.length !== A) throw new Error("unexpected assignment count");
  return state;
}

/* ------------------------------------------------------------------ */
/* Frozen S6-F-1-era reference restore body (verbatim loop structure   */
/* from the pre-S7-F-1 code, but calling PRODUCTION                    */
/* validateExperimentPlan so the delta isolates S7-F-1 alone).         */
/* ------------------------------------------------------------------ */
function refRestoreShadowS6F1(serialized: ShadowState, expected: ExperimentPlan): ShadowState {
  if (typeof serialized !== "object" || serialized === null) {
    throw new DomainValidationError("shadow state is required");
  }
  validateExperimentPlan(serialized.plan);
  if (serialized.plan.experimentId !== expected.experimentId) {
    throw new DomainValidationError("restored plan does not match runner");
  }
  if (serialized.plan.mode !== "shadow") {
    throw new DomainValidationError("shadow restore requires mode \"shadow\"");
  }
  if (typeof serialized.halted !== "boolean") {
    throw new DomainValidationError("halted must be a boolean");
  }
  if (!Array.isArray(serialized.assignments) || !Array.isArray(serialized.outcomes)) {
    throw new DomainValidationError("assignments and outcomes must be arrays");
  }
  if (!Number.isInteger(serialized.guardrailBreaches) || serialized.guardrailBreaches < 0) {
    throw new DomainValidationError("guardrailBreaches must be an integer >= 0");
  }
  if (typeof serialized.startedAtMs !== "number" || !Number.isFinite(serialized.startedAtMs)) {
    throw new DomainValidationError("startedAtMs must be a finite number");
  }
  if (typeof serialized.elapsedMs !== "number" || !Number.isFinite(serialized.elapsedMs) || serialized.elapsedMs < 0) {
    throw new DomainValidationError("elapsedMs must be a finite number >= 0");
  }
  let structuralFault: DomainValidationError | undefined;
  const pending = new Set<string>();
  for (const assignment of serialized.assignments) {
    if (assignment.liveAction !== "baseline" || assignment.changedLiveAction !== false) {
      structuralFault = new DomainValidationError("shadow state must not change the live action");
      break;
    }
    if (assignment.shadowDecision !== "baseline" && assignment.shadowDecision !== "candidate") {
      structuralFault = new DomainValidationError("invalid shadowDecision");
      break;
    }
    if (typeof assignment.episodeHash !== "string" || assignment.episodeHash.trim() === "") {
      structuralFault = new DomainValidationError("episodeHash is required");
      break;
    }
    pending.add(assignment.episodeHash);
  }
  const target = pending.size;
  if (target > 0) {
    let found = 0;
    for (const hash of serialized.plan.population) {
      if (pending.has(hash)) {
        found += 1;
        if (found === target) {
          break;
        }
      }
    }
    if (found !== target) {
      const population = new Set(serialized.plan.population);
      for (const assignment of serialized.assignments) {
        requirePopulationMember(population, assignment.episodeHash);
      }
    }
  }
  if (structuralFault !== undefined) {
    throw structuralFault;
  }
  const haltReason = canonicalHaltReason(serialized.haltReason, serialized.halted);
  return {
    plan: serialized.plan,
    halted: serialized.halted,
    haltReason,
    assignments: [...serialized.assignments],
    outcomes: [...serialized.outcomes],
    guardrailBreaches: serialized.guardrailBreaches,
    startedAtMs: serialized.startedAtMs,
    elapsedMs: serialized.elapsedMs,
  };
}

function makeAlignedState(fractionNumer: number, fractionDenom: number): ShadowState {
  const cut = Math.round((A * fractionNumer) / fractionDenom);
  const assignments: ShadowAssignment[] = [];
  for (let i = 0; i < A; i++) {
    // aligned prefix: hash equals population[i]; misaligned suffix: a
    // population member from the second half (index >= A), never at its own
    // index, all unique.
    const hash = i < cut ? (anchorPopulation[i] as string) : (anchorPopulation[A + (i - cut)] as string);
    assignments.push({
      episodeHash: hash,
      liveAction: "baseline",
      shadowDecision: "baseline",
      changedLiveAction: false,
    });
  }
  return {
    plan: anchorPlan,
    halted: false,
    haltReason: undefined,
    assignments,
    outcomes: [],
    guardrailBreaches: 0,
    startedAtMs: 0,
    elapsedMs: 0,
  };
}

const HALT_K = 500;
const haltPlanPopulation = makePopulation(P, "h");

let activeTwin = haltRunner.start(0);

  const h = haltHashes[k] as string;
  haltedState = haltRunner.assign(haltedState, h, 0);
  activeTwin = haltRunner.assign(activeTwin, h, 0);
  const breach = k === HALT_K - 1;
  haltedState = haltRunner.recordOutcome(
    haltedState,
    { episodeHash: h, utility: 0.5, costUsd: 0.01, guardrailBreached: breach },
    0
  );
  activeTwin = haltRunner.recordOutcome(
    activeTwin,
    { episodeHash: h, utility: 0.5, costUsd: 0.01, guardrailBreached: false },
    0
  );
}
if (!haltedState.halted || haltedState.haltReason !== `guardrail: ${haltHashes[HALT_K - 1] as string}`) {
  throw new Error(`unexpected halt state: halted=${haltedState.halted} reason=${haltedState.haltReason}`);
}
if (activeTwin.halted) throw new Error("active twin unexpectedly halted");

const OPS = 200;
const nextHash = haltHashes[HALT_K] as string;
// post-halt assign: full restore (validate + alignment + haltReason trim) +
// clock, then early return — membership includes / uniqueness scan / rng all
// skipped. Verify the early return leaves the ledger unchanged.
const postHaltOut = haltRunner.assign(haltedState, nextHash, 0);
if (postHaltOut.assignments.length !== HALT_K || !postHaltOut.halted) {
  throw new Error("post-halt assign must be a no-op on the ledger");
}
let sink3 = 0;
const tPostHaltAssign = timeMs(() => {
  for (let r = 0; r < OPS; r++) sink3 += haltRunner.assign(haltedState, nextHash, 0).assignments.length;
}, 5);
const tActiveAssign = timeMs(() => {
  for (let r = 0; r < OPS; r++) sink3 += haltRunner.assign(activeTwin, nextHash, 0).assignments.length;
}, 5);
// post-halt cancel: idempotent (restore + clock, halted short-circuit)
const cancelOut = haltRunner.cancel(haltedState, 0);
if (cancelOut.haltReason !== haltedState.haltReason) throw new Error("cancel must preserve haltReason");
const tPostHaltCancel = timeMs(() => {
  for (let r = 0; r < OPS; r++) sink3 += haltRunner.cancel(haltedState, 0).assignments.length;
}, 5);
// halted restore alone (haltReason trim probe live)
const tHaltedRestore = timeMs(() => {
  for (let r = 0; r < OPS; r++) sink3 += haltRunner.restore(haltedState).assignments.length;
}, 5);
const tActiveRestore = timeMs(() => {
  for (let r = 0; r < OPS; r++) sink3 += haltRunner.restore(activeTwin).assignments.length;
}, 5);
process.stdout.write(
  `axis 3 (post-halt ops, P=${P}, halted at k=${HALT_K}, per-op µs over ${OPS} ops):\n` +
    `axis 3 halted-restore=${((tHaltedRestore / OPS) * 1e3).toFixed(1)}µs active-restore=${((tActiveRestore / OPS) * 1e3).toFixed(1)}µs ` +
    `(haltReason trim-probe increment=${(((tHaltedRestore - tActiveRestore) / OPS) * 1e3).toFixed(2)}µs)\n` +
    `axis 3 post-halt-assign=${((tPostHaltAssign / OPS) * 1e3).toFixed(1)}µs active-assign=${((tActiveAssign / OPS) * 1e3).toFixed(1)}µs ` +
    `(early-return saving=${(((tActiveAssign - tPostHaltAssign) / OPS) * 1e3).toFixed(2)}µs — membership+uniqueness+rng skipped)\n` +
    `axis 3 post-halt-cancel(idempotent)=${((tPostHaltCancel / OPS) * 1e3).toFixed(1)}µs\n`
);
void sink3;
