### 2a. 理论：缝普查与调用站点穷举

`this.now` / `this.generateId` 进入执行流的站点穷举（14 文件逐行
对差）：

| 站点 | id 调用 | now 调用 | 备注 |
| --- | --- | --- | --- |

branded id 当后缀，产出双前缀 44 字符 id，战役首钉）；(iii) id
几何（44 vs 10 字符）流经正则闸 / Map 键哈希 / 持久化字节的下游
载体——(iii) 经 B 臂受控隔离并归档 R14-D/R16-D 既裁域。

### 2e. 精确普查、税表与 10 ms 越线求解（S1e）

- **计数见证**（委托真默认实现的计数包装 = 生产臂行为 + 计数，
  全部断言过）：registerBaseline **1 id + 1 now**；createCandidate
  **1 id + 1 now**；promoteWithRegistry **1 id + 3 now**；
  rollback（user CAS）**0 id + 1 now**；retire **0 + 0**。
- **税表**：promote 税 = 1×(263.5–298.1) + 3×(471.3–627.0) =
  **1.68–2.18 µs/promote**；全生命周期 3.15–4.03 µs；rollback 税
  ~0.5–0.6 µs；save 侧 randomUUID（S7-D-3 机制）夹具/生产同价
  **零偏差**。
- **频率端**：adapt promote 命令恰 1 promote；auto-loop 每 run
  ≤1 提议（首 run +1 baseline）；adapt eval **零缝调用**；
  rollback 命令 1 now——R13-D 相位账上界个位数缝敏感 op/命令 ⇒
  税 = once-per-command 微秒类，**拒绝**。
- **越线求解**：10 ms 需每命令 ~4.6–6.0×10³ 次 promote（或
  ~1.6–2.1×10⁴ 次裸 now 调用）——先撞 R13-D 相位账与 R21-D
  批准门频率论证（同一资源每 promote 过显式批准）。**频率 ×
  单价 × 合法性三端封死。**

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** Recursively sort object keys so JSON compare ignores key order. */
function canonicalJson(value: unknown): string {
  return JSON.stringify(value, (_key, val: unknown) => {
    if (val !== null && typeof val === "object" && !Array.isArray(val)) {
      const sorted: Record<string, unknown> = {};
      for (const k of Object.keys(val as Record<string, unknown>).sort()) {
        sorted[k] = (val as Record<string, unknown>)[k];
      }
      return sorted;
    }
    return val;
  });
}

function buildRoutingRegistry(

  candidatePolicy = CANDIDATE_POLICY,

/* Configured-state registry (same construction as R8-D..R25-D; optional
 * RegistryOptions selects the injection arm, defaulting to the historical
 * fixture arm A). */
function buildConfiguredRegistry(
  tag: string,
  baselinePolicy = CONFIGURED_BASELINE_POLICY,
  candidatePolicy = CONFIGURED_CANDIDATE_POLICY,
  options?: RegistryOptions
): {
  registry: ResourceRegistry;
  candidateId: CandidateId;
  baseline: ResourceVersion;
  identity: ResourceIdentity;
} {
  const built = buildRoutingRegistry(tag, baselinePolicy, candidatePolicy, options);
  const { registry } = built;
  const versions: {
    identity: ResourceIdentity;
    v1: ResourceVersion;
    v2?: ResourceVersion;
    candidate: ImprovementCandidate;
  }[] = [];
  for (let i = 0; i < 8; i += 1) {
    const identity: ResourceIdentity = {
      kind: "prompt",
      name: `agent-prompt-${i}`,
      scope: { kind: "project", projectId: createProjectId(() => `${tag}proj`) }
    };
    const body = `# prompt ${i}\n${`Follow the review checklist, cite evidence for every claim, and keep the diff bounded to the stated scope (${i}).\n`.repeat(30 + i * 8)}`;
    const v1 = registry.registerBaseline({ identity, content: body, author: HUMAN });
    const candidate = registry.createCandidate({
      identity,
      content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    const entry: {
      identity: ResourceIdentity;
      v1: ResourceVersion;
      v2?: ResourceVersion;
      candidate: ImprovementCandidate;
    } = { identity, v1, candidate };
    if (i < 5) {
      const result = promoteWithRegistry(registry, {
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
      entry.v2 = result.newVersion as ResourceVersion;
    } else if (i === 5) {
      registry.beginPromotion({
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
    }
    versions.push(entry);
  }
  const first = versions[0] as (typeof versions)[number];
  registry.rollback({
    identity: first.identity,
    expectedCurrentVersionId: (first.v2 as ResourceVersion).versionId,
    targetVersionId: first.v1.versionId,
    reason: "degradation",
    evidence: ["ev:configured-degradation"],
    automatic: false
  });
  const second = versions[1] as (typeof versions)[number];
  registry.rollback({
    identity: second.identity,
    expectedCurrentVersionId: (second.v2 as ResourceVersion).versionId,
    targetVersionId: second.v1.versionId,
    reason: "user",
    evidence: ["ev:configured-user"],
    automatic: false
  });
  registry.retire((versions[2] as { v1: ResourceVersion }).v1.versionId);
  registry.retire((versions[3] as { v1: ResourceVersion }).v1.versionId);
  return built;
}

function runPromoteRollbackAnchor(
  fixtures: { registry: ResourceRegistry; input: PromoteInput }[]
): { promoteOpUs: number; rollbackOpUs: number } {
  const K = fixtures.length;
  const t0 = performance.now();
  const promoted: { registry: ResourceRegistry; newVersion: ResourceVersion; identity: ResourceIdentity; baselineId: ResourceVersionId }[] = [];
  for (const fixture of fixtures) {
    const result = promoteWithRegistry(fixture.registry, fixture.input);
    promoted.push({
      registry: fixture.registry,
      newVersion: result.newVersion as ResourceVersion,
      identity: (result.newVersion as ResourceVersion).identity,
      baselineId: fixture.input.expectedCurrentVersionId
    });
  }
  const promoteOpUs = ((performance.now() - t0) / K) * 1e3;
  check("promote anchor: all promotes succeeded", promoted.length === K && promoted.every((p) => p.newVersion !== undefined));
  const t1 = performance.now();
  for (const p of promoted) {
    p.registry.rollback({
      identity: p.identity,
      expectedCurrentVersionId: p.newVersion.versionId,
      targetVersionId: p.baselineId,
      reason: "user",
      evidence: ["ev:r26d-op-anchor"],
      automatic: false
    });
  }
  const rollbackOpUs = ((performance.now() - t1) / K) * 1e3;
  return { promoteOpUs, rollbackOpUs };
}

interface FleetRow {
  readonly identity: ResourceIdentity;
  readonly v1: ResourceVersion;
  readonly candidate: ImprovementCandidate;
  readonly input: PromoteInput;
}
interface Fleet {
  readonly registry: ResourceRegistry;
  readonly rows: readonly FleetRow[];
}

/** N identities at the historical fixture coordinate (prompt x project).
 * Names and contents are byte-identical across arms; only the injection arm
 * varies. All promotes are explicit-mode over the default profile. */
function buildFleet(tag: string, n: number, arm: ArmKind): Fleet {
  const registry = new ResourceRegistry(armOptions(arm, tag));
  const rows: FleetRow[] = [];
  for (let i = 0; i < n; i += 1) {
    const identity: ResourceIdentity = { kind: "prompt", name: `res-${String(i).padStart(4, "0")}`, scope: PROJECT_SCOPE };
    const body = `# resource row ${String(i).padStart(4, "0")}\nKeep the diff bounded to the stated scope and cite evidence for every claim.\n`;
    const v1 = registry.registerBaseline({ identity, content: body, author: HUMAN });
    const content = `${body}\n## amendment\nAdd an explicit check for the attributable failure (row ${String(i).padStart(4, "0")}).\n`;
    const candidate = registry.createCandidate({
      identity,
      content,

      candidateId: candidate.candidateId,
      expectedCurrentVersionId: v1.versionId,

      review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}${i}`),
      changeNote: mkNote(`row:${String(i).padStart(4, "0")}`, v1.versionId),
      explicitApproval: true
    };
    rows.push({ identity, v1, candidate, input });
  }
  return { registry, rows };
}
