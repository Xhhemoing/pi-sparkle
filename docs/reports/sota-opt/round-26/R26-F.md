    if (!check.halted || check.haltReason !== "budget-exhausted" || check.assignments.length !== A) {
      throw new Error(`${armName}: bad final state in child`);
    }
    const samples: number[] = [];
    for (let pass = 0; pass < PASSES; pass++) {
      samples.push(onceMs(() => {
        runInterleaved(arm.plan, arm.stream, arm.freezeChain);
      }));
    }
    process.stdout.write(`CHILD e2e ${armName} ${median(samples).toFixed(2)} ms\n`);
    process.exit(0);
  }

  if (cell === "restore") {
    const { runner, state } = buildActiveFullLedger();
    const variant = buildRestoreVariant(armName, state);
    if (ledgerDigest(runner.restore(variant)) !== ledgerDigest(runner.restore(state))) {
      throw new Error(`${armName}: restore output diverged in child`);
    }
    for (let c = 0; c < 20; c++) runner.restore(variant);
    const samples: number[] = [];
    for (let pass = 0; pass < PASSES; pass++) {
      samples.push(onceMs(() => {
        for (let c = 0; c < RESTORE_CALLS; c++) runner.restore(variant);
      }));
    }
    process.stdout.write(`CHILD restore ${armName} ${((median(samples) * 1000) / RESTORE_CALLS).toFixed(2)} us/restore\n`);
    process.exit(0);
  }

  if (cell === "probe") {
    // Mechanism attribution: for..of vs index traversal of the same P=2000
    // string array under one elements kind / integrity level.
    let arr: readonly string[];
    if (armName === "packed") arr = [...basePopulation];
    else if (armName === "sealed") arr = Object.seal([...basePopulation]);
    else if (armName === "frozen") arr = Object.freeze([...basePopulation]);
    else if (armName === "holey") arr = holeyCopy(basePopulation);
    else throw new Error(`unknown probe arm: ${armName}`);
    if (probe !== undefined) {
      process.stdout.write(`CHILD-KIND ${armName} array=${probe(arr)}\n`);
    }
    let sink = 0;
    const forOf = (): void => {
      for (const value of arr) {
        sink += value.length;
      }
    };
    const indexed = (): void => {
      for (let i = 0; i < arr.length; i++) {
        sink += (arr[i] as string).length;
      }
    };
    for (let c = 0; c < 500; c++) {
      forOf();
      indexed();
    }
    const PROBE_CALLS = 2000;
    const samplesF: number[] = [];
    const samplesI: number[] = [];
    for (let pass = 0; pass < PASSES; pass++) {
      samplesF.push(onceMs(() => {
        for (let c = 0; c < PROBE_CALLS; c++) forOf();
      }));
      samplesI.push(onceMs(() => {
        for (let c = 0; c < PROBE_CALLS; c++) indexed();
      }));
    }
    process.stdout.write(
      `CHILD probe ${armName} for-of=${((median(samplesF) * 1e6) / PROBE_CALLS / P).toFixed(3)} ns/elem ` +
        `index=${((median(samplesI) * 1e6) / PROBE_CALLS / P).toFixed(3)} ns/elem sink=${sink > 0 ? "ok" : "zero"}\n`
    );
    process.exit(0);
  }

  throw new Error(`unknown cell: ${cell}`);
}

      throw new Error(`${arm.name}: expected budget-exhausted final state`);
    }
    if (state.assignments.length !== A || state.outcomes.length !== A) {
      throw new Error(`${arm.name}: unexpected ledger sizes`);
    }
    for (let i = 0; i < A; i++) {
      const assignment = state.assignments[i] as { episodeHash: string };

      }
    }
    if (state.guardrailBreaches !== 0 || state.startedAtMs !== 0 || state.elapsedMs !== 0) {
      throw new Error(`${arm.name}: unexpected counters/clock fields`);
    }
    digests.push(ledgerDigest(state));
  }
  if (digests[0] !== digests[1]) {
    throw new Error(`${arm.name}: nondeterministic ledger`);
  }
  if (arm.name === BASELINE) {
    anchorDigest = digests[0] as string;
  }
}
for (const arm of armsA) {
  if (ledgerDigest(runInterleaved(arm.plan, arm.stream, arm.freezeChain)) !== anchorDigest) {

  if (arm.plan.population[0] !== basePopulation[0] || arm.plan.population[P - 1] !== basePopulation[P - 1]) {

}

/* ----------------------- isolated child-cell driver ------------------------ */
function runChild(cell: string, arm: string): string[] {
  const child = spawnSync(
    process.execPath,
    ["--import", "tsx", "--allow-natives-syntax", process.argv[1] as string, `--cell=${cell}`, `--arm=${arm}`],
    { encoding: "utf8", timeout: 180_000 }
  );
  if (child.status !== 0) {
    return [`CHILD ${cell} ${arm} FAILED status=${String(child.status)} ${child.stderr.slice(0, 200)}`];
  }
  return child.stdout.trim().split("\n");
}

function runIsolatedCell(title: string, cell: string, arms: readonly string[]): Map<string, number> {
  process.stdout.write(`${title}\n`);
  const values = new Map<string, number>();
  for (const arm of arms) {
    for (const line of runChild(cell, arm)) {
      process.stdout.write(`  ${line}\n`);
      const match = line.match(/^CHILD \S+ \S+ ([0-9.]+)/);
      if (match !== null) {
        values.set(arm, Number(match[1]));
      }
    }
  }
  return values;
}
