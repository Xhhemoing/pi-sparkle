let failures = 0;
function check(name: string, ok: boolean, detail?: string): void {
  if (!ok) {
    failures += 1;
    console.error(`FAIL ${name}${detail === undefined ? "" : `: ${detail}`}`);
  }
}
function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function pick<T>(rng: () => number, items: readonly T[]): T {
  return items[Math.floor(rng() * items.length)] as T;
}
function bench(fn: () => void, reps: number): number {
  fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}

function genObjective(rng: () => number): string {
  if (rng() < 0.5) return pick(rng, OBJECTIVE_TEMPLATES);
  let text = "";
  const parts = 1 + Math.floor(rng() * 8);
  for (let i = 0; i < parts; i += 1) text += pick(rng, FRAGMENTS) + " ";
  return text;
}
function genTasks(rng: () => number, n: number): AssignableTask[] {
  return Array.from({ length: n }, (_, i) => ({
    taskId: `tsk_${i.toString().padStart(6, "0")}` as AssignableTask["taskId"],
    role: pick(rng, ROLES),
    objective: genObjective(rng),
    ...(rng() < 0.1 ? { contractRisk: rng() < 0.5 } : {}),
    ...(rng() < 0.3 ? { contextTokens: Math.floor(rng() * 100000) } : {}),
    ...(rng() < 0.3 ? { outputTokens: Math.floor(rng() * 4000) } : {})
  }));
}
/** Replay-faithful shape: every production caller maps only taskId/role/objective. */
function genReplayTasks(rng: () => number, n: number): AssignableTask[] {

    taskId: `tsk_${i.toString().padStart(6, "0")}` as AssignableTask["taskId"],
    role: pick(rng, ROLES),
    objective: genObjective(rng)
  }));
}

    const out = assignTasks({ catalog: forbidden10, tasks: bare });
    const forbiddenIds = new Set(["m0", "m1", "m2"]);
    let withProviderRow = 0;
    let selectedForbidden = 0;
    for (const row of out) {
      if (row.decision.rejections.some((r) => r.constraint === "provider-policy")) withProviderRow += 1;
      if (forbiddenIds.has(row.decision.model)) selectedForbidden += 1;
    }
    check(
      "privacy W3: forbidden dose k=3 -> every decision carries provider-policy rejection rows",
      withProviderRow === out.length,
      `rows=${withProviderRow}/${out.length}`
    );
    check("privacy W3: forbidden ids never selected", selectedForbidden === 0, `selected=${selectedForbidden}`);
    console.log(
      `privacy W3 census [M=10 forbidden k=3]: provider-policy rejection rows ${withProviderRow}/${out.length} | forbidden selections ${selectedForbidden}`
    );
  }

  // W1: double-run determinism at the most loaded arm (declared + forbidden).
  {
    const loaded: ModelRouterConfig = {
      policyVersion: forbidden10.policyVersion,
      models: forbidden10.models.map((model) => ({ ...model, privacyClass: "cloud-approved" as PrivacyClass }))
    };
    const runA = JSON.stringify(assignTasks({ catalog: loaded, tasks: bare }));
    const runB = JSON.stringify(assignTasks({ catalog: loaded, tasks: bare }));
    check("privacy W1 [M=10 declared+forbidden]: double run byte-identical", runA === runB);
  }

  // Timing lattice: catalog-privacy column at M=2 (same objective stream,
  // same window), declared cross-section at M=7 configured, forbidden dose
  // at M=10, real face N=30, 10x stress N=20000.
  const tBare = bench(() => assignTasks({ catalog: base2, tasks: bare }), 30);
  const tCG = bench(() => assignTasks({ catalog: declaredCG2, tasks: bare }), 30);
  const tLocal = bench(() => assignTasks({ catalog: declaredLocal2, tasks: bare }), 30);
  const tMixed = bench(() => assignTasks({ catalog: mixed2, tasks: bare }), 30);
  const line = (label: string, cost: number): string =>
    `privacy [M=2 ${label}]: ${(cost * 1e3).toFixed(1)}us per call (${(cost * 2).toFixed(2)}ms per eval x2; delta vs bare ${((cost - tBare) * 2).toFixed(2)}ms)`;
  console.log(`privacy [M=2 undeclared production corner]: ${(tBare * 1e3).toFixed(1)}us per call (${(tBare * 2).toFixed(2)}ms per eval x2)`);
  console.log(line("declared cloud-general@100%", tCG));
  console.log(line("declared local@100%", tLocal));
  console.log(line("declared mixed 50/50", tMixed));
  const tCfgBare = bench(() => assignTasks({ catalog: baseCfg, tasks: bare }), 30);
  const tCfgDecl = bench(() => assignTasks({ catalog: declaredCfg, tasks: bare }), 30);
  console.log(
    `privacy spot [M=7 configured]: bare=${(tCfgBare * 2).toFixed(2)}ms declared@100%=${(tCfgDecl * 2).toFixed(2)}ms (delta ${((tCfgDecl - tCfgBare) * 2).toFixed(2)}ms per eval x2)`
  );
  const t10Bare = bench(() => assignTasks({ catalog: base10, tasks: bare }), 30);
  const t10Forbidden = bench(() => assignTasks({ catalog: forbidden10, tasks: bare }), 30);
  console.log(
    `privacy forbidden dose [M=10]: approved=${(t10Bare * 2).toFixed(2)}ms forbidden-k3=${(t10Forbidden * 2).toFixed(2)}ms (delta ${((t10Forbidden - t10Bare) * 2).toFixed(2)}ms per eval x2)`
  );
  conclusions.push(
    `privacy lattice M=2 per-eval: bare=${(tBare * 2).toFixed(2)} declaredCG=${(tCG * 2).toFixed(2)} declaredLocal=${(tLocal * 2).toFixed(2)} mixed=${(tMixed * 2).toFixed(2)}ms | M=7 delta=${((tCfgDecl - tCfgBare) * 2).toFixed(2)}ms | M=10 forbidden-k3 delta=${((t10Forbidden - t10Bare) * 2).toFixed(2)}ms`
  );

  // Real face N=30 + 10x stress at the declared corner (production callers
  // can never reach it -- any delta is a synthetic-only surcharge).
  {
    const bare30 = bare.slice(0, 30);
    const tBare30 = bench(() => assignTasks({ catalog: base2, tasks: bare30 }), 2000);
    const tDecl30 = bench(() => assignTasks({ catalog: declaredCG2, tasks: bare30 }), 2000);
    console.log(
      `privacy REAL face N=30 [M=2]: bare=${(tBare30 * 1e3).toFixed(1)}us declared=${(tDecl30 * 1e3).toFixed(1)}us per call (delta ${((tDecl30 - tBare30) * 1e3).toFixed(1)}us)`
    );
    const bareStress = genReplayTasks(mulberry32(0xb44b01), 20000);
    const tBareStress = bench(() => assignTasks({ catalog: base2, tasks: bareStress }), 5);
    const tDeclStress = bench(() => assignTasks({ catalog: declaredCG2, tasks: bareStress }), 5);
    console.log(
      `privacy 10x stress N=20000 [M=2]: bare=${(tBareStress * 2).toFixed(1)}ms declared=${(tDeclStress * 2).toFixed(1)}ms per eval x2 (delta ${((tDeclStress - tBareStress) * 2).toFixed(1)}ms)`
    );
    conclusions.push(
      `privacy real-face N=30 delta=${((tDecl30 - tBare30) * 1e3).toFixed(1)}us | stress N=20000 delta=${((tDeclStress - tBareStress) * 2).toFixed(1)}ms per eval`
    );
  }

  // Request-side dose on the route() unit face (the only face that can carry
  // privacyRequired) + satisfiesPrivacy leg micro-attribution.
  {
    const routerCfgDecl = createModelRouter(declaredCfg);
    const routerCfgBare = createModelRouter(baseCfg);
    const mkInput = (privacyRequired?: PrivacyClass): RouteTaskInput => ({
      taskId: "tsk_unit_01" as TaskId,
      role: "actor",
      complexity: "MEDIUM",
      modelPolicy: { allowedModels: baseCfg.models.map((m) => m.id) },
      family: "edit",
      requiredCapabilities: ["tool-use"],
      ...(privacyRequired !== undefined ? { privacyRequired } : {}),
      limits: { remainingTimeMs: Number.MAX_SAFE_INTEGER }
    });
    const tDefault = bench(() => routerCfgBare.route(mkInput()), 20000);
    const tExplicitCG = bench(() => routerCfgBare.route(mkInput("cloud-general")), 20000);
    const tCA = bench(() => routerCfgDecl.route(mkInput("cloud-approved")), 20000);
    console.log(
      `privacy request-side unit [M=7 route()]: default=${(tDefault * 1e6).toFixed(0)}ns explicit-cloud-general=${(tExplicitCG * 1e6).toFixed(0)}ns cloud-approved-on-declared=${(tCA * 1e6).toFixed(0)}ns per route`
    );
    const undeclaredDesc: ModelDescriptor = {
      modelId: "u1", providerId: "p", version: "v1", capabilities: ["tool-use"],
      providerPolicy: "approved", inputCostPerMTok: 1, outputCostPerMTok: 3, latencyMsPer1K: 80
    };
    const declaredDesc: ModelDescriptor = { ...undeclaredDesc, modelId: "d1", privacyClass: "cloud-approved" };
    const tUndeclLeg = bench(() => {
      satisfiesPrivacy(undeclaredDesc, "cloud-general");
    }, 1000000);
    const tDeclLeg = bench(() => {
      satisfiesPrivacy(declaredDesc, "cloud-general");
    }, 1000000);
    const tRejectLeg = bench(() => {
      satisfiesPrivacy(declaredDesc, "local");
    }, 1000000);
    console.log(
      `privacy leg micro: satisfiesPrivacy undeclared=${(tUndeclLeg * 1e6).toFixed(2)}ns declared=${(tDeclLeg * 1e6).toFixed(2)}ns declared-reject=${(tRejectLeg * 1e6).toFixed(2)}ns per call`
    );
    check("privacy micro sanity: undeclared leg passes cloud-general", satisfiesPrivacy(undeclaredDesc, "cloud-general"));
    check("privacy micro sanity: undeclared leg fail-closes local", !satisfiesPrivacy(undeclaredDesc, "local"));
    check("privacy micro sanity: local-declared serves cloud-approved", satisfiesPrivacy({ ...undeclaredDesc, privacyClass: "local" }, "cloud-approved"));
    check("privacy micro sanity: cloud-general-declared rejected for cloud-approved", !satisfiesPrivacy({ ...undeclaredDesc, privacyClass: "cloud-general" }, "cloud-approved"));
    conclusions.push(
      `privacy request-side unit per-route: default=${(tDefault * 1e6).toFixed(0)}ns explicitCG=${(tExplicitCG * 1e6).toFixed(0)}ns CA-on-declared=${(tCA * 1e6).toFixed(0)}ns | leg micro undeclared=${(tUndeclLeg * 1e6).toFixed(2)}ns declared=${(tDeclLeg * 1e6).toFixed(2)}ns`
    );
  }
}

console.log(`\nCONCLUSIONS: ${conclusions.join(" | ")}`);
if (failures > 0) {

}
