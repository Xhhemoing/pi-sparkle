function buildContractFn(canon: RequirementContract): RequirementContract {
  const criteria: unknown[] = [];
  for (let i = 0; i < canon.acceptanceCriteria.length; i += 1) criteria.push({ ...canon.acceptanceCriteria[i] });
  const assumptions: unknown[] = [];
  for (let i = 0; i < canon.assumptions.length; i += 1) assumptions.push({ ...canon.assumptions[i] });
  const questions: unknown[] = [];
  for (let i = 0; i < canon.questions.length; i += 1) questions.push({ ...canon.questions[i] });
  const contract = { ...canon } as Record<string, unknown>;
  contract.acceptanceCriteria = criteria;
  contract.assumptions = assumptions;
  contract.questions = questions;
  return contract as unknown as RequirementContract;
}
function buildTasksFn(contract: RequirementContract, count: number): CoverageTaskRef[] {
  const tasks: unknown[] = [];
  for (let t = 0; t < count; t += 1) {
    const acs: unknown[] = [];
    for (let i = 0; i < contract.acceptanceCriteria.length; i += 1) acs.push({ id: contract.acceptanceCriteria[i]?.id });
    tasks.push({ id: "tsk_0000000" + String(t), acceptanceCriteria: acs });
  }
  return tasks as CoverageTaskRef[];
}
function buildQuestionArrFn(canon: RequirementContract): unknown[] {
  const questions: unknown[] = [];
  for (let i = 0; i < canon.questions.length; i += 1) questions.push({ ...canon.questions[i] });
  return questions;
}
function buildOptionsFn(contract: RequirementContract): { resolvedQuestionIds: string[] } {
  const ids: string[] = [];
  for (let i = 0; i < contract.questions.length; i += 1) ids.push(String(contract.questions[i]?.id));
  return { resolvedQuestionIds: ids };
}

function realmOk(arm: ArmName, set: OperandSet, index: number): boolean {
  const kit: RealmKit = arm === "FR" || arm === "FR-CT" ? F1 : arm === "FR2" ? (index % 2 === 0 ? F1 : F2) : MAIN;
  const tasksKit: RealmKit = arm === "FR" || arm === "FR-TK" ? F1 : arm === "FR2" ? (index % 2 === 0 ? F1 : F2) : MAIN;
  const contractSide =
    protoOf(set.contract.acceptanceCriteria) === kit.arrProto &&
    protoOf(set.contract.questions) === kit.arrProto &&
    protoOf(set.contract.acceptanceCriteria[0] ?? {}) === kit.objProto &&
    (arm === "SR" || protoOf(set.contract) === kit.objProto);
  const tasksSide =
    protoOf(set.tasks) === tasksKit.arrProto &&
    protoOf(set.tasks[0] ?? {}) === tasksKit.objProto &&
    set.tasks.every((task) => protoOf(task.acceptanceCriteria) === tasksKit.arrProto);
  const pinned =
    storagePinned(set.contract.acceptanceCriteria) &&
    storagePinned(set.tasks) &&
    set.tasks.every((task) => storagePinned(task.acceptanceCriteria));
  return contractSide && tasksSide && pinned;
}

/* Leg EX: isolated-child cross-check for the worst E1 arm (gate+precedence)

    let worstArm = arms[1] as string;
    let worstVal = -Infinity;
    for (const arm of arms.slice(1)) {
      if ((result[arm] as number) > worstVal) {
        worstVal = result[arm] as number;
        worstArm = arm;
      }
    }
    return worstArm;
  };

  const worstText = worstOf(e3Result, ["MB", "HO", "DI"]);
  const mkChildScript = (arm: string, mode: "gate" | "text"): string => `
