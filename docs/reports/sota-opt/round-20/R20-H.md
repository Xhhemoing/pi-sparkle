 *                       Array.fill().join("") — two earlier draft forms
 *                       were caught by the dose-verification gate:
 *                       String.prototype.repeat builds an O(log n) shared
 *                       cons DAG that retains almost nothing, and
 *                       Buffer.toString("latin1") yields Node EXTERNAL
 *                       strings whose payload lives outside heapUsed)
 *   D2 SMALLOBJ ~100MB (1M small objects: pointer-dense old-gen graph,
 *                       worst case for marking)
 *   D3 SMALLOBJ ~300MB (3M small objects: dose-response scaling point)
 * Adjudication: all doses must stay µs-class; any ambient tax is
 * engine-owned, once-per-run, and unfixable from H code — no candidate.
 * ================================================================ */
const mkPressureScript = (dose: string): string => `
  const { extractHeuristicContract } = await import("/workspace/dist/requirement/heuristic.js");
  const { assertCoverageAllowsStart } = await import("/workspace/dist/requirement/coverage.js");
  const { applyPrecedence } = await import("/workspace/dist/requirement/precedence.js");
  const objective = "fix the login retry bug in src/auth/session.ts and keep tests green";
  const args = { objective };
  const tasks = Array.from({ length: 5 }, (_, i) => ({
    id: "tsk_0000000" + i,
    acceptanceCriteria: [{ id: "ac-objective" }, { id: "ac-tests" }]
  }));
  const heapBeforeDose = process.memoryUsage().heapUsed;
  let hold;
  ${dose === "D0" ? "hold = null;" : ""}
  ${dose === "D1"
    ? `hold = new Array(16384); for (let i = 0; i < 16384; i += 1) hold[i] = new Array(512).fill("abcdefg" + String.fromCharCode(97 + (i % 26))).join("") + i;`
    : ""}
  ${dose === "D2"
    ? `hold = new Array(1000000); for (let i = 0; i < 1000000; i += 1) hold[i] = { a: i, b: "s" + (i % 1000), c: i * 0.5 };`
    : ""}
  ${dose === "D3"
    ? `hold = new Array(3000000); for (let i = 0; i < 3000000; i += 1) hold[i] = { a: i, b: "s" + (i % 1000), c: i * 0.5 };`
    : ""}
  const heapAfterDose = process.memoryUsage().heapUsed;
  const first = JSON.stringify(await extractHeuristicContract(args));
  const N = 4096;
  const durations = new Array(N);
  for (let i = 0; i < N; i += 1) {
    const t0 = process.hrtime.bigint();
    const cand = await extractHeuristicContract(args);
    assertCoverageAllowsStart(cand.contract, tasks);
    applyPrecedence(cand.contract, "user-first");
    const t1 = process.hrtime.bigint();
    durations[i] = Number(t1 - t0);
  }
  const last = JSON.stringify(await extractHeuristicContract(args));
  const tail = durations.slice(N - 2048).sort((a, b) => a - b);
  const q = (p) => tail[Math.min(tail.length - 1, Math.floor(p * tail.length))];
  const heapEnd = process.memoryUsage().heapUsed;
  // keep the dose live to the very end so GC cannot collect it mid-harness
  if (hold !== null && hold.length === 0) process.stderr.write("unreachable");

    dose: "${dose}",
    doseMB: (heapAfterDose - heapBeforeDose) / (1 << 20),
    heapEndMB: heapEnd / (1 << 20),
    p50: q(0.5), p90: q(0.9), p99: q(0.99), max: tail[tail.length - 1],
    call1Us: durations[0] / 1e3,
    deterministic: first === last,
    candidate: first
  }));
`;
log("--- AMBIENT-STATE CENSUS, leg P: retained-heap dose-response (fresh dist processes, 4096 calls each) ---");
const doseCandidates: Record<string, string> = {};
{
  const expectedMinDoseMB: Record<string, number> = { D0: -1, D1: 50, D2: 60, D3: 180 };
  for (const dose of ["D0", "D1", "D2", "D3"]) {

      encoding: "utf8",
      maxBuffer: 64 * 1024 * 1024
    });

    if (proc.status !== 0) continue;
    const r = JSON.parse(proc.stdout) as {
      dose: string; doseMB: number; heapEndMB: number;
      p50: number; p90: number; p99: number; max: number;
      call1Us: number; deterministic: boolean; candidate: string;
    };
    doseCandidates[dose] = r.candidate;
    log(
      `P ${dose}: retained dose=${r.doseMB.toFixed(1)}MB (heap end ${r.heapEndMB.toFixed(0)}MB); steady last-2048: p50=${(r.p50 / 1e3).toFixed(2)}µs p90=${(r.p90 / 1e3).toFixed(2)}µs p99=${(r.p99 / 1e3).toFixed(2)}µs max=${(r.max / 1e3).toFixed(1)}µs; call#1=${r.call1Us.toFixed(0)}µs`
    );
    check(`P ${dose} retained dose held (>= ${expectedMinDoseMB[dose]}MB)`, r.doseMB >= (expectedMinDoseMB[dose] ?? 0), String(r.doseMB));
    check(`P ${dose} steady p50 is µs-class (<100µs)`, r.p50 < 1e5, String(r.p50));
    check(`P ${dose} steady p99 is µs-class (<1ms)`, r.p99 < 1e6, String(r.p99));
    check(`P ${dose} output byte-determinism (first vs last call)`, r.deterministic);

  }
  const base = doseCandidates["D0"];
  for (const dose of ["D1", "D2", "D3"]) {
    check(`P ${dose} candidate JSON byte-equal to D0 (dose cannot change output)`, doseCandidates[dose] === base);
  }
}
