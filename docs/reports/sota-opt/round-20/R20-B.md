function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);

  };

/* ============================================================

 * Content shapes:
 *  - MISS: uniform tokens that match none of the 8 analyze-task patterns ->
 *    every regex runs its full-length scan (worst steady-state for misses).
 *  - NEARMISS: tokens that are maximal proper prefixes of pattern keywords
 *    (deplo/producti/credentia/...) -> forces deep alternation attempts at
 *    high density (adversarial backtracking census; every pattern is a
 *    bounded alternation with no nested quantifiers, so this must stay
 *    within a constant factor of MISS -- superlinear blowup impossible).
 *  - HITLATE: MISS body with " production deploy" appended at the very end
 *    -> HIGH_RISK_RE + deploy gate must scan the whole string, then family
 *    resolution stops at "deploy" (the 6 remaining family regexes skipped).
 *  - PAD: whitespace-padded short MISS core -> isolates the trim() scan
 *    (post-trim text is tiny so regex work is constant).

const MISS_TOKENS = ["zorv", "quel", "brint", "moxa", "felp", "durn", "sakle", "wemb"] as const;
const NEARMISS_TOKENS = [
  "deplo", "producti", "credentia", "revie", "verif", "surve",
  "investiga", "refacto", "implemen", "decompo", "critiqu", "integrat"
] as const;
type LShape = "miss" | "nearmiss" | "hitlate" | "pad";
function textOfLength(rng: () => number, targetLen: number, shape: LShape): string {
  if (shape === "pad") {
    const core = "zorv quel brint moxa";
    if (targetLen <= core.length) return core.slice(0, targetLen);
    const padTotal = targetLen - core.length;
    const left = Math.floor(padTotal / 2);
    return " ".repeat(left) + core + " ".repeat(padTotal - left);
  }
  const tokens: readonly string[] = shape === "nearmiss" ? NEARMISS_TOKENS : MISS_TOKENS;
  let out = "";
  while (out.length < targetLen) out += `${pick(rng, tokens)} `;
  let text = out.slice(0, targetLen);
  // Replace any truncated trailing token with a keyword-free x-run: slicing a
  // near-miss token can otherwise mint a real keyword ("producti" -> "prod").
  const lastSpace = text.lastIndexOf(" ");
  if (lastSpace < 0) {
    text = "x".repeat(text.length);
  } else if (lastSpace < text.length - 1) {
    text = text.slice(0, lastSpace + 1) + "x".repeat(text.length - lastSpace - 1);
  }
  if (shape === "hitlate") {
    const tail = " production deploy";
    text = text.slice(0, Math.max(0, targetLen - tail.length)) + tail;
  }
  return text;
}

}
