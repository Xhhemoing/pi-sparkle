function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

  candidatePolicy = CONFIGURED_CANDIDATE_POLICY

    v1: ResourceVersion;
    v2?: ResourceVersion;

    body: string;
  }[] = [];
  for (let i = 0; i < 8; i += 1) {

      kind: "prompt",
      name: `agent-prompt-${i}`,
      scope: { kind: "project", projectId: createProjectId(() => `${tag}proj`) }
    };
    const body = `# prompt ${i}\n${`Follow the review checklist, cite evidence for every claim, and keep the diff bounded to the stated scope (${i}).\n`.repeat(30 + i * 8)}`;
    const v1 = registry.registerBaseline({ identity, content: body, author: HUMAN });

      identity,
      content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    const entry: {
      identity: ResourceIdentity;
      v1: ResourceVersion;
      v2?: ResourceVersion;
      candidate: ImprovementCandidate;
      body: string;
    } = { identity, v1, candidate, body };
    if (i < 5) {
      const result = promoteWithRegistry(registry, {
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
      entry.v2 = result.newVersion as ResourceVersion;
    } else if (i === 5) {
      registry.beginPromotion({
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
    }
    versions.push(entry);
  }
  const first = versions[0] as (typeof versions)[number];
  registry.rollback({
    identity: first.identity,
    expectedCurrentVersionId: (first.v2 as ResourceVersion).versionId,
    targetVersionId: first.v1.versionId,
    reason: "degradation",
    evidence: ["ev:configured-degradation"],
    automatic: false
  });
  const second = versions[1] as (typeof versions)[number];
  registry.rollback({
    identity: second.identity,
    expectedCurrentVersionId: (second.v2 as ResourceVersion).versionId,
    targetVersionId: second.v1.versionId,
    reason: "user",
    evidence: ["ev:configured-user"],
    automatic: false
  });
  registry.retire((versions[2] as { v1: ResourceVersion }).v1.versionId);
  registry.retire((versions[3] as { v1: ResourceVersion }).v1.versionId);
  return {
    ...built,
    prompts: versions.map(({ identity, v1, candidate, body }) => ({ identity, v1, candidate, body }))
  };
}

/** Spawn a child that acquires the registry lock, runs `bodyJs` (an async JS
 *  snippet with `root` in scope, executed INSIDE the lock after printing
 *  LOCKED), prints OPEND at body end, then releases and prints RELEASED. */
function spawnHolder(
  root: string,
  bodyJs: string,
  onLocked: () => void
): Promise<HolderResult> {
  const script = `
import { withAdaptationRegistryLock } from ${JSON.stringify(join(DIST, "adaptation/promotion.js"))};
import { loadAdaptationRegistry, promoteWithRegistry, saveAdaptationRegistry } from ${JSON.stringify(join(DIST, "adaptation/promotion.js"))};
import { readFileSync } from "node:fs";
const root = ${JSON.stringify(root)};
await withAdaptationRegistryLock(root, async () => {
  console.log("LOCKED " + Date.now());
  ${bodyJs}
  console.log("OPEND " + Date.now());
});
console.log("RELEASED " + Date.now());
`;
  return new Promise((resolveResult, rejectResult) => {
    const child = spawn(NODE, ["--input-type=module", "-e", script], { stdio: ["ignore", "pipe", "pipe"] });
    const result: HolderResult = { lockedAt: 0, opEndAt: 0, releasedAt: 0, exitCode: -1 };
    let stderr = "";
    let buffer = "";
    child.stdout.on("data", (chunk: Buffer) => {
      buffer += chunk.toString();
      let idx = buffer.indexOf("\n");
      while (idx >= 0) {
        const line = buffer.slice(0, idx).trim();
        buffer = buffer.slice(idx + 1);
        const [tag, ts] = line.split(" ");
        if (tag === "LOCKED") {
          result.lockedAt = Number(ts);
          onLocked();
        } else if (tag === "OPEND") {
          result.opEndAt = Number(ts);
        } else if (tag === "RELEASED") {
          result.releasedAt = Number(ts);
        }
        idx = buffer.indexOf("\n");
      }
    });
    child.stderr.on("data", (chunk: Buffer) => {
      stderr += chunk.toString();
    });
    child.on("close", (code) => {
      result.exitCode = code ?? -1;
      if (code !== 0) {
        rejectResult(new Error(`holder exited ${code}: ${stderr.slice(0, 300)}`));
      } else {
        resolveResult(result);
      }
    });
  });
}

/* 1a: cross-process contention queueing quantization (default options:
 * timeoutMs=5000, retryMs=10). Holder holds H ms; contender measures wait. */
{
  const root = join(workRoot, "contention-1a");
  await mkdir(root, { recursive: true });
  const HOLDS = [0, 3, 5, 20, 35] as const;
  const K = 6;
  console.log(`S1a cross-process contention queueing (holder holds H ms; contender uses default options retryMs=10; ${K} fresh holder processes per H; wait = contender acquisition wall):`);
  for (const holdMs of HOLDS) {
    const waits: number[] = [];
    let taxSum = 0;
    for (let i = 0; i < K; i += 1) {
      let contenderPromise: Promise<{ wait: number; acquiredAt: number }> | undefined;
      let contenderStart = () => {};
      const started = new Promise<void>((r) => {
        contenderStart = r;
      });
      const holderPromise = spawnHolder(
        root,
        `await new Promise((r) => setTimeout(r, ${holdMs}));`,
        () => {
          contenderPromise = (async () => {
            const t0 = performance.now();
            let acquiredAt = 0;
            await withAdaptationRegistryLock(root, async () => {
              acquiredAt = Date.now();
            });
            return { wait: performance.now() - t0, acquiredAt };
          })();
          contenderStart();
        }
      );
      await started;
      const [holder, contender] = await Promise.all([holderPromise, contenderPromise as Promise<{ wait: number; acquiredAt: number }>]);
      check(
        `S1a mutual exclusion witness (H=${holdMs}, rep=${i})`,
        contender.acquiredAt >= holder.opEndAt - 2,
        `contender acquired at ${contender.acquiredAt} but holder operation ended at ${holder.opEndAt}`
      );
      waits.push(contender.wait);
      taxSum += contender.wait - holdMs;
    }
    waits.sort((a, b) => a - b);
    console.log(
      `S1a   H=${String(holdMs).padStart(2)}ms -> contender wait min=${(waits[0] as number).toFixed(1)} med=${quantile(waits, 0.5).toFixed(1)} max=${(waits[waits.length - 1] as number).toFixed(1)} ms | mean quantization tax (wait - H) = ${(taxSum / K).toFixed(1)} ms`
    );
  }
}

  // Seed the storm root with a real save of variant A.
