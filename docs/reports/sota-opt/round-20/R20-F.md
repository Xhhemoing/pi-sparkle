type StreamKind =
  | { readonly kind: "clean4" }
  | { readonly kind: "clean6" }
  | { readonly kind: "mix46" }
  | { readonly kind: "missingEvery"; readonly stride: number } // stride=1 → all missing
  | { readonly kind: "breachEvery"; readonly stride: number };

function buildStream(hashes: readonly string[], shape: StreamKind): ExperimentOutcome[] {
  return hashes.map((h, i) => {
    switch (shape.kind) {
      case "clean4":
        return clean4(h);
      case "clean6":
        return clean6(h);
      case "mix46":
        return i % 2 === 0 ? clean4(h) : clean6(h);
      case "missingEvery":
        return i % shape.stride === 0 ? missing5(h) : clean4(h);
      case "breachEvery":
        return i % shape.stride === 0 ? breach4(h) : clean4(h);
    }
  });
}

/* ------------------------------ experiment loops -------------------------- */
/** Production interleave (evaluateProposalShadow shape): assign+record per entry, break on halt. */
function runInterleaved(plan: ExperimentPlan, stream: readonly ExperimentOutcome[]): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (const outcome of stream) {
    if (state.halted) {
      break;
    }
    state = runner.assign(state, outcome.episodeHash, 0);
    state = runner.recordOutcome(state, outcome, 0);
  }
  return state;
}

/** Phase-separated: all assigns first, then all records (optionally reversed record order). */
function runPhased(
  plan: ExperimentPlan,
  stream: readonly ExperimentOutcome[],
  reverseRecords: boolean
): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (const outcome of stream) {
    if (state.halted) break;
    state = runner.assign(state, outcome.episodeHash, 0);
  }
  if (reverseRecords) {
    for (let i = stream.length - 1; i >= 0; i--) {
      if (state.halted) break;
      state = runner.recordOutcome(state, stream[i] as ExperimentOutcome, 0);
    }
  } else {
    for (const outcome of stream) {
      if (state.halted) break;
      state = runner.recordOutcome(state, outcome, 0);
    }
  }
  return state;
}
