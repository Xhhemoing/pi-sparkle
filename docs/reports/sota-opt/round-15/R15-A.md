function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function fitQuadratic(
  points: ReadonlyArray<{ x: number; t: number }>
): { a: number; b: number; c: number } {
  // exact solve through the three given points (Vandermonde 3x3)
  const [p0, p1, p2] = points;
  if (p0 === undefined || p1 === undefined || p2 === undefined) {
    throw new Error("need exactly 3 points");
  }
  const d01 = (p1.t - p0.t) / (p1.x - p0.x);
  const d12 = (p2.t - p1.t) / (p2.x - p1.x);
  const c = (d12 - d01) / (p2.x - p0.x);
  const b = d01 - c * (p0.x + p1.x);
  const a = p0.t - b * p0.x - c * p0.x * p0.x;
  return { a, b, c };
}

function crossingX(fit: { a: number; b: number; c: number }, perGateMsLine: number): number {
  // solve a + b*x + c*x^2 = perGateMsLine (ms), positive root
  const { a, b, c } = fit;
  const rhs = perGateMsLine;
  if (c <= 0) return Number.POSITIVE_INFINITY;
  const disc = b * b - 4 * c * (a - rhs);
  return (-b + Math.sqrt(disc)) / (2 * c);
}
