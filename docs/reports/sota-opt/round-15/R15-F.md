function timeMs(fn: () => void, rounds = 5): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

function medianMs(fn: () => void, rounds: number): number {
  const samples: number[] = [];
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    samples.push(Number(t1 - t0) / 1e6);
  }
  samples.sort((a, b) => a - b);
  return samples[Math.floor(samples.length / 2)] as number;
}

const P = 2000;

function fullExperiment(plan: ExperimentPlan, hashes: readonly string[]): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (let k = 0; k < A; k++) {
    const h = hashes[k] as string;
    state = runner.assign(state, h, 0);
    const outcome: ExperimentOutcome = {
      episodeHash: h,
      utility: 0.5,
      costUsd: 0.01,
      guardrailBreached: false,
    };
    state = runner.recordOutcome(state, outcome, 0);
  }
  if (state.assignments.length !== A) throw new Error("unexpected assignment count");
  return state;
}

const deltas: number[] = [];

    fullExperiment(anchorPlan, anchorHashes);
  }, PER_ARM);
  const tY = medianMs(() => {
    fullExperiment(anchorPlan, anchorHashes);
  }, PER_ARM);
  deltas.push(tX - tY);
}

const positive = deltas.filter((d) => d > 0).length;
// longest same-sign streak — how many consecutive windows a pure-noise
// comparison can appear "stable" for
let streak = 1;
let bestStreak = 1;
for (let i = 1; i < deltas.length; i++) {
  if (Math.sign(deltas[i] as number) === Math.sign(deltas[i - 1] as number)) {
    streak += 1;
    bestStreak = Math.max(bestStreak, streak);
  } else {
    streak = 1;
  }
}

);

  validateExperimentPlan(plan);
  const interned = population.slice(0, A);
  const deinterned = JSON.parse(JSON.stringify(interned)) as string[];
  for (let i = 0; i < A; i++) {
    if (deinterned[i] !== interned[i]) throw new Error("round-trip mismatch");
  }

  // determinism: interned and de-interned runs are field-identical
  const sInt = fullExperiment(plan, interned);
  const sDeint = fullExperiment(plan, deinterned);
  if (
    sInt.halted !== sDeint.halted ||
    sInt.haltReason !== sDeint.haltReason ||
    sInt.guardrailBreaches !== sDeint.guardrailBreaches ||
    sInt.assignments.length !== sDeint.assignments.length ||
    sInt.outcomes.length !== sDeint.outcomes.length

    throw new Error(`state divergence at L=${L}`);
  }

    const a = sInt.assignments[i];
    const b = sDeint.assignments[i];
    if (a?.episodeHash !== b?.episodeHash || a?.shadowDecision !== b?.shadowDecision) {
      throw new Error(`assignment divergence at L=${L} index=${i}`);
    }
  }

  const tInt = timeMs(() => {
    fullExperiment(plan, interned);
  }, 5);
  const tDeint = timeMs(() => {
    fullExperiment(plan, deinterned);
  }, 5);

  /* component view A: assign-site includes scan (requirePopulationEpisode shape) */
  let includesSink = 0;
  const tInc = timeMs(() => {
    for (let k = 0; k < A; k++) {
      if (population.includes(interned[k] as string)) includesSink += 1;
    }
  }, 5);
  const tIncDeint = timeMs(() => {
    for (let k = 0; k < A; k++) {
      if (population.includes(deinterned[k] as string)) includesSink += 1;

  }, 5);

  /* component view B: aligned-compare scan (S7-F-1 phase-one shape), 2A restores */
  let alignSink = 0;
  const REP = 2000;

    for (let r = 0; r < REP; r++) {
      for (let i = 0; i < A; i++) {
        if (interned[i] === population[i]) alignSink += 1;
      }
    }
  }, 5);
  const tAlnDeint = timeMs(() => {
    for (let r = 0; r < REP; r++) {
      for (let i = 0; i < A; i++) {
        if (deinterned[i] === population[i]) alignSink += 1;
      }
    }
  }, 5);

  /* component view C: validate ×2A (provenance-invariant; L-sensitive) */
  const tValidate = timeMs(() => {
    for (let r = 0; r < REP; r++) validateExperimentPlan(plan);
  }, 3);

  process.stdout.write(
    `axis 2 L=${String(L).padStart(2)}: anchor interned=${tInt.toFixed(2)}ms ` +
      `de-interned=${tDeint.toFixed(2)}ms provenance-delta=${(tDeint - tInt).toFixed(2)}ms | ` +
      `includes ${(tInc * 1000).toFixed(0)}/${(tIncDeint * 1000).toFixed(0)}µs | ` +
      `aligned-compare x${REP} ${tAln.toFixed(2)}/${tAlnDeint.toFixed(2)}ms | ` +
      `validate x${REP} ${tValidate.toFixed(2)}ms\n`
  );
  void includesSink;
  void alignSink;
}
