### A.3 A/A 噪声地板 harness（/tmp/r15i-bench/aa.mjs；§1.3 轴 1 证据）

```javascript
// R15-I axis 1: A/A noise floor for the spawn matrix. Both arms run the SAME
// binary on the SAME fixture (run-children@custom); 12 interleaved windows,
// each arm's stat = median of 3 spawns. Any nonzero "delta" is pure
// measurement noise, quantifying the MDE for spawn-to-exit cells.
import { spawnSync } from "node:child_process";
import { mkdirSync, rmSync, cpSync } from "node:fs";
import { join } from "node:path";
import { performance } from "node:perf_hooks";

const CLI = "/workspace/dist/cli/main.js";
const ROOT = "/tmp/r15i-bench";
const FIX = join(ROOT, "fixtures");
const WORK = join(ROOT, "work-aa");
const nodeBin = process.argv[2];
const stateName = process.argv[3] ?? "custom";
const WINDOWS = 12, PER_ARM = 3;

const PROJ = join(FIX, "project");
const CHILDREN = join(FIX, "children.json");

let iter = 0;
function freshState() {
  const dir = join(WORK, `it-${iter++}`);
  rmSync(dir, { recursive: true, force: true });
  cpSync(join(FIX, `state-${stateName}`), dir, { recursive: true });
  return dir;
}
function once() {
  const sd = freshState();
  const env = { ...process.env };
  delete env.PI_PROVIDER; delete env.PI_MODEL; delete env.PI_API_KEY;
  const args = ["run", "--project", PROJ, "--objective", "bench objective", "--state-root", sd, "--children", CHILDREN];
  const t0 = performance.now();
  const r = spawnSync(nodeBin, [CLI, ...args], { env, encoding: "utf8", timeout: 120000 });
  const dt = performance.now() - t0;
  if (r.status !== 0) { console.error(`FAIL: ${(r.stderr ?? "").slice(0, 200)}`); process.exit(1); }
  return dt;
}
function median3() {
  const xs = [once(), once(), once()].sort((a, b) => a - b);
  return xs[1];
}
rmSync(WORK, { recursive: true, force: true });
mkdirSync(WORK, { recursive: true });
once(); once(); // warmup
const deltas = [];
for (let w = 0; w < WINDOWS; w++) {
  const a = median3();
  const b = median3();
  deltas.push(a - b);
}
const abs = deltas.map(Math.abs).sort((x, y) => x - y);
const medianAbs = abs[Math.floor(abs.length / 2)];
const maxAbs = abs[abs.length - 1];
let longest = 0, cur = 0, prev = 0;
for (const d of deltas) {
  const s = Math.sign(d);
  cur = s !== 0 && s === prev ? cur + 1 : 1;
  if (s !== 0 && cur > longest) longest = cur;
  prev = s;
}
const pos = deltas.filter((d) => d > 0).length;
console.log(`A/A run-children@${stateName}: medianAbs=${medianAbs.toFixed(2)}ms maxAbs=${maxAbs.toFixed(2)}ms signs=+${pos}/-${WINDOWS - pos} longestRun=${longest}`);
console.log(`  deltas: ${deltas.map((d) => d.toFixed(1)).join(" ")}`);
```
