 * pass's NEW completeness axis for this slice: the A/A DUEL NOISE-FLOOR
 * GRID (mirroring R15-B §2 / R15-C AAFLR / R15-D / R15-E in their slices).
 * Fourteen passes adjudicated dozens of candidates through paired duel
 * readings down to single-digit ns; the noise floor of that duel protocol
 * ITSELF was never directly measured on this slice. Here, two BYTE-IDENTICAL
 * arms (independent closure identities, true effect exactly zero) are
 * duelled with position alternation across 8 windows per cell, on four
 * cells spanning the slice's production cost range:
 *
 *   cell CHAIN  — full default extractHeuristicContract chain (µs-scale)
 *   cell GATE   — assertCoverageAllowsStart leg (sub-µs scale)
 *   cell PREC   — applyPrecedence leg (~100ns scale)
 *   cell CRIT   — critiqueContract leg (~600ns scale)
 *
 * Purpose: empirically bracket the historical ns-level duel readings behind
 * this slice's rejection ledger (S6-H-1 133-136ns, S8-H-2 103-182ns,
 * S1-H-2 ~80ns, S2-H-1/2 ~44ns, S4-H-1 2-9ns, S2-H-5 9-13ns, S9-H-2
 * 9.0-9.5ns, ...) against the floor the protocol can actually resolve —
 * if the A/A floor is at or above those magnitudes, every such rejection is
 * confirmed a fortiori (the readings could not have hidden a real win), and
 * the "re-measure a closed micro-duel hoping for a different sign" candidate
 * class is closed as a class, without minting IDs.
 *
 * Reference = production imports everywhere; no candidate is proposed, so
 * there is no randomized fuzz — every sanity/fail-closed check below is
 * fully deterministic (fixed inputs; includes the three S5-H-1 semantic
 * attestations required by the dispatch), and three runs are exact

 *

let failures = 0;
let checks = 0;
function check(name: string, ok: boolean, detail?: string): void {
  checks += 1;
  if (!ok) {
    failures += 1;
    process.stderr.write(`FAIL ${name}${detail === undefined ? "" : `: ${detail}`}\n`);
  }
}
function log(line: string): void {
  process.stdout.write(`${line}\n`);
}
function median(xs: number[]): number {
  const sorted = [...xs].sort((a, b) => a - b);
  return sorted[Math.floor(sorted.length / 2)] ?? Number.NaN;
}
function bench(fn: () => void, reps: number): number {
  fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}
async function benchAsync(fn: () => Promise<void>, reps: number): Promise<number> {
  await fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) await fn();
  return (performance.now() - t0) / reps;
}

async function measureChain(label: string, objective: string, habits: HeuristicHabits | undefined): Promise<void> {
  const args = { objective, ...(habits !== undefined ? { habits } : {}) };
  const chain = await benchAsync(async () => {
    await extractHeuristicContract(args);
  }, 3000);
  const prod = await extractHeuristicContract(args);

    id: `tsk_0000000${i}` as TaskId,

  }));
  const options: CoverageStartOptions | undefined =
    real.questions.length > 0

      : undefined;
  const gate = bench(() => assertCoverageAllowsStart(real, tasks, options), 50000);
  const precedence = bench(() => void applyPrecedence(real, "user-first"), 100000);
  const total = chain + gate + precedence;
  log(
    `CEILING ${label}: extractHeuristicContract=${(chain * 1e6).toFixed(0)}ns + run-start gate(Q=${real.questions.length}${options !== undefined ? ", all resolved" : ", no options"})=${(gate * 1e6).toFixed(0)}ns + applyPrecedence=${(precedence * 1e6).toFixed(0)}ns = ${(total * 1e6).toFixed(0)}ns once-per-run production total`
  );
}

/* ================================================================
 * NEW AXIS: A/A duel noise-floor grid (mirrors R15-B / R15-C AAFLR /
 * R15-D / R15-E). Two byte-identical arms — independent closure identities,
 * true effect exactly zero — duelled with position alternation across 8
 * windows per cell. Reports per-window deltas, max-abs pseudo-effect, and
 * sign-stability (8/8 same sign = a pseudo-stable set the protocol could
 * mistake for a real effect).
 * ================================================================ */
{
  const objective = "fix the login retry bug in src/auth/session.ts and keep tests green";
  const prod = await extractHeuristicContract({ objective });
  const real = prod.contract;
  const tasks: CoverageTaskRef[] = Array.from({ length: 5 }, (_, i) => ({
    id: `tsk_0000000${i}` as TaskId,
    acceptanceCriteria: real.acceptanceCriteria.map((criterion) => ({ id: criterion.id }))
  }));

  interface AsyncCell {
    readonly name: string;
    readonly kind: "async";
    readonly armA: () => Promise<void>;
    readonly armB: () => Promise<void>;
    readonly reps: number;
  }
  interface SyncCell {
    readonly name: string;
    readonly kind: "sync";
    readonly armA: () => void;
    readonly armB: () => void;
    readonly reps: number;
  }
  // Byte-identical arm pairs: same body, distinct closure identities (so
  // each arm carries its own FeedbackCell, exactly like a real duel where
  // candidate and reference are distinct functions).
  const chainA = async (): Promise<void> => {
    await extractHeuristicContract({ objective });
  };
  const chainB = async (): Promise<void> => {
    await extractHeuristicContract({ objective });
  };
  const gateA = (): void => {
    assertCoverageAllowsStart(real, tasks);
  };
  const gateB = (): void => {
    assertCoverageAllowsStart(real, tasks);
  };
  const precA = (): void => {
    void applyPrecedence(real, "user-first");
  };
  const precB = (): void => {
    void applyPrecedence(real, "user-first");
  };
  const critA = (): void => {
    void critiqueContract(real);
  };
  const critB = (): void => {
    void critiqueContract(real);
  };
  const cells: (AsyncCell | SyncCell)[] = [
    { name: "CHAIN default (µs-scale)", kind: "async", armA: chainA, armB: chainB, reps: 1500 },
    { name: "GATE leg (sub-µs)", kind: "sync", armA: gateA, armB: gateB, reps: 20000 },
    { name: "PREC leg (~100ns)", kind: "sync", armA: precA, armB: precB, reps: 50000 },
    { name: "CRIT leg (~600ns)", kind: "sync", armA: critA, armB: critB, reps: 20000 }
  ];

  for (const cell of cells) {
    // Warm both arms identically before the windows.
    if (cell.kind === "async") {
      for (let i = 0; i < 200; i += 1) {
        await cell.armA();
        await cell.armB();
      }
    } else {
      for (let i = 0; i < 2000; i += 1) {
        cell.armA();
        cell.armB();
      }
    }
    const deltasNs: number[] = [];
    for (let w = 0; w < 8; w += 1) {
      let tA: number;
      let tB: number;
      if (cell.kind === "async") {
        if (w % 2 === 0) {
          tA = await benchAsync(cell.armA, cell.reps);
          tB = await benchAsync(cell.armB, cell.reps);
        } else {
          tB = await benchAsync(cell.armB, cell.reps);
          tA = await benchAsync(cell.armA, cell.reps);
        }
      } else if (w % 2 === 0) {
        tA = bench(cell.armA, cell.reps);
        tB = bench(cell.armB, cell.reps);
      } else {
        tB = bench(cell.armB, cell.reps);
        tA = bench(cell.armA, cell.reps);
      }
      deltasNs.push((tA - tB) * 1e6);
    }
    const maxAbs = Math.max(...deltasNs.map((d) => Math.abs(d)));
    const positives = deltasNs.filter((d) => d > 0).length;
    const signStable = positives === 8 || positives === 0;
    log(
      `AA-FLOOR ${cell.name}: deltas(ns/call)=[${deltasNs.map((d) => d.toFixed(1)).join(", ")}] median=${median(deltasNs).toFixed(1)} max-abs=${maxAbs.toFixed(1)} signs=${positives}+/${8 - positives}- ${signStable ? "PSEUDO-STABLE(!)" : "mixed(ok)"}`
    );
    check(`AA floor produced finite deltas (${cell.name})`, deltasNs.every(Number.isFinite));
  }
}
