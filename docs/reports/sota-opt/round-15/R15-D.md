每批恰 **45 次 GC 事件**（三次运行逐次相同——分配流确定性的旁证），
合计 22.8–26.2 ms = **152–175 µs/eval**，占批内命令时长
（3.12–3.25 ms/eval）的 **4.7–5.5%**。据 §2b 的切片内分配占比
（23.4–24.5%）计算，「消除全部切片内分配」（不可能——钉名项含明令

**37–43 µs/eval**；叠加 R13-D 已闭合的切片内 CPU 池 <0.9 ms，
分配/GC 维度不改变任何既有裁决的量级，距 10 ms 落地线下沿仍差
1.5–2 个数量级。**验收标准第 3 条在内存维度上对本切片独立成立。**

### 2d. A/A 噪声地板 / 最小可检出效应（MDE）

顺序均衡配对设计（每 rep 对内交替先后次序消除位置偏置；12 窗 ×
12 reps/侧，窗口中位差；三次独立运行）：

| 锚点 | 窗口中位差范围 | max\|Δ\| | sign+ | 最长同号连串 |
| --- | --- | --- | --- | --- |
| eval E=200 配置态 | −435..+304 µs | **263–435 µs** | 3–5/12 | 3–6 窗 |
| save(+fsync) 配置态 | −408..+178 µs | **219–408 µs** | 3–7/12 | 3–4 窗 |

另以朴素 A-先行设计（A 恒在 rep 对内先跑）做过一次试点运行：eval
锚点出现 **−448..+128 µs、sign+ 2/12、8 窗同号**的稳定伪差——纯粹的
位置偏置在 ~3.7 ms 的命令上制造了一个「看起来稳定为负 ~0.4 ms」的
假赢家。两个设计合起来给出本切片的测量端结论：

1. **MDE ≈ 0.26–0.44 ms**（窗口中位口径）：切片内全部可寻址 CPU 池
   （0.53–0.54 ms @ 生产锚，R14-D 实测）只有 MDE 的 ~1.2–2×，任何
   单点微观候选（池的真子集）都落在噪声内，无法诚实测出；
2. 纯噪声下同号连串可达 6–8 窗（H0 下 5 窗同号本就有 6.25% 概率，
   且位置偏置可把它推成系统性假稳定）——「符号稳定」单独不构成落地
   证据，R12-B/S13-B-1 的 15 轮符号纪律与「band-overlap 不是落地」
   规则获得本切片测量端的第一性支持；
3. save 侧 fsync 方差（max|Δ| 408 µs）与 R8-D 以来「fsync 方差支配
   持久化路径」的定性结论定量吻合。

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);

/** Clean-sample allocation measurement: heapUsed delta for one invocation,
 *  keeping only samples with zero GC events delivered for the window. */
async function measureAllocSync(fn: () => void, samples = 15): Promise<{ kb: number; clean: number }> {
  fn(); // warm
  const clean: number[] = [];
  const dirty: number[] = [];
  for (let i = 0; i < samples; i += 1) {
    gcNow();
    await settle();
    const before = gcLog.length;
    const h0 = process.memoryUsage().heapUsed;
    fn();
    const h1 = process.memoryUsage().heapUsed;
    await settle();
    const gcDuring = gcLog.length - before;
    const kb = (h1 - h0) / 1024;
    if (gcDuring === 0 && kb >= 0) clean.push(kb);
    else dirty.push(kb);
  }
  const source = clean.length > 0 ? clean : dirty;
  return { kb: median(source), clean: clean.length };
}

interface EpisodeShape {
  readonly episodeHash: string;
  readonly taskId: string;
  readonly role: AgentRole;
  readonly objective: string;
  readonly taskFamily?: string | undefined;
  readonly taskSuccess?: "PASS" | "FAIL" | undefined;

interface ReplayActionShape {
  readonly episodeHash: string;
  readonly taskFamily: string;
  readonly taskSuccess: "PASS" | "FAIL" | "UNOBSERVED";
  readonly baselineModel: string;
  readonly candidateModel: string;
  readonly baselineCostUsd: number;
  readonly candidateCostUsd: number;
}
const FORBIDDEN_POLICY_FIELDS = new Set([
  "permission",
  "credential",
  "secret",
  "token",
  "apikey",
  "password",
  "authorization"
]);
function assertNoForbiddenFieldsReplica(value: unknown): void {
  if (Array.isArray(value)) {
    for (const entry of value) {
      assertNoForbiddenFieldsReplica(entry);
    }
    return;
  }
  if (!isRecord(value)) {
    return;
  }
  for (const [key, child] of Object.entries(value)) {
    if (FORBIDDEN_POLICY_FIELDS.has(key.toLowerCase())) {
      throw new DomainValidationError(`routing-policy contains forbidden field ${key}`);
    }
    assertNoForbiddenFieldsReplica(child);
  }
}
function parseEpisodeReplica(value: unknown, index: number): EpisodeShape {
  if (!isRecord(value)) {
    throw new DomainValidationError(`dataset episodes[${index}] must be an object`);
  }
  if (typeof value.episodeHash !== "string" || value.episodeHash.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires episodeHash`);
  }
  if (typeof value.taskId !== "string") {
    throw new DomainValidationError(`dataset episodes[${index}] requires taskId`);
  }
  parseTaskId(value.taskId);
  if (!isAgentRole(value.role)) {
    throw new DomainValidationError(`dataset episodes[${index}] has invalid role`);
  }
  if (typeof value.objective !== "string" || value.objective.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires objective`);
  }
  if (typeof value.originalWorkspace !== "string" || value.originalWorkspace.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires originalWorkspace`);
  }
  let taskSuccess: "PASS" | "FAIL" | undefined;
  if (value.taskSuccess !== undefined) {
    if (value.taskSuccess !== "PASS" && value.taskSuccess !== "FAIL") {
      throw new DomainValidationError(
        `dataset episodes[${index}] taskSuccess must be PASS or FAIL when present`
      );
    }
    taskSuccess = value.taskSuccess;
  }
  const taskFamily =
    typeof value.taskFamily === "string" && value.taskFamily.trim() !== ""
      ? value.taskFamily
      : undefined;
  return {
    episodeHash: value.episodeHash,
    taskId: value.taskId,
    role: value.role,
    objective: value.objective,
    originalWorkspace: value.originalWorkspace,
    ...(taskFamily !== undefined ? { taskFamily } : {}),
    ...(taskSuccess !== undefined ? { taskSuccess } : {})
  };
}
function catalogCostReplica(catalog: ModelRouterConfig, modelId: string): number {
  const model = catalog.models.find((entry) => entry.id === modelId);
  if (model === undefined) {
    throw new DomainValidationError(`selected model ${modelId} is not in the catalog`);
  }
  return model.estimatedCostUsd;
}

function parseManifestHead(parsed: unknown): ManifestShape {

  });

      datasetId: manifest.datasetId,
      cache: {
        candidateHash: candidate.contentHash,
        environmentVersion: manifest.environmentVersion,
        evaluatorVersion: ROUTING_EVALUATOR_VERSION
      },
      actions
    })
  );
  return {
    candidateId,
    contentHash: candidate.contentHash,
    cacheKey,
    stages: ["static", "replay"],

        candidateCostUsd: action.candidateCostUsd
      });
    }
  });
  await addPhase("report.gatedComparisonReport", "outside slice / S7-D-2", false, () => {
    gatedComparisonReport({ records, claims: [], config: REPLAY_CONFIG, difficultyTier: "replay" });
  });
  await addPhase("report.rerunHash (stableStringify+hash32)", "S6-D-5", true, () => {
    hash32(
      stableStringify({
        datasetId: manifest.datasetId,
        cache: {
          candidateHash: candidate.contentHash,
          environmentVersion: manifest.environmentVersion,
          evaluatorVersion: ROUTING_EVALUATOR_VERSION
        },
        actions
      })
    );
  });

  console.log("S2b per-phase allocation attribution (KB per eval, clean-sample medians):");
  let inSliceKb = 0;
  let totalKb = 0;
  for (const phase of phases) {
    totalKb += phase.kb;
    if (phase.inSlice) inSliceKb += phase.kb;
    console.log(
      `S2b   ${phase.name.padEnd(44)} ${phase.kb.toFixed(1).padStart(8)}KB  clean=${phase.clean}/15  ${phase.inSlice ? "IN-SLICE" : "callee/floor"}  [${phase.pin}]`
    );
  }
  console.log(
    `S2b   in-slice allocation total=${inSliceKb.toFixed(0)}KB/eval of phase-sum ${totalKb.toFixed(0)}KB (${((inSliceKb / totalKb) * 100).toFixed(1)}%); every KB pinned to a closed ID`
  );

  // (c) GC amortization over a batch of eval commands
  const BATCH = 150;
  await evalRoutingPolicy({ stateRoot, candidateId, datasetDir }); // warm
  gcNow();
  await settle();
  const gcBefore = gcLog.length;
  const t0 = performance.now();
  for (let i = 0; i < BATCH; i += 1) {
    await evalRoutingPolicy({ stateRoot, candidateId, datasetDir });
  }
  const batchMs = performance.now() - t0;
  await settle();
  const events = gcLog.slice(gcBefore);
  const gcTotalMs = events.reduce((sum, event) => sum + event.durationMs, 0);
  const perEvalGcUs = (gcTotalMs / BATCH) * 1e3;
  const perEvalMs = batchMs / BATCH;
  const gcShare = (gcTotalMs / batchMs) * 100;
  // Upper bound: even if ALL in-slice allocation vanished (impossible -- the
  // pins include mandated keeps), GC savings <= GC-total x in-slice share.
  const upperBoundUs = perEvalGcUs * (inSliceKb / totalKb);
  console.log(
    `S2c GC amortization (${BATCH}-eval batch): ${events.length} GC events, ${gcTotalMs.toFixed(1)}ms total = ${perEvalGcUs.toFixed(0)}us/eval (${gcShare.toFixed(1)}% of ${perEvalMs.toFixed(2)}ms/eval) | eliminate-ALL-in-slice-allocation upper bound ~= ${upperBoundUs.toFixed(0)}us/eval GC-side + in-slice CPU pool (R13-D: <0.9ms) -- 1.5-2 orders below the 10ms landing line`
  );
  check("S2c sanity: GC share is a small fraction of the command", gcShare < 50, `gcShare=${gcShare.toFixed(1)}%`);
}

/* ============================================================
 * Section 3 (NEW AXIS B): A/A noise floor / minimum detectable effect.
 * Paired identical-code windows (order-balanced A/A' reps, window medians)
 * for the configured eval command and for save(+fsync). Quantifies the max
 * |delta| pure noise can show on this slice's anchors and the frequency of
 * pseudo-stable sign runs.
 * ============================================================ */
{
  const stateRoot = join(workRoot, "state-aa");
  const { registry, candidateId } = buildConfiguredRegistry("aa");
  await saveAdaptationRegistry(stateRoot, registry);
  const datasetDir = join(workRoot, "dataset-aa");
  await writeDataset(datasetDir, 200, mulberry32(0xd15d04 ^ 0x5a5a));

  async function aaWindows(
    label: string,
    op: () => Promise<void>,
    windows: number,
    repsPerSide: number
  ): Promise<void> {
    await op(); // warm
    const deltas: number[] = [];
    for (let w = 0; w < windows; w += 1) {
      const aTimes: number[] = [];
      const bTimes: number[] = [];
      // Alternate which side runs first inside each rep pair so systematic
      // position-within-pair bias cancels (a naive A-first design showed a
      // stable ~0.4ms pseudo-delta on this very command -- see report).
      for (let r = 0; r < repsPerSide; r += 1) {
        if (r % 2 === 0) {
          const t0 = performance.now();
          await op();
          aTimes.push(performance.now() - t0);
          const t1 = performance.now();
          await op();
          bTimes.push(performance.now() - t1);
        } else {
          const t0 = performance.now();
          await op();
          bTimes.push(performance.now() - t0);
          const t1 = performance.now();
          await op();
          aTimes.push(performance.now() - t1);
        }
      }
      deltas.push(median(aTimes) - median(bTimes));
    }
    const maxAbsUs = Math.max(...deltas.map((delta) => Math.abs(delta))) * 1e3;
    const minUs = Math.min(...deltas) * 1e3;
    const maxUs = Math.max(...deltas) * 1e3;
    const positives = deltas.filter((delta) => delta > 0).length;
    let streak = 1;
    let bestStreak = 1;
    for (let i = 1; i < deltas.length; i += 1) {
      if (Math.sign(deltas[i]!) === Math.sign(deltas[i - 1]!) && deltas[i]! !== 0) {
        streak += 1;
        bestStreak = Math.max(bestStreak, streak);
      } else {
        streak = 1;
      }
    }
    console.log(
      `S3 A/A [${label}]: ${windows} windows x ${repsPerSide} reps/side -> window-median deltas ${minUs.toFixed(0)}..${maxUs.toFixed(0)}us | max|delta|=${maxAbsUs.toFixed(0)}us | sign+ ${positives}/${windows} | longest same-sign streak=${bestStreak} (H0: 5-window pseudo-stable sign occurs w.p. 6.25%)`
    );
  }

  await aaWindows(
    "eval E=200 configured",
    async () => {
      await evalRoutingPolicy({ stateRoot, candidateId, datasetDir });
    },
    12,
    12
  );
  await aaWindows(
    "save(+fsync) configured",
    async () => {
      await saveAdaptationRegistry(stateRoot, registry);
    },
    12,
    12
  );
}
