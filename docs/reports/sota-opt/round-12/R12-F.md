```ts
/**
 * R12-F pool-size accounting (temporary; NOT a candidate adjudication and NOT
 * committed — no winner this round). Completes the per-pool decomposition of
 * the full-experiment anchor by single-listing the last two line items that
 * earlier rounds carried by subtraction or by a nine-round-old measurement:
 *
 *   pool D  per-call object-construction constant residual — the
 *           "clock/budget/return-spread" line every round since R2-F carried
 *           as "O(field-count), constant residual" without a direct number:
 *           D1 = 2A restore-return constructions (canonicalHaltReason mirror
 *                + 8-field literal; arrays aliased NOT copied — the copies
 *                are R11-F pool B1, already measured at 0.98–1.06 ms).
 *           D2 = 2A applyExperimentClock spreads ({...state, elapsedMs}).
 *           D3 = A assign-site object spreads ({...next, assignments}) plus
 *                the haltOnAssignmentBudget not-halted pass-through check
 *                (array append copies are R10-F pool 1, excluded here).
 *           D4 = A recordOutcome return spreads
 *                ({...next, outcomes, halted, haltReason, guardrailBreaches}).
 *           Elimination mechanisms are all locked (R2-F pool + X1-2 parallel
 *           path + S5-D-3 intermediate-object elision, adjudicated R9-F §0);
 *           this measures the magnitude ceiling only.
 *
 *   pool E  shadowDecisionAt rng replay (production growth profile: assign k
 *           allocates a fresh mulberry32 closure and steps it k+1 times, so
 *           the experiment pays Σ(k+1) = A(A+1)/2 steps + A closure allocs).
 *           The O(1) jump/carried-state mechanisms are S2-F-1 / X1-1 / X0-4
 *           locked; the ledger number (~3 ms) dates from R2-F on an older
 *           VM — this refreshes it on THIS VM. Ceiling only.
 *
 * Fixtures are fixed/deterministic; timing lines are informational. Run:
 *   npx tsx tmp-r12f-pools.ts
 */

function timeMs(fn: () => void, rounds = 7): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

interface MiniAssignment {
  readonly episodeHash: string;
  readonly liveAction: "baseline";
  readonly shadowDecision: "baseline" | "candidate";
  readonly changedLiveAction: false;
}
interface MiniOutcome {
  readonly episodeHash: string;
  readonly utility: number;
  readonly costUsd: number;
  readonly guardrailBreached: boolean;
}
interface MiniState {
  readonly plan: { readonly budget: { readonly maxAssignments: number; readonly maxWallClockMs: number } };
  readonly halted: boolean;
  readonly haltReason: string | undefined;
  readonly assignments: readonly MiniAssignment[];
  readonly outcomes: readonly MiniOutcome[];
  readonly guardrailBreaches: number;
  readonly startedAtMs: number;
  readonly elapsedMs: number;
}

const A = 1000;
const hashes: string[] = [];
for (let i = 0; i < A; i++) hashes.push(`ep_${i.toString(36).padStart(8, "0")}`);
const assignmentsFull: MiniAssignment[] = hashes.map((episodeHash, i) => ({
  episodeHash,
  liveAction: "baseline",
  shadowDecision: i % 2 === 0 ? "candidate" : "baseline",
  changedLiveAction: false,
}));
const outcomesFull: MiniOutcome[] = hashes.map((episodeHash) => ({
  episodeHash,
  utility: 0.5,
  costUsd: 0.01,
  guardrailBreached: false,
}));
const plan = { budget: { maxAssignments: A, maxWallClockMs: 1e12 } };
const baseState: MiniState = {
  plan,
  halted: false,
  haltReason: undefined,
  assignments: assignmentsFull,
  outcomes: outcomesFull,
  guardrailBreaches: 0,
  startedAtMs: 0,
  elapsedMs: 0,
};

/* canonicalHaltReason mirror (production not-halted path: value undefined) */
function canonicalHaltReasonMirror(value: unknown, halted: boolean): string | undefined {
  if (value === undefined || value === null) {
    if (halted) throw new Error("halted experiment is missing haltReason");
    return undefined;
  }
  if (typeof value !== "string" || (value as string).trim() === "") {
    throw new Error("haltReason must be a non-empty string when present");
  }
  return value as string;
}

let sink = 0;

/* pool D — object-construction constant residual (arrays aliased throughout) */
const tPoolD = timeMs(() => {
  sink = 0;
  for (let k = 0; k < A; k++) {
    // assign leg: restore-return (D1) + clock spread (D2) + assign spread (D3)
    const haltReason = canonicalHaltReasonMirror(undefined, false);
    const restored: MiniState = {
      plan: baseState.plan,
      halted: baseState.halted,
      haltReason,
      assignments: baseState.assignments,
      outcomes: baseState.outcomes,
      guardrailBreaches: baseState.guardrailBreaches,
      startedAtMs: baseState.startedAtMs,
      elapsedMs: baseState.elapsedMs,
    };
    const clocked: MiniState = { ...restored, elapsedMs: k };
    const appended: MiniState = { ...clocked, assignments: assignmentsFull };
    // haltOnAssignmentBudget not-halted pass-through
    const budgeted =
      appended.assignments.length >= appended.plan.budget.maxAssignments && !appended.halted
        ? { ...appended, halted: true, haltReason: "budget-exhausted" }
        : appended;
    sink += budgeted.elapsedMs;

    // record leg: restore-return (D1) + clock spread (D2) + record spread (D4)
    const haltReason2 = canonicalHaltReasonMirror(undefined, false);
    const restored2: MiniState = {
      plan: baseState.plan,
      halted: baseState.halted,
      haltReason: haltReason2,
      assignments: baseState.assignments,
      outcomes: baseState.outcomes,
      guardrailBreaches: baseState.guardrailBreaches,
      startedAtMs: baseState.startedAtMs,
      elapsedMs: baseState.elapsedMs,
    };
    const clocked2: MiniState = { ...restored2, elapsedMs: k };
    const recorded: MiniState = {
      ...clocked2,
      outcomes: outcomesFull,
      halted: false,
      haltReason: undefined,
      guardrailBreaches: clocked2.guardrailBreaches,
    };
    sink += recorded.guardrailBreaches;
  }
});
/* floor: same control flow, no object construction */
const tPoolDFloor = timeMs(() => {
  sink = 0;
  for (let k = 0; k < A; k++) {
    const haltReason = canonicalHaltReasonMirror(undefined, false);
    sink += haltReason === undefined ? k : 0;
    if (assignmentsFull.length >= plan.budget.maxAssignments && baseState.halted) sink += 1;
    const haltReason2 = canonicalHaltReasonMirror(undefined, false);
    sink += haltReason2 === undefined ? 0 : 1;
  }
});
process.stdout.write(
  `pool D (A=${A}, 6A object constructions): full=${(tPoolD * 1000).toFixed(0)}µs ` +
    `control-flow floor=${(tPoolDFloor * 1000).toFixed(0)}µs ` +
    `(ceiling ${((tPoolD - tPoolDFloor) * 1000).toFixed(0)}µs per experiment)\n`
);

/* pool E — shadowDecisionAt rng replay (production growth profile) */
function createSeededRngMirror(seed: number): () => number {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
const SEED = 42;
let rngSink = 0;
const tRngReplay = timeMs(() => {
  rngSink = 0;
  for (let k = 0; k < A; k++) {
    // landed shadowDecisionAt(seed, k): fresh closure, k+1 steps
    const rng = createSeededRngMirror(SEED);
    let value = 0;
    for (let i = 0; i <= k; i++) value = rng();
    rngSink += value < 0.5 ? 1 : 0;
  }
});
const tRngCarried = timeMs(() => {
  // O(1)-per-assign floor (carried state — mechanism locked by S2-F-1/X1-1/X0-4)
  rngSink = 0;
  const rng = createSeededRngMirror(SEED);
  for (let k = 0; k < A; k++) {
    const value = rng();
    rngSink += value < 0.5 ? 1 : 0;
  }
});
process.stdout.write(
  `pool E (A=${A}, Σ=A(A+1)/2 steps): rng replay=${(tRngReplay * 1000).toFixed(0)}µs ` +
    `carried-state floor=${(tRngCarried * 1000).toFixed(0)}µs ` +
    `(ceiling ${((tRngReplay - tRngCarried) * 1000).toFixed(0)}µs per experiment)\n`
);
process.stdout.write(`sinks: ${sink} ${rngSink}\n`);
```
