## 4. 关键裁决细节

### 4.1 S10-B-1：死计算真实存在，量级差一个数量级

活性论证完备：`complexityOf` 的分支序是 highRisk/deploy → long → role →
family → 默认，`long` 仅在第一分支不取时被读；`text.match(/\n/g)` 的结果是
局部值、分类正则无 /g 状态，惰性化在语言语义上不重排任何可观察效应。12000 组
fuzz（replica 对真实 `analyzeTask` 逐字节保真先行，再证 lazy ≡ eager）与
确定性死计算份额计数（4525/12000，三次逐位一致）双证等价。fuzz 语料的 37.7%
HIGH 份额是**含 30% contractRisk 注入的对抗混合**——真实 replay 语料的命中率
更低，实测省 40.2–57.9µs/批；即便 contractRisk=true 使 100% 任务命中，上界
也只有 109.4–124.7µs/批。两个口径都落在 S4-B-4（143–215µs）已裁决噪声带
**之下**，且被 analyzeTask 全量 2.18–2.24ms/eval 的天花板支配（R1-B §4.1
的 6% 捆绑都不达线，本候选 <1.5%）。等价证据已备，满足结构性重开条件时可
直接引用。封死。

### 4.2 S10-B-2：穷举位同一是最强证据等级，输在 ns 与护栏

本战役首个**穷举域**候选：`coldStartRoutingScore` 的输入域恰是
3 complexity × 2 preferred = 6 点，`Object.is` 位同一 + `<1` 全部成立——
无需 fuzz 的完备证明（0.9+0.04 = 0.9400000000000001 是域上最大值）。但：

1. **量级**：8.27–9.45ns/call（每 route 一次；microbench 的 5× 相对差含
   跨模块调用未内联成分，绝对差取保守口径），N=2000 批上界 ~17–19µs——
   S9-B-2（2.9–22.3ns/route）同带之下的深度噪声。
2. **护栏**：`Math.min(1, ·)` 是 `ConfidenceScore` 类型（≤1 语义）在本站点
   的唯一运行时强制。删除后该不变式只存在于「常量表凑巧 <0.96」这一隐式
   事实里——未来任何人把 LOW base 上调到 0.97 都会无声产出 >1 的
   confidence（事件契约字段）。与 S5-B-1（谓词是分类意图唯一显式记载）、
   S9-D-3（零成本防御纵深保留）同判：删卫不值 ns。

### 4.3 S10-B-3：教训正向成立，规模照旧封死

S1-A-4/S1-B-6/S3-B-5/S8-B-2 反例链证明「微阵列上加索引更慢」；本候选是链上
第一次**反向**尝试——把既有的 Set 分配拆回线性扫描。方向确认为正
（M=2 173→72ns、M=10 707→434ns，三次同向），教训的对称形式成立。但站点
规模与 S9-B-1（同函数上一轮的单探针候选，33–69ns）完全同类：每
createModelRouter 一次＝每 assignTasks 批一次，createModelRouter M=2 全程
598–713ns（S2-B-2/S7-B-6/S9-B-1 之后第四次复核同带）。占 N=2000 批
<0.01%。封死；等价与方向证据已备。

/* ============================================================

 * classification patterns; the match(/\n/g) result is local), so lazification

 * ============================================================ */
const HIGH_RISK_RE_R =
  /\b(deploy(?:ing|ment|s)?|production|prod|credentials?|secrets?|privileged?|rm\s+-[a-z]*|drop\s+(table|database)|privilege\s+escalat\w*)\b/i;
const REVIEW_RE_R = /\b(review|audit|critique|nits?)\b/i;
const TEST_RE_R = /\b(tests?|spec|coverage|qa|verify|validation)\b/i;
const PLAN_RE_R = /\b(plan|decompos|roadmap|break down|design)\b/i;
const RESEARCH_RE_R = /\b(survey|research|investigat|scout|explor|compar)\b/i;
const REFACTOR_RE_R = /\b(refactor|cleanup|rename|extract)\b/i;
const IMPLEMENT_RE_R = /\b(implement|add |fix |integrate|migrate|write |build )\b/i;
const ROLE_FAMILY_R: Record<AgentRole, TaskFamily> = {
  worker: "edit",
  scout: "research",
  planner: "plan",
  implementer: "edit",
  reviewer: "review",
  tester: "test",
  debugger: "edit"
};
function familyOfR(text: string, role: AgentRole): TaskFamily {
  if (HIGH_RISK_RE_R.test(text) && /\b(deploy|production|prod\b)\b/i.test(text)) return "deploy";
  if (PLAN_RE_R.test(text) || role === "planner") return "plan";
  if (RESEARCH_RE_R.test(text) || role === "scout") return "research";
  if (TEST_RE_R.test(text) || role === "tester") return "test";
  if (REVIEW_RE_R.test(text) || role === "reviewer") return "review";
  if (REFACTOR_RE_R.test(text)) return "refactor";
  if (IMPLEMENT_RE_R.test(text) || role === "implementer" || role === "worker") return "edit";
  return ROLE_FAMILY_R[role] ?? "unknown";
}
function complexityTail(role: AgentRole, family: TaskFamily, long: boolean): TaskComplexity {
  if (long) return "MEDIUM";
  if (role === "scout" || role === "tester") return "LOW";
  if (role === "planner" || role === "debugger" || role === "reviewer") return "MEDIUM";
  if (family === "plan" || family === "research") return "MEDIUM";
  return "MEDIUM";
}

/* ============================================================
 * S10-B-3: validateConfig per-model duplicate-role check — new Set(roles)
 * allocation replaced by an indexOf linear scan (the S1-A-4/S1-B-6 tiny-array
 * lesson applied proactively). Fault order (dup-id, roles-empty, dup-role,
 * unknown-role) and messages preserved by construction; adjudicated over a
 * fault-mix fuzz. Once per createModelRouter (per assignTasks batch).
 * ============================================================ */
const SIM_ROLES = ["actor", "critic", "router", "judge", "tool", "human"] as const;
function validateConfigReplica(models: readonly CatalogModel[], linearRoleDup: boolean): void {
  const ids = new Set<string>();
  for (const model of models) {
    if (ids.has(model.id)) {
      throw new DomainValidationError("ModelRouter model ids must be unique and non-empty");
    }
    ids.add(model.id);
    if (!Array.isArray(model.roles) || model.roles.length === 0) {
      throw new DomainValidationError(`ModelRouter model ${model.id} must declare roles`);
    }
    if (linearRoleDup) {
      const roles = model.roles;
      for (let i = 1; i < roles.length; i += 1) {
        if (roles.indexOf(roles[i]!) !== i) {
          throw new DomainValidationError(`ModelRouter model ${model.id} declares duplicate roles`);
        }
      }
    } else if (new Set(model.roles).size !== model.roles.length) {
      throw new DomainValidationError(`ModelRouter model ${model.id} declares duplicate roles`);
    }
    const unknownRole = model.roles.find((role) => !(SIM_ROLES as readonly string[]).includes(role));
    if (unknownRole !== undefined) {
      throw new DomainValidationError(`ModelRouter model ${model.id} declares unknown role: ${String(unknownRole)}`);
    }
  }
}
