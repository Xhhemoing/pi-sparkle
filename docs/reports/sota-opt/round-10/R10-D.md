## 3. 关键裁决细节

### S10-D-1：mkdir 消除的健全形态无处安放

十轮以来 save 分解（S8-D 域）锚过 stringify/fsync/rename，但
`mkdir` 这一行从未获 ID。本轮补上：它每存执行、目录首存后恒存在，
表面是纯重复。但消除的两个形态各自独立死刑——（a）天真形态（直接
删行）在新 stateRoot 首存路径确定性 ENOENT：`open(tempPath, "wx")`
的父目录不存在。这不是构造边缘态——`loadAdaptationRegistryOrNew`
之后首存正是 auto-loop / from-episode 空仓初始化的**公开主流程**。
（b）健全形态必须记住"哪些目录已确保存在"，跨调用状态 = X1-1
模块级隐藏缓存禁族的教科书形态（外部 `rm -rf` 状态目录后缓存即
陈旧，save 从 fail-closed 变 fail-crash-later）。首存后按注册表
实例携带标志？——save 是自由函数、registry 不持有 stateRoot，
加字段即公开契约变更（X0-4 类）。收益端：mkdir(既存目录) 实测
24.7–26.1µs，占配置态整存 1.3–1.7%；整存 A/B 三次运行 delta
−300/+272/+301µs——**fsync 方差比候选收益大一个量级**，命令级
符号不稳即验收标准第 3 条的直接否决。

### S10-D-2：命中率恒为 0 的等价快路径

第十遍新透镜问的是："eval 的双 assignTasks 在什么输入下退化？"
答案：candidateContent === baselineContent 时两次调用输入逐位相同、
`assignTasks` 为纯确定函数，第二次是纯重复（命中态省 319–463µs，
等价三 seeds 逐位验证含 rerunHash 序列化路径）。但生产反向证据
完备：路由政策候选只有两个生产者，**双双结构性拒绝该输入**——
auto-loop.ts:178 只在 `parent.contentHash !== contentHash` 时
createCandidate；from-episode.ts:123 在相等时早退。hash32 是内容
函数（同内容 ⇒ 同哈希），故哈希不等 ⇒ 内容必不等；候选的
parentVersionId 创建后不可变，eval 时重取的 parent 即创建时的
parent。于是快路径在生产可达状态空间内**永不触发**，其真实期望
收益精确为 0，而唯一真实路径（内容不等）反要为每次 eval 付一次
9.2KB 字符串比较（实测零附近噪声 −18.6~+8.0µs，字符串比较在长度
相同前缀不同处早退故廉价，但方向为负担）。哈希门形态的补充裁决:
在 eval-routing 内部哈希门恰好健全（两内容经同一 contentsByHash
槽位解析，哈希相等 ⇒ 取同一字符串），但把这个"恰好"写进代码
等于把 S7-D-2 已裁决的 hash32 碰撞语义搬进新站点做隐式前提——
字符串门是唯一规范健全形态，而它已被命中率 0 否决。

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/* ============================================================
 * S10-D-1: saveAdaptationRegistry per-save mkdir(recursive) elision.
 * saveCurrent below is a VERBATIM replica of saveAdaptationRegistry;
 * saveNoMkdir drops the mkdir call. On an existing dir the bytes must be
 * identical; on a FRESH stateRoot the no-mkdir variant must fail (ENOENT)
 * -> the sound form needs cross-call "dir exists" state = X1-1 family.
 * ============================================================ */
async function saveCurrent(stateRoot: string, registry: ResourceRegistry): Promise<void> {
  const path = adaptationRegistryPath(stateRoot);
  const serialized = `${JSON.stringify(registry.snapshot(), null, 2)}\n`;
  await mkdir(dirname(path), { recursive: true });
  const tempPath = `${path}.${process.pid}.${randomUUID()}.tmp`;
  const handle = await open(tempPath, "wx");
  try {
    await handle.writeFile(serialized, "utf8");
    await handle.sync();
  } finally {
    await handle.close();
  }
  try {
    await rename(tempPath, path);
  } catch (error) {
    await rm(tempPath, { force: true });
    throw error;
  }
}

/* ============================================================
 * S10-D-2: replayAssignments degenerate-equality fast path.
 * replayCurrent is a VERBATIM replica of the current replayAssignments +
 * catalogCost; replayFast adds: if candidate content === baseline content,
 * run assignTasks ONCE and reuse the result for both sides.
 * Production hit rate is zero (auto-loop.ts:178 creates a candidate only
 * when parent.contentHash !== contentHash; from-episode.ts:123 returns
 * early on equality), so even a proven-equivalent fast path has no
 * real-scale gain. Cheap hash-equality gating additionally sits in the
 * S7-D-2 hash32-collision domain at any other site; inside evalRoutingPolicy
 * both contents resolve through the same contentsByHash entry, which is why
 * the string-equality form is the only canonical sound one.
 * ============================================================ */
interface EpisodeShape {
  readonly episodeHash: string;
  readonly taskId: string;
  readonly role: AgentRole;
  readonly objective: string;
  readonly taskFamily?: string | undefined;
  readonly taskSuccess?: "PASS" | "FAIL" | undefined;
  readonly originalWorkspace: string;
}
interface ReplayActionShape {
  readonly episodeHash: string;
  readonly taskFamily: string;
  readonly taskSuccess: "PASS" | "FAIL" | "UNOBSERVED";
  readonly baselineModel: string;
  readonly candidateModel: string;
  readonly baselineCostUsd: number;
  readonly candidateCostUsd: number;
}
function catalogCostReplica(catalog: ModelRouterConfig, modelId: string): number {
  const model = catalog.models.find((entry) => entry.id === modelId);
  if (model === undefined) {
    throw new DomainValidationError(`selected model ${modelId} is not in the catalog`);
  }
  return model.estimatedCostUsd;
}
function replayCurrent(
  episodes: readonly EpisodeShape[],
  catalog: ModelRouterConfig,
  baselinePolicy: LearnedRoutingPolicy,
  candidatePolicy: LearnedRoutingPolicy
): readonly ReplayActionShape[] {
  const tasks = episodes.map((episode) => ({
    taskId: parseTaskId(episode.taskId),
    role: episode.role,
    objective: episode.objective
  }));
  const baseline = assignTasks({ catalog, tasks, learned: baselinePolicy });
  const candidate = assignTasks({ catalog, tasks, learned: candidatePolicy });
  return episodes.map((episode, index) => {
    const baselineAssignment = baseline[index];
    const candidateAssignment = candidate[index];
    if (baselineAssignment === undefined || candidateAssignment === undefined) {
      throw new DomainValidationError(`missing assignment for ${episode.episodeHash}`);
    }
    return {
      episodeHash: episode.episodeHash,
      taskFamily: episode.taskFamily ?? baselineAssignment.analysis.family,
      taskSuccess: episode.taskSuccess ?? "UNOBSERVED",
      baselineModel: baselineAssignment.decision.model,
      candidateModel: candidateAssignment.decision.model,
      baselineCostUsd: catalogCostReplica(catalog, baselineAssignment.decision.model),
      candidateCostUsd: catalogCostReplica(catalog, candidateAssignment.decision.model)
    };
  });
}
function replayFast(
  episodes: readonly EpisodeShape[],
  catalog: ModelRouterConfig,
  baselinePolicy: LearnedRoutingPolicy,
  candidatePolicy: LearnedRoutingPolicy,
  baselineContent: string,
  candidateContent: string
): readonly ReplayActionShape[] {
  const tasks = episodes.map((episode) => ({
    taskId: parseTaskId(episode.taskId),
    role: episode.role,
    objective: episode.objective
  }));
  const baseline = assignTasks({ catalog, tasks, learned: baselinePolicy });
  const candidate =
    baselineContent === candidateContent ? baseline : assignTasks({ catalog, tasks, learned: candidatePolicy });
  return episodes.map((episode, index) => {
    const baselineAssignment = baseline[index];
    const candidateAssignment = candidate[index];
    if (baselineAssignment === undefined || candidateAssignment === undefined) {
      throw new DomainValidationError(`missing assignment for ${episode.episodeHash}`);
    }
    return {
      episodeHash: episode.episodeHash,
      taskFamily: episode.taskFamily ?? baselineAssignment.analysis.family,
      taskSuccess: episode.taskSuccess ?? "UNOBSERVED",
      baselineModel: baselineAssignment.decision.model,
      candidateModel: candidateAssignment.decision.model,
      baselineCostUsd: catalogCostReplica(catalog, baselineAssignment.decision.model),
      candidateCostUsd: catalogCostReplica(catalog, candidateAssignment.decision.model)
    };
  });
}
const REPLAY_CONFIG: ComparisonReportConfig = {
  ...DEFAULT_COMPARISON_REPORT_CONFIG,
  evidenceClass: "simulation"
};
function pairedRecordsReplica(
  episodes: readonly EpisodeShape[],
  actions: readonly ReplayActionShape[]
): PairedEvaluationRecord[] {
  const records: PairedEvaluationRecord[] = [];
  for (const [index, episode] of episodes.entries()) {
    if (episode.taskSuccess !== "PASS" && episode.taskSuccess !== "FAIL") {
      continue;
    }
    const action = actions[index];
    if (action === undefined) {
      throw new DomainValidationError(`missing replay action for ${episode.episodeHash}`);
    }
    const utility = episode.taskSuccess === "PASS" ? 1 : 0;
    records.push({
      episodeHash: episode.episodeHash,
      taskFamily: episode.taskFamily ?? action.taskFamily,
      baselineUtility: utility,
      candidateUtility: utility,
      baselineCostUsd: action.baselineCostUsd,
      candidateCostUsd: action.candidateCostUsd
    });
  }
  return records;
}
function gatedReplica(records: readonly PairedEvaluationRecord[]): ComparisonReport {
  return gatedComparisonReport({ records, claims: [], config: REPLAY_CONFIG, difficultyTier: "replay" });
}

  }));
}
