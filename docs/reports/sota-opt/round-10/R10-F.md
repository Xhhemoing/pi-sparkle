## 附录 A：池测量脚本（完整，可复现；无赢家按纪律不入库）

保存为 `tmp-r10f-pools.ts` 后 `npx tsx tmp-r10f-pools.ts` 运行。这不是候选裁决仿真（本轮无候选），是 §1.3 两项池的天花板单列测量。

```ts
/**
 * R10-F pool-size accounting (temporary; NOT a candidate adjudication and NOT
 * committed — no winner this round). Isolates two residual per-experiment
 * pools that earlier rounds carried inside aggregate accounting, to confirm
 * neither reaches the tens-of-ms landing line:
 *
 *   pool 1  growth-append copies on the assign / recordOutcome paths
 *           ([...next.assignments, assignment] and [...next.outcomes, o]):
 *           sum over A calls = A^2/2 element writes each. Elision is covered
 *           by X4-2 / S1-F-3 (append-copy -> mutable push, readonly public
 *           contract) plus X1-2 (a runner-private parallel path would be
 *           needed to exploit restore-fresh arrays); this measures the
 *           magnitude ceiling only.
 *
 *   pool 2  accumulatedCostUsd(outcomes) full re-sum per recordOutcome
 *           (S1-F-8 locked: incrementalization = hidden state / public state
 *           shape change). Sum over A calls = A^2/2 float adds + missing
 *           checks. Magnitude ceiling only.
 */

function timeMs(fn: () => void, rounds = 7): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

interface MiniAssignment {
  readonly episodeHash: string;
  readonly liveAction: "baseline";
  readonly shadowDecision: "baseline" | "candidate";
  readonly changedLiveAction: false;
}
interface MiniOutcome {
  readonly episodeHash: string;
  readonly utility: number;
  readonly costUsd: number;
  readonly guardrailBreached: boolean;
  readonly missing?: boolean | undefined;
}

const A = 1000;
const hashes: string[] = [];
for (let i = 0; i < A; i++) hashes.push(`ep_${i.toString(36).padStart(8, "0")}`);

/* pool 1: growth-append copies (assignments + outcomes shapes) */
let sink = 0;
const tAppendCopies = timeMs(() => {
  let assignments: readonly MiniAssignment[] = [];
  let outcomes: readonly MiniOutcome[] = [];
  for (let k = 0; k < A; k++) {
    const a: MiniAssignment = {
      episodeHash: hashes[k] as string,
      liveAction: "baseline",
      shadowDecision: k % 2 === 0 ? "candidate" : "baseline",
      changedLiveAction: false,
    };
    assignments = [...assignments, a];
    const o: MiniOutcome = {
      episodeHash: hashes[k] as string,
      utility: 0.5,
      costUsd: 0.01,
      guardrailBreached: false,
    };
    outcomes = [...outcomes, o];
  }
  sink += assignments.length + outcomes.length;
});
const tMutablePush = timeMs(() => {
  const assignments: MiniAssignment[] = [];
  const outcomes: MiniOutcome[] = [];
  for (let k = 0; k < A; k++) {
    assignments.push({
      episodeHash: hashes[k] as string,
      liveAction: "baseline",
      shadowDecision: k % 2 === 0 ? "candidate" : "baseline",
      changedLiveAction: false,
    });
    outcomes.push({
      episodeHash: hashes[k] as string,
      utility: 0.5,
      costUsd: 0.01,
      guardrailBreached: false,
    });
  }
  sink += assignments.length + outcomes.length;
});

/* pool 2: accumulatedCostUsd full re-sum per recordOutcome */
const outcomesFull: MiniOutcome[] = hashes.map((episodeHash) => ({
  episodeHash,
  utility: 0.5,
  costUsd: 0.01,
  guardrailBreached: false,
}));
let costSink = 0;
const tCostResum = timeMs(() => {
  costSink = 0;
  for (let k = 1; k <= A; k++) {
    let total = 0;
    for (let i = 0; i < k; i++) {
      const item = outcomesFull[i] as MiniOutcome;
      if (item.missing === true) continue;
      total += item.costUsd;
    }
    costSink += total;
  }
});
const tCostIncr = timeMs(() => {
  costSink = 0;
  let total = 0;
  for (let k = 0; k < A; k++) {
    const item = outcomesFull[k] as MiniOutcome;
    if (item.missing !== true) total += item.costUsd;
    costSink += total;
  }
});

process.stdout.write(
  `pool 1 (A=${A}, both stations): growth-append copies=${(tAppendCopies * 1000).toFixed(0)}µs ` +
    `mutable-push floor=${(tMutablePush * 1000).toFixed(0)}µs ` +
    `(ceiling ${((tAppendCopies - tMutablePush) * 1000).toFixed(0)}µs per experiment)\n`
);
process.stdout.write(
  `pool 2 (A=${A}): accumulatedCostUsd re-sum=${(tCostResum * 1000).toFixed(0)}µs ` +
    `incremental floor=${(tCostIncr * 1000).toFixed(0)}µs ` +
    `(ceiling ${((tCostResum - tCostIncr) * 1000).toFixed(0)}µs per experiment)\n`
);
process.stdout.write(`sinks: ${sink} ${costSink.toFixed(2)}\n`);
```
