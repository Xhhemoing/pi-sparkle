## 3. 关键裁决细节

### S10-A-1 的定位（第十遍还剩什么）

九遍搜查分别点名了输入侧（S1-A-2 重复 prescore、S8-A-3 死输入字段）、
输出侧（S9-A-1 死输出尾）、扫描侧（S1-A-1/S1-A-9/S8-A-1）、human 管道
（S7-A-1 常量折叠），但**没人点名门控结局本身的值域坍缩**：S7-A-1 证明了
进 evaluateGates 的 human 侧输入是常量，却仍把 evaluateGates 当黑盒保留；
本轮引理 C 追加了**得分下界**（PASSED 时 P ≥ 2/3，源于 from-child 构造
不变量：completed = required 拷贝、retained = 全量、escaped ≡ false、
stalledTurns ≡ 0、claims 至多一条）与 **openMinors 恒空**两个事实，使
evaluateGates 的软阈值比较（0.6667 > 0.55 恒真）与 minor 升级检查
（空集恒假）双双坍缩——门控结局成为 verification.kind 的纯函数。fuzz
中 min PASSED P **恰好打到 0.6667 下界**，证明引理是紧的。

淘汰三重落锤与 S9-A-1 完全同构：(1) **落地面**——利用坍缩需调用点特化
（X1-2 平行入口）或公开旗标（S5-A-3/S6-A-3/S7-A-1/S9-A-1 同结构第五例）；
(2) **量级**——孤立折叠 18–21 ns/turn ⇒ ~0.1 µs/run，比落地线低约五个
量级，甚至比 S9-A-1（~3.5–4.5 µs/run）还低一个量级半；e2e 符号翻转确认
其在端到端噪声内不可测；(3) **支配**——被 §1 整片预算收口覆盖。

- **矩阵收紧**：门控结局 × 合同 2×2 满矩阵之外，soft / minor-escalated /
  wait_user 行被证明生产不可达（经 applyChildThreeLine）——未来提案若以
  这些行的成本为前提可直接否决，与 R9-A 的 supervisor no-op /

- **RUN_WAITING_FOR_USER 与 createMessageId 在本切片生产路径上零流量**
  ——gate-apply 的 wait_user 分支是 supervisor 侧带 assessment 的
  settle 才可达的面，而该面生产为 no-op（R9-A 事实）。
- 与 S7-A-4（加权组合重排在 H=0.35 生产常量上发散）对照：引理 C 依赖的

  在**真代码**上验证而非纸面推导——下界 2/3 由 `P = qualitySum /
  applicableWeight`（quality×coverage 的约分形式）给出，applicable ∈
  {3,4,5}、PASS 数 ≥ 2 且仅 progress 可失分。

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function derivedCVWCReplica(input: TrackingTurnInput): boolean {
  if (input.gateFacts?.claimedVerificationWithoutChecks !== undefined) {
    return input.gateFacts.claimedVerificationWithoutChecks;

  const completed = input.prescoreInput.completedChecks;
  const requiredCheckGap =
    required.length > 0 && !required.every((id) => completed.includes(id));
  return input.prescoreInput.claims.some(isSuccessClaim) && requiredCheckGap;
}

function candidateTurnBinaryGate(input: TrackingTurnInput): TrackingTurnResult {
  const openMinors = mergeOpenMinors(input.window.previous?.openMinors ?? [], input.window.openMinors);
  const lightMinorCount =
    input.prescoreInput.lightMinorCount ??
    openMinors.filter((item) => item.status === "verified-true").length;
  const prescore = computePrescore({
    ...input.prescoreInput,
    lightMinorCount
  });
  const userText = input.humanInput.userText ?? input.window.userText;
  const human = extractHumanScore({
    ...(input.humanInput.list !== undefined ? { list: input.humanInput.list } : {}),
    ...(userText !== undefined ? { userText } : {})
  });
  const obviousProblem = hasObviousHumanProblem(human);
  const score = combineScore({ P: prescore.P, human, obviousProblem });

  // S10-A-1 FOLD: binary gate dispatch (Lemma C).
  // deterministicFail === (verification.kind === "FAILED") on this path.
  let gate: GateDecision;
  if (input.gateFacts?.deterministicFail === true) {
    const codes: AnomalyCode[] = derivedCVWCReplica(input)
      ? ["deterministic-fail", "claimed-verification-without-checks"]
      : ["deterministic-fail"];
    gate = {
      kind: "hard",
      codes,
      wakeAnalysis: true,
      expandDetail: true,
      askUser: false,
      openMinors
    };
  } else {
    gate = {
      kind: "none",
      codes: [],
      wakeAnalysis: false,
      expandDetail: false,
      askUser: false,
      openMinors
    };
  }

  let readersInvoked: TrackingTurnResult["readersInvoked"] = {
    toolBodies: false,
    chainOfThought: false
  };
  let toolBodies: readonly string[] | undefined;
  if (gate.expandDetail && input.readers?.readToolBodies !== undefined) {
    toolBodies = input.readers.readToolBodies();
    readersInvoked = { toolBodies: true, chainOfThought: false };
  }

  const anomalyCodes = [...gate.codes];

  const rolled = rollSummary({
    window: { ...input.window, openMinors },
    prescore: prescore.P,
    human,
    score,
    anomalyCodes,
    evidenceRefs: collectEvidenceReplica(input.window),
    openMinors,
    ...(input.maxItems !== undefined ? { maxItems: input.maxItems } : {})
  });

  let summary = rolled.summary;
  if (summary.failClosed) {
    const codes: AnomalyCode[] = uniqueCodesReplica([...summary.anomalyCodes, "mandatory-omission"]);
    summary = { ...summary, anomalyCodes: codes };
    gate = { ...gate, askUser: true, codes };
  }

  let packet: AnomalyPacket | undefined;
  if (gate.wakeAnalysis) {
    const windowDetail: AnomalyPacketWindow = {
      contextFacts: input.window.contextFacts,
      toolSituations: input.window.toolSituations,
      ...(input.window.userText !== undefined
        ? { userText: input.window.userText, userTextTrust: "UNTRUSTED_TEXT" as const }
        : {}),
      ...(input.window.aiText !== undefined ? { aiText: input.window.aiText } : {}),
      ...(toolBodies !== undefined ? { toolBodies } : {})
    };
    packet = {
      summary,
      window: windowDetail,
      P: prescore.P,
      H: humanScoreValue(human),
      score,
      gate: gate.codes[0] ?? "soft-threshold",
      evidenceRefs: summary.evidenceRefs
    };
  }

  return {
    summary,
    P: prescore.P,
    human,
    score,
    gate,
    ...(packet !== undefined ? { packet } : {}),
    readersInvoked
  };
}

function candidateAssess(input: {
  readonly observation: ChildObservation;
  readonly episodeId: string;
  readonly runId: string;
}): { readonly apply: false } | { readonly apply: true; readonly assessment: TrackingAssessment } {
  const verification = input.observation.verification;
  if (verification === undefined || (verification.kind !== "PASSED" && verification.kind !== "FAILED")) {
    return { apply: false };
  }
  const prescoreInput = prescoreInputFromObservation(input.observation);
  const prescore = computePrescore(prescoreInput);
  const hasHardPassOrFail = prescore.dimensions.some(
    (dimension) => dimension.hardRelated && (dimension.outcome === "PASS" || dimension.outcome === "FAIL")
  );
  if (
    !shouldApplyThreeLine({
      verificationKind: verification.kind,
      coverage: prescore.coverage,
      hasHardPassOrFail
    })
  ) {
    return { apply: false };
  }
  const failRefs = evidenceRefsOfReplica(input.observation);
  if (prescore.dimensions.some((dimension) => dimension.outcome === "FAIL") && failRefs.length === 0) {
    return { apply: false };
  }
  const window = {
    contextFacts: [`role ${input.observation.role}`, `task ${input.observation.taskId}`],
    toolSituations: prescoreInput.toolSituations,
    constraints: input.observation.constraints,
    unresolvedDecisions: [],
    confirmedDecisions: [],
    openMinors: []
  };
  const turn = candidateTurnBinaryGate({
    window,
    prescoreInput,
    humanInput: {},
    gateFacts: { deterministicFail: verification.kind === "FAILED" }
  });
  return {
    apply: true,
    assessment: {
      schemaVersion: 1,
      episodeId: input.episodeId,
      runId: input.runId,
      turnId: input.observation.taskId,
      prescore: prescore.displayPrescore,
      quality: prescore.quality,
      coverage: prescore.coverage,
      human: turn.human,
      score: turn.score,
      dimensions: prescore.dimensions.map((dimension) => {
        const verdict = dimension.outcome === "ABSTAIN" ? "UNOBSERVED" : dimension.outcome;
        if (verdict === "FAIL") {
          return { id: dimension.id, verdict, evidenceRefs: failRefs };
        }
        return { id: dimension.id, verdict };
      }),
      gate: turn.gate,
      evidenceRefs: failRefs
    }
  };
}
