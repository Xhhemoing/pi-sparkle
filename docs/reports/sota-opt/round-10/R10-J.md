    for (let i = 0; i < iters; i++) {
      seedFeedback(stateRoot, true); // untimed re-seed
      const t0 = hr();
      const out = await cascadeFeedbackTombstones(stateRoot, "eps_hit" as EpisodeId);
      const t1 = hr();
      if (out.length !== 20) throw new Error(`expected 20 cascaded, got ${out.length}`);
      total += t1 - t0;
    }
    matchResults.push(Number(total) / 1000 / iters);
    rmSync(stateRoot, { recursive: true, force: true });
  }
  console.log(`deletion cascade N=200 20-hit MATCH: ${matchResults.map(fmt).join(" / ")} us/cascade`);

  // no-match cell: 0 hits -> two reads, no rewrite
  const nomatchResults: number[] = [];
  for (let run = 0; run < 3; run++) {
    const stateRoot = mkdtempSync(join(tmpdir(), "r10j-casc-"));
    seedFeedback(stateRoot, true);
    const iters = 200;
    for (let i = 0; i < 20; i++) await cascadeFeedbackTombstones(stateRoot, "eps_none" as EpisodeId);

    rmSync(stateRoot, { recursive: true, force: true });
  }
  console.log(`deletion cascade N=200 NO-MATCH: ${nomatchResults.map(fmt).join(" / ")} us/cascade`);
}
