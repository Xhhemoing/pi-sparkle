/** 200 records, 20 hits; hit ids either distinct or ONE duplicated id. */
function seedStore(stateRoot: string, duplicateHitId: boolean): void {
  const log = feedbackLogPath(stateRoot);
  mkdirSync(dirname(log), { recursive: true });
  const lines: string[] = [];
  for (let i = 0; i < 200; i++) {
    const hit = i % 10 === 0;
    const id = hit && duplicateHitId ? "fb-dup" : `fb-${i}`;
    lines.push(makeRecord(id, hit ? "eps_hit" : `eps_other${i % 7}`));
  }
  writeFileSync(log, lines.join("\n") + "\n", "utf8");
  writeFileSync(
    feedbackTombstonesPath(stateRoot),
    JSON.stringify(["fb-pre-a", "fb-pre-b", "fb-pre-c", "fb-pre-d", "fb-pre-e"], null, 2) + "\n",
    "utf8"
  );
}

async function cellDeleteRedelivery(): Promise<void> {
  console.log("[A1 deleteEpisodeRecords redelivery dose K=1,2,4,8 + state fixpoint]");
  const stateRoot = mkdtempSync(join(tmpdir(), "r19j-redel-"));
  for (const K of [1, 2, 4, 8]) {
    const iters = 10;
    let firstTotal = 0;
    let noopTotal = 0;
    let noopCount = 0;
    for (let it = 0; it < iters; it++) {
      seedStore(stateRoot, false);
      seedEpisodeFiles(stateRoot);
      let afterFirstRecords = "";
      let afterFirstTombs = "";
      for (let k = 1; k <= K; k++) {
        const t0 = hr();
        const out = await deleteEpisodeRecords(stateRoot, "eps_hit" as EpisodeId);
        const t1 = hr();
        if (k === 1) {
          if (out.cascadedFeedbackTombstones.length !== 20 || out.removedPaths.length !== 2) {
            throw new Error("unexpected first-delivery outcome");
          }
          firstTotal += usOf(t0, t1);
          afterFirstRecords = readFileSync(feedbackLogPath(stateRoot), "utf8");
          afterFirstTombs = readFileSync(feedbackTombstonesPath(stateRoot), "utf8");
        } else {
          // Redelivery: both episode files already gone; cascade finds the
          // 20 already-stripped records again (episodeId shell is retained
          // for audit), re-asserts the same tombstones.
          if (out.removedPaths.length !== 0) throw new Error("redelivery must remove nothing");
          noopTotal += usOf(t0, t1);
          noopCount += 1;
        }
      }
      const recordsNow = readFileSync(feedbackLogPath(stateRoot), "utf8");
      const tombsNow = readFileSync(feedbackTombstonesPath(stateRoot), "utf8");
      if (recordsNow !== afterFirstRecords || tombsNow !== afterFirstTombs) {
        throw new Error(`state fixpoint violated at K=${K}`);
      }
    }
    const first = firstTotal / iters;
    const noop = noopCount > 0 ? noopTotal / noopCount : 0;
    console.log(
      `  K=${K}: first delivery ${f(first)} us; redelivery ${noopCount > 0 ? `${f(noop)} us/redelivery` : "n/a"}; state fixpoint after K deliveries BYTE-IDENTICAL to after 1 (W-A1)`
    );
  }
  rmSync(stateRoot, { recursive: true, force: true });
}

async function cellAppendDuplicateDose(): Promise<void> {
  console.log("[A2 appendFeedback duplicate-id dose + cascade duplicate-collapse witness]");
  // duplicate-id appends vs distinct-id appends: per-append price must be
  // dose-invariant (there is no dedup machinery to trigger).
  for (const [label, distinct] of [["same-id x2000", false], ["distinct-id x2000", true]] as const) {
    const stateRoot = mkdtempSync(join(tmpdir(), "r19j-dupapp-"));
    const results: number[] = [];
    for (let run = 0; run < 3; run++) {
      const iters = 2000;
      for (let i = 0; i < 100; i++) {
        await appendFeedback(stateRoot, recordOf(distinct ? `fb-warm-${run}-${i}` : "fb-dup"));
      }
      const t0 = hr();
      for (let i = 0; i < iters; i++) {
        await appendFeedback(stateRoot, recordOf(distinct ? `fb-${run}-${i}` : "fb-dup"));
      }
      const t1 = hr();
      results.push(usOf(t0, t1) / iters);
    }
    console.log(`  append ${label}: ${results.map(f).join(" / ")} us/append`);
    rmSync(stateRoot, { recursive: true, force: true });
  }

  // cascade duplicate-collapse: 20 hit records sharing ONE id vs 20 distinct.
  for (const [label, duplicateHitId, expectNewTombs] of [
    ["20 distinct hit ids", false, 20],
    ["20 hits sharing ONE id", true, 1]
  ] as const) {
    const stateRoot = mkdtempSync(join(tmpdir(), "r19j-dupcasc-"));
    const results: number[] = [];
    for (let run = 0; run < 3; run++) {
      seedStore(stateRoot, duplicateHitId);
      await cascadeFeedbackTombstones(stateRoot, "eps_warm" as EpisodeId); // warm no-match
      const iters = 30;
      let total = 0n;
      for (let i = 0; i < iters; i++) {
        seedStore(stateRoot, duplicateHitId); // untimed re-seed
        const t0 = hr();
        const out = await cascadeFeedbackTombstones(stateRoot, "eps_hit" as EpisodeId);
        const t1 = hr();
        if (out.length !== 20) throw new Error(`expected 20 cascaded entries, got ${out.length}`);
        total += t1 - t0;
      }
      results.push(Number(total) / 1000 / iters);
    }
    const tombs = JSON.parse(readFileSync(feedbackTombstonesPath(stateRoot), "utf8")) as string[];
    const newTombs = tombs.filter((id) => !id.startsWith("fb-pre-")).length;
    if (newTombs !== expectNewTombs) {
      throw new Error(`expected ${expectNewTombs} new tombstone ids, got ${newTombs}`);
    }
    console.log(
      `  cascade MATCH ${label}: ${results.map(f).join(" / ")} us/cascade  (sidecar +${newTombs} id${newTombs === 1 ? "" : "s"} — Set semantics collapse duplicates, W-A2)`
    );
    rmSync(stateRoot, { recursive: true, force: true });
  }
}

function mulberry32(seed: number): () => number {
  return () => {
    seed |= 0;
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function shuffled<T>(items: readonly T[], seed: number): T[] {
  const out = [...items];
  const rand = mulberry32(seed);

    const j = Math.floor(rand() * (i + 1));
    const tmp = out[i]!;
    out[i] = out[j]!;
    out[j] = tmp;
  }
  return out;
}
