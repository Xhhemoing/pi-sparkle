/* ================================================================
 * NEW AXIS, leg M: protocol-contamination probe. Four fresh processes on
 * dist: three ISOLATED (exactly one state measured per process =

const STATE_SNIPPETS: Record<string, string> = {

  const OBJ = "fix the login retry bug in src/auth/session.ts and keep tests green";

  const out = {};
  for (const state of states) {
    const args = state.habits !== undefined ? { objective: state.objective, habits: state.habits } : { objective: state.objective };
    await extractHeuristicContract(args);

    t0 = performance.now();
    for (let i = 0; i < 50000; i += 1) runGate();
    const gate = (performance.now() - t0) / 50000;
    applyPrecedence(prod.contract, "user-first");
    t0 = performance.now();
    for (let i = 0; i < 100000; i += 1) applyPrecedence(prod.contract, "user-first");
    const precedence = (performance.now() - t0) / 100000;

log("--- EXECUTION-HISTORY CENSUS, leg M: isolated-process vs shared-matrix protocol (dist) ---");
{

  for (const key of ["default", "A", "B"]) {

    check(`M state ${key} matrix is µs-class (<100µs)`, mat < 1e5, String(mat));
