 * (a) Leg-attribution census across every anchored matrix cell via a

 *     also the lexicographic minimum of every tie band), asserted per task.
 *     The measured delta is therefore the pure cost of pushing the scan
 *     from leg 2 (double compare) onto leg 3 (ICU localeCompare).
 * (c) Unit prices: localeCompare vs codepoint compare per call on short ids
 *     and on provider-prefixed ids (context pricing only — S3-B-3 and
 *     S8-B-3 remain closed; no mechanism is being re-proposed).
 * (d) Real-scale N=30 face on both arms + cascade-side cheapFirstTiers sort
 *     under the 8-way tie.
 * ============================================================ */
interface LegCensus { comparisons: number; leg1: number; leg2: number; leg3: number; tasks: number }
function selectWithCensus(
  eligible: readonly CatalogModel[],
  preferred: string | undefined,
  census: LegCensus
): CatalogModel {
  let best = eligible[0]!;
  for (let index = 1; index < eligible.length; index += 1) {
    const candidate = eligible[index]!;
    census.comparisons += 1;
    // compareLiveCandidates(candidate, best, preferred) replica w/ attribution
    const preferredDifference =
      Number(best.id === preferred) - Number(candidate.id === preferred);
    if (preferredDifference !== 0) {
      census.leg1 += 1;
      if (preferredDifference < 0) best = candidate;
      continue;
    }
    const costDifference = candidate.estimatedCostUsd - best.estimatedCostUsd;
    if (costDifference !== 0) {
      census.leg2 += 1;
      if (costDifference < 0) best = candidate;
      continue;
    }
    census.leg3 += 1;
    if (candidate.id.localeCompare(best.id) < 0) best = candidate;
  }
  return best;
}
function censusBatch(
  catalog: ModelRouterConfig,
  assignments: readonly TaskAssignment[],
  label: string
): LegCensus {
  const byId = new Map(createModelRouter(catalog).config.models.map((m) => [m.id, m]));
  const census: LegCensus = { comparisons: 0, leg1: 0, leg2: 0, leg3: 0, tasks: assignments.length };
  let replicaMatches = true;
  for (const item of assignments) {
    const eligible = item.decision.eligibleModels.map((id) => byId.get(id)!);
    const winner = selectWithCensus(eligible, item.preferredModel, census);
    if (winner.id !== item.decision.model) replicaMatches = false;
  }
  check(`ordering census [${label}]: leg-attribution replica reproduces every production selection`, replicaMatches);
  console.log(
    `ordering census [${label}]: comparisons=${census.comparisons} leg1(preferred)=${census.leg1} leg2(cost)=${census.leg2} leg3(localeCompare)=${census.leg3} (${(census.leg3 / census.tasks).toFixed(2)} per task)`
  );
  return census;
}
