### 2.1 plan 内容比对有效性证明：为什么 S7-F-1 引理不能上提到 validate 站点

本轮枚举中纸面收益最高的角度：runner 的 `expected` plan 在 `createShadowRunner` 时被 validateExperimentPlan 完整验证过；restore 时若 `serialized.plan` 与 `expected` 同下标逐字段（含 population 逐条目）相等，似乎即可传递「已验证」属性——Ω(P) 内容重读保留（每条目仍被读取比较），失配回退逐字。这正是 S7-F-1 的证明结构（同下标相等 vs 刚验证过的参照）。**致命差别在参照的验证新鲜度**：S7-F-1 的参照（population）是**同一次调用内**刚被 validateExperimentPlan 重验的内容；本角度的参照（`expected`）是**创建时那一次调用**验证的内容——跨调用携带「已验证」属性即 X1-1 的陈旧缓存本体。构造性反例（S3-F-2 反例逐字适用）：生产路径 `evaluateProposalShadow` 中 runner 返回态内嵌的 plan 就是闭包里的同一个对象（`serialized.plan === expected`），对该共享对象做就地变异（如 `population.push(重复项)`——TS readonly 仅编译期），自比较恒真、快路径放行，而生产 validateExperimentPlan 会抛 `population contains a duplicate: …`——**fail-open 发散**。要求 `serialized.plan !== expected` 才走快路径则生产命中率恰为 0（S10-D-2 同型自败）；每次 restore 先重验 `expected` 再比较则严格更慢。等价性在理论门即不成立，按 S10-C-3 先例不测量；判为 S3-F-2 + X1-1 换址重命名，不立 ID。

### 2.2 WeakSet 品牌态 restore 跳过：硬不变量正面违反

第二个纸面高收益角度：runner 闭包持 WeakSet 记录自己返回过的 state 对象，restore 对品牌对象跳过重验（生产路径 state 恰是上一调用返回的原对象，命中率 100%——S8-G-2 当年 0/64 命中的反面）。三重死因，任一独立成立：(1) 正面违反两条硬不变量——「validateExperimentPlan 每调用照跑」与「每条 assignment 逐调用全检」；(2) fail-open——调用方对返回态就地变异（品牌不灭），篡改内容被静默接受，生产 restore 全检会抛；(3) 机制上是 X1-1 闭包隐藏状态 + X3-3/X4-1 fail-open 增量 restore + S8-G-2 WeakSet 身份 memo 的三重同族。理论门淘汰，不测量，不立 ID。

 * R11-F pool-size accounting (temporary; NOT a candidate adjudication and NOT
 * committed — no winner this round). Completes the per-pool decomposition of
 * the ~120 ms full-experiment anchor by single-listing the three line items
 * that earlier rounds carried inside aggregate accounting:
 *
 *   pool A  recordOutcome duplicate-outcome check `outcomes.some(...)`
 *           (absence proof — S1-F-8 locked: a miss must scan the whole array
 *           or use a hash table). Production pattern: at record k the array
 *           holds k-1 entries and the scan always misses, so
 *           sum over A records = A(A-1)/2 full-scan compares. R10-F carried
 *           this pool as "sub-ms"; its Σ is the same order as the
 *           assignments.some pool R9-F measured at 1.54–1.57 ms, so this
 *           measurement corrects the ledger entry. Ceiling only — the O(1)
 *           elimination mechanism (hash table / incremental state) is S1-F-8.
 *
 *   pool B  restore-side split of the ~7 ms "Ω(A) alignment scan + defensive
 *           copies" aggregate (R8-F..R10-F carried it as one number):
 *           B1 = defensive copies [...assignments] + [...outcomes] over the
 *                production growth profile (2A restores over arrays growing
 *                0..A). Hard-invariant-locked (defensive copies stay).
 *           B2 = the aligned structural scan (same growth profile): per entry
 *                liveAction + changedLiveAction + shadowDecision loads/checks
 *                plus the same-index hash compare. Contract read obligation
 *                (every assignment fully checked per call, Ω(A)).
 *
 *   pool C  validateExperimentPlan constant (non-Ω(P)) part: planVersion,
 *           experimentId regex, mode, id predicates, thresholds, budget,
 *           randomization, stopPolicy, missingOutcomePolicy, canary-shape
 *           checks — everything except the two assertUniqueNonEmpty sites.
 *           2A calls per experiment. Skip/hoist mechanisms are S3-F-2 / X1-1;
 *           magnitude ceiling only.
 *
 * Fixtures are fixed/deterministic; timing lines are informational. Run:
 *   npx tsx tmp-r11f-pools.ts
 */

function timeMs(fn: () => void, rounds = 7): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

const A = 1000;
const P = 2 * A;
const hashes: string[] = [];
for (let i = 0; i < P; i++) hashes.push(`ep_${i.toString(36).padStart(8, "0")}`);

interface MiniAssignment {
  readonly episodeHash: string;
  readonly liveAction: "baseline";
  readonly shadowDecision: "baseline" | "candidate";
  readonly changedLiveAction: false;
}
interface MiniOutcome {
  readonly episodeHash: string;
  readonly utility: number;
  readonly costUsd: number;
  readonly guardrailBreached: boolean;
}

const assignmentsFull: MiniAssignment[] = [];
const outcomesFull: MiniOutcome[] = [];
for (let i = 0; i < A; i++) {
  assignmentsFull.push({
    episodeHash: hashes[i] as string,
    liveAction: "baseline",
    shadowDecision: i % 2 === 0 ? "candidate" : "baseline",
    changedLiveAction: false,
  });

    utility: 0.5,
    costUsd: 0.01,

  });
}

/* ------------------------------------------------------------------ */
/* pool A — outcomes.some absence proof (production full-miss pattern) */
/* ------------------------------------------------------------------ */
let sinkA = 0;
const tOutcomesSome = timeMs(() => {
  // At record k the outcomes array holds k-1 entries and the duplicate scan
  // misses every time (production mirror of recordExperimentOutcome's
  // `next.outcomes.some((item) => item.episodeHash === outcome.episodeHash)`).
  for (let k = 0; k < A; k++) {
    const prefix = outcomesFull.slice(0, k);
    void prefix;
  }
});
// Separate the slice-fixture cost from the scan cost: prebuild prefixes once.
const outcomePrefixes: MiniOutcome[][] = [];
for (let k = 0; k < A; k++) outcomePrefixes.push(outcomesFull.slice(0, k));
const tScan = timeMs(() => {
  sinkA = 0;
  for (let k = 0; k < A; k++) {
    const arr = outcomePrefixes[k] as MiniOutcome[];
    const query = (outcomesFull[k] as MiniOutcome).episodeHash;
    if (arr.some((item) => item.episodeHash === query)) sinkA += 1;
  }
});
const tSkip = timeMs(() => {
  // O(1) floor (what a hash-table absence proof would approach; S1-F-8).
  sinkA = 0;
  for (let k = 0; k < A; k++) {
    const query = (outcomesFull[k] as MiniOutcome).episodeHash;
    if (query.length === 0) sinkA += 1;
  }
});

);

const assignmentPrefixes: MiniAssignment[][] = [];

    for (let r = 0; r < 2; r++) {

      const co = [...o];

/** Frozen mirror of validateExperimentPlan WITHOUT the two
 *  assertUniqueNonEmpty sites — the constant part only. */
function validateConstantPart(plan: ExperimentPlan): void {
  if (typeof plan !== "object" || plan === null) {
    throw new DomainValidationError("experiment plan is required");
  }
  if (plan.planVersion !== 1) {
    throw new DomainValidationError(`unsupported planVersion: ${String(plan.planVersion)}`);
  }
  if (typeof plan.experimentId !== "string" || !EXPERIMENT_ID_PATTERN.test(plan.experimentId)) {
    throw new DomainValidationError(`invalid experimentId: ${String(plan.experimentId)}`);
  }
  if (plan.mode !== "shadow" && plan.mode !== "canary") {
    throw new DomainValidationError(`invalid mode: ${String(plan.mode)}`);
  }
  if (!isResourceVersionId(plan.baselineVersionId)) {
    throw new DomainValidationError(`invalid baselineVersionId: ${String(plan.baselineVersionId)}`);
  }
  if (!isCandidateId(plan.candidateId)) {
    throw new DomainValidationError(`invalid candidateId: ${String(plan.candidateId)}`);
  }
  if (typeof plan.thresholds !== "object" || plan.thresholds === null) {
    throw new DomainValidationError("thresholds are required");
  }
  if (
    typeof plan.thresholds.maxGuardrailBreaches !== "number" ||
    !Number.isInteger(plan.thresholds.maxGuardrailBreaches) ||
    plan.thresholds.maxGuardrailBreaches < 0
  ) {
    throw new DomainValidationError("maxGuardrailBreaches must be an integer >= 0");
  }
  if (
    typeof plan.thresholds.maxCostUsd !== "number" ||
    !Number.isFinite(plan.thresholds.maxCostUsd) ||
    plan.thresholds.maxCostUsd < 0
  ) {
    throw new DomainValidationError("maxCostUsd must be a finite number >= 0");
  }
  if (typeof plan.budget !== "object" || plan.budget === null) {
    throw new DomainValidationError("budget is required");
  }
  if (
    typeof plan.budget.maxAssignments !== "number" ||
    !Number.isInteger(plan.budget.maxAssignments) ||
    plan.budget.maxAssignments < 1
  ) {
    throw new DomainValidationError("maxAssignments must be an integer >= 1");
  }
  if (
    typeof plan.budget.maxWallClockMs !== "number" ||
    !Number.isInteger(plan.budget.maxWallClockMs) ||
    plan.budget.maxWallClockMs < 1
  ) {
    throw new DomainValidationError("maxWallClockMs must be an integer >= 1");
  }
  if (typeof plan.randomization !== "object" || plan.randomization === null) {
    throw new DomainValidationError("randomization is required");
  }
  if (!Number.isInteger(plan.randomization.seed)) {
    throw new DomainValidationError("seed must be an integer");
  }
  if (typeof plan.stopPolicy !== "object" || plan.stopPolicy === null) {
    throw new DomainValidationError("stopPolicy is required");
  }
  if (plan.stopPolicy.onGuardrail !== "halt") {
    throw new DomainValidationError(`invalid onGuardrail: ${String(plan.stopPolicy.onGuardrail)}`);
  }
  if (plan.stopPolicy.onBudgetExhausted !== "halt") {
    throw new DomainValidationError(
      `invalid onBudgetExhausted: ${String(plan.stopPolicy.onBudgetExhausted)}`
    );
  }
  if (
    plan.missingOutcomePolicy !== "exclude" &&
    plan.missingOutcomePolicy !== "treat-as-failure" &&
    plan.missingOutcomePolicy !== "abort"
  ) {
    throw new DomainValidationError(
      `invalid missingOutcomePolicy: ${String(plan.missingOutcomePolicy)}`
    );
  }
  if (plan.mode === "shadow" && plan.canary !== undefined) {
    throw new DomainValidationError("shadow plans must not include a canary block");
  }
}

const plan = makeShadowPlan(P, A);
validateExperimentPlan(plan); // sanity: fixture is valid
