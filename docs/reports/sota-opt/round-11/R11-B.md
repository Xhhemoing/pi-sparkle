## 2c. 新测量格：任务行 PIC 形状探针（输入侧收口）

方法：同一 seeded 值元组（seed `0xbbbb01`，N=2000）做两种物化——
**mixed**（条件 spread，undefined 可选键缺席；真实调用方 eval-routing/cli
构造任务行的形状）与 **uniform**（六键恒在、可持 undefined；整批单一
隐藏类）。`analyzeTask` 的 options 消费与 `assignPlanned` 的下游传递全部
按 `!== undefined` 判值，两口径 `assignTasks` 输出 **JSON 逐字节相等**
（每次运行均验证通过）。时延差五次独立运行：

```text
delta(mixed − uniform) = −184.4 / +21.4 / +73.1 / +51.8 / −161.8 us per call（−4.4% ~ +1.5%）
```

**符号不稳定，纯噪声带**——与 R9-B 目录行探针「形状多态在本切片规模上
无可利用成本」的结论在输入侧对称成立。不立候选的三重理由：(1) 测量本身
证明无收益池；(2) 任务行形状是切片外调用方的表面，切片内改写等于把
S8-E-2 已裁决的 spread→命令式镜像换址重提；(3) 属性存在性可观察
（S1-C-10 类），强行全键化会改变 `Object.keys`/序列化可见形状。输入侧
形状通道就此**测量收口**，与目录侧（R9-B）合并为全输入面闭合。

## 4. 关键裁决细节

### 4.1 S11-B-1：等价完美，输在「肥循环融合悖论」

等价论证完备：`selectLiveModel` 的扫描序与融合循环的 push 序都是目录序，
比较器调用序列逐次相同，平手分支（`<0` 不取等）保早序语义在两版中按构造
一致——fuzz 用 40% 成本平手 + 混大小写 id 专门轰炸该契约面，3 seeds ×
2000 组全部**引用同一**（比逐字节更强）。replica 对真实 router 的保真
先行（含 4185 个 refusal 路径三次逐位一致），排除了「测的不是生产语义」
的可能。

但基准淘汰且机制清晰：过滤循环体本就**肥**（13 键 options 字面量 +
`evaluateLiveCandidate` 跨模块调用 + refusals 展开 push），把比较器调用与
preferred 分支再塞进去增大热循环的代码体积与活跃寄存器压力；而被省掉的
第二遍扫描跑在**刚物化、cache 热**的紧凑数组上，且 `selectLiveModel` 是
独立小函数、单态调用点、V8 可整体内联——省的是最便宜的一遍，付的是最贵
的循环。M 越大（过滤循环占比越高）负效应越显：M=2 微正（+0.4~1.0%）、
M=7 符号翻转、M=10 五次全负（−1.0~−2.4%）。这与 S3-B-1「键构造成本 >
本体」同族但站点/机制均新：**跨阶段融合在肥循环上是负优化**，本轮以
符号稳定的负数据首次记档。即便取 M=2 的微正口径，+19~48 µs/批也在
S4-B-4（143–215 µs）已裁决噪声带之下。封死。

/* ============================================================
 * S11-B-1: partition + selection min-fusion. route() today runs
 * partitionLiveCandidates (one pass over the in-policy catalog slice building
 * eligible + refusals) and then selectLiveModel (a second pass over eligible
 * taking the min under compareLiveCandidates). The candidate tracks the
 * min WHILE partitioning: eligible is pushed in catalog order, and
 * selectLiveModel's scan starts at eligible[0] and compares each later
 * candidate against the running best with compareLiveCandidates(candidate,
 * best) < 0 — exactly the comparison sequence the fused loop performs, so the
 * selected object is reference-identical by construction. The eligible array
 * itself cannot be dropped (eligibleModels + behaviorDistribution consume it),
 * so the saving is only the second O(E) scan.

interface ResolvedReplica {
  readonly highRisk: boolean;
  readonly family: string;
  readonly featureVersion: string;
  readonly privacyRequired: PrivacyClass;
  readonly requiredCapabilities: readonly string[];
  readonly contextNeeded: number;
  readonly outputNeeded: number;
  readonly budgetUsd: number;
  readonly deadlineMs: number;
}
function resolveReplica(input: RouteTaskInput): ResolvedReplica {
  return {
    highRisk: input.highRisk === true,
    family: input.family ?? "unknown",
    featureVersion: input.featureVersion ?? FLOWCHART_FEATURE_VERSION,
    privacyRequired: input.privacyRequired ?? "cloud-general",
    requiredCapabilities: input.requiredCapabilities ?? ["tool-use"],
    contextNeeded: input.contextNeeded ?? 0,
    outputNeeded: input.outputNeeded ?? 0,
    budgetUsd: input.limits.remainingCostUsd ?? Number.POSITIVE_INFINITY,
    deadlineMs: input.limits.remainingTimeMs
  };
}
interface PartitionOutcome {
  readonly eligible: readonly CatalogModel[];
  readonly refusals: readonly RoutingRefusal[];
  readonly selected: CatalogModel | undefined;
}
function partitionTwoPass(
  models: readonly CatalogModel[],
  input: RouteTaskInput,
  resolved: ResolvedReplica
): PartitionOutcome {
  const allowed = new Set(input.modelPolicy.allowedModels);
  const eligible: CatalogModel[] = [];
  const refusals: RoutingRefusal[] = [];
  for (const model of models) {
    if (!allowed.has(model.id)) continue;
    const checkResult = evaluateLiveCandidate(model, {
      role: input.role,
      complexity: input.complexity,
      taskFamily: resolved.family,
      privacyRequired: resolved.privacyRequired,
      requiredCapabilities: resolved.requiredCapabilities,
      contextNeeded: resolved.contextNeeded,
      outputNeeded: resolved.outputNeeded,
      budgetUsd: resolved.budgetUsd,
      deadlineMs: resolved.deadlineMs,
      highRisk: resolved.highRisk,
      fixedCostUsd: model.estimatedCostUsd,
      fixedLatencyMs: model.estimatedDurationMs
    });
    if (checkResult.eligible) {
      eligible.push(model);
    } else {
      refusals.push(...checkResult.failures);
    }
  }
  const selected =
    eligible.length === 0 ? undefined : selectLiveModel(eligible, input.modelPolicy.preferredModel);
  return { eligible, refusals, selected };
}
function partitionFused(
  models: readonly CatalogModel[],
  input: RouteTaskInput,
  resolved: ResolvedReplica
): PartitionOutcome {
  const preferredModel = input.modelPolicy.preferredModel;
  const allowed = new Set(input.modelPolicy.allowedModels);
  const eligible: CatalogModel[] = [];
  const refusals: RoutingRefusal[] = [];
  let best: CatalogModel | undefined;
  for (const model of models) {
    if (!allowed.has(model.id)) continue;
    const checkResult = evaluateLiveCandidate(model, {
      role: input.role,
      complexity: input.complexity,
      taskFamily: resolved.family,
      privacyRequired: resolved.privacyRequired,
      requiredCapabilities: resolved.requiredCapabilities,
      contextNeeded: resolved.contextNeeded,
      outputNeeded: resolved.outputNeeded,
      budgetUsd: resolved.budgetUsd,
      deadlineMs: resolved.deadlineMs,
      highRisk: resolved.highRisk,
      fixedCostUsd: model.estimatedCostUsd,
      fixedLatencyMs: model.estimatedDurationMs
    });
    if (checkResult.eligible) {
      eligible.push(model);
      if (best === undefined) {
        best = model;
      } else if (compareLiveCandidates(model, best, preferredModel) < 0) {
        best = model;
      }
    } else {
      refusals.push(...checkResult.failures);
    }
  }
  return { eligible, refusals, selected: best };
