### 2.2 猎漏结果：恰两处从未点名位点（均不铸 ID）

**位点 A：`context/index.ts` L170/L191 同谓词双扫。**
`buildProjectContextIndex` 中 `hasTestRoute` 经
`snapshot.commands.some((c) => c.name === "test")`（L170），21 行后
`testCommand` 再经 `snapshot.commands.find((c) => c.name === "test")`
（L191）——同一纯谓词对同一未变异数组扫描两遍（两点之间仅 push
facts / 构建 manifests，`snapshot.commands` 零触碰）。候选：合并为
单次 `.find`，`hasTestRoute = testCommand !== undefined`。
- 理论：`some(p) ≡ (find(p) !== undefined)` 对纯谓词逐位成立；谓词为
  内联箭头非扩展点，调用次数非可观察契约（与 S1-F-2 的 propensityFor
  扩展点情形相反）；等价平凡。
- 探针实测（附录 B，生产形状 3 命令夹具，两次独立运行 × 3 rep）：
  双扫 vs 单扫 delta = **+0.7 / −7.3 / +5.1ns**（run1）、
  **+0.6 / −7.5 / +5.2ns**（run2）每构建——**符号翻转，纯抖动**；
  每 run 一次，against S0d 地板 40.2–40.3µs 差 3 个数量级，距落地线
  6–7 个数量级。
- 裁决：**换位点不铸 ID**——同函数双遍扫描域已由 S9-J-3
  （architecture/risks 双遍融合，21–88ns，同函数同机制更大站点已淘汰）
  与 S8-J-1（同文件跨调用 CSE 收口）双重覆盖；量级低于 S9-J-3 本体。
  R10-J 完备性声明的字面漏项，实质收口之内。

**位点 B：`cluster/host.ts` `send()` L126–128 死复核。**

if (role === undefined) throw ...`——函数头部已有

 * Site B: host.send() role-cast dead `role === undefined` recheck (L126-128).
 * Deterministic fixtures; three timed reps each; equivalence asserted inline.

const hr = (): bigint => process.hrtime.bigint();
const ns = (a: bigint, b: bigint, n: number): number => Number(b - a) / n;

// ---- Site A: two scans vs one over the production-shaped 3-command array
interface Cmd { readonly name: string; readonly command: string }
const commands: readonly Cmd[] = [
  { name: "test", command: "pnpm test" },
  { name: "lint", command: "pnpm lint" },
  { name: "build", command: "pnpm build" }
];
const commandsNoTest: readonly Cmd[] = [
  { name: "lint", command: "pnpm lint" },
  { name: "build", command: "pnpm build" }
];

function twoScan(cmds: readonly Cmd[]): { hasTestRoute: boolean; tests: string[] } {
  const hasTestRoute = cmds.some((command) => command.name === "test");
  const testCommand = cmds.find((command) => command.name === "test");
  const tests = testCommand !== undefined ? [testCommand.command] : [];
  return { hasTestRoute, tests };
}
function oneScan(cmds: readonly Cmd[]): { hasTestRoute: boolean; tests: string[] } {
  const testCommand = cmds.find((command) => command.name === "test");
  const hasTestRoute = testCommand !== undefined;
  const tests = testCommand !== undefined ? [testCommand.command] : [];
  return { hasTestRoute, tests };
}
// equivalence: some(p) === (find(p) !== undefined) for identical pure predicate over
// an unmutated array (no interleaving mutation between L170 and L191 in production).
for (const fixture of [commands, commandsNoTest, [] as readonly Cmd[]]) {
  const a = twoScan(fixture);
  const b = oneScan(fixture);
  if (a.hasTestRoute !== b.hasTestRoute || JSON.stringify(a.tests) !== JSON.stringify(b.tests)) {
    throw new Error("site A equivalence violated");
  }
}
{
  const N = 5_000_000;
  let sink = 0;
  for (let rep = 0; rep < 3; rep++) {
    for (let i = 0; i < N / 10; i++) sink += twoScan(commands).tests.length + oneScan(commands).tests.length;
    let t0 = hr();
    for (let i = 0; i < N; i++) sink += twoScan(commands).tests.length;
    let t1 = hr();
    const two = ns(t0, t1, N);
    t0 = hr();
    for (let i = 0; i < N; i++) sink += oneScan(commands).tests.length;
    t1 = hr();
    const one = ns(t0, t1, N);
    console.log(`site A rep${rep}: two-scan=${two.toFixed(1)}ns one-scan=${one.toFixed(1)}ns delta=${(two - one).toFixed(1)}ns/build [sink=${sink}]`);
  }
}

// ---- Site B: dead recheck branch cost (guarded vs unguarded queued-path body)
interface SendInput { readonly to?: string; readonly addressRole?: string; readonly from: string; readonly body: string }
function queuedWithRecheck(input: SendInput): string {
  // mirrors host.send() L125-137 shape: local alias + undefined recheck + throw arm
  const role = input.addressRole;
  if (role === undefined) {
    throw new Error("cluster send requires to or addressRole");
  }
  return `${input.from}:${role}:${input.body}`;
}
function queuedNoRecheck(input: SendInput): string {
  const role = input.addressRole as string;
  return `${input.from}:${role}:${input.body}`;
}
{
  const input: SendInput = { addressRole: "tester", from: "agent-1", body: "ping" };
  if (queuedWithRecheck(input) !== queuedNoRecheck(input)) throw new Error("site B equivalence violated");
  const N = 20_000_000;

    let t1 = hr();

    t0 = hr();
    for (let i = 0; i < N; i++) sink += queuedNoRecheck(input).length;
    t1 = hr();
    const bare = ns(t0, t1, N);
    console.log(`site B rep${rep}: with-recheck=${guarded.toFixed(2)}ns without=${bare.toFixed(2)}ns delta=${(guarded - bare).toFixed(2)}ns/send [sink=${sink}]`);
  }
}
