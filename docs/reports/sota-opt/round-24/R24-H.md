/* Leg OR: identity rotation at fixed warmth/age. Both arms share the SAME
 * loop body (indexed slot fetch + gate + precedence); arm S aliases all
 * slots to one set, arm R holds 1024 distinct sets. */
{
  const single: OperandSet[] = Array.from({ length: POOL_SIZE }, () => pool[0] as OperandSet);
  const mask = POOL_SIZE - 1;
  let cursorS = 0;
  let cursorR = 0;
  const armS = () => {
    const set = single[cursorS & mask] as OperandSet;
    cursorS += 1;
    assertCoverageAllowsStart(set.contract, set.tasks);
    sink += applyPrecedence(set.contract, "user-first").acceptanceCriteria.length;
  };
  const armR = () => {
    const set = pool[cursorR & mask] as OperandSet;
    cursorR += 1;
    assertCoverageAllowsStart(set.contract, set.tasks);
    sink += applyPrecedence(set.contract, "user-first").acceptanceCriteria.length;

  check("OR consumer stages stay sub-10µs under rotation", r < 10e-3, `${(r * 1e6).toFixed(0)}ns`);
