## 4. 重命名检查（派单强制项）

- **OCCX vs SCALEX（R20-C）**：SCALEX 在冻结组成/种子/次序上动
  **输入体量**——其剂量 (6,8,400)→(8,10,640)→(9,12,900) 中基数与
  占据随同一生成器**联动**（同占据纪律，K 随体量与基数一起动）；
  OCCX 冻结体量 400、基数 (4,6,8)、组成与种子，**只动 400 行在
  192 格胞上的联合分布**——把 SCALEX 中联动的 K 与 p 在固定体量下
  **解耦隔离**（conc8 vs spread51 隔离 K@p=17；pairs48 vs even192
  隔离 K@p=64）。对象（体量 vs 分布）、剂量单元、结论域（体量越线
  点 vs 占据无越线点+双通道响应律）均不同。非重命名，SCALEX 未
  重测（零尺寸剂量触碰）。
- **OCCX vs ORDX（R23-C）**：ORDX 置换**同一个固定多重集**并证明
  p/distinctKeys **置换不变**——次序只经迭代计数一条通道进池；
  OCCX 的每个剂量是**不同的多重集**（占据轮廓不同，p/K 被移动），
  恰好定价 ORDX 证明次序**够不着**的那组量（p 与 K）。ORDX 的结论
  （置换不变）与 OCCX 的结论（占据双通道响应）互为正交补。各 OCCX
  剂量在其自身规范发射序上单次定价，未置换任何固定多重集——ORDX
  未重测。非重命名。
- **OCCX vs YMIX（R21-C）/ SEEDX（R22-C）**：YMIX 逐位冻结三元组
  结构只翻 y 向量；OCCX 冻结 y 边际**计数**（ones=200 每剂量断言）
  只动三元组结构——两轴互为镜像、正交；SEEDX 只动 options.seed，
  OCCX 种子冻结 20260818。五轴（体量 × 组成 × 身份 × 次序 × 占据）
  两两正交，张成输入多重集在冻结契约下的全部自由度——**方法线至此
  完备**。非重命名，YMIX/SEEDX 零剂量重测。
- **OCCX vs ITERX / FITQ / ICOL / S7-C 本体**：ITERX（R16-C）是
  锚点占据上的迭代计数普查（收敛缩短类无逐位合法形态——答案维持
  无）；FITQ（R19-C）是锚点占据上的拟合粒度墙钟分布；ICOL（R16-C）

  已落地的去重 eta 计算。OCCX 只问**池量与池结构对占据分布的响应**

function sigmoid(z: number): number {
  if (z >= 0) return 1 / (1 + Math.exp(-z));
  const e = Math.exp(z);
  return e / (1 + e);
}

function rng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function buildDesign(rows: readonly Row[]): Design {
  const scenarios = [...new Set(rows.map((r) => r.scenarioId))];
  const models = [...new Set(rows.map((r) => r.modelVersion))];
  const projects = [...new Set(rows.map((r) => r.projectId))];
  const dropLast = (levels: readonly string[]): string[] =>
    levels.slice(0, Math.max(0, levels.length - 1));
  const scenarioLevels = dropLast(scenarios);
  const modelLevels = dropLast(models);
  const projectLevels = dropLast(projects);

  const pairCounts = new Map<string, number>();
  for (const row of rows) {
    const key = `${row.modelVersion}|${row.projectId}`;
    pairCounts.set(key, (pairCounts.get(key) ?? 0) + 1);
  }
  const interactionPairs = [...pairCounts.entries()]
    .filter(([, n]) => n >= INTERACTION_MIN_N)
    .map(([key]) => key);

  const names = [
    "intercept",
    ...scenarioLevels.map((s) => `a:${s}`),
    ...modelLevels.map((m) => `u:${m}`),
    ...projectLevels.map((p) => `v:${p}`),
    ...interactionPairs.map((key) => `w:${key}`)
  ];
  const columnIndex = new Map(names.map((name, index) => [name, index] as const));
  const interactionPairSet = new Set(interactionPairs);

  const referenceLevels: Array<{ factor: "a" | "u" | "v"; name: string }> = [];
  const lastModel = models[models.length - 1];
  const lastProject = projects[projects.length - 1];
  if (lastModel !== undefined) referenceLevels.push({ factor: "u", name: lastModel });
  if (lastProject !== undefined) referenceLevels.push({ factor: "v", name: lastProject });

  return {
    names,
    columnIndex,
    referenceLevels,
    build(row: Row, skip?: string): number[] {
      const vec = new Array<number>(names.length).fill(0);
      vec[0] = 1;
      const set = (name: string): void => {
        if (name === skip) return;
        const index = columnIndex.get(name);
        if (index !== undefined && index > 0) vec[index] = 1;
      };
      if (row.scenarioId !== scenarios[scenarios.length - 1]) set(`a:${row.scenarioId}`);
      if (row.modelVersion !== models[models.length - 1]) set(`u:${row.modelVersion}`);
      if (row.projectId !== projects[projects.length - 1]) set(`v:${row.projectId}`);
      const pairKey = `${row.modelVersion}|${row.projectId}`;
      if (interactionPairSet.has(pairKey)) set(`w:${pairKey}`);
      return vec;
    }
  };
}

function computeSupports(vectors: readonly number[][]): number[][] {
  return vectors.map((vector) => {
    const active: number[] = [];
    for (let j = 0; j < vector.length; j++) {
      if (vector[j] !== 0) active.push(j);
    }
    return active;
  });
}

function canonicalRowKeys(rows: readonly Row[]): number[] {
  const byScenario = new Map<string, Map<string, Map<string, number>>>();
  const keys = new Array<number>(rows.length);
  for (let i = 0; i < rows.length; i++) {
    const row = rows[i]!;
    let byModel = byScenario.get(row.scenarioId);
    if (byModel === undefined) {
      byModel = new Map();
      byScenario.set(row.scenarioId, byModel);
    }
    let byProject = byModel.get(row.modelVersion);
    if (byProject === undefined) {
      byProject = new Map();
      byModel.set(row.modelVersion, byProject);
    }
    let canonical = byProject.get(row.projectId);
    if (canonical === undefined) {
      canonical = i;
      byProject.set(row.projectId, canonical);
    }
    keys[i] = canonical;
  }
  return keys;
}

/* verbatim landed solveSymmetric (lin-alg.ts) — plain, no timers */
function solvePlain(a: readonly (readonly number[])[], b: readonly number[]): number[] | null {
  const n = b.length;
  if (a.length !== n || a.some((row) => row.length !== n)) {
    throw new DomainValidationError("solveSymmetric requires a square matrix matching b");
  }
  const m: number[][] = a.map((row) => [...row]);
  const x: number[] = [...b];
  const eps = 1e-12;

  for (let col = 0; col < n; col++) {
    let pivotRow = col;
    let pivotAbs = Math.abs(m[col]![col]!);
    for (let row = col + 1; row < n; row++) {
      const abs = Math.abs(m[row]![col]!);
      if (abs > pivotAbs) {
        pivotAbs = abs;
        pivotRow = row;
      }
    }
    if (pivotAbs < eps) return null;
    if (pivotRow !== col) {
      const tmp = m[col]!;
      m[col] = m[pivotRow]!;
      m[pivotRow] = tmp;
      const tb = x[col]!;
      x[col] = x[pivotRow]!;
      x[pivotRow] = tb;
    }
    const colArr = m[col]!;
    const pivot = colArr[col]!;
    const xCol = x[col]!;
    for (let row = col + 1; row < n; row++) {
      const rowArr = m[row]!;
      const factor = rowArr[col]! / pivot;
      if (factor === 0) continue;
      let k = col;
      const stop = n - 3;
      for (; k < stop; k += 4) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
        rowArr[k + 1] = rowArr[k + 1]! - factor * colArr[k + 1]!;
        rowArr[k + 2] = rowArr[k + 2]! - factor * colArr[k + 2]!;
        rowArr[k + 3] = rowArr[k + 3]! - factor * colArr[k + 3]!;
      }
      for (; k < n; k++) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
      }
      x[row] = x[row]! - factor * xCol;
    }
  }

  const solution = new Array<number>(n).fill(0);
  for (let row = n - 1; row >= 0; row--) {
    const rowArr = m[row]!;
    let sum = x[row]!;
    for (let k = row + 1; k < n; k++) sum -= rowArr[k]! * solution[k]!;
    const diag = rowArr[row]!;
    if (Math.abs(diag) < eps) return null;
    solution[row] = sum / diag;
  }
  for (const value of solution) {
    if (!Number.isFinite(value)) return null;
  }
  return solution;
}

function makeIrls(instrumented: boolean): IrlsImpl {
  return (design, rows, _vectors, supports, keys, keySpace, maxIter) => {
    const tIrls0 = instrumented ? performance.now() : 0;
    if (instrumented) STATS.fits += 1;
    const p = design.names.length;
    const n = rows.length;
    const eta = new Array<number>(n).fill(0);
    const mu = new Array<number>(n).fill(0);
    const xtwx: number[][] = Array.from({ length: p }, () => new Array<number>(p).fill(0));
    const xtwz: number[] = new Array<number>(p).fill(0);
    const stamp = new Int32Array(keySpace);
    const etaByKey = new Float64Array(keySpace);
    const muByKey = new Float64Array(keySpace);
    let beta = new Array<number>(p).fill(0);
    for (let iter = 0; iter < maxIter; iter++) {
      if (instrumented) {
        STATS.irlsIterations += 1;
        STATS.rowVisits += n;
      }
      const mark = iter + 1;
      for (let i = 0; i < n; i++) {
        const key = keys[i]!;
        if (stamp[key] !== mark) {
          stamp[key] = mark;
          if (instrumented) STATS.sigmoidIrls += 1;
          let value = 0;
          const active = supports[i]!;
          for (let ai = 0; ai < active.length; ai++) value += beta[active[ai]!]!;
          etaByKey[key] = value;
          muByKey[key] = sigmoid(value);
        }
        eta[i] = etaByKey[key]!;
        mu[i] = muByKey[key]!;
      }
      for (let d = 0; d < p; d++) xtwx[d]!.fill(0);
      xtwz.fill(0);
      for (let i = 0; i < n; i++) {
        const w = Math.max(mu[i]! * (1 - mu[i]!)!, 1e-10);
        const z = eta[i]! + ((rows[i]!.y - mu[i]!) / w);
        const wz = w * z;
        const active = supports[i]!;

            const a2 = active[2]!;
            const a3 = active[3]!;

            r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w; r0[a2] = r0[a2]! + w; r0[a3] = r0[a3]! + w; r0[a4] = r0[a4]! + w;

          }
          default: {

              const a = active[ai]!;
              xtwz[a] = xtwz[a]! + wz;
              const rowA = xtwx[a]!;

      if (next === null) {

        return { coefficients: null };
      }
      if (l2 < TOL) break;
    }
    if (instrumented) STATS.irlsMs += performance.now() - tIrls0;
    return { coefficients: beta };
  };

function dot(a: readonly number[], b: readonly number[]): number {
  let sum = 0;
  for (let i = 0; i < a.length; i++) sum += a[i]! * b[i]!;
  return sum;
}

function percentile(sortedValues: readonly number[], q: number): number {
  if (sortedValues.length === 0) return Number.NaN;
  const index = (sortedValues.length - 1) * q;
  const lo = Math.floor(index);
  const hi = Math.ceil(index);
  if (lo === hi) return sortedValues[lo]!;
  return sortedValues[lo]! * (hi - index) + sortedValues[hi]! * (index - lo);
}

function modelOf(effectName: string): string {
  return effectName.slice("u:".length);
}

/* betaQuantileLcb replica (posterior.ts verbatim) — keeps the replica lanes
 * self-contained exactly like R14-C..R23-C. */
function lnGamma(z: number): number {
  const g = 7;
  const c = [
    0.99999999999980993, 676.5203681218851, -1259.1392167224028, 771.32342877765313,
    -176.61502916214059, 12.507343278686905, -0.13857109526572012, 9.9843695780195716e-6,
    1.5056327351493116e-7
  ];
  if (z < 0.5) {
    return Math.log(Math.PI / Math.sin(Math.PI * z)) - lnGamma(1 - z);
  }
  const x = z - 1;
  let a = c[0]!;
  for (let i = 1; i < g + 2; i++) a += c[i]! / (x + i);
  const t = x + g + 0.5;
  return 0.5 * Math.log(2 * Math.PI) + (x + 0.5) * Math.log(t) - t + Math.log(a);
}

function lnBetaFunction(a: number, b: number): number {
  return lnGamma(a) + lnGamma(b) - lnGamma(a + b);
}

function betacf(a: number, b: number, x: number): number {
  const maxIt = 200;
  const eps = 3e-12;
  const qab = a + b;
  const qap = a + 1;
  const qam = a - 1;
  let c = 1;
  let d = 1 - (qab * x) / qap;
  if (Math.abs(d) < 1e-30) d = 1e-30;
  d = 1 / d;
  let h = d;
  for (let m = 1; m <= maxIt; m++) {
    const m2 = 2 * m;
    let aa = (m * (b - m) * x) / ((qam + m2) * (a + m2));
    d = 1 + aa * d;
    if (Math.abs(d) < 1e-30) d = 1e-30;
    c = 1 + aa / c;
    if (Math.abs(c) < 1e-30) c = 1e-30;
    d = 1 / d;
    h *= d * c;
    aa = (-(a + m) * (qab + m) * x) / ((a + m2) * (qap + m2));
    d = 1 + aa * d;
    if (Math.abs(d) < 1e-30) d = 1e-30;
    c = 1 + aa / c;
    if (Math.abs(c) < 1e-30) c = 1e-30;
    d = 1 / d;
    const del = d * c;
    h *= del;
    if (Math.abs(del - 1) < eps) break;
  }
  return h;
}

function regularizedIncompleteBeta(x: number, a: number, b: number, lnBeta: number): number {
  if (x <= 0) return 0;
  if (x >= 1) return 1;
  const prefix = Math.exp(a * Math.log(x) + b * Math.log(1 - x) - lnBeta);
  if (x < (a + 1) / (a + b + 2)) {
    return (prefix * betacf(a, b, x)) / a;
  }
  return 1 - (prefix * betacf(b, a, 1 - x)) / b;
}

function inverseRegularizedIncompleteBeta(p: number, a: number, b: number): number {
  const lnBeta = lnBetaFunction(a, b);
  let lo = 0;
  let hi = 1;
  for (let i = 0; i < 80; i++) {
    const mid = (lo + hi) / 2;
    if (regularizedIncompleteBeta(mid, a, b, lnBeta) > p) hi = mid;
    else lo = mid;
  }
  return (lo + hi) / 2;
}

function betaQuantileLcbReplica(posterior: { alpha: number; beta: number }, p = 0.05): number {
  if (!(p > 0 && p < 1)) return p <= 0 ? 0 : 1;
  if (posterior.alpha <= 0 || posterior.beta <= 0) return 0;
  return inverseRegularizedIncompleteBeta(p, posterior.alpha, posterior.beta);
}

function makeFit(irlsImpl: IrlsImpl, instrumented: boolean) {
  const onProb = (vectors: readonly number[][], coefficients: readonly number[]): number[] => {
    if (!instrumented) return vectors.map((vector) => sigmoid(dot(coefficients, vector)));
    const t0 = performance.now();
    const result = vectors.map((vector) => sigmoid(dot(coefficients, vector)));
    STATS.onProbMs += performance.now() - t0;
    STATS.onProbSigmoids += vectors.length;
    return result;
  };

  const apc = (
    design: Design,
    rows: readonly Row[],
    vectors: readonly number[][],
    coefficients: readonly number[],
    onProbabilities: readonly number[],
    column: string
  ): number => {
    const t0 = instrumented ? performance.now() : 0;
    if (instrumented) STATS.apcCalls += 1;
    let sum = 0;
    const columnIdx = design.columnIndex.get(column);
    if (columnIdx !== undefined && columnIdx !== 0) {
      for (let i = 0; i < rows.length; i++) {
        if (vectors[i]![columnIdx] === 0) continue;
        if (instrumented) STATS.apcActivePairs += 1;
        const on = onProbabilities[i]!;
        const offVector = vectors[i]!.slice();
        offVector[columnIdx] = 0;
        const off = sigmoid(dot(coefficients, offVector));
        sum += on - off;
      }
    }
    const result = rows.length === 0 ? 0 : sum / rows.length;
    if (instrumented) STATS.apcMs += performance.now() - t0;
    return result;
  };

  const pushValue = (map: Map<string, number[]>, key: string, value: number): void => {
    if (instrumented) STATS.drawsMapOps += 1;
    const list = map.get(key);
    if (list === undefined) map.set(key, [value]);
    else list.push(value);
  };

  return function fit(
    rows: readonly OfflineRow[],
    options?: { readonly maxIter?: number; readonly bootstrap?: number; readonly seed?: number }
  ): AttributionReport {
    const tTotal0 = instrumented ? performance.now() : 0;
    const maxIter = options?.maxIter ?? MAX_ITER_DEFAULT;
    const bootstrapDraws = options?.bootstrap ?? BOOTSTRAP_DEFAULT;
    const seed = options?.seed ?? SEED_DEFAULT;

    const tBase0 = instrumented ? performance.now() : 0;
    const baseRows: Row[] = rows.map((r) => ({
      scenarioId: r.scenarioId,
      modelVersion: r.modelVersion,
      projectId: r.projectId,
      y: r.y
    }));
    if (instrumented) STATS.baseCopyMs += performance.now() - tBase0;
    const effects: Array<{ name: string; point: number; lcb: number; ucb: number }> = [];

    if (baseRows.length === 0 || baseRows.every((r) => r.y === 0) || baseRows.every((r) => r.y === 1)) {
      return uncertainReport(baseRows.length, "INVALID_ESTIMATE: degenerate or empty design");
    }

    const design = buildDesign(baseRows);
    const vectors = baseRows.map((r) => design.build(r));
    const supports = computeSupports(vectors);
    const keys = canonicalRowKeys(baseRows);
    const fit = irlsImpl(design, baseRows, vectors, supports, keys, baseRows.length, maxIter);
    if (fit.coefficients === null) {
      return uncertainReport(baseRows.length, "INVALID_ESTIMATE: singular or non-finite Hessian");
    }

    const onProbabilities = onProb(vectors, fit.coefficients);
    const pointEffects = new Map<string, number>();
    for (const name of design.names) {
      if (name === "intercept") continue;
      pointEffects.set(
        name,
        apc(design, baseRows, vectors, fit.coefficients, onProbabilities, name)
      );
    }
    for (const ref of design.referenceLevels) {
      if (!pointEffects.has(`${ref.factor}:${ref.name}`)) {
        pointEffects.set(`${ref.factor}:${ref.name}`, 0);
      }
    }

    const random = rng(seed);
    const draws = new Map<string, number[]>();
    let successful = 0;
    for (let draw = 0; draw < bootstrapDraws; draw++) {
      const tSample0 = instrumented ? performance.now() : 0;
      const sample: Row[] = [];
      const sampleVectors: number[][] = [];
      const sampleSupports: number[][] = [];
      const sampleKeys: number[] = [];
      for (let i = 0; i < baseRows.length; i++) {
        const index = Math.floor(random() * baseRows.length);
        sample.push(baseRows[index]!);
        sampleVectors.push(vectors[index]!);
        sampleSupports.push(supports[index]!);
        sampleKeys.push(keys[index]!);
      }
      const collapsed = sample.every((r) => r.y === 0) || sample.every((r) => r.y === 1);
      if (instrumented) STATS.samplingMs += performance.now() - tSample0;
      if (collapsed) {
        if (instrumented) STATS.collapsedDraws += 1;
        continue;
      }
      const bootFit = irlsImpl(design, sample, sampleVectors, sampleSupports, sampleKeys, baseRows.length, maxIter);
      if (bootFit.coefficients === null) {
        if (instrumented) STATS.singularDraws += 1;
        continue;
      }
      successful += 1;
      if (instrumented) STATS.successfulDraws += 1;
      const sampleOnProbabilities = onProb(sampleVectors, bootFit.coefficients);
      for (const [name] of pointEffects.entries()) {
        const value = apc(
          design,
          sample,
          sampleVectors,
          bootFit.coefficients,
          sampleOnProbabilities,
          name
        );
        pushValue(draws, name, value);
      }
    }

    const tTail0 = instrumented ? performance.now() : 0;
    for (const [name, point] of pointEffects.entries()) {
      const values = draws.get(name) ?? [];
      if (successful < MIN_SUCCESSFUL_DRAWS || values.length < MIN_SUCCESSFUL_DRAWS) {
        return uncertainReport(baseRows.length, "INVALID_ESTIMATE: fewer than 20 successful bootstrap draws");
      }
      values.sort((a, b) => a - b);
      effects.push({
        name,
        point,
        lcb: percentile(values, 0.025),
        ucb: percentile(values, 0.975)
      });
    }

    const n = baseRows.length;
    const mean = baseRows.reduce((acc, r) => acc + r.y, 0) / n;
    const muPosterior = { alpha: 1 + n * mean, beta: 1 + n * (1 - mean) };
    const muLcb = betaQuantileLcbReplica(muPosterior, 0.05);
    const models = new Set(baseRows.map((r) => r.modelVersion)).size;
    const projects = new Set(baseRows.map((r) => r.projectId)).size;
    const ZERO_EPS = 0.005 * ATTRIBUTION_EFFECT;
    const containsZero = (e: { lcb: number; ucb: number }): boolean =>
      e.lcb <= ZERO_EPS && e.ucb >= -ZERO_EPS;
    const interactionsFor = (prefix: string): Array<{ name: string; lcb: number; ucb: number; point: number }> =>
      effects.filter((e) => e.name.startsWith(`w:${prefix}`));

    let diagnosis: AttributionReport["diagnosis"] = "uncertain";
    const scenarioHard = muLcb < QUALITY_FLOOR && models >= 2 && projects >= 3;
    if (
      effects.some(
        (e) => e.name.startsWith("u:") && e.lcb < -ATTRIBUTION_EFFECT && interactionsFor(modelOf(e.name)).every(containsZero)
      )
    ) {
      diagnosis = "model-problem";
    } else if (
      effects.some(
        (e) =>
          e.name.startsWith("v:") &&
          e.lcb < -ATTRIBUTION_EFFECT &&
          effects
            .filter((w) => w.name.startsWith("w:") && w.name.endsWith(`|${e.name.slice("v:".length)}`))
            .every(containsZero)
      )
    ) {
      diagnosis = "project-problem";
    } else if (effects.some((e) => e.name.startsWith("w:") && e.lcb < -ATTRIBUTION_EFFECT)) {
      diagnosis = "interaction-only";
    } else if (scenarioHard) {
      diagnosis = "scenario-hard";
    }

    const report: AttributionReport = {
      estimator: "logit-additive",
      rowsUsed: baseRows.length,
      effects,
      diagnosis,
      reason: diagnosis === "uncertain" ? "no effect beyond threshold or intervals too wide" : `${diagnosis} beyond the ${ATTRIBUTION_EFFECT} effect threshold`,
      writesActivePointer: false
    };
    if (instrumented) {
      STATS.tailMs += performance.now() - tTail0;
      STATS.totalMs += performance.now() - tTotal0;
    }
    return report;
  };
}

const fitInst = makeFit(irlsInstrumented, true);

function check(label: string, ok: boolean): void {

  else {

function compareReports(label: string, expected: AttributionReport, actual: AttributionReport): void {
  check(`${label}.estimator`, expected.estimator === actual.estimator);
  check(`${label}.rowsUsed`, expected.rowsUsed === actual.rowsUsed);
  check(`${label}.diagnosis`, expected.diagnosis === actual.diagnosis);
  check(`${label}.reason`, expected.reason === actual.reason);
  check(`${label}.writesActivePointer`, expected.writesActivePointer === actual.writesActivePointer);
  check(`${label}.effects.length`, expected.effects.length === actual.effects.length);
  if (expected.effects.length === actual.effects.length) {
    for (let i = 0; i < expected.effects.length; i++) {
      const e = expected.effects[i]!;
      const g = actual.effects[i]!;
      check(`${label}.effects[${i}].name`, e.name === g.name);
      check(`${label}.effects[${i}].point`, Object.is(e.point, g.point));
      check(`${label}.effects[${i}].lcb`, Object.is(e.lcb, g.lcb));
      check(`${label}.effects[${i}].ucb`, Object.is(e.ucb, g.ucb));
    }
  }
}

function fixtureRng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function pick<T>(r: () => number, items: readonly T[]): T {
  return items[Math.floor(r() * items.length)]!;
}

function randomRows(
  r: () => number,
  options: {
    scenarios: number;
    models: number;
    projects: number;
    rows: number;
    passRate: number;
    pipeInModel?: boolean;
  }
): OfflineRow[] {
  const scenarios = Array.from({ length: options.scenarios }, (_, i) => `fam${i}|role${i % 2}`);
  const models = Array.from({ length: options.models }, (_, i) =>
    options.pipeInModel === true && i === 0 ? "weird|model-0" : `model-${i}`
  );
  const projects = Array.from({ length: options.projects }, (_, i) => `prj_${i}`);
  const rows: OfflineRow[] = [];
  for (let i = 0; i < options.rows; i++) {
    const modelVersion = pick(r, models);
    const rate = modelVersion === models[0] ? options.passRate * 0.4 : options.passRate;
    rows.push({
      scenarioId: pick(r, scenarios),
      modelVersion,
      projectId: pick(r, projects),
      y: r() < rate ? 1 : 0,
      occurredAtMs: 1_000 + i,
    });
  }
  return rows;
}

      projectId: `p${i}`,
      y: r() < 0.6 ? 1 : 0,

  return rows;
}

/* --------- scenario 4: OCCX cell-occupancy concentration dose response --- *
 * NEW completeness axis (C-first; the FIFTH input dimension of the C-slice
 * dose-response method line — SCALEX moved volume with cardinality coupled,
 * YMIX moved the y marginal at a bitwise-frozen triple structure, SEEDX
 * moved the resample-identity stream, ORDX permuted a fixed multiset and
 * PROVED p/distinctKeys permutation-invariant). Question never measured in
 * C: every prior round priced the pools at ONE cell-occupancy profile
 * (K=167 distinct triples, p=60). Does ANY occupancy profile of 400 rows at
 * the frozen level sets, y marginal, and seed inflate a count-driven
 * sub-band pool over the ±35 ms band — and through WHICH channels does the
 * occupancy reach the frozen pools (kernel width p for solve; dedup key
 * count K for the irls sigmoid population; and whether the maxIter cap fires
 * when K < p and the fit is only ridge-identified)? */

const OCC_SCENARIOS = ["fam0|role0", "fam1|role1", "fam2|role0", "fam3|role1"] as const;
const OCC_MODELS = ["model-0", "model-1", "model-2", "model-3", "model-4", "model-5"] as const;
const OCC_PROJECTS = ["prj_0", "prj_1", "prj_2", "prj_3", "prj_4", "prj_5", "prj_6", "prj_7"] as const;

interface OccCell {
  readonly s: string;
  readonly m: string;
  readonly p: string;
}

function bresenhamCounts(total: number, cells: number): number[] {
  const counts = new Array<number>(cells);
  for (let i = 0; i < cells; i++) {
    counts[i] = Math.floor(((i + 1) * total) / cells) - Math.floor((i * total) / cells);
  }
  return counts;
}

/** Deterministic dose builder: rows emitted cell-by-cell; the y marginal is
 * spread evenly over the emission sequence (Bresenham), summing exactly to
 * `ones`. */
function buildDoseRows(cells: readonly OccCell[], counts: readonly number[], ones: number): OfflineRow[] {
  const total = counts.reduce((a, b) => a + b, 0);
  const rows: OfflineRow[] = [];
  let g = 0;
  for (let c = 0; c < cells.length; c++) {
    for (let r = 0; r < counts[c]!; r++) {
      const y = Math.floor(((g + 1) * ones) / total) - Math.floor((g * ones) / total) === 1 ? 1 : 0;
      rows.push({
        scenarioId: cells[c]!.s,
        modelVersion: cells[c]!.m,
        projectId: cells[c]!.p,
        y: y as 0 | 1,
        occurredAtMs: 1_000 + g,
      });
      g++;
    }
  }
  return rows;
}

function conc8Rows(ones: number): OfflineRow[] {
  const cells: OccCell[] = [];
  for (let i = 0; i < 8; i++) {
    cells.push({ s: OCC_SCENARIOS[i % 4]!, m: OCC_MODELS[i % 6]!, p: OCC_PROJECTS[i]! });
  }
  return buildDoseRows(cells, [2, 2, 2, 2, 2, 2, 2, 386], ones);
}

function spread51Rows(ones: number): OfflineRow[] {
  const cells: OccCell[] = [];
  const counts: number[] = [];
  for (let j = 0; j < 6; j++) {
    for (let k = 0; k < 8; k++) {
      if (j === 0 && k === 0) continue;
      cells.push({ s: OCC_SCENARIOS[(j * 8 + k) % 4]!, m: OCC_MODELS[j]!, p: OCC_PROJECTS[k]! });
      counts.push(2);
    }
  }
  const heavy = bresenhamCounts(306, 4);
  for (let si = 0; si < 4; si++) {
    cells.push({ s: OCC_SCENARIOS[si]!, m: OCC_MODELS[0]!, p: OCC_PROJECTS[0]! });
    counts.push(heavy[si]!);
  }
  return buildDoseRows(cells, counts, ones);
}

function pairs48Rows(ones: number): OfflineRow[] {
  const cells: OccCell[] = [];
  for (let j = 0; j < 6; j++) {
    for (let k = 0; k < 8; k++) {
      cells.push({ s: OCC_SCENARIOS[(j * 8 + k) % 4]!, m: OCC_MODELS[j]!, p: OCC_PROJECTS[k]! });
    }
  }
  return buildDoseRows(cells, bresenhamCounts(400, 48), ones);
}

function even192Rows(ones: number): OfflineRow[] {
  const cells: OccCell[] = [];
  for (let si = 0; si < 4; si++) {
    for (let j = 0; j < 6; j++) {
      for (let k = 0; k < 8; k++) {
        cells.push({ s: OCC_SCENARIOS[si]!, m: OCC_MODELS[j]!, p: OCC_PROJECTS[k]! });
      }
    }
  }
  return buildDoseRows(cells, bresenhamCounts(400, 192), ones);
}

function designWidthAndKeys(rows: readonly OfflineRow[]): { p: number; distinctKeys: number } {
  const baseRows: Row[] = rows.map((r) => ({
    scenarioId: r.scenarioId,
    modelVersion: r.modelVersion,
    projectId: r.projectId,
    y: r.y
  }));
  const design = buildDesign(baseRows);
  const keys = canonicalRowKeys(baseRows);
  return { p: design.names.length, distinctKeys: new Set(keys).size };
}

function setEq(a: ReadonlySet<string>, b: ReadonlySet<string>): boolean {
  if (a.size !== b.size) return false;
  for (const v of a) if (!b.has(v)) return false;
  return true;
}

function runOccDose(dose: OccDose, anchorOnes: number, anchorLevels: {
  s: ReadonlySet<string>; m: ReadonlySet<string>; p: ReadonlySet<string>;
}): DoseResult {
  check(`R24C-occx[${dose.label}].volume`, dose.rows.length === 400);
  const ones = dose.rows.reduce((acc, r) => acc + r.y, 0);
  check(`R24C-occx[${dose.label}].y-marginal-invariant`, ones === anchorOnes);
  check(`R24C-occx[${dose.label}].scenario-set`, setEq(new Set(dose.rows.map((r) => r.scenarioId)), anchorLevels.s));
  check(`R24C-occx[${dose.label}].model-set`, setEq(new Set(dose.rows.map((r) => r.modelVersion)), anchorLevels.m));
  check(`R24C-occx[${dose.label}].project-set`, setEq(new Set(dose.rows.map((r) => r.projectId)), anchorLevels.p));
  const { p, distinctKeys } = designWidthAndKeys(dose.rows);
  check(`R24C-occx[${dose.label}].p-expected`, p === dose.expectedP);
  check(`R24C-occx[${dose.label}].keys-expected`, distinctKeys === dose.expectedKeys);
  const expected = fitLogitAdditive(dose.rows, PERF_OPTIONS);
  const prodMs: number[] = [];
  for (let rep = 0; rep < 2; rep++) {
    const t0 = performance.now();
    fitLogitAdditive(dose.rows, PERF_OPTIONS);
    prodMs.push(performance.now() - t0);
  }
  const pools: PoolReading[] = [];
  for (let rep = 0; rep < 2; rep++) {
    STATS.reset();
    const actual = fitInst(dose.rows, PERF_OPTIONS);
    if (rep === 0) compareReports(`R24C-occx[${dose.label}]`, expected, actual);
    pools.push(snapshotStats());
  }
  const first = pools[0]!;
  const second = pools[1]!;
  check(`R24C-occx[${dose.label}].count-determinism`,
    first.fits === second.fits &&
    first.irlsIterations === second.irlsIterations &&
    first.successfulDraws === second.successfulDraws &&
    first.collapsedDraws === second.collapsedDraws &&
    first.singularDraws === second.singularDraws &&
    first.apcActivePairs === second.apcActivePairs &&
    first.onProbSigmoids === second.onProbSigmoids &&
    first.drawsMapOps === second.drawsMapOps &&
    first.sigmoidIrls === second.sigmoidIrls);
  check(`R24C-occx[${dose.label}].onprob-count-identity`,
    first.onProbSigmoids === 400 * (1 + first.successfulDraws));
  check(`R24C-occx[${dose.label}].onprob-cap`, first.onProbSigmoids <= 80_400);
  check(`R24C-occx[${dose.label}].tail-count-identity`,
    first.drawsMapOps === (p + 1) * first.successfulDraws);
  check(`R24C-occx[${dose.label}].tail-cap`, first.drawsMapOps <= (64 + 1) * 200);
  check(`R24C-occx[${dose.label}].iter-cap`, first.irlsIterations <= first.fits * 50);
  check(`R24C-occx[${dose.label}].apc-pairs-cap`, first.apcActivePairs <= first.fits * 400 * 4);
  return { label: dose.label, prodMs, pools, diagnosis: expected.diagnosis, p, distinctKeys };
}

function meanOf(values: number[]): number {
  return values.reduce((a, b) => a + b, 0) / values.length;
}

function scenarioOccx(): void {
  out(`probe OCCX (cell-occupancy / concentration dose response; NEW axis, not a lane):`);
  const anchorRows = perfRows();
  const anchorOnes = anchorRows.reduce((acc, r) => acc + r.y, 0);
  const anchorLevels = {
    s: new Set(anchorRows.map((r) => r.scenarioId)),
    m: new Set(anchorRows.map((r) => r.modelVersion)),
    p: new Set(anchorRows.map((r) => r.projectId)),
  };
  const { p: anchorP, distinctKeys: anchorKeys } = designWidthAndKeys(anchorRows);
  out(`  frozen dims: volume=400, levels=(4,6,8) anchor name sets, ones=${anchorOnes}, seed=${SEED_DEFAULT}, bootstrap=200`);
  out(`  anchor occupancy: p=${anchorP} distinctKeys=${anchorKeys}`);
  const doses: OccDose[] = [
    { label: "anchor", rows: anchorRows, expectedP: anchorP, expectedKeys: anchorKeys },
    { label: "conc8", rows: conc8Rows(anchorOnes), expectedP: 17, expectedKeys: 8 },
    { label: "spread51", rows: spread51Rows(anchorOnes), expectedP: 17, expectedKeys: 51 },
    { label: "pairs48", rows: pairs48Rows(anchorOnes), expectedP: 64, expectedKeys: 48 },
    { label: "even192", rows: even192Rows(anchorOnes), expectedP: 64, expectedKeys: 192 },
  ];
  const results: DoseResult[] = [];
  for (const dose of doses) {
    results.push(runOccDose(dose, anchorOnes, anchorLevels));
  }
  const poolKeys = [
    ["solve", (r: PoolReading) => r.solveMs],
    ["irls-other", (r: PoolReading) => r.irlsMs - r.solveMs],
    ["onProb", (r: PoolReading) => r.onProbMs],
    ["APC", (r: PoolReading) => r.apcMs],
    ["sampling", (r: PoolReading) => r.samplingMs],
    ["tail", (r: PoolReading) => r.tailMs],
    ["rest", (r: PoolReading) => r.totalMs - r.irlsMs - r.onProbMs - r.apcMs - r.samplingMs - r.tailMs],
  ] as const;
  for (const res of results) {
    const m = (f: (r: PoolReading) => number): number => meanOf(res.pools.map(f));
    const first = res.pools[0]!;
    out(
      `  dose occ=${res.label}${res.label === "anchor" ? " (anchor)" : ""}: p=${res.p} keys=${res.distinctKeys} ` +
      `fits=${first.fits} iters=${first.irlsIterations} irlsSigmoids=${first.sigmoidIrls} successful=${first.successfulDraws} ` +
      `collapsed=${first.collapsedDraws} singular=${first.singularDraws} apcPairs=${first.apcActivePairs} ` +
      `onProbSigmoids=${first.onProbSigmoids} drawsMapOps=${first.drawsMapOps} diagnosis=${res.diagnosis}; ` +
      `prod ${res.prodMs.map((t) => t.toFixed(1)).join("/")} ms; inst total ${m((r) => r.totalMs).toFixed(1)} ms | ` +
      poolKeys.map(([name, f]) => `${name} ${m(f).toFixed(1)}`).join(" | ")
    );
  }
  // Dose-response arithmetic: (a) worst COUNT-DRIVEN sub-band pool reading
  // across every dose vs the 35 ms band (solve/irls are the frozen pools
  // and the APC combination is floor-anchored by the R14-C rule at the
  // production shape — their rows are descriptive, exactly like SCALEX's /
  // YMIX's / SEEDX's / ORDX's); (b) the per-iteration prices of the frozen
  // pools per dose — the occupancy channels: kernel width p (solve),
  // per-iteration sigmoid population = irlsSigmoids/iters (irls dedup), and
  // the maxIter cap (rank-deficient doses).
  out(`  occupancy-dose arithmetic (35 ms band; count-driven sub-band pools only):`);
  for (const [name, f] of poolKeys) {
    if (name === "solve" || name === "irls-other" || name === "APC") continue;
    let worst = 0;
    let worstLabel = "";
    for (const res of results) {
      const v = meanOf(res.pools.map(f));
      if (v > worst) {
        worst = v;
        worstLabel = res.label;
      }
    }
    out(`    ${name}: worst ${worst.toFixed(1)} ms at occ=${worstLabel} (${worst < 35 ? "sub-band" : "OVER BAND"}, x${(35 / worst).toFixed(1)} below 35)`);
  }
  {
    const f = poolKeys[3][1];
    const values = results.map((res) => meanOf(res.pools.map(f)));
    const pairs = results.map((res) => res.pools[0]!.apcActivePairs);
    out(
      `    APC (descriptive; closure floor-anchored at the production shape, not count-based): ` +
      results.map((res, i) => `${res.label}:${values[i]!.toFixed(1)}`).join(" ") +
      ` ms — active pairs ${pairs.join("/")} (structural cap 321,600 = x${(321600 / Math.max(...pairs)).toFixed(2)} of worst dose)`
    );
  }
  {
    const iters = results.map((res) => res.pools[0]!.irlsIterations);
    const cap = results[0]!.pools[0]!.fits * 50;
    let worstIdx = 0;
    for (let i = 1; i < iters.length; i++) if (iters[i]! > iters[worstIdx]!) worstIdx = i;
    const kltp = results.filter((res) => res.distinctKeys < res.p).map((res) => res.label);
    out(
      `    iteration counts across doses: ${results.map((res, i) => `${res.label}:${iters[i]}(${(100 * iters[i]! / cap).toFixed(1)}% of cap)`).join(" ")} — ` +
      `cap ${cap} (fits x 50); worst dose = ${results[worstIdx]!.label} at ${(100 * iters[worstIdx]! / cap).toFixed(1)}%; ` +
      `the ridge-identified K<p doses (${kltp.join(", ")}) converge under tol BEFORE the cap — the pre-registered ` +
      `non-convergence expectation is FALSIFIED (deterministic witness, not a lane)`
    );
  }
  for (const [name, f] of [poolKeys[0], poolKeys[1]] as const) {
    const prices = results.map((res) => {
      const v = meanOf(res.pools.map(f));
      const iters = res.pools[0]!.irlsIterations;
      return (v / iters) * 1e3; // µs per iteration
    });
    out(
      `    ${name} per-iteration price (µs/iter across doses): ` +
      results.map((res, i) => `${res.label}[p=${res.p},K=${res.distinctKeys}]:${prices[i]!.toFixed(1)}`).join(" ") +
      ` — occupancy reaches this frozen pool through ${name === "solve" ? "the kernel width p (and the iteration count)" : "the dedup key population K (and the iteration count)"}`
    );
  }
  {
    const sigPerIter = results.map((res) => res.pools[0]!.sigmoidIrls / res.pools[0]!.irlsIterations);
    out(
      `    irls sigmoid population per iteration (dedup channel witness): ` +
      results.map((res, i) => `${res.label}[K=${res.distinctKeys}]:${sigPerIter[i]!.toFixed(1)}`).join(" ") +
      ` — tracks the per-resample distinct-key count, bounded by min(K, 400)`
    );
  }
}

/* ------ scenario 5: APCX/FLOORX ceiling re-measure (R14-C method) ------- */

function scenarioApcFloor(): void {
  const rows = perfRows();
  const baseRows: Row[] = rows.map((r) => ({
    scenarioId: r.scenarioId,
    modelVersion: r.modelVersion,
    projectId: r.projectId,
    y: r.y
  }));
  const design = buildDesign(baseRows);
  const vectors = baseRows.map((r) => design.build(r));
  const supports = computeSupports(vectors);
  const keys = canonicalRowKeys(baseRows);
  const fit = irlsPlain(design, baseRows, vectors, supports, keys, baseRows.length, MAX_ITER_DEFAULT);
  const coefficients = fit.coefficients!;
  const onProbabilities = vectors.map((vector) => sigmoid(dot(coefficients, vector)));
  const names = design.names.filter((name) => name !== "intercept");
  let apcSink = 0;
  const sweepReps = 60;
  for (let rep = 0; rep < 5; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const on = onProbabilities[i]!;
          const offVector = vectors[i]!.slice();
          offVector[columnIdx] = 0;
          const off = sigmoid(dot(coefficients, offVector));
          sum += on - off;
        }
      }
      apcSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const tSweep0 = performance.now();
  for (let rep = 0; rep < sweepReps; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const on = onProbabilities[i]!;
          const offVector = vectors[i]!.slice();
          offVector[columnIdx] = 0;
          const off = sigmoid(dot(coefficients, offVector));
          sum += on - off;
        }
      }
      apcSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const perSweepMs = (performance.now() - tSweep0) / sweepReps;
  out(
    `probe APCX: standalone timer-free APC sweep (59 names x 400 rows, base-fit coefficients) ` +
      `${perSweepMs.toFixed(3)} ms/fit -> x201 fits = ${(perSweepMs * 201).toFixed(1)} ms/report ` +
      `(base-shape approximation; sink=${apcSink.toFixed(3)})`
  );

  let floorSink = 0;
  for (let rep = 0; rep < 5; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const active = supports[i]!;
          let value = 0;
          for (let ai = 0; ai < active.length; ai++) {
            const j = active[ai]!;
            if (j === columnIdx) continue;
            value += coefficients[j]!;
          }
          sum += onProbabilities[i]! - sigmoid(value);
        }
      }
      floorSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const tFloor0 = performance.now();
  for (let rep = 0; rep < sweepReps; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const active = supports[i]!;
          let value = 0;
          for (let ai = 0; ai < active.length; ai++) {
            const j = active[ai]!;
            if (j === columnIdx) continue;
            value += coefficients[j]!;
          }
          sum += onProbabilities[i]! - sigmoid(value);
        }
      }
      floorSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const perFloorMs = (performance.now() - tFloor0) / sweepReps;
  out(
    `probe FLOORX: cheapest bitwise-legal sweep ${perFloorMs.toFixed(3)} ms/fit -> x201 fits = ` +
      `${(perFloorMs * 201).toFixed(1)} ms/report floor; APC combination ceiling on THIS VM = ` +
      `${(perSweepMs * 201 - perFloorMs * 201).toFixed(1)} ms (floorSink=${floorSink.toFixed(3)})`
  );
}
