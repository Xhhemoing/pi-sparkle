const FRAGMENTS = [
  "deploy", "production", "credentials", "review", "tests", "coverage", "plan",
  "design", "survey", "research", "refactor", "implement", "add ", "fix ",
  "migrate", "verify", "\n", "the module", "碎片", " "

function genObjective(rng: () => number): string {
  if (rng() < 0.5) return pick(rng, OBJECTIVE_TEMPLATES);
  let text = "";
  const parts = 1 + Math.floor(rng() * 8);
  for (let i = 0; i < parts; i += 1) text += pick(rng, FRAGMENTS) + " ";
  return text;
}
function genTasks(rng: () => number, n: number): AssignableTask[] {
  return Array.from({ length: n }, (_, i) => ({
    taskId: `tsk_${i.toString().padStart(6, "0")}` as AssignableTask["taskId"],
    role: pick(rng, ROLES),
    objective: genObjective(rng),
    ...(rng() < 0.1 ? { contractRisk: rng() < 0.5 } : {}),
    ...(rng() < 0.3 ? { contextTokens: Math.floor(rng() * 100000) } : {}),
    ...(rng() < 0.3 ? { outputTokens: Math.floor(rng() * 4000) } : {})
  }));
}

    `collapse same-window anchor [M=2 default, no learned]: replay N=2000 ${(m2NoLearnedSW * 1e3).toFixed(1)}us per call (${(m2NoLearnedSW * 2).toFixed(2)}ms per eval x2)`
  );
  console.log(
    `collapse marginal per-model at axis bottom (M=2 - M=1, no learned, same window): ${((m2NoLearnedSW - m1NoLearned) * 2).toFixed(2)}ms per eval x2`
  );
  console.log(
    `collapse analyzeTask share [M=1 custom, no learned]: ${(analyzeReplay * 1e3).toFixed(1)}us (${((analyzeReplay / m1NoLearned) * 100).toFixed(0)}% of the M=1 call)`
  );
  conclusions.push(
    `collapse M=1 custom evalface: no-learned per-eval=${(m1NoLearned * 2).toFixed(2)}ms baseline-learned=${(m1WithBaseline * 2).toFixed(2)}ms proposal-learned=${(m1WithProposal * 2).toFixed(2)}ms | same-window M=2=${(m2NoLearnedSW * 2).toFixed(2)}ms | marginal per-model=${((m2NoLearnedSW - m1NoLearned) * 2).toFixed(2)}ms per eval | analyzeTask share=${((analyzeReplay / m1NoLearned) * 100).toFixed(0)}%`
  );

  // M=1 10x stress + real live face.
  {
    const stress = genReplayTasks(mulberry32(0xb44b01), 20000);
    const stressM1 = bench(() => assignTasks({ catalog: catalogM1, tasks: stress }), 5);
    console.log(
      `collapse 10x stress N=20000 [M=1 custom]: ${stressM1.toFixed(1)}ms per call (${(stressM1 * 2).toFixed(1)}ms per eval x2)`
    );
    const cli = genTasks(mulberry32(0xb44b01), 30);
    const cliM1 = bench(() => assignTasks({ catalog: catalogM1, tasks: cli, learned: m1Baseline }), 2000);
    console.log(`collapse REAL face N=30 [M=1 custom + baseline learned]: ${(cliM1 * 1e3).toFixed(1)}us per call`);
    conclusions.push(
      `collapse M=1 stress N=20000=${(stressM1 * 2).toFixed(1)}ms per eval | real face N=30=${(cliM1 * 1e3).toFixed(1)}us per call`
    );
  }

  // The "cheap" collapse state (baseline primary literally "cheap"): sole
  // row is MEDIUM-cap and NOT high-risk approved. W5 existence probes pin
  // the fail-closed semantics; the timing cell runs on the low-risk-only
  // sub-corpus (every task whose analysis is routable under a MEDIUM cap).
  {
    const catalogCheap = catalogFromPrimary({ primaryModelId: "cheap" });
    check("collapse: catalogFromPrimary('cheap') also collapses to one model", catalogCheap.models.length === 1);
    const lowRisk = replayTasks.filter((task) => {
      const a = analyzeTask(task.objective, task.role);
      return !a.highRisk && a.complexity !== "HIGH";
    });
    console.log(`collapse [M=1 cheap]: low-risk routable sub-corpus = ${lowRisk.length}/2000 tasks`);
    check("collapse [M=1 cheap]: low-risk sub-corpus is non-trivial", lowRisk.length > 500, `${lowRisk.length}`);
    const okOut = assignTasks({ catalog: catalogCheap, tasks: [lowRisk[0]!] });
    check("collapse W5a [M=1 cheap]: a low-risk task routes to the sole cheap row", okOut[0]!.decision.model === "cheap");
    const risky = replayTasks.find((task) => analyzeTask(task.objective, task.role).highRisk);
    check("collapse W5 precondition: corpus contains a high-risk task", risky !== undefined);
    let threw = false;
    let message = "";
    try {
      assignTasks({ catalog: catalogCheap, tasks: [risky!] });
    } catch (error) {
      threw = error instanceof RoutingRefusalError;
      message = error instanceof Error ? error.message : String(error);
    }
    check(
      "collapse W5b [M=1 cheap]: a high-risk task fails closed with the high-risk-approval refusal",
      threw && message === "No allowed model is approved for high-risk tasks",
      message
    );
    const cheapCost = bench(() => assignTasks({ catalog: catalogCheap, tasks: lowRisk }), 30);
    console.log(
      `collapse unit [M=1 cheap MEDIUM-cap, low-risk N=${lowRisk.length}]: ${(cheapCost * 1e3).toFixed(1)}us per call (${((cheapCost / lowRisk.length) * 1e6).toFixed(0)}ns per task)`
    );
    conclusions.push(
      `collapse M=1 cheap low-risk-only: ${(cheapCost * 1e3).toFixed(1)}us per call over N=${lowRisk.length} (${((cheapCost / lowRisk.length) * 1e6).toFixed(0)}ns per task; high-risk fails closed W5b)`
    );
  }
}

console.log(`\nCONCLUSIONS: ${conclusions.join(" | ")}`);
