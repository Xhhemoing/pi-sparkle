 * R21-F new-axis measurements (temporary; NOT candidate adjudications and NOT
 * committed — no winner this round). One axis no earlier F round measured
 * directly: the TIME-CHANNEL (clock input face) COMPOSITION dimension. Every
 * runner operation threads a per-op scalar clock — assign/recordOutcome/
 * cancel run applyExperimentClock (one subtraction, one negative guard, one
 * always-on `elapsedMs >= budget.maxWallClockMs` compare, one spread), start
 * pins startedAtMs, and every fail-closed restore re-validates the two time
 * fields — yet twenty-one rounds of measurement pinned the channel to one
 * point: nowMs = 0 constant, startedAtMs = 0, elapsedMs = 0 (SMI zeros) and
 * maxWallClockMs = 1e12 (HeapNumber, never crossed). That is also the
 * production form (`evaluateProposalShadow` defaults nowMs = 0 and no src/
 * caller passes anything else), but the public runner signature accepts any
 * finite clock, the unit tests feed small advancing SMIs, and a realistic
 * caller would feed Date.now() values (~1.76e12 — beyond SMI range, so the
 * whole channel switches numeric representation to HeapNumber). R12-F priced
 * the clock MACHINERY (pool D2: the 2A clock spreads, at fixed value 0);
 * R17-F priced the non-finite-nowMs REJECTION; R18-F priced the halted state
 * via the GUARDRAIL endpoint — the "timeout" halt endpoint has never been
 * exercised in any F measurement, and no round ever varied the VALUES flowing
 * through the channel. Four cells close the dimension:
 *
 *   cell a  clock value-class × advancement grid (end-to-end, P=2000/A=1000,
 *           production interleave, budget 1e12): Z = constant 0 (anchor form
 *           = production form), E = constant epoch 1.756e12 (HeapNumber
 *           representation, zero elapsed), EP = epoch plateau (+8 ms every
 *           8 entries — Date.now() granularity: many ops share one ms),
 *           EA = epoch strictly advancing (+1 ms per entry — realistic
 *           caller), SA = small SMI strictly advancing from 0 (advancement
 *           without the representation switch; the unit-test form at anchor
 *           scale). Deltas vs Z. The time-projected ledgers (assignments,
 *           outcomes, breaches, halt) must be identical across all five arms.
 *
 *   cell b  wall-clock budget operand representation & margin on the EA
 *           clock: B-SMI = maxWallClockMs 1e9 (SMI operand), B-TIGHT =
 *           maxWallClockMs 1000 (margin 1 ms above the final elapsed 999 —
 *           the compare is always-on and always false in every arm; only the
 *           operand representation and the margin vary). Deltas vs EA.
 *
 *   cell c  the "timeout" halt endpoint, first direct exercise: EA clock with
 *           maxWallClockMs = 500 halts the entry-500 assign inside
 *           applyExperimentClock BEFORE the assignment is appended (unlike
 *           the budget-exhausted halt, which appends first). Canonical usage
 *           (halt checked after assign) stops at 500/500 with haltReason
 *           "timeout"; the EXACT production interleave (evaluateProposalShadow
 *           checks halted only at loop top) turns a mid-experiment timeout
 *           into a reject: the paired recordOutcome throws "outcome for
 *           unassigned episode" (R17-F reject class) — asserted, untimed.
 *
 *   cell d  per-op micro decomposition: the real applyExperimentClock on a
 *           minimal state at the four value classes (constant SMI 0, constant
 *           epoch HeapNumber, advancing epoch, advancing SMI), ns/op — the
 *           mechanism account for cell a's expected null and the value-side
 *           cross-check of R12-F pool D2.
 *
 * Every arm is verified deterministically before timing (final ledger sizes,
 * halt reasons, exact final startedAtMs/elapsedMs per arm, outcome
 * pass-through identity, cross-arm time-projected ledger digest equality),
 * then all end-to-end arms are timed ROUND-ROBIN (1 warmup + 5 passes) so
 * process drift spreads evenly across arms instead of biasing late blocks.
 *
 * Production pins the channel to nowMs=0 (zero dirty flow), so these are
 * informational floors and completeness closures, not pools.

function onceMs(fn: () => void): number {
  const t0 = process.hrtime.bigint();
  fn();
  const t1 = process.hrtime.bigint();
  return Number(t1 - t0) / 1e6;
}

function median(xs: readonly number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  return s[Math.floor(s.length / 2)] as number;
}

/** Historical fixture shape: `ep_` + zero-padded base36 index, L=11. */
function makePopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

function makeShadowPlan(
  population: readonly string[],
  maxAssignments: number,
  maxWallClockMs: number,
  id: string
): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: id,
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r21fbase"),
    candidateId: createCandidateId(() => "r21fcand"),
    population,
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments, maxWallClockMs },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

const P = 2000;
const A = 1000;
/** Epoch-scale clock base: beyond SMI range, the channel goes HeapNumber. */
const EPOCH = 1_756_000_000_000;

const population = makePopulation(P);
const hashes = population.slice(0, A);
const cleanStream: ExperimentOutcome[] = hashes.map((h) => ({
  episodeHash: h,
  utility: 0.5,
  costUsd: 0.01,
  guardrailBreached: false,
}));

/** Per-entry clock schedules (entry i's assign and record share times[i]). */
type ClockSchedule = { readonly startMs: number; readonly times: readonly number[] };

function constantSchedule(value: number): ClockSchedule {
  return { startMs: value, times: new Array<number>(A).fill(value) };
}
function advancingSchedule(base: number): ClockSchedule {
  const times: number[] = [];
  for (let i = 0; i < A; i++) times.push(base + i);
  return { startMs: base, times };
}
function plateauSchedule(base: number, width: number): ClockSchedule {
  const times: number[] = [];
  for (let i = 0; i < A; i++) times.push(base + Math.floor(i / width) * width);
  return { startMs: base, times };
}

/** Production interleave shape with the canonical between-op halt check. */
function runClocked(plan: ExperimentPlan, schedule: ClockSchedule): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(schedule.startMs);
  const times = schedule.times;
  for (let i = 0; i < cleanStream.length; i++) {
    if (state.halted) break;
    const outcome = cleanStream[i] as ExperimentOutcome;
    state = runner.assign(state, outcome.episodeHash, times[i] as number);
    if (state.halted && state.haltReason === "timeout") break;
    state = runner.recordOutcome(state, outcome, times[i] as number);
  }
  return state;
}

/** EXACT evaluateProposalShadow shape (halt checked only at loop top). */
function runProductionShape(plan: ExperimentPlan, schedule: ClockSchedule): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(schedule.startMs);
  const times = schedule.times;
  for (let i = 0; i < cleanStream.length; i++) {
    if (state.halted) break;
    const outcome = cleanStream[i] as ExperimentOutcome;
    state = runner.assign(state, outcome.episodeHash, times[i] as number);
    state = runner.recordOutcome(state, outcome, times[i] as number);
  }
  return state;
}

/** Digest over the time-projected ledger (time fields and plan excluded). */
function ledgerDigest(state: ShadowState): string {
  return stableStringify({
    assignments: state.assignments,
    outcomes: state.outcomes,
    guardrailBreaches: state.guardrailBreaches,
    halted: state.halted,
    haltReason: state.haltReason,
  });
}

interface Arm {
  readonly name: string;
  readonly cell: "a" | "b" | "c";
  readonly exec: () => ShadowState;
  readonly verify: (state: ShadowState) => void;
}

function expectBudgetHalt(
  state: ShadowState,
  startMs: number,
  finalElapsed: number
): void {
  if (!state.halted || state.haltReason !== "budget-exhausted") {
    throw new Error("expected budget-exhausted final state");
  }
  if (state.assignments.length !== A || state.outcomes.length !== A) {
    throw new Error("unexpected ledger sizes");
  }
  if (state.startedAtMs !== startMs) throw new Error("startedAtMs mismatch");
  if (state.elapsedMs !== finalElapsed) throw new Error("elapsedMs mismatch");
  for (let i = 0; i < A; i += 97) {
    if (state.outcomes[i] !== cleanStream[i]) throw new Error("outcome not pass-through");
  }
}
