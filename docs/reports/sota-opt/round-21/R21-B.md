type PadPosition = "tail" | "head";
const PRIOR_SOURCES = [
  "aider-polyglot",
  "swe-bench-verified-mini",
  "terminal-bench-2.1-fixed-harness",
  "arena-coding"
] as const;
function paddedPrior(rows: number, aliasEntries: number, position: PadPosition): PublicPriorSnapshot {
  const base = realisticPrior();
  const padRowCount = rows - base.scores.length;
  if (padRowCount < 0) throw new Error(`rows ${rows} below base ${base.scores.length}`);
  const padRows: PublicPriorScore[] = Array.from({ length: padRowCount }, (_, i) => ({
    sourceId: PRIOR_SOURCES[i % PRIOR_SOURCES.length]!,
    modelAliases: [`pad/m${String(i).padStart(5, "0")}`, `Pad Model ${i}`],
    raw: 0.5,
    unit: "pass_rate" as const,
    fetchedAt: TS as PublicPriorScore["fetchedAt"],
    sourceUrl: "https://example.com/pad"
  }));
  const padAliasCount = aliasEntries - (base.aliasMap?.length ?? 0);
  if (padAliasCount < 0) throw new Error(`aliasMap ${aliasEntries} below base`);
  const padAliases = Array.from({ length: padAliasCount }, (_, i) => ({
    canonicalId: `pad/alias-${String(i).padStart(5, "0")}`,
    alias: `Pad Alias ${i}`
  }));
  return {
    ...base,
    scores: position === "tail" ? [...base.scores, ...padRows] : [...padRows, ...base.scores],
    aliasMap: position === "tail"
      ? [...(base.aliasMap ?? []), ...padAliases]
      : [...padAliases, ...(base.aliasMap ?? [])]
  };
}
