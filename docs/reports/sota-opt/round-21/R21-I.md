```javascript
// R21-I forensic: attribute the superlinear e2e growth of flowchart cells to
// the G-owned execution plane by counting persisted bytes/events per run.
// One run per cell, then inspect runtime/runs/<runId>/.
import { spawnSync } from "node:child_process";
import { mkdirSync, rmSync, cpSync, readdirSync, statSync, readFileSync } from "node:fs";
import { join } from "node:path";

const CLI = "/workspace/dist/cli/main.js";
const ROOT = "/tmp/r21i-bench";
const FIX = join(ROOT, "fixtures");
const env = { ...process.env };
delete env.PI_PROVIDER; delete env.PI_MODEL; delete env.PI_API_KEY;

for (const name of ["flat16", "flat64", "flat128", "chain30"]) {
  const sd = join(ROOT, "work-forensic", name);
  rmSync(sd, { recursive: true, force: true });
  mkdirSync(sd, { recursive: true });
  cpSync(join(FIX, "state-custom"), sd, { recursive: true });
  const r = spawnSync(process.execPath, [CLI, "run", "--project", join(FIX, "project"), "--objective", "bench objective", "--state-root", sd,
    "--flowchart", join(FIX, `flow-${name}.json`), "--results", join(FIX, `results-${name}.json`)], { env, encoding: "utf8", timeout: 300000 });
  const runsDir = join(sd, "runtime", "runs");
  const runId = readdirSync(runsDir)[0];
  const evPath = join(runsDir, runId, "events.jsonl");
  const cpPath = join(runsDir, runId, "checkpoint.json");
  const evBytes = statSync(evPath).size;
  const evLines = readFileSync(evPath, "utf8").split("\n").filter((l) => l.trim() !== "").length;
  const cpBytes = statSync(cpPath).size;
  console.log(`${name}: exit=${r.status} events=${evLines} (${(evBytes / 1024).toFixed(0)}KB) checkpoint=${(cpBytes / 1024).toFixed(0)}KB`);
}
```
