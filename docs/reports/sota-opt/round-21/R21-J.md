/* ========== K3: deletion cascade episode fan-out (E-dose, I/O path) ========== */
/* N=200 records, target episode hits h=20 fixed. The remaining 180 records
 * spread across E_bg distinct background episode ids (fixed byte width).
 * Theory: per-record equality filter is cardinality-blind => flat. */
{
  const N = 200;
  const HITS = 20;
  const DOSES = [1, 9, 180];
  const ITERS = 30;
  const target = "ep_target000000" as EpisodeId;

  // Same seeding pattern as the anchor cascade cell (direct JSONL write,
  // untimed re-seed per iteration), with the BACKGROUND episode-id
  // cardinality parametrized and every field fixed-width.
  function seedFeedback(stateRoot: string, Ebg: number): void {
    const log = feedbackLogPath(stateRoot);
    mkdirSync(dirname(log), { recursive: true });
    const lines: string[] = [];
    for (let i = 0; i < N; i++) {
      const hit = i % 10 === 0; // 20 hits, position-determined (E-invariant)
      lines.push(
        JSON.stringify({
          id: `fb_${pad(i, 6)}`,
          episodeId: hit ? target : `ep_bg${pad(i % Ebg, 10)}`,
          kind: "human",
          rubricVersion: "r1",
          score: 50 + (i % 40),
          evidenceRefs: [],
          redacted: false,
          createdAt: "2026-08-25T00:00:00.000Z",
          body: `free text feedback payload number ${pad(i, 4)} `.repeat(3),
          summary: `summary ${pad(i, 4)}`
        })
      );
    }
    writeFileSync(log, lines.join("\n") + "\n", "utf8");
    writeFileSync(
      feedbackTombstonesPath(stateRoot),
      JSON.stringify(["fb-pre-a", "fb-pre-b", "fb-pre-c", "fb-pre-d", "fb-pre-e"], null, 2) + "\n",
      "utf8"
    );
  }

  const outBytes: string[] = [];
  const cascadedAccounts: string[] = [];

    outBytes.push(`${recBytes}/${Buffer.byteLength(tsRaw)}`);
