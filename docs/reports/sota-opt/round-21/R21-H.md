### 2.1 Y 腿：微任务检查点普查（确定性结构常数，16 重复 × 3 运行）

操作定义：与一次链执行并发地泵一个自重排队计数微任务；FIFO 微任务
序使计数器与链的续体严格交替，计数器轮数即链的**插入窗计数**（含

```text

               extractor.extract 单独        = 0   （async 体全同步 ⇒ 返回时已决议 ⇒ 零插入窗）
               critic.critique   单独        = 0   （同上）
Y 空闲单跳价: 140.1 / 143.3 / 144.5 ns  ⇒ 调度面 = 3 × ~142ns ≈ 420 / 430 / 434 ns/run
```

   单跳基准含 Promise 分配）与 S7-H-1 实测同步化差价 167–175ns/run
   同居 10²ns 级——互为第二证人**：机器全部调度面亚 µs、
   once-per-run、已被既有档案定价。
2. **唯一收缩杠杆已被禁**：削检查点 = 消异步机器 = S7-H-1（差价已
   测且拒列）+ X0-4（扩展点 Promise 签名）。本轮新分解出的

```ts
  let turns = 0;
  const p = run();
  void p.then(() => {
    settled = true;
  });
  await new Promise<void>((resolve) => {
    const pump = (): void => {
      if (settled) {
        resolve();
        return;
      }
      turns += 1;
      queueMicrotask(pump);
    };
    queueMicrotask(pump);
  });
  return turns;
}

  const surfaceNs = hopNs * (countsFull[0] ?? 0);
  log(
    `Y per-checkpoint hop price (idle)=${hopNs.toFixed(1)}ns; full-chain scheduling surface = ${countsFull[0]} x ${hopNs.toFixed(1)}ns = ${surfaceNs.toFixed(0)}ns/run (S7-H-1 sync-ification delta was 167-175ns/run)`
  );
  check("Y scheduling surface is sub-µs class (<1000ns)", surfaceNs < 1000, String(surfaceNs));
  check(
    "Y surface estimate and S7-H-1 delta share the 10^2 ns class (30..1000ns)",
    surfaceNs >= 30 && surfaceNs <= 1000,
    String(surfaceNs)
  );
}

/* ================================================================
 * NEW AXIS, leg QM: foreign-microtask dose-response (fresh dist children).
 * Before each timed chain call, queue K one-shot foreign microtasks (fixed
 * small spin each). FIFO ordering drains the whole foreign batch inside the
 * chain's FIRST checkpoint window, so chain wall = own price + foreign
 * batch. Adjudication: inflation must equal the STANDALONE foreign batch
 * price (zero-sum interleaving, no superadditive H-attributable tax), and
 * candidate bytes must be dose-invariant. Doses K = 0 / 64 / 512.
 * ================================================================ */
const mkMicrotaskDoseScript = (k: number): string => `
  const { extractHeuristicContract } = await import("/workspace/dist/requirement/heuristic.js");
  const { assertCoverageAllowsStart } = await import("/workspace/dist/requirement/coverage.js");
  const { applyPrecedence } = await import("/workspace/dist/requirement/precedence.js");
  const objective = "fix the login retry bug in src/auth/session.ts and keep tests green";
  const args = { objective };
  const tasks = Array.from({ length: 5 }, (_, i) => ({
    id: "tsk_0000000" + i,
    acceptanceCriteria: [{ id: "ac-objective" }, { id: "ac-tests" }]
  }));
  const K = ${k};
  let sink = 0;
  const foreign = () => { let s = 0; for (let j = 0; j < 32; j += 1) s += j; sink += s; };
  // Standalone foreign batch price: drain K one-shot microtasks with nothing
  // else queued, median of 512 samples.
  const drainOnce = async () => {
    const t0 = process.hrtime.bigint();
    if (K > 0) {
      for (let j = 0; j < K; j += 1) queueMicrotask(foreign);
      await new Promise((res) => queueMicrotask(res));
    }
    return Number(process.hrtime.bigint() - t0);
  };
  const drains = [];
  for (let i = 0; i < 512; i += 1) drains.push(await drainOnce());
  drains.sort((a, b) => a - b);
  const foreignP50 = drains[256];
  const first = JSON.stringify(await extractHeuristicContract(args));
  const N = 4096;
  const runChain = async () => {
    const cand = await extractHeuristicContract(args);
    assertCoverageAllowsStart(cand.contract, tasks);
    applyPrecedence(cand.contract, "user-first");
  };
  // Idle baseline (K queued nowhere).
  const idle = new Array(N);
  for (let i = 0; i < N; i += 1) {
    const t0 = process.hrtime.bigint();
    await runChain();
    idle[i] = Number(process.hrtime.bigint() - t0);
  }
  // Dosed: queue K foreign one-shot microtasks immediately before each call.
  const dosed = new Array(N);
  for (let i = 0; i < N; i += 1) {
    for (let j = 0; j < K; j += 1) queueMicrotask(foreign);
    const t0 = process.hrtime.bigint();
    await runChain();
    dosed[i] = Number(process.hrtime.bigint() - t0);
  }
  const last = JSON.stringify(await extractHeuristicContract(args));
  const p50 = (xs) => { const t = xs.slice(xs.length / 2).sort((a, b) => a - b); return t[Math.floor(t.length / 2)]; };
  process.stdout.write(JSON.stringify({
    k: K, foreignP50, idleP50: p50(idle), dosedP50: p50(dosed),
    deterministic: first === last, candidate: first, sinkNonzero: K === 0 || sink > 0
  }));
`;
log("--- SCHEDULING-SURFACE CENSUS, leg QM: foreign-microtask dose-response (fresh dist processes) ---");
const doseCandidates: Record<string, string> = {};
{
  for (const k of [0, 64, 512]) {
    const proc = spawnSync(process.execPath, ["--input-type=module", "-e", mkMicrotaskDoseScript(k)], {
      encoding: "utf8",
      maxBuffer: 64 * 1024 * 1024
    });
    check(`QM K=${k} child exits 0`, proc.status === 0, proc.stderr);
    if (proc.status !== 0) continue;
    const r = JSON.parse(proc.stdout) as {
      k: number; foreignP50: number; idleP50: number; dosedP50: number;
      deterministic: boolean; candidate: string; sinkNonzero: boolean;
    };
    doseCandidates[`K${k}`] = r.candidate;
    const inflation = r.dosedP50 - r.idleP50;
    log(
      `QM K=${k}: idle p50=${(r.idleP50 / 1e3).toFixed(2)}µs dosed p50=${(r.dosedP50 / 1e3).toFixed(2)}µs inflation=${(inflation / 1e3).toFixed(2)}µs vs standalone foreign batch p50=${(r.foreignP50 / 1e3).toFixed(2)}µs`
    );
    check(`QM K=${k} foreign tasks actually ran`, r.sinkNonzero);
    check(`QM K=${k} output byte-determinism (first vs last call)`, r.deterministic);
    check(`QM K=${k} dosed p50 stays µs-class (<100µs)`, r.dosedP50 < 1e5, String(r.dosedP50));
    if (k > 0) {
      // Zero-sum adjudication: inflation ~= standalone foreign price. A
      // superadditive H-attributable tax would push inflation well past the
      // standalone batch. Bound: inflation <= foreign + max(2µs, 1.0x foreign).
      const bound = r.foreignP50 + Math.max(2000, r.foreignP50);
      check(
        `QM K=${k} zero superadditive tax (inflation ${inflation}ns <= foreign ${r.foreignP50}ns + slack)`,
        inflation <= bound,
        `inflation=${inflation} foreign=${r.foreignP50}`
      );
    }
  }
  const base = doseCandidates["K0"];
  for (const key of ["K64", "K512"]) {
    check(`QM ${key} candidate JSON byte-equal to K0 (dose cannot change output)`, doseCandidates[key] === base);
  }
}

/* ================================================================
 * NEW AXIS, leg QT: macrotask / I-O non-interleaving attestation (fresh
 * dist child). A due 0ms timer, a pending setImmediate, and an in-flight
 * fs.readFile completion are all armed BEFORE the chain starts; the chain
 * executes entirely within one macrotask turn's microtask drain, so ALL
 * of them must observe-order strictly AFTER chain settle. Consequence:
 * ambient I/O state CANNOT inflate in-flight chain latency at all — the
 * only interleaving face is same-turn queued microtasks (leg QM), and
 * production call sites (track/clarify.ts direct await, track/loop.ts
 * sequential) queue none.
 * ================================================================ */
const QT_SCRIPT = `
  const { extractHeuristicContract } = await import("/workspace/dist/requirement/heuristic.js");
  const { assertCoverageAllowsStart } = await import("/workspace/dist/requirement/coverage.js");

    id: "tsk_0000000" + i,
    acceptanceCriteria: [{ id: "ac-objective" }, { id: "ac-tests" }]
  }));
  // Warm one call so module-level lazies are settled, then wait a macrotask
  // so the loop is quiescent before arming.
  await extractHeuristicContract(args);

  const order = [];
  setImmediate(() => order.push("immediate"));
  setTimeout(() => order.push("timer0"), 0);
  fs.readFile("/workspace/package.json", () => order.push("io"));
  // Spin >1ms so the 0ms timer is DUE and the readFile has time to complete
  // in the thread pool while we are still synchronous.
  const spinUntil = process.hrtime.bigint() + 5000000n;
  while (process.hrtime.bigint() < spinUntil) { /* hold the turn */ }
  const p = (async () => {
    const cand = await extractHeuristicContract(args);
    assertCoverageAllowsStart(cand.contract, tasks);

  await new Promise((res) => setTimeout(res, 50));
  process.stdout.write(JSON.stringify({ order }));
`;
log("--- SCHEDULING-SURFACE CENSUS, leg QT: macrotask/I-O non-interleaving attestation (fresh dist process) ---");
{
  const proc = spawnSync(process.execPath, ["--input-type=module", "-e", QT_SCRIPT], {
    encoding: "utf8",
    maxBuffer: 16 * 1024 * 1024
  });
  check("QT child exits 0", proc.status === 0, proc.stderr);
  if (proc.status === 0) {
    const r = JSON.parse(proc.stdout) as { order: string[] };
    log(`QT observed order: ${r.order.join(" -> ")}`);
    const settleIdx = r.order.indexOf("chain-settled");
    check("QT chain settled before the due 0ms timer fired", settleIdx !== -1 && settleIdx < r.order.indexOf("timer0"), r.order.join(","));
    check("QT chain settled before the pending setImmediate fired", settleIdx < r.order.indexOf("immediate"), r.order.join(","));
    check("QT chain settled before the in-flight fs completion fired", settleIdx < r.order.indexOf("io"), r.order.join(","));
    check("QT all four events observed", r.order.length === 4, r.order.join(","));
  }
}
```
