function buildDesign(rows: readonly Row[]): Design {
  const scenarios = [...new Set(rows.map((r) => r.scenarioId))];
  const models = [...new Set(rows.map((r) => r.modelVersion))];
  const projects = [...new Set(rows.map((r) => r.projectId))];

    levels.slice(0, Math.max(0, levels.length - 1));
  const scenarioLevels = dropLast(scenarios);
  const modelLevels = dropLast(models);
  const projectLevels = dropLast(projects);

  const pairCounts = new Map<string, number>();
  for (const row of rows) {
    const key = `${row.modelVersion}|${row.projectId}`;
    pairCounts.set(key, (pairCounts.get(key) ?? 0) + 1);
  }
  const interactionPairs = [...pairCounts.entries()]
    .filter(([, n]) => n >= INTERACTION_MIN_N)
    .map(([key]) => key);

    "intercept",
    ...scenarioLevels.map((s) => `a:${s}`),
    ...modelLevels.map((m) => `u:${m}`),
    ...projectLevels.map((p) => `v:${p}`),
    ...interactionPairs.map((key) => `w:${key}`)
  ];
  const columnIndex = new Map(names.map((name, index) => [name, index] as const));
  const interactionPairSet = new Set(interactionPairs);

  const referenceLevels: Array<{ factor: "a" | "u" | "v"; name: string }> = [];
  const lastModel = models[models.length - 1];
  const lastProject = projects[projects.length - 1];
  if (lastModel !== undefined) referenceLevels.push({ factor: "u", name: lastModel });
  if (lastProject !== undefined) referenceLevels.push({ factor: "v", name: lastProject });

  return {
    names,
    columnIndex,
    referenceLevels,
    build(row: Row, skip?: string): number[] {
      const vec = new Array<number>(names.length).fill(0);
      vec[0] = 1;
      const set = (name: string): void => {
        if (name === skip) return;
        const index = columnIndex.get(name);
        if (index !== undefined && index > 0) vec[index] = 1;
      };
      if (row.scenarioId !== scenarios[scenarios.length - 1]) set(`a:${row.scenarioId}`);

/* verbatim landed solveSymmetric (lin-alg.ts) — plain, no timers */
function solvePlain(a: readonly (readonly number[])[], b: readonly number[]): number[] | null {
  const n = b.length;
  if (a.length !== n || a.some((row) => row.length !== n)) {
    throw new DomainValidationError("solveSymmetric requires a square matrix matching b");
  }
  const m: number[][] = a.map((row) => [...row]);
  const x: number[] = [...b];
  const eps = 1e-12;

  for (let col = 0; col < n; col++) {
    let pivotRow = col;
    let pivotAbs = Math.abs(m[col]![col]!);
    for (let row = col + 1; row < n; row++) {
      const abs = Math.abs(m[row]![col]!);
      if (abs > pivotAbs) {
        pivotAbs = abs;
        pivotRow = row;
      }
    }
    if (pivotAbs < eps) return null;
    if (pivotRow !== col) {
      const tmp = m[col]!;
      m[col] = m[pivotRow]!;
      m[pivotRow] = tmp;
      const tb = x[col]!;
      x[col] = x[pivotRow]!;
      x[pivotRow] = tb;
    }
    const colArr = m[col]!;
    const pivot = colArr[col]!;
    const xCol = x[col]!;
    for (let row = col + 1; row < n; row++) {
      const rowArr = m[row]!;
      const factor = rowArr[col]! / pivot;
      if (factor === 0) continue;
      let k = col;
      const stop = n - 3;
      for (; k < stop; k += 4) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
        rowArr[k + 1] = rowArr[k + 1]! - factor * colArr[k + 1]!;
        rowArr[k + 2] = rowArr[k + 2]! - factor * colArr[k + 2]!;
        rowArr[k + 3] = rowArr[k + 3]! - factor * colArr[k + 3]!;
      }
      for (; k < n; k++) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
      }
      x[row] = x[row]! - factor * xCol;
    }
  }

  const solution = new Array<number>(n).fill(0);
  for (let row = n - 1; row >= 0; row--) {
    const rowArr = m[row]!;
    let sum = x[row]!;
    for (let k = row + 1; k < n; k++) sum -= rowArr[k]! * solution[k]!;
    const diag = rowArr[row]!;
    if (Math.abs(diag) < eps) return null;
    solution[row] = sum / diag;
  }
  for (const value of solution) {
    if (!Number.isFinite(value)) return null;
  }
  return solution;
}

            const a0 = active[0]!;
            const a1 = active[1]!;
            const a2 = active[2]!;
            const a3 = active[3]!;
            xtwz[a0] = xtwz[a0]! + wz;
            const r0 = xtwx[a0]!;
            r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w; r0[a2] = r0[a2]! + w; r0[a3] = r0[a3]! + w;
            xtwz[a1] = xtwz[a1]! + wz;
            const r1 = xtwx[a1]!;
            r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w; r1[a2] = r1[a2]! + w; r1[a3] = r1[a3]! + w;
            xtwz[a2] = xtwz[a2]! + wz;
            const r2 = xtwx[a2]!;
            r2[a0] = r2[a0]! + w; r2[a1] = r2[a1]! + w; r2[a2] = r2[a2]! + w; r2[a3] = r2[a3]! + w;
            xtwz[a3] = xtwz[a3]! + wz;
            const r3 = xtwx[a3]!;
            r3[a0] = r3[a0]! + w; r3[a1] = r3[a1]! + w; r3[a2] = r3[a2]! + w; r3[a3] = r3[a3]! + w;
            break;
          }
          case 3: {
            const a0 = active[0]!;
            const a1 = active[1]!;
            const a2 = active[2]!;
            xtwz[a0] = xtwz[a0]! + wz;
            const r0 = xtwx[a0]!;
            r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w; r0[a2] = r0[a2]! + w;
            xtwz[a1] = xtwz[a1]! + wz;
            const r1 = xtwx[a1]!;
            r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w; r1[a2] = r1[a2]! + w;
            xtwz[a2] = xtwz[a2]! + wz;
            const r2 = xtwx[a2]!;
            r2[a0] = r2[a0]! + w; r2[a1] = r2[a1]! + w; r2[a2] = r2[a2]! + w;
            break;
          }
          case 2: {
            const a0 = active[0]!;
            const a1 = active[1]!;
            xtwz[a0] = xtwz[a0]! + wz;
            const r0 = xtwx[a0]!;
            r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w;
            xtwz[a1] = xtwz[a1]! + wz;
            const r1 = xtwx[a1]!;
            r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w;
            break;
          }
          default: {
            for (let ai = 0; ai < active.length; ai++) {
              const a = active[ai]!;
              xtwz[a] = xtwz[a]! + wz;
              const rowA = xtwx[a]!;
              for (let bi = 0; bi < active.length; bi++) {
                const b = active[bi]!;
                rowA[b] = rowA[b]! + w;
              }
            }
          }
        }
      }
      for (let d = 0; d < p; d++) xtwx[d]![d] = xtwx[d]![d]! + 1e-6;
      let next: number[] | null;
      if (instrumented) {
        const tSolve0 = performance.now();
        next = solvePlain(xtwx, xtwz);
        STATS.solveMs += performance.now() - tSolve0;
        STATS.solveCalls += 1;
      } else {
        next = solvePlain(xtwx, xtwz);
      }
      if (next === null) {
        if (instrumented) STATS.irlsMs += performance.now() - tIrls0;
        return { coefficients: null };
      }
      const delta = next.map((value, index) => value - beta[index]!);
      const l2 = Math.sqrt(delta.reduce((acc, d) => acc + d * d, 0));
      beta = next;
      if (!beta.every(Number.isFinite)) {
        if (instrumented) STATS.irlsMs += performance.now() - tIrls0;
        return { coefficients: null };
      }
      if (l2 < TOL) break;
    }
    if (instrumented) STATS.irlsMs += performance.now() - tIrls0;
    return { coefficients: beta };
  };
}

function dot(a: readonly number[], b: readonly number[]): number {
  let sum = 0;
  for (let i = 0; i < a.length; i++) sum += a[i]! * b[i]!;
  return sum;
}

function percentile(sortedValues: readonly number[], q: number): number {
  if (sortedValues.length === 0) return Number.NaN;
  const index = (sortedValues.length - 1) * q;
  const lo = Math.floor(index);
  const hi = Math.ceil(index);
  if (lo === hi) return sortedValues[lo]!;
  return sortedValues[lo]! * (hi - index) + sortedValues[hi]! * (index - lo);
}

function makeFit(irlsImpl: IrlsImpl, instrumented: boolean) {
  const onProb = (vectors: readonly number[][], coefficients: readonly number[]): number[] => {
    if (!instrumented) return vectors.map((vector) => sigmoid(dot(coefficients, vector)));
    const t0 = performance.now();
    const result = vectors.map((vector) => sigmoid(dot(coefficients, vector)));
    STATS.onProbMs += performance.now() - t0;
    STATS.onProbSigmoids += vectors.length;
    return result;
  };

  const apc = (
    design: Design,
    rows: readonly Row[],
    vectors: readonly number[][],
    coefficients: readonly number[],
    onProbabilities: readonly number[],
    column: string
  ): number => {
    const t0 = instrumented ? performance.now() : 0;
    if (instrumented) STATS.apcCalls += 1;
    let sum = 0;
    const columnIdx = design.columnIndex.get(column);
    if (columnIdx !== undefined && columnIdx !== 0) {
      for (let i = 0; i < rows.length; i++) {
        if (vectors[i]![columnIdx] === 0) continue;
        if (instrumented) STATS.apcActivePairs += 1;
        const on = onProbabilities[i]!;
        const offVector = vectors[i]!.slice();
        offVector[columnIdx] = 0;
        const off = sigmoid(dot(coefficients, offVector));
        sum += on - off;
      }
    }
    const result = rows.length === 0 ? 0 : sum / rows.length;
    if (instrumented) STATS.apcMs += performance.now() - t0;
    return result;
  };

  const pushValue = (map: Map<string, number[]>, key: string, value: number): void => {
    if (instrumented) STATS.drawsMapOps += 1;
    const list = map.get(key);
    if (list === undefined) map.set(key, [value]);
    else list.push(value);
  };

  return function fit(
    rows: readonly OfflineRow[],
    options?: { readonly maxIter?: number; readonly bootstrap?: number; readonly seed?: number }
  ): AttributionReport {
    const tTotal0 = instrumented ? performance.now() : 0;
    const maxIter = options?.maxIter ?? MAX_ITER_DEFAULT;
    const bootstrapDraws = options?.bootstrap ?? BOOTSTRAP_DEFAULT;
    const seed = options?.seed ?? SEED_DEFAULT;

    const tBase0 = instrumented ? performance.now() : 0;
    const baseRows: Row[] = rows.map((r) => ({
      scenarioId: r.scenarioId,
      modelVersion: r.modelVersion,
      projectId: r.projectId,
      y: r.y
    }));
    if (instrumented) STATS.baseCopyMs += performance.now() - tBase0;
    const effects: Array<{ name: string; point: number; lcb: number; ucb: number }> = [];

    if (baseRows.length === 0 || baseRows.every((r) => r.y === 0) || baseRows.every((r) => r.y === 1)) {
      return uncertainReport(baseRows.length, "INVALID_ESTIMATE: degenerate or empty design");
    }

    const design = buildDesign(baseRows);
    const vectors = baseRows.map((r) => design.build(r));
    const supports = computeSupports(vectors);
    const keys = canonicalRowKeys(baseRows);
    const fit = irlsImpl(design, baseRows, vectors, supports, keys, baseRows.length, maxIter);
    if (fit.coefficients === null) {
      return uncertainReport(baseRows.length, "INVALID_ESTIMATE: singular or non-finite Hessian");
    }

    const onProbabilities = onProb(vectors, fit.coefficients);
    const pointEffects = new Map<string, number>();
    for (const name of design.names) {
      if (name === "intercept") continue;
      pointEffects.set(
        name,
        apc(design, baseRows, vectors, fit.coefficients, onProbabilities, name)
      );
    }
    for (const ref of design.referenceLevels) {
      if (!pointEffects.has(`${ref.factor}:${ref.name}`)) {
        pointEffects.set(`${ref.factor}:${ref.name}`, 0);
      }
    }

    const random = rng(seed);
    const draws = new Map<string, number[]>();
    let successful = 0;
    for (let draw = 0; draw < bootstrapDraws; draw++) {
      const tSample0 = instrumented ? performance.now() : 0;
      const sample: Row[] = [];
      const sampleVectors: number[][] = [];
      const sampleSupports: number[][] = [];
      const sampleKeys: number[] = [];
      for (let i = 0; i < baseRows.length; i++) {
        const index = Math.floor(random() * baseRows.length);
        sample.push(baseRows[index]!);
        sampleVectors.push(vectors[index]!);
        sampleSupports.push(supports[index]!);
        sampleKeys.push(keys[index]!);
      }
      const collapsed = sample.every((r) => r.y === 0) || sample.every((r) => r.y === 1);
      if (instrumented) STATS.samplingMs += performance.now() - tSample0;
      if (collapsed) {
        if (instrumented) STATS.collapsedDraws += 1;
        continue;
      }
      const bootFit = irlsImpl(design, sample, sampleVectors, sampleSupports, sampleKeys, baseRows.length, maxIter);
      if (bootFit.coefficients === null) {
        if (instrumented) STATS.singularDraws += 1;
        continue;
      }
      successful += 1;
      if (instrumented) STATS.successfulDraws += 1;
      const sampleOnProbabilities = onProb(sampleVectors, bootFit.coefficients);
      for (const [name] of pointEffects.entries()) {
        const value = apc(
          design,
          sample,
          sampleVectors,
          bootFit.coefficients,
          sampleOnProbabilities,
          name
        );
        pushValue(draws, name, value);
      }
    }

    const tTail0 = instrumented ? performance.now() : 0;
    for (const [name, point] of pointEffects.entries()) {
      const values = draws.get(name) ?? [];
      if (successful < MIN_SUCCESSFUL_DRAWS || values.length < MIN_SUCCESSFUL_DRAWS) {
        return uncertainReport(baseRows.length, "INVALID_ESTIMATE: fewer than 20 successful bootstrap draws");
      }
      values.sort((a, b) => a - b);
      effects.push({
        name,
        point,
        lcb: percentile(values, 0.025),
        ucb: percentile(values, 0.975)
      });
    }

    const n = baseRows.length;
    const mean = baseRows.reduce((acc, r) => acc + r.y, 0) / n;
    const muPosterior = { alpha: 1 + n * mean, beta: 1 + n * (1 - mean) };
    const muLcb = betaQuantileLcb(muPosterior, 0.05);
    const models = new Set(baseRows.map((r) => r.modelVersion)).size;
    const projects = new Set(baseRows.map((r) => r.projectId)).size;
    const ZERO_EPS = 0.005 * ATTRIBUTION_EFFECT;
    const containsZero = (e: { lcb: number; ucb: number }): boolean =>
      e.lcb <= ZERO_EPS && e.ucb >= -ZERO_EPS;
    const interactionsFor = (prefix: string): Array<{ name: string; lcb: number; ucb: number; point: number }> =>
      effects.filter((e) => e.name.startsWith(`w:${prefix}`));

    let diagnosis: AttributionReport["diagnosis"] = "uncertain";
    const scenarioHard = muLcb < QUALITY_FLOOR && models >= 2 && projects >= 3;
    if (
      effects.some(
        (e) => e.name.startsWith("u:") && e.lcb < -ATTRIBUTION_EFFECT && interactionsFor(modelOf(e.name)).every(containsZero)
      )
    ) {
      diagnosis = "model-problem";
    } else if (
      effects.some(
        (e) =>
          e.name.startsWith("v:") &&
          e.lcb < -ATTRIBUTION_EFFECT &&
          effects
            .filter((w) => w.name.startsWith("w:") && w.name.endsWith(`|${e.name.slice("v:".length)}`))
            .every(containsZero)
      )
    ) {
      diagnosis = "project-problem";
    } else if (effects.some((e) => e.name.startsWith("w:") && e.lcb < -ATTRIBUTION_EFFECT)) {
      diagnosis = "interaction-only";
    } else if (scenarioHard) {
      diagnosis = "scenario-hard";
    }

    const report: AttributionReport = {
      estimator: "logit-additive",
      rowsUsed: baseRows.length,
      effects,
      diagnosis,
      reason: diagnosis === "uncertain" ? "no effect beyond threshold or intervals too wide" : `${diagnosis} beyond the ${ATTRIBUTION_EFFECT} effect threshold`,
      writesActivePointer: false
    };
    if (instrumented) {
      STATS.tailMs += performance.now() - tTail0;
      STATS.totalMs += performance.now() - tTotal0;
    }
    return report;
  };
}

const fitInst = makeFit(irlsInstrumented, true);

function check(label: string, ok: boolean): void {
  if (ok) checksPassed++;
  else {
    failures++;
    process.stderr.write(`FAIL ${label}\n`);

function compareReports(label: string, expected: AttributionReport, actual: AttributionReport): void {
  check(`${label}.estimator`, expected.estimator === actual.estimator);
  check(`${label}.rowsUsed`, expected.rowsUsed === actual.rowsUsed);
  check(`${label}.diagnosis`, expected.diagnosis === actual.diagnosis);
  check(`${label}.reason`, expected.reason === actual.reason);
  check(`${label}.writesActivePointer`, expected.writesActivePointer === actual.writesActivePointer);
  check(`${label}.effects.length`, expected.effects.length === actual.effects.length);
  if (expected.effects.length === actual.effects.length) {
    for (let i = 0; i < expected.effects.length; i++) {
      const e = expected.effects[i]!;
      const g = actual.effects[i]!;
      check(`${label}.effects[${i}].name`, e.name === g.name);
      check(`${label}.effects[${i}].point`, Object.is(e.point, g.point));
      check(`${label}.effects[${i}].lcb`, Object.is(e.lcb, g.lcb));
      check(`${label}.effects[${i}].ucb`, Object.is(e.ucb, g.ucb));
    }
  }
}

function fixtureRng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function pick<T>(r: () => number, items: readonly T[]): T {
  return items[Math.floor(r() * items.length)]!;
}

function randomRows(
  r: () => number,
  options: {
    scenarios: number;
    models: number;
    projects: number;
    rows: number;
    passRate: number;
    pipeInModel?: boolean;
  }
): OfflineRow[] {
  const scenarios = Array.from({ length: options.scenarios }, (_, i) => `fam${i}|role${i % 2}`);
  const models = Array.from({ length: options.models }, (_, i) =>
    options.pipeInModel === true && i === 0 ? "weird|model-0" : `model-${i}`
  );
  const projects = Array.from({ length: options.projects }, (_, i) => `prj_${i}`);
  const rows: OfflineRow[] = [];
  for (let i = 0; i < options.rows; i++) {
    const modelVersion = pick(r, models);
    const rate = modelVersion === models[0] ? options.passRate * 0.4 : options.passRate;
    rows.push({
      scenarioId: pick(r, scenarios),
      modelVersion,
      projectId: pick(r, projects),
      y: r() < rate ? 1 : 0,
      occurredAtMs: 1_000 + i,
    });
  }
  return rows;
}

function allUniqueRows(r: () => number, count: number): OfflineRow[] {
  const rows: OfflineRow[] = [];
  for (let i = 0; i < count; i++) {
    rows.push({
      scenarioId: `s${i % 2}`,
      modelVersion: `m${i}`,
      projectId: `p${i}`,
      y: r() < 0.6 ? 1 : 0,
      occurredAtMs: 1_000 + i,
    });
  }
  return rows;
}

    out(`  shape perf (6 models x 8 projects, 400 rows): production median ${median(times).toFixed(1)} ms/report (7 reps)`);
    out(`  perf reps: ${times.map((t) => t.toFixed(1)).join(" ")}`);
  }

  // Instrumented pool table on the perf shape (3 runs; medians reported).
  const rows = perfRows();
  const expected = fitLogitAdditive(rows, PERF_OPTIONS);
  const runs: PoolReading[] = [];
  for (let rep = 0; rep < 3; rep++) {
    STATS.reset();
    const actual = fitInst(rows, PERF_OPTIONS);
    if (rep === 0) {
      compareReports("R21C-perf-fixture", expected, actual);
    }
    runs.push(snapshotStats());
  }
  const med = (key: keyof PoolReading): number => median(runs.map((r) => r[key]));
  const total = med("totalMs");
  const solveMs = med("solveMs");
  const irlsOther = med("irlsMs") - solveMs;
  const rest = total - med("irlsMs") - med("onProbMs") - med("apcMs") - med("samplingMs") - med("tailMs");
  APC_POOL_MEDIAN = med("apcMs");
  out(`instrumented pool table (perf shape, median of 3 instrumented runs; timers only, bitwise-checked):`);
  out(`  total                 ${total.toFixed(1)} ms`);
  out(`  solveSymmetric        ${solveMs.toFixed(1)} ms  (${(100 * solveMs / total).toFixed(1)}%)  [${med("solveCalls")} calls]`);
  out(`  irls minus solve      ${irlsOther.toFixed(1)} ms  (${(100 * irlsOther / total).toFixed(1)}%)  [${med("irlsIterations")} iterations, ${med("rowVisits")} row visits, ${med("sigmoidIrls")} irls sigmoids]`);
  out(`  onProbabilitiesFor    ${med("onProbMs").toFixed(1)} ms  (${(100 * med("onProbMs") / total).toFixed(1)}%)  [${med("onProbSigmoids")} sigmoid+dot]`);
  out(`  APC (all sweeps)      ${med("apcMs").toFixed(1)} ms  (${(100 * med("apcMs") / total).toFixed(1)}%)  [${med("apcActivePairs")} active pairs, ${med("apcCalls")} calls]`);

  out(`  outside pools (build/supports/keys/base copies) ${rest.toFixed(1)} ms  (${(100 * rest / total).toFixed(1)}%)`);
  out(`  fits ${med("fits")}  draws successful=${med("successfulDraws")} collapsed=${med("collapsedDraws")} singular=${med("singularDraws")}`);

/* -------- scenario 4: YMIX outcome-mix composition dose response -------- *
 * NEW completeness axis (C-first; composition family: R20-F outcome-stream
 * composition, R20-J input-class composition dose face precedents in other
 * slices). Question never measured in C: every prior round froze the
 * outcome composition at passRate=0.6 (SCALEX's size doses included). Does
 * ANY outcome composition, at frozen size and shape, inflate a size-closed
 * sub-band pool over the ±35 ms band — or reach the frozen pools through
 * any channel other than the ITERX-frozen iteration count?
 *
 * Dose grid: passRate ∈ {0.05, 0.30, 0.60 anchor, 0.80, 0.95} at the
 * frozen perf shape (4 scenarios, 6 models, 8 projects, 400 rows) and
 * bootstrap=200. The fixture consumes exactly 4 rng draws per row
 * regardless of passRate, so the categorical triples — hence the design,
 * supports, keys, p, and the APC active-pair structure — are bitwise
 * identical across doses; only y flips. Both facts are asserted per dose
 * (design-freeze witness + per-dose bitwise fidelity of the instrumented
 * lane against the production import). Per dose: 2 uninstrumented
 * production reps cross-check the total; 2 instrumented reps (mean) give
 * the per-pool readings, per-dose iteration counts, and the derived
 * per-iteration price of the frozen pools. No lane is built and no closed
 * duel is re-run. */

interface MixResult {
  readonly passRate: number;
  readonly ones: number;
  readonly prodMs: number[];
  readonly pools: PoolReading[];
  readonly p: number;
  readonly diagnosis: string;
}

function mixFixture(passRate: number): OfflineRow[] {
  const r = fixtureRng(0xbeef);
  return randomRows(r, { scenarios: 4, models: 6, projects: 8, rows: 400, passRate });
}

function runMix(passRate: number, anchorRows: readonly OfflineRow[]): MixResult {
  const rows = mixFixture(passRate);
  // Design-freeze witness: identical categorical triples vs the anchor.
  let tripleMatches = 0;
  for (let i = 0; i < rows.length; i++) {
    const a = anchorRows[i]!;
    const b = rows[i]!;
    if (a.scenarioId === b.scenarioId && a.modelVersion === b.modelVersion && a.projectId === b.projectId) {
      tripleMatches += 1;
    }
  }
  check(`R21C-ymix[${passRate}].design-freeze`, tripleMatches === rows.length);
  const ones = rows.reduce((acc, r) => acc + r.y, 0);

  const expected = fitLogitAdditive(rows, PERF_OPTIONS);
  const prodMs: number[] = [];
  for (let rep = 0; rep < 2; rep++) {
    const t0 = performance.now();
    fitLogitAdditive(rows, PERF_OPTIONS);
    prodMs.push(performance.now() - t0);
  }
  const pools: PoolReading[] = [];
  for (let rep = 0; rep < 2; rep++) {
    STATS.reset();
    const actual = fitInst(rows, PERF_OPTIONS);
    if (rep === 0) compareReports(`R21C-ymix[${passRate}]`, expected, actual);
    pools.push(snapshotStats());
  }
  const baseRows: Row[] = rows.map((r) => ({
    scenarioId: r.scenarioId, modelVersion: r.modelVersion, projectId: r.projectId, y: r.y
  }));
  const p = buildDesign(baseRows).names.length;
  return { passRate, ones, prodMs, pools, p, diagnosis: expected.diagnosis };
}

function meanOf(values: number[]): number {
  return values.reduce((a, b) => a + b, 0) / values.length;
}

function scenarioYmix(): void {
  out(`probe YMIX (outcome-mix / class-balance composition dose response; NEW axis, not a lane):`);
  const anchorRows = perfRows();
  const doses = [0.05, 0.3, 0.6, 0.8, 0.95];
  const results: MixResult[] = [];
  for (const passRate of doses) {
    results.push(runMix(passRate, anchorRows));
  }
  const poolKeys = [
    ["solve", (r: PoolReading) => r.solveMs],
    ["irls-other", (r: PoolReading) => r.irlsMs - r.solveMs],
    ["onProb", (r: PoolReading) => r.onProbMs],
    ["APC", (r: PoolReading) => r.apcMs],
    ["sampling", (r: PoolReading) => r.samplingMs],
    ["tail", (r: PoolReading) => r.tailMs],
    ["rest", (r: PoolReading) => r.totalMs - r.irlsMs - r.onProbMs - r.apcMs - r.samplingMs - r.tailMs],
  ] as const;
  for (const res of results) {
    const m = (f: (r: PoolReading) => number): number => meanOf(res.pools.map(f));
    const first = res.pools[0]!;
    out(
      `  dose passRate=${res.passRate.toFixed(2)}${res.passRate === 0.6 ? " (anchor)" : ""}: ones=${res.ones}/400 p=${res.p} ` +
      `fits=${first.fits} iters=${first.irlsIterations} successful=${first.successfulDraws} collapsed=${first.collapsedDraws} ` +
      `singular=${first.singularDraws} apcPairs=${first.apcActivePairs} diagnosis=${res.diagnosis}; ` +
      `prod ${res.prodMs.map((t) => t.toFixed(1)).join("/")} ms; inst total ${m((r) => r.totalMs).toFixed(1)} ms | ` +
      poolKeys.map(([name, f]) => `${name} ${m(f).toFixed(1)}`).join(" | ")
    );
  }
  // Composition-response arithmetic: (a) worst SIZE-CLOSED sub-band pool
  // reading across every dose vs the 35 ms band (solve/irls are frozen
  // pools and the APC combination is floor-anchored by the R14-C rule at
  // the production shape — their rows are descriptive only, exactly like
  // SCALEX's); (b) per-iteration price of the frozen pools per dose
  // (composition-invariance witness: composition reaches the frozen pools
  // only through the iteration count).
  out(`  composition arithmetic (35 ms band; size-closed sub-band pools only):`);
  for (const [name, f] of poolKeys) {
    if (name === "solve" || name === "irls-other" || name === "APC") continue;
    let worst = 0;
    let worstDose = 0;
    for (const res of results) {
      const v = meanOf(res.pools.map(f));
      if (v > worst) {
        worst = v;
        worstDose = res.passRate;
      }
    }
    out(`    ${name}: worst ${worst.toFixed(1)} ms at passRate=${worstDose.toFixed(2)} (${worst < 35 ? "sub-band" : "OVER BAND"}, x${(35 / worst).toFixed(1)} below 35)`);
  }
  {
    const f = poolKeys[3][1];
    const values = results.map((res) => meanOf(res.pools.map(f)));
    out(
      `    APC (descriptive; closure floor-anchored, not size-based): ` +
      results.map((res, i) => `${res.passRate.toFixed(2)}:${values[i]!.toFixed(1)}`).join(" ") +
      ` ms — active-pair structure frozen at ${results[0]!.pools[0]!.apcActivePairs} pairs across doses`
    );
  }
  for (const [name, f] of [poolKeys[0], poolKeys[1]] as const) {
    const prices = results.map((res) => {
      const v = meanOf(res.pools.map(f));
      const iters = res.pools[0]!.irlsIterations;
      return (v / iters) * 1e3; // µs per iteration
    });
    const lo = Math.min(...prices);
    const hi = Math.max(...prices);
    out(
      `    ${name} per-iteration price (µs/iter across doses): ` +
      results.map((res, i) => `${res.passRate.toFixed(2)}:${prices[i]!.toFixed(1)}`).join(" ") +
      ` — spread ${(100 * (hi - lo) / lo).toFixed(1)}% (composition reaches the frozen pools only through the iteration count)`
    );
  }
}

/* --------------------------------- main --------------------------------- */

scenarioEquivalence();
scenarioDeterminism();
scenarioProfile();
scenarioYmix();

}
