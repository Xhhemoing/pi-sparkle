async function driveCell(spec: CellSpec, index: number): Promise<CellOut> {
  const digests: string[] = [];
  let out: CellOut | undefined;
  for (let rep = 0; rep < REPS; rep++) {
    const stateRoot = mkdtempSync(join(tmpdir(), `r27g-joinface-c${index}-`));
    try {
      const deps = {
        stateRoot, router: router(), now: () => T0,
        generateId: sequenceGenerator(String(index + 1)),
        pause: createFilePauseController(stateRoot, () => T0)
      };
      const phases: Phase[] = [];

      resetCounts();
      let started = performance.now();
      const startOut = await startFlowchartRun(deps, {
        projectRoot, flowchart: spec.flowchart,
        objective: "joinface drive", childResults: spec.childResults
      });
      phases.push({ label: "start", counts: countsLine(), wallMs: performance.now() - started });
      if (startOut.status !== spec.expectStartStatus) {
        throw new Error(`${spec.name}: start status ${startOut.status} != ${spec.expectStartStatus}`);
      }

      let ledgerNote = "-";
      for (const step of spec.injects ?? []) {
        resetCounts();
        started = performance.now();
        const injectOut = await injectFlowchartRun(deps, startOut.runId, step.request);
        phases.push({ label: "inject", counts: countsLine(), wallMs: performance.now() - started });
        if (injectOut.status !== step.expectStatus) {
          throw new Error(`${spec.name}: inject status ${injectOut.status} != ${step.expectStatus}`);
        }
        const snap = injectOut.snapshot as unknown as Rec;
        const nodes = snap.nodes as Record<string, { state: string }>;
        const ledger = snap.ledger as { consecutiveStalls: number; isBlocked: boolean };
        ledgerNote =
          `after-inject j=${nodes.j!.state} supervisor-status=${snap.status} ` +
          `ledger{stalls=${ledger.consecutiveStalls} blocked=${ledger.isBlocked}}`;
        witness(`    INJECT ${spec.name} :: ${step.note} -> ${ledgerNote}`);
      }

      let finalOut = startOut;
      if (spec.resume !== undefined) {
        resetCounts();
        started = performance.now();
        finalOut = await resumeFlowchartRun(deps, startOut.runId,
          spec.resume.withResults !== undefined ? { childResults: spec.resume.withResults } : {});
        phases.push({ label: "resume", counts: countsLine(), wallMs: performance.now() - started });
        if (finalOut.status !== spec.resume.expectStatus) {
          throw new Error(`${spec.name}: resume status ${finalOut.status} != ${spec.resume.expectStatus}`);
        }
      }

      const dir = join(stateRoot, "runtime", "runs", startOut.runId);
      const normalize = normalizeFactory(stateRoot);
      const eventsRaw = await readFile(join(dir, "events.jsonl"), "utf8");
      const checkpointRaw = await readFile(join(dir, "checkpoint.json"), "utf8");
      digests.push(createHash("sha256").update(normalize(eventsRaw)).update(normalize(checkpointRaw)).digest("hex"));

      const nodes = finalOut.snapshot.nodes as unknown as Record<string, { state: string }>;
      for (const [id, fate] of Object.entries(spec.expectFates)) {
        if (nodes[id]!.state !== fate) {
          throw new Error(`${spec.name}: node ${id} fate ${nodes[id]!.state} != ${fate}`);
        }
      }
      out = {
        name: spec.name,
        phases,
        eventTypes: eventsRaw.trim().split("\n").map((line) => (JSON.parse(line) as { type: string }).type).join(","),
        carry: Object.keys(spec.expectFates).map((id) => `${id}=${nodes[id]!.state}`).join(" "),
        ledgerNote,
        digest: digests[digests.length - 1]!,
        bytes: checkpointRaw.length
      };
    } finally {
      rmSync(stateRoot, { recursive: true, force: true });
    }
  }
  if (!digests.every((d) => d === digests[0])) throw new Error(`${spec.name}: unstable artifacts`);
  return out!;
}
