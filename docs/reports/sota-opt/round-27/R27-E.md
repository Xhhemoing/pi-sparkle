## 2. 新轴 RVL：TASK_RESULT 裁决格面

### 2.1 定义与盲区

受控变量：**一条 TASK_RESULT 落在 (outcome × verification) 裁决格的
哪一格**。格集 = `TASK_OUTCOMES`（SUCCESS/PARTIAL/FAILURE/CANCELLED，
v1.ts:47）× `VERIFICATION_KINDS`（PASSED/FAILED/UNOBSERVED，
v1.ts:50）= 12 格。消费该格的裁决函数恰三个：

- **V1** `scoreTaskResult`（`signals.ts:48-55`）：6 腿梯 → score。
  第 1 腿析取（FAILURE ∨ FAILED → 15）使其 **outcome 主导**；
  尾腿 `return 70` 枚举内不可达（R11-E 已记，维持不立项）。
- **V2** `taskSuccessFromResult`（`task-success.ts:26-34`）：
  criterion 落笔命运。首腿 PARTIAL/CANCELLED 一票否决 → 仅
  {SUCCESS,FAILURE}×{PASSED,FAILED} 4 格落笔。
- **V3** `outcomeKindFromResult`（`from-episode.ts:291-298`）：
  行铸造命运，**纯 verification**（outcome 形参死置，R10-E「死形参」
  已记）；FAIL 行再经切片内调用位点（`from-episode.ts:218-226`）付
  `classifyTaskFailure`（`routing/failure-class.ts:22-46`，实现体
  切片外）取 failureClass 腿。

盲区证明（为什么 27 轮没人控过这一格）：R17-E 新轴 B 的价表把
TASK_RESULT 当**一个**事件类（1.12–1.18 µs/事件），类内 12 格从未
分开；S9-E-3 的 12 组合枚举是**裸 (V1,V2) 函数对**的等价性 fuzz
（为分派合一淘汰案作证，对价平坦 ~27ns/call——不过铸造位点、不看
V3、无下游命运）；R24-E JB 在 outcomes 铸造位点分的是 **join 命运**
（no-route / 守卫拦死 / mint），其 die-unobs / mint-pass /
mint-fail+classify 三格只是 verification 列在固定 outcome 下的标本，
outcome 维与 classify 腿从未变动；R25-E CG 头注释显式钉死命运
（"Fate pinned: every arm mints exactly 1 outcome (PASS...)")只动
链态。**没有任何一轮以裁决格为受控变量、没有任何一轮画过跨铸造
一致性矩阵、没有任何一轮把 classify 税分解到腿。**

/* Verbatim in-lock bandit build replica (R2-E S2-E-3 reference form). */
function banditBuild(previous: BanditState | undefined, signals: readonly ObservedSignal[]): BanditState {
  const arms = new Set(previous?.arms ?? []);
  for (const signal of signals) {
    if (signal.modelId !== undefined && signal.modelId.trim() !== "") arms.add(signal.modelId);
  }
  const armList = [...arms];
  let state = createBanditState(armList);
  if (previous !== undefined) {
    const pulls: Record<string, number> = {};
    const rewardSum: Record<string, number> = {};
    for (const arm of armList) {
      pulls[arm] = previous.pulls[arm] ?? 0;
      rewardSum[arm] = previous.rewardSum[arm] ?? 0;
    }
    state = {
      arms: armList,
      pulls,
      rewardSum,
      explorationsUsed: previous.explorationsUsed,
      highRiskExplorations: previous.highRiskExplorations
    };
  }
  for (const signal of signals) {
    if (signal.criterion !== "taskSuccess") continue;
    if (signal.source === "user" || signal.kind === "human" || signal.kind !== "deterministic") continue;
    if (signal.modelId === undefined || !state.arms.includes(signal.modelId)) continue;
    if (signal.outcomeKind === "PASS") state = recordReward(state, signal.modelId, 1);
    else if (signal.outcomeKind === "FAIL") state = recordReward(state, signal.modelId, 0);
  }
  return state;
}
