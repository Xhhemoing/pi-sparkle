function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

}

  const cli2 = bench(() => assignTasks({ catalog: catalog2, tasks: cli, prior }), 2000);
  const cli7 = bench(() => assignTasks({ catalog: catalogCfg, tasks: cli, prior, learned: configuredLearned() }), 2000);
  console.log(
    `matrix prior REAL-SCALE live face N=30: M=2+prior ${(cli2 * 1e3).toFixed(1)}us | M=7+prior+learned ${(cli7 * 1e3).toFixed(1)}us per call`
  );
  const models7 = createModelRouter(catalogCfg).config.models;
  const plan7 = planAssignmentPolicy(models7, catalogCfg.models.map((m) => m.id));
  const perPick = bench(() => {
    pickFromPublicPrior(prior, "edit", plan7.assignableModels);
  }, 20000);
  console.log(`prior unit cost: pickFromPublicPrior(edit, M=7 pool, ${prior.scores.length} rows) = ${(perPick * 1e6).toFixed(0)}ns per call`);
  conclusions.push(
    `prior-cells real-scale-live-face M=7+prior+learned=${(cli7 * 1e3).toFixed(1)}us hypothetical-N2000 M=7+prior per-eval=${(p7 * 2).toFixed(1)}ms per-pick=${(perPick * 1e6).toFixed(0)}ns`
  );
}

/* ============================================================
 * §0d NEW measurement cell: batch-cardinality axis bottom + per-call
 * batch-invariant prelude intercept face.
 * Production state (grep evidence, re-taken this round): planFromContract
 * emits C in [1,5] children per track batch; smartChildPlan routes N<=30
 * (N=0 reachable via role filtering); eval-routing is the only N>=2000
 * caller. The prelude (createModelRouter + catalogIds map +
 * planAssignmentPolicy + limits) is structurally learned/prior-free
 * (assign.ts:62-68 vs :100-105).
 * ============================================================ */
{
  const NS = [0, 1, 2, 3, 5, 8, 16, 30] as const;
  const repsFor = (n: number): number =>
    n === 0 ? 30000 : n <= 3 ? 20000 : n <= 8 ? 10000 : n <= 16 ? 5000 : 2500;
  const catalog2 = catalogFromPrimary({ primaryModelId: "premium" });
  const catalogCfg = configuredCatalog();
  const prior = realisticPrior();
  const stream = genReplayTasks(mulberry32(0xb44b01), 30);
  /** Homogeneous median task: plain MEDIUM/edit, no keyword hits beyond IMPLEMENT_RE. */
  const median: AssignableTask = {
    taskId: "tsk_median" as AssignableTask["taskId"],
    role: "implementer",
    objective: "Implement retry logic for the ledger sync"
  };
  const homogeneous = (n: number): AssignableTask[] =>
    Array.from({ length: n }, (_, i) => ({ ...median, taskId: `tsk_${i.toString().padStart(6, "0")}` as AssignableTask["taskId"] }));

  interface Arm {
    readonly label: string;
    readonly catalog: ModelRouterConfig;
    readonly learned: LearnedRoutingPolicy | undefined;
    readonly withPrior: boolean;
  }
  const arms: readonly Arm[] = [
    { label: "M=2 default, none", catalog: catalog2, learned: undefined, withPrior: false },
    { label: "M=2 default, learned", catalog: catalog2, learned: learnedM2(), withPrior: false },
    { label: "M=7 configured, learned", catalog: catalogCfg, learned: configuredLearned(), withPrior: false },
    { label: "M=7 configured, learned+prior", catalog: catalogCfg, learned: configuredLearned(), withPrior: true }
  ];
  const runArm = (arm: Arm, tasksFor: (n: number) => readonly AssignableTask[], leg: string): { intercept: number; slope: number } => {
    const points: { n: number; us: number }[] = [];
    for (const n of NS) {
      const tasks = tasksFor(n);
      const cost = bench(
        () =>
          assignTasks({
            catalog: arm.catalog,
            tasks,
            ...(arm.learned !== undefined ? { learned: arm.learned } : {}),
            ...(arm.withPrior ? { prior } : {})
          }),
        repsFor(n)
      );
      points.push({ n, us: cost * 1e3 });
    }
    // Least-squares affine fit over N>=1 (N=0 kept as the direct intercept probe).
    const fitPoints = points.filter((p) => p.n >= 1);
    const k = fitPoints.length;
    const sumN = fitPoints.reduce((acc, p) => acc + p.n, 0);
    const sumY = fitPoints.reduce((acc, p) => acc + p.us, 0);
    const sumNN = fitPoints.reduce((acc, p) => acc + p.n * p.n, 0);
    const sumNY = fitPoints.reduce((acc, p) => acc + p.n * p.us, 0);
    const slope = (k * sumNY - sumN * sumY) / (k * sumNN - sumN * sumN);
    const intercept = (sumY - slope * sumN) / k;
    const meanY = sumY / k;
    const ssTot = fitPoints.reduce((acc, p) => acc + (p.us - meanY) ** 2, 0);
    const ssRes = fitPoints.reduce((acc, p) => acc + (p.us - (intercept + slope * p.n)) ** 2, 0);
    const r2 = ssTot === 0 ? 1 : 1 - ssRes / ssTot;
    const maxResidual = Math.max(...fitPoints.map((p) => Math.abs(p.us - (intercept + slope * p.n))));
    const n0 = points[0]!.us;
    console.log(
      `nbottom ${leg} [${arm.label}]: ${points.map((p) => `N=${p.n}:${p.us.toFixed(2)}us`).join(" ")}`
    );
    console.log(
      `nbottom ${leg} [${arm.label}] fit(N>=1): intercept=${intercept.toFixed(2)}us slope=${slope.toFixed(2)}us/task R2=${r2.toFixed(4)} maxResid=${maxResidual.toFixed(2)}us | measured N=0=${n0.toFixed(2)}us`
    );
    return { intercept: n0, slope };
  };

  // Leg A: homogeneous repeated median task (cleanest affine read).
  const homo: Record<string, { intercept: number; slope: number }> = {};
  for (const arm of arms) homo[arm.label] = runArm(arm, homogeneous, "homogeneous");
  // Leg B: production-faithful heterogeneous prefixes of the replay stream.
  for (const arm of arms) runArm(arm, (n) => stream.slice(0, n), "prefix");

  // Prelude stage micro-attribution (batch-invariant work only).
  for (const [label, catalog] of [["M=2", catalog2], ["M=7", catalogCfg]] as const) {
    const tRouter = bench(() => createModelRouter(catalog), 20000);
    const router = createModelRouter(catalog);
    const tIds = bench(() => catalog.models.map((model) => model.id), 200000);
    const catalogIds = catalog.models.map((model) => model.id);
    const tPlan = bench(() => planAssignmentPolicy(router.config.models, catalogIds), 100000);
    const sum = (tRouter + tIds + tPlan) * 1e3;
    console.log(
      `nbottom stage micro [${label}]: createModelRouter=${(tRouter * 1e3).toFixed(2)}us idsMap=${(tIds * 1e6).toFixed(0)}ns planAssignmentPolicy=${(tPlan * 1e6).toFixed(0)}ns | staged sum=${sum.toFixed(2)}us`
    );
  }

  // N=0 input-arm invariance: learned/prior are per-task layers, structurally
  // absent from the prelude; the four input arms should price identically
  // (any residual delta is the input-object shape PIC class, R25-B family).
  {
    const learned = configuredLearned();
    const t0none = bench(() => assignTasks({ catalog: catalogCfg, tasks: [] }), 30000);
    const t0learned = bench(() => assignTasks({ catalog: catalogCfg, tasks: [], learned }), 30000);
    const t0prior = bench(() => assignTasks({ catalog: catalogCfg, tasks: [], prior }), 30000);
    const t0both = bench(() => assignTasks({ catalog: catalogCfg, tasks: [], learned, prior }), 30000);
    console.log(
      `nbottom N=0 input-arm invariance [M=7]: none=${(t0none * 1e3).toFixed(2)}us learned=${(t0learned * 1e3).toFixed(2)}us prior=${(t0prior * 1e3).toFixed(2)}us learned+prior=${(t0both * 1e3).toFixed(2)}us`
    );
  }

  // Track-loop-faithful cell: the exact plan.ts child shapes (C in {1,2,5}).
  {
    const objective = "Add retry logic to the ledger sync and cover it with tests";
    const full: AssignableTask[] = [
      { taskId: "tsk_plan01" as AssignableTask["taskId"], role: "planner", objective: `Decompose and route: ${objective}` },
      { taskId: "tsk_scout1" as AssignableTask["taskId"], role: "scout", objective: `Survey the workspace and constraints for: ${objective}` },
      { taskId: "tsk_impl01" as AssignableTask["taskId"], role: "implementer", objective: `Implement: ${objective}` },
      { taskId: "tsk_revw01" as AssignableTask["taskId"], role: "reviewer", objective: `Review the change against: ${objective}` },
      { taskId: "tsk_test01" as AssignableTask["taskId"], role: "tester", objective: `Verify tests for: ${objective}` }
    ];
    const invest1 = [
      { taskId: "tsk_plan01" as AssignableTask["taskId"], role: "planner" as const, objective: `Decompose and route: ${objective}` }
    ];
    const invest2 = [
      invest1[0]!,
      { taskId: "tsk_scout1" as AssignableTask["taskId"], role: "scout" as const, objective: `Investigate without writing files: ${objective}` }
    ];
    const learned = configuredLearned();
    for (const [label, tasks] of [["C=1 investigation", invest1], ["C=2 investigation", invest2], ["C=5 full", full]] as const) {
      const noPrior = bench(() => assignTasks({ catalog: catalogCfg, tasks, learned }), 20000);
      const withPrior = bench(() => assignTasks({ catalog: catalogCfg, tasks, learned, prior }), 5000);
      console.log(
        `nbottom track-faithful [${label}] M=7+learned: ${(noPrior * 1e3).toFixed(2)}us | +prior: ${(withPrior * 1e3).toFixed(2)}us per call`
      );
      if (label === "C=5 full") {
        conclusions.push(
          `track-faithful C=5 M=7+learned=${(noPrior * 1e3).toFixed(2)}us (+prior=${(withPrior * 1e3).toFixed(2)}us)`
        );
      }
    }
  }

  // W2: batch-decomposition purity witness -- for each arm the batch output is
  // element-wise byte-identical to the fold of singleton calls.
  {
    const w2arms: readonly Arm[] = [
      arms[0]!,
      arms[2]!,
      arms[3]!
    ];
    for (const arm of w2arms) {
      const batch = assignTasks({
        catalog: arm.catalog,
        tasks: stream,
        ...(arm.learned !== undefined ? { learned: arm.learned } : {}),
        ...(arm.withPrior ? { prior } : {})
      });
      let identical = batch.length === stream.length;
      for (let i = 0; i < stream.length && identical; i += 1) {
        const singleton = assignTasks({
          catalog: arm.catalog,
          tasks: [stream[i]!],
          ...(arm.learned !== undefined ? { learned: arm.learned } : {}),
          ...(arm.withPrior ? { prior } : {})
        });
        identical = singleton.length === 1 && JSON.stringify(singleton[0]) === JSON.stringify(batch[i]);
      }
      check(`nbottom W2 [${arm.label}]: batch(N=30) element-wise byte-identical to fold of singleton calls`, identical);
    }
  }

  // W5: N=1 singleton equals element 0 of the N=30 batch at the M=2 corner.
  {
    const batch = assignTasks({ catalog: catalog2, tasks: stream });
    const single = assignTasks({ catalog: catalog2, tasks: [stream[0]!] });
    check(
      "nbottom W5 [M=2]: N=1 singleton byte-identical to element 0 of the N=30 batch",
      JSON.stringify(single[0]) === JSON.stringify(batch[0])
    );
  }

  // W1: double-run byte identity at the most loaded arm.
  {
    const runA = JSON.stringify(assignTasks({ catalog: catalogCfg, tasks: stream, learned: configuredLearned(), prior }));
    const runB = JSON.stringify(assignTasks({ catalog: catalogCfg, tasks: stream, learned: configuredLearned(), prior }));
    check("nbottom W1 [M=7 learned+prior, N=30]: double run byte-identical", runA === runB);
  }

  // W3: N=0 fail-closed prelude ordering. assignTasks({tasks: []}) returns []
  // AND still validates the catalog: a duplicate-id catalog and an empty
  // catalog both throw DomainValidationError even with zero tasks. An "N=0
  // early return before the prelude" would silently accept invalid catalogs.
  {
    const empty = assignTasks({ catalog: catalog2, tasks: [] });
    check("nbottom W3a: assignTasks(tasks=[]) returns an empty array", Array.isArray(empty) && empty.length === 0);
    const dupCatalog: ModelRouterConfig = {
      policyVersion: "sim-r27b-dup",
      models: [...catalog2.models, { ...catalog2.models[0]! }]
    };
    let dupThrew = false;
    try {
      assignTasks({ catalog: dupCatalog, tasks: [] });
    } catch (error) {
      dupThrew = error instanceof DomainValidationError;
    }
    check("nbottom W3b: duplicate-id catalog throws DomainValidationError even at N=0", dupThrew);
    let emptyThrew = false;
    try {
      assignTasks({ catalog: { policyVersion: "sim-r27b-empty", models: [] }, tasks: [] });
    } catch (error) {
      emptyThrew = error instanceof DomainValidationError;
    }
    check("nbottom W3c: empty catalog throws DomainValidationError even at N=0", emptyThrew);
  }

  conclusions.push(
    `nbottom homogeneous intercept/slope per-call: ${arms.map((arm) => `${arm.label}: N0=${homo[arm.label]!.intercept.toFixed(2)}us slope=${homo[arm.label]!.slope.toFixed(2)}us/task`).join(" | ")}`
  );
}

console.log(`\nCONCLUSIONS: ${conclusions.join(" | ")}`);
if (failures > 0) {
  console.error(`\n${failures} check(s) FAILED`);
  process.exit(1);
}
console.log("ALL CHECKS PASSED (measurement harness + batch-cardinality-bottom fidelity witnesses; corpora seeded + deterministic)");
