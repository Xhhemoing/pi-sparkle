### 2.2 与近邻轴的区分（换名自查核心）

- **vs R16-D payload 形态轴（P 轴）**：P 轴 = 固定几何、字节剂量

- **vs R21-D 状态增长方向面**：其「深度」是**注册表状态**维度
  （同一身份的版本链长，O(n²) restore 拷贝）；本轴深度是**单个
  内容 blob 内部**的 JSON 树深度，状态维度全程恒定（每臂恒
  1 身份 × 1 baseline + 1 candidate）。
- **vs R17-D 拒绝路径普查**：其普查对象是 `DomainValidationError`

  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;

interface Census {
  objects: number;
  keys: number;
  arrays: number;
  elements: number;
  strings: number;
  maxDepth: number;
  keyBytes: number;
}
function censusOf(root: unknown): Census {
  const c: Census = { objects: 0, keys: 0, arrays: 0, elements: 0, strings: 0, maxDepth: 0, keyBytes: 0 };
  const stack: { v: unknown; d: number }[] = [{ v: root, d: 1 }];
  while (stack.length > 0) {
    const { v, d } = stack.pop() as { v: unknown; d: number };
    if (d > c.maxDepth) c.maxDepth = d;
    if (Array.isArray(v)) {
      c.arrays += 1;
      c.elements += v.length;
      for (const e of v) stack.push({ v: e, d: d + 1 });
    } else if (isRecordReplica(v)) {
      c.objects += 1;
      for (const [k, child] of Object.entries(v)) {
        c.keys += 1;
        c.keyBytes += k.length;
        stack.push({ v: child, d: d + 1 });
      }
    } else if (typeof v === "string") {
      c.strings += 1;
    }
  }
  return c;
}

/** Envelope: identical schema-valid policy skeleton across all arms; rider
 * carries the dosed geometry; zpad (escape-free) absorbs residual bytes so
 * every arm serializes to exactly `target` bytes. */
function armContent(rider: unknown, target: number): string {
  const mk = (pad: string): string =>
    JSON.stringify({
      primaryModelId: "premium",
      avoid: [{ modelId: "fast", family: "edit", reason: "deterministic FAIL on edit replay" }],
      prefer: [{ family: "test", modelId: "fast" }],

  const base = mk("");

  if (deficit < 0) throw new Error(`arm overflows target by ${-deficit} bytes`);
  const content = mk("x".repeat(deficit));
  if (content.length !== target) throw new Error(`pad mismatch: ${content.length} != ${target}`);
  return content;
}

function maxOkDepth(test: (d: number) => boolean, cap: number): number {
  let lo = 1;
  if (!test(lo)) return 0;
  let hi = 2;
  while (hi <= cap && test(hi)) {
    lo = hi;
    hi *= 2;
  }
  if (hi > cap) return lo; // no cliff found up to cap
  // invariant: test(lo) ok, test(hi) fails
  while (hi - lo > 1) {
    const mid = Math.floor((lo + hi) / 2);
    if (test(mid)) lo = mid;
    else hi = mid;
  }
  return lo;
}

function fitLeastSquares(points: { x: number; y: number }[]): { slope: number; intercept: number } {
  const n = points.length;
  const sx = points.reduce((s, p) => s + p.x, 0);
  const sy = points.reduce((s, p) => s + p.y, 0);
  const sxx = points.reduce((s, p) => s + p.x * p.x, 0);
  const sxy = points.reduce((s, p) => s + p.x * p.y, 0);
  const slope = (n * sxy - sx * sy) / (n * sxx - sx * sx);
  return { slope, intercept: (sy - slope * sx) / n };
}
