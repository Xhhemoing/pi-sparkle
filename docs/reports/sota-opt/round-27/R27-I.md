```javascript
    let err = "";
    child.stderr.on("data", (d) => { err += d; });
    child.on("close", (code) => {
      if (fd !== undefined) closeSync(fd);
      if (code !== 0) reject(new Error(`probe exit ${code}`));
      else resolve(JSON.parse(err.trim().split("\n").pop()));
    });
  });
}
```

### A.5 行为见证（/tmp/r27i-bench/equiv.mjs，双引擎各执行一次）

```javascript
// R27-I behavioral witnesses for the output-channel disposition face.
//   W1 disposition byte-invariance: the byte stream a bearing command emits
//      must be identical whether stdout is a pipe or a regular file
//      (normalized for fresh run/episode ids) — structurally guaranteed by
//      zero isTTY/color sensitivity in the slice, witnessed end-to-end.
//   W2 exit parity: every disposition (pipe / file / ignore=/dev/null) exits
//      with the same code on the same command.
//   W3 EPIPE quiet-exit: a bearing command piped into an immediately-exiting
//      consumer must exit 0 via the slice-owned handler (main.ts L1489-1495)
//      instead of crashing with an unhandled EPIPE.
//   W4 stderr independence: the error channel keeps its bytes and exit code
//      when stdout is wired to /dev/null.
import { spawnSync } from "node:child_process";
import { mkdirSync, rmSync, cpSync, openSync, closeSync, readFileSync } from "node:fs";
import { join } from "node:path";

const CLI = "/workspace/dist/cli/main.js";
const FIX = "/tmp/r27i-bench/fixtures";
const WORK = "/tmp/r27i-bench/equiv-work";
const PROJ = join(FIX, "project");
const CHILDREN = join(FIX, "children.json");

function normalize(text, stateDir) {
  return text
    .replaceAll(stateDir, "<STATE>")
    .replace(/run_[0-9a-f-]+/g, "run_X")
    .replace(/ep_[0-9a-f-]+/g, "ep_X")
    .replace(/evd_[0-9a-f-]+/g, "evd_X")
    .replace(/cnd_[0-9a-f-]+/g, "cnd_X")
    .replace(/agt_[0-9a-f-]+/g, "agt_X");
}
function cleanEnv() {
  const env = { ...process.env };
  delete env.PI_PROVIDER; delete env.PI_MODEL; delete env.PI_API_KEY;
  delete env.PI_FAST_MODEL; delete env.SPARKLE_AUTO_ADAPT; delete env.PI_THINKING_LEVEL;
  return env;
}
function fresh(state, tag) {
  const dir = join(WORK, tag);
  rmSync(dir, { recursive: true, force: true });
  cpSync(join(FIX, state), dir, { recursive: true });
  return dir;
}
const childrenArgs = (s) => ["run", "--project", PROJ, "--objective", "bench objective", "--state-root", s, "--children", CHILDREN];
const flowArgs = (s) => ["run", "--project", PROJ, "--objective", "bench objective", "--state-root", s, "--flowchart", join(FIX, "flow-flat64.json"), "--results", join(FIX, "results-flat64.json")];

rmSync(WORK, { recursive: true, force: true });
mkdirSync(WORK, { recursive: true });
let checks = 0, failures = 0;
function assertEq(a, b, label) {
  checks++;
  if (a !== b) { failures++; console.error(`FAIL ${label}\n  a=${String(a).slice(0, 300)}\n  b=${String(b).slice(0, 300)}`); }
}

// W1 + W2: pipe vs file byte-invariance and exit parity, 3 runs x bearing cells.
for (let i = 0; i < 3; i++) {
  for (const [state, argsOf, tag] of [["state-builtin", childrenArgs, "cb"], ["state-custom", childrenArgs, "cc"], ["state-custom", flowArgs, "fc"]]) {
    const dirP = fresh(state, `${tag}-pipe-${i}`);
    const rp = spawnSync(process.execPath, [CLI, ...argsOf(dirP)], { env: cleanEnv(), encoding: "utf8", timeout: 180000 });
    const dirF = fresh(state, `${tag}-file-${i}`);
    const outPath = join(WORK, `${tag}-file-${i}.out`);
    const fd = openSync(outPath, "w");
    const rf = spawnSync(process.execPath, [CLI, ...argsOf(dirF)], { env: cleanEnv(), timeout: 180000, stdio: ["ignore", fd, "pipe"] });
    closeSync(fd);
    assertEq(rp.status, rf.status, `W2 exit parity pipe/file ${tag} run${i}`);
    assertEq(normalize(rp.stdout, dirP), normalize(readFileSync(outPath, "utf8"), dirF), `W1 byte-invariance ${tag} run${i}`);
    const dirN = fresh(state, `${tag}-null-${i}`);
    const rn = spawnSync(process.execPath, [CLI, ...argsOf(dirN)], { env: cleanEnv(), timeout: 180000, stdio: ["ignore", "ignore", "ignore"] });
    assertEq(rp.status, rn.status, `W2 exit parity pipe/null ${tag} run${i}`);
  }
}

// W3: EPIPE quiet-exit through the slice-owned handler. The consumer exits
// immediately, so by the time the child (a bearing children run, ~100 ms of
// work before its first write) writes, the pipe's read end is closed.
for (let i = 0; i < 3; i++) {
  const dir = fresh("state-builtin", `epipe-${i}`);
  const cmd = `${process.execPath} ${CLI} run --project '${PROJ}' --objective 'bench objective' --state-root '${dir}' --children '${CHILDREN}' | true; exit \${PIPESTATUS[0]}`;
  const r = spawnSync("bash", ["-c", cmd], { env: cleanEnv(), encoding: "utf8", timeout: 180000 });
  assertEq(r.status, 0, `W3 EPIPE quiet-exit run${i}`);
  assertEq(/EPIPE|Error/.test(r.stderr), false, `W3 no crash text on stderr run${i}`);
}

// W4: stderr independence — unknown command keeps its stderr bytes and exit
// code when stdout goes to /dev/null.
{
  const r1 = spawnSync(process.execPath, [CLI, "definitely-not-a-command"], { env: cleanEnv(), encoding: "utf8", timeout: 60000 });
  const r2 = spawnSync(process.execPath, [CLI, "definitely-not-a-command"], { env: cleanEnv(), encoding: "utf8", timeout: 60000, stdio: ["ignore", "ignore", "pipe"] });
  assertEq(r1.status, r2.status, "W4 exit parity stdout-null");
  assertEq(r1.stderr, r2.stderr, "W4 stderr bytes identical");
}

if (failures > 0) process.exit(1);
```
