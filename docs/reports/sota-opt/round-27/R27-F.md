 *            object access paths; ns/element for plain vs proxied array

function onceMs(fn: () => void): number {
  const t0 = process.hrtime.bigint();
  fn();
  const t1 = process.hrtime.bigint();
  return Number(t1 - t0) / 1e6;
}

function median(xs: readonly number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  return s[Math.floor(s.length / 2)] as number;
}

/** Historical fixture shape: `ep_` + zero-padded base36 index, L=11. */
function makePackedPopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

/* ------------------------- access-path constructors ----------------------- */
/** Same keys, same order, same pinned values — but every read is a getter. */
function accessorize<T extends object>(source: T): T {
  const out: Record<string, unknown> = {};
  for (const key of Object.keys(source)) {
    const pinned = (source as Record<string, unknown>)[key];
    Object.defineProperty(out, key, {
      get: () => pinned,
      enumerable: true,
      configurable: true,
    });
  }
  return out as T;
}

/** Own-empty instance; every field lives on the prototype. */
function protoHost<T extends object>(source: T): T {
  return Object.create(source) as T;
}

/** Same data fields, null-prototype receiver. */
function nullProto<T extends object>(source: T): T {
  return Object.assign(Object.create(null) as T, source);
}

/** Transparent no-trap membrane; [[Get]] falls through to the same target. */
function membrane<T extends object>(target: T): T {
  return new Proxy(target, {}) as T;
}

/** Accessor-backed plan: plan + 4 nested config objects getter-ized. */
function buildAccessorPlan(): ExperimentPlan {
  const base = makePlanFields([...basePopulation]);
  const nested = {
    ...base,
    thresholds: accessorize(base.thresholds),
    budget: accessorize(base.budget),
    randomization: accessorize(base.randomization),
    stopPolicy: accessorize(base.stopPolicy),
  };
  return accessorize(nested);
}

/** Null-prototype plan: plan + 4 nested config objects on null-proto bases. */
function buildNullProtoPlan(): ExperimentPlan {
  const base = makePlanFields([...basePopulation]);
  return nullProto({
    ...base,
    thresholds: nullProto(base.thresholds),
    budget: nullProto(base.budget),
    randomization: nullProto(base.randomization),
    stopPolicy: nullProto(base.stopPolicy),
  });
}

function cleanStream(): ExperimentOutcome[] {
  return baseHashes.map((h) => ({
    episodeHash: h,
    utility: 0.5,
    costUsd: 0.01,
    guardrailBreached: false,
  }));
}

/* ----------------------------- e2e arm builders --------------------------- */
interface ArmA {
  readonly name: string;
  readonly plan: ExperimentPlan;
  readonly stream: readonly ExperimentOutcome[];
  /** Membrane-pipeline arms re-wrap the state handed into every op. */
  readonly wrapState: ((state: ShadowState) => ShadowState) | undefined;
}

function buildE2eArm(name: string): ArmA {
  switch (name) {
    case "BASE":
      return {
        name,
        plan: makePlanFields([...basePopulation]),
        stream: sharedStream,
        wrapState: undefined,
      };
    case "ACC-PLAN":
      return { name, plan: buildAccessorPlan(), stream: sharedStream, wrapState: undefined };
    case "PROTO-PLAN":
      return {
        name,
        plan: protoHost(makePlanFields([...basePopulation])),
        stream: sharedStream,
        wrapState: undefined,
      };
    case "NULLPROTO-PLAN":
      return { name, plan: buildNullProtoPlan(), stream: sharedStream, wrapState: undefined };
    case "PXY-PLAN":
      return {
        name,
        plan: membrane(makePlanFields([...basePopulation])),
        stream: sharedStream,
        wrapState: undefined,
      };
    case "PXY-POP":
      return {
        name,
        plan: makePlanFields(membrane([...basePopulation])),
        stream: sharedStream,
        wrapState: undefined,
      };
    case "PXY-OUT": {
      // Same outcome TARGET objects as the shared stream; only the membrane
      // moves. The runner aliases these into the ledger, so every later
      // dedupe/cost scan re-reads through the membrane (Θ(A²) reads).
      const stream = sharedStream.map((o) => membrane(o));
      return {
        name,
        plan: makePlanFields([...basePopulation]),
        stream,
        wrapState: undefined,
      };
    }
    case "PXY-STATE":
      return {
        name,
        plan: makePlanFields([...basePopulation]),
        stream: sharedStream,
        wrapState: (state) => membrane(state),
      };
    default:
      throw new Error(`unknown e2e arm: ${name}`);
  }
}

function buildValidatePlan(name: string): ExperimentPlan {
  switch (name) {
    case "BASE":
      return makePlanFields([...basePopulation]);
    case "ACC-PLAN":
      return buildAccessorPlan();
    case "PROTO-PLAN":
      return protoHost(makePlanFields([...basePopulation]));
    case "NULLPROTO-PLAN":
      return buildNullProtoPlan();
    case "PXY-PLAN":
      return membrane(makePlanFields([...basePopulation]));
    case "PXY-POP":
      return makePlanFields(membrane([...basePopulation]));
    default:
      throw new Error(`unknown validate arm: ${name}`);
  }
}

/** Arm-construction fidelity witnesses (printed and asserted per child). */
function assertArmFidelity(name: string, plan: ExperimentPlan): void {
  if (name === "ACC-PLAN") {
    const desc = Object.getOwnPropertyDescriptor(plan, "experimentId");
    if (desc?.get === undefined) throw new Error("ACC-PLAN: experimentId is not an accessor");
  }
  if (name === "PROTO-PLAN") {
    if (Object.hasOwn(plan, "experimentId") || plan.experimentId !== "exp_r27fbase") {
      throw new Error("PROTO-PLAN: fields are not prototype-hosted");
    }
  }
  if (name === "NULLPROTO-PLAN" && Object.getPrototypeOf(plan) !== null) {
    throw new Error("NULLPROTO-PLAN: prototype is not null");
  }
  if (name === "PXY-PLAN" && !nodeTypes.isProxy(plan)) {
    throw new Error("PXY-PLAN: plan is not a proxy");
  }
  if (name === "PXY-POP" && !nodeTypes.isProxy(plan.population)) {
    throw new Error("PXY-POP: population is not a proxy");
  }
  if (name === "PXY-POP" && !Array.isArray(plan.population)) {
    throw new Error("PXY-POP: proxied population must still satisfy Array.isArray");
  }
}

/** Production interleave (evaluateProposalShadow shape), clock pinned to 0. */
function runInterleaved(arm: ArmA): ShadowState {
  const runner = createShadowRunner(arm.plan);
  let state = runner.start(0);
  for (const outcome of arm.stream) {
    if (state.halted) {
      break;
    }
    const input = arm.wrapState === undefined ? state : arm.wrapState(state);
    state = runner.assign(input, outcome.episodeHash, 0);
    const mid = arm.wrapState === undefined ? state : arm.wrapState(state);
    state = runner.recordOutcome(mid, outcome, 0);
  }
  return state;
}

function ledgerDigest(state: ShadowState): string {
  return stableStringify({
    assignments: state.assignments,
    outcomes: state.outcomes,
    guardrailBreaches: state.guardrailBreaches,
    halted: state.halted,
    haltReason: state.haltReason,
  });
}

/* ------------------- restore-microscope state construction ----------------- */
function buildActiveFullLedger(): {
  runner: ReturnType<typeof createShadowRunner>;
  state: ShadowState;
} {
  const plan: ExperimentPlan = {
    ...makePlanFields([...basePopulation]),
    budget: { maxAssignments: 2 * A, maxWallClockMs: 1e12 },
  };
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (const outcome of sharedStream) {
    state = runner.assign(state, outcome.episodeHash, 0);
    state = runner.recordOutcome(state, outcome, 0);
  }
  if (state.halted || state.assignments.length !== A) {
    throw new Error("restore microscope: expected an active full-ledger state");
  }
  return { runner, state };
}

function buildRestoreVariant(name: string, state: ShadowState): ShadowState {
  switch (name) {
    case "mem":
      return state;
    case "ACC-STATE":
      return accessorize({ ...state });
    case "PXY-STATE":
      return membrane(state);
    case "PXY-LEDGER":
      return {
        ...state,
        assignments: membrane([...state.assignments]),
        outcomes: membrane([...state.outcomes]),
      };
    default:
      throw new Error(`unknown restore arm: ${name}`);
  }
}

/* ----------------------- isolated child-cell driver ------------------------ */
function runChild(cell: string, arm: string): string[] {
  const child = spawnSync(
    process.execPath,
    ["--import", "tsx", process.argv[1] as string, `--cell=${cell}`, `--arm=${arm}`],
    { encoding: "utf8", timeout: 300_000 }
  );
  if (child.status !== 0) {
    return [`CHILD ${cell} ${arm} FAILED status=${String(child.status)} ${child.stderr.slice(0, 300)}`];
  }
  return child.stdout.trim().split("\n");
}

function runIsolatedCell(title: string, cell: string, arms: readonly string[]): Map<string, number> {
  process.stdout.write(`${title}\n`);
  const values = new Map<string, number>();
  for (const arm of arms) {
    for (const line of runChild(cell, arm)) {
      process.stdout.write(`  ${line}\n`);
      const match = line.match(/^CHILD \S+ \S+ ([0-9.]+)/);
      if (match !== null) {
        values.set(arm, Number(match[1]));
      }
    }
  }
  return values;
}
