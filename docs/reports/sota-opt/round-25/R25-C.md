## 4. 重命名检查（派单强制项）

- **AGEX vs SCALEX（R20-C）**：SCALEX 动**输入体量**（基数与占据随
  生成器联动）；AGEX 冻结体量（A 面 400 行、B 面 4,000 obs / 480
  episode 每剂量恒等断言）只动时间字段的值。对象（体量 vs 时间）、
  剂量单元、结论域（体量有越线点 vs 时间无越线点+归因面逐位零）
  均不同。非重命名，SCALEX 未重测（零尺寸剂量触碰）。
- **AGEX vs YMIX（R21-C）**：YMIX 冻结三元组结构只翻 y 向量；AGEX
  的 A 面每剂量断言 (三元组, y) 序列逐位冻结、B 面每键 20 PASS +
  5 FAIL[model] 组成冻结——y/结局边际零移动。B 面后验质量的移动
  经由**权重**（时间）而非**计数**（组成）：betacf 结构硬顶
  16,000/分位只能被组成/体量接近，这是 YMIX/SCALEX 的裁决域，
  本轮明示不越界。非重命名，YMIX 未重测。
- **AGEX vs SEEDX（R22-C）/ ORDX（R23-C）/ OCCX（R24-C）**：SEEDX
  只动 options.seed（本轮 A 面种子冻结 20260818、B 面无 RNG）；
  ORDX 置换**同一个固定多重集**（本轮每剂量发射序冻结，各剂量是
  **不同多重集**——时间字段不同——在其自身规范序上单次定价，
  零置换决斗，OCCX 先例）；OCCX 动行在格胞上的联合分布（本轮 A 面
  (三元组,y) 序列逐位冻结 ⇒ 占据轮廓逐位冻结；B 面键结构冻结）。
  R24-C「五自由度定价完毕」的裁决域是 (三元组,y) 多重集——AGEX
  定价的是该裁决域**之外**的第六坐标（时间字段），恰好是五轴全部
  冻结在 `1_000+i` 上的那一维；六轴两两正交。非重命名，三轴未重测。
- **AGEX vs FITQ（R19-C）/ ITERX（R16-C）**：FITQ 是锚点上 IRLS
  拟合粒度墙钟分布普查、ITERX 是锚点上 IRLS 迭代计数普查——两者

  迭代 + 阈值控制流——**零字符串字节移动**（A/B 面层名/键名集合
  每剂量断言冻结）、零扫描纹理、零长度移动。字段类不同、机制不同、
  面不同：非移植、非重命名。（若未来有人试图以「数值 value-domain」

function buildDesign(rows: readonly Row[]): Design {
  const scenarios = [...new Set(rows.map((r) => r.scenarioId))];
  const models = [...new Set(rows.map((r) => r.modelVersion))];
  const projects = [...new Set(rows.map((r) => r.projectId))];
  const dropLast = (levels: readonly string[]): string[] =>
    levels.slice(0, Math.max(0, levels.length - 1));
  const scenarioLevels = dropLast(scenarios);
  const modelLevels = dropLast(models);
  const projectLevels = dropLast(projects);

  const pairCounts = new Map<string, number>();
  for (const row of rows) {
    const key = `${row.modelVersion}|${row.projectId}`;
    pairCounts.set(key, (pairCounts.get(key) ?? 0) + 1);
  }
  const interactionPairs = [...pairCounts.entries()]
    .filter(([, n]) => n >= INTERACTION_MIN_N)
    .map(([key]) => key);

  const names = [
    "intercept",
    ...scenarioLevels.map((s) => `a:${s}`),
    ...modelLevels.map((m) => `u:${m}`),
    ...projectLevels.map((p) => `v:${p}`),
    ...interactionPairs.map((key) => `w:${key}`)
  ];
  const columnIndex = new Map(names.map((name, index) => [name, index] as const));
  const interactionPairSet = new Set(interactionPairs);

  const referenceLevels: Array<{ factor: "a" | "u" | "v"; name: string }> = [];
  const lastModel = models[models.length - 1];
  const lastProject = projects[projects.length - 1];
  if (lastModel !== undefined) referenceLevels.push({ factor: "u", name: lastModel });
  if (lastProject !== undefined) referenceLevels.push({ factor: "v", name: lastProject });

  return {
    names,
    columnIndex,
    referenceLevels,
    build(row: Row, skip?: string): number[] {
      const vec = new Array<number>(names.length).fill(0);
      vec[0] = 1;
      const set = (name: string): void => {
        if (name === skip) return;
        const index = columnIndex.get(name);
        if (index !== undefined && index > 0) vec[index] = 1;
      };
      if (row.scenarioId !== scenarios[scenarios.length - 1]) set(`a:${row.scenarioId}`);
      if (row.modelVersion !== models[models.length - 1]) set(`u:${row.modelVersion}`);
      if (row.projectId !== projects[projects.length - 1]) set(`v:${row.projectId}`);
      const pairKey = `${row.modelVersion}|${row.projectId}`;
      if (interactionPairSet.has(pairKey)) set(`w:${pairKey}`);
      return vec;
    }
  };
}

/* verbatim landed solveSymmetric (lin-alg.ts) — plain, no timers */
function solvePlain(a: readonly (readonly number[])[], b: readonly number[]): number[] | null {
  const n = b.length;
  if (a.length !== n || a.some((row) => row.length !== n)) {
    throw new DomainValidationError("solveSymmetric requires a square matrix matching b");
  }
  const m: number[][] = a.map((row) => [...row]);
  const x: number[] = [...b];
  const eps = 1e-12;

  for (let col = 0; col < n; col++) {
    let pivotRow = col;
    let pivotAbs = Math.abs(m[col]![col]!);
    for (let row = col + 1; row < n; row++) {
      const abs = Math.abs(m[row]![col]!);
      if (abs > pivotAbs) {
        pivotAbs = abs;
        pivotRow = row;
      }
    }
    if (pivotAbs < eps) return null;
    if (pivotRow !== col) {
      const tmp = m[col]!;
      m[col] = m[pivotRow]!;
      m[pivotRow] = tmp;
      const tb = x[col]!;
      x[col] = x[pivotRow]!;
      x[pivotRow] = tb;
    }
    const colArr = m[col]!;
    const pivot = colArr[col]!;
    const xCol = x[col]!;
    for (let row = col + 1; row < n; row++) {
      const rowArr = m[row]!;
      const factor = rowArr[col]! / pivot;
      if (factor === 0) continue;
      let k = col;
      const stop = n - 3;
      for (; k < stop; k += 4) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
        rowArr[k + 1] = rowArr[k + 1]! - factor * colArr[k + 1]!;
        rowArr[k + 2] = rowArr[k + 2]! - factor * colArr[k + 2]!;
        rowArr[k + 3] = rowArr[k + 3]! - factor * colArr[k + 3]!;
      }
      for (; k < n; k++) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
      }
      x[row] = x[row]! - factor * xCol;
    }
  }

  const solution = new Array<number>(n).fill(0);
  for (let row = n - 1; row >= 0; row--) {
    const rowArr = m[row]!;
    let sum = x[row]!;
    for (let k = row + 1; k < n; k++) sum -= rowArr[k]! * solution[k]!;
    const diag = rowArr[row]!;
    if (Math.abs(diag) < eps) return null;
    solution[row] = sum / diag;
  }
  for (const value of solution) {
    if (!Number.isFinite(value)) return null;
  }
  return solution;
}

            xtwz[a3] = xtwz[a3]! + wz;
            const r3 = xtwx[a3]!;
            r3[a0] = r3[a0]! + w; r3[a1] = r3[a1]! + w; r3[a2] = r3[a2]! + w; r3[a3] = r3[a3]! + w;
            break;
          }
          case 3: {
            const a0 = active[0]!;
            const a1 = active[1]!;
            const a2 = active[2]!;
            xtwz[a0] = xtwz[a0]! + wz;
            const r0 = xtwx[a0]!;
            r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w; r0[a2] = r0[a2]! + w;
            xtwz[a1] = xtwz[a1]! + wz;
            const r1 = xtwx[a1]!;
            r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w; r1[a2] = r1[a2]! + w;
            xtwz[a2] = xtwz[a2]! + wz;
            const r2 = xtwx[a2]!;
            r2[a0] = r2[a0]! + w; r2[a1] = r2[a1]! + w; r2[a2] = r2[a2]! + w;
            break;
          }
          case 2: {
            const a0 = active[0]!;
            const a1 = active[1]!;
            xtwz[a0] = xtwz[a0]! + wz;
            const r0 = xtwx[a0]!;
            r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w;
            xtwz[a1] = xtwz[a1]! + wz;
            const r1 = xtwx[a1]!;
            r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w;
            break;
          }
          default: {
            for (let ai = 0; ai < active.length; ai++) {
              const a = active[ai]!;
              xtwz[a] = xtwz[a]! + wz;
              const rowA = xtwx[a]!;
              for (let bi = 0; bi < active.length; bi++) {
                const b = active[bi]!;
                rowA[b] = rowA[b]! + w;
              }
            }
          }
        }
      }
      for (let d = 0; d < p; d++) xtwx[d]![d] = xtwx[d]![d]! + 1e-6;
      let next: number[] | null;
      if (instrumented) {
        const tSolve0 = performance.now();
        next = solvePlain(xtwx, xtwz);
        STATS.solveMs += performance.now() - tSolve0;
        STATS.solveCalls += 1;
      } else {
        next = solvePlain(xtwx, xtwz);
      }
      if (next === null) {
        if (instrumented) STATS.irlsMs += performance.now() - tIrls0;
        return { coefficients: null };
      }
      const delta = next.map((value, index) => value - beta[index]!);
      const l2 = Math.sqrt(delta.reduce((acc, d) => acc + d * d, 0));
      beta = next;
      if (!beta.every(Number.isFinite)) {
        if (instrumented) STATS.irlsMs += performance.now() - tIrls0;
        return { coefficients: null };
      }
      if (l2 < TOL) break;
    }
    if (instrumented) STATS.irlsMs += performance.now() - tIrls0;
    return { coefficients: beta };
  };
}

function dot(a: readonly number[], b: readonly number[]): number {
  let sum = 0;
  for (let i = 0; i < a.length; i++) sum += a[i]! * b[i]!;
  return sum;
}

function percentile(sortedValues: readonly number[], q: number): number {
  if (sortedValues.length === 0) return Number.NaN;
  const index = (sortedValues.length - 1) * q;
  const lo = Math.floor(index);
  const hi = Math.ceil(index);
  if (lo === hi) return sortedValues[lo]!;
  return sortedValues[lo]! * (hi - index) + sortedValues[hi]! * (index - lo);
}

/* betaQuantileLcb replica (posterior.ts verbatim) — keeps the replica lanes
 * self-contained exactly like R14-C..R23-C. */
function lnGamma(z: number): number {
  const g = 7;
  const c = [
    0.99999999999980993, 676.5203681218851, -1259.1392167224028, 771.32342877765313,
    -176.61502916214059, 12.507343278686905, -0.13857109526572012, 9.9843695780195716e-6,
    1.5056327351493116e-7
  ];
  if (z < 0.5) {
    return Math.log(Math.PI / Math.sin(Math.PI * z)) - lnGamma(1 - z);
  }
  const x = z - 1;
  let a = c[0]!;
  for (let i = 1; i < g + 2; i++) a += c[i]! / (x + i);
  const t = x + g + 0.5;
  return 0.5 * Math.log(2 * Math.PI) + (x + 0.5) * Math.log(t) - t + Math.log(a);
}

function lnBetaFunction(a: number, b: number): number {
  return lnGamma(a) + lnGamma(b) - lnGamma(a + b);
}

function betacf(a: number, b: number, x: number): number {
  const maxIt = 200;
  const eps = 3e-12;
  const qab = a + b;
  const qap = a + 1;
  const qam = a - 1;
  let c = 1;
  let d = 1 - (qab * x) / qap;
  if (Math.abs(d) < 1e-30) d = 1e-30;
  d = 1 / d;
  let h = d;
  for (let m = 1; m <= maxIt; m++) {
    const m2 = 2 * m;
    let aa = (m * (b - m) * x) / ((qam + m2) * (a + m2));
    d = 1 + aa * d;
    if (Math.abs(d) < 1e-30) d = 1e-30;
    c = 1 + aa / c;
    if (Math.abs(c) < 1e-30) c = 1e-30;
    d = 1 / d;
    h *= d * c;
    aa = (-(a + m) * (qab + m) * x) / ((a + m2) * (qap + m2));
    d = 1 + aa * d;
    if (Math.abs(d) < 1e-30) d = 1e-30;
    c = 1 + aa / c;
    if (Math.abs(c) < 1e-30) c = 1e-30;
    d = 1 / d;
    const del = d * c;
    h *= del;
    if (Math.abs(del - 1) < eps) break;
  }
  return h;
}

function regularizedIncompleteBeta(x: number, a: number, b: number, lnBeta: number): number {
  if (x <= 0) return 0;
  if (x >= 1) return 1;
  const prefix = Math.exp(a * Math.log(x) + b * Math.log(1 - x) - lnBeta);
  if (x < (a + 1) / (a + b + 2)) {
    return (prefix * betacf(a, b, x)) / a;
  }
  return 1 - (prefix * betacf(b, a, 1 - x)) / b;
}

function inverseRegularizedIncompleteBeta(p: number, a: number, b: number): number {
  const lnBeta = lnBetaFunction(a, b);
  let lo = 0;
  let hi = 1;
  for (let i = 0; i < 80; i++) {
    const mid = (lo + hi) / 2;
    if (regularizedIncompleteBeta(mid, a, b, lnBeta) > p) hi = mid;
    else lo = mid;
  }
  return (lo + hi) / 2;
}

function betaQuantileLcbReplica(posterior: { alpha: number; beta: number }, p = 0.05): number {
  if (!(p > 0 && p < 1)) return p <= 0 ? 0 : 1;
  if (posterior.alpha <= 0 || posterior.beta <= 0) return 0;
  return inverseRegularizedIncompleteBeta(p, posterior.alpha, posterior.beta);
}

function makeFit(irlsImpl: IrlsImpl, instrumented: boolean) {
  const onProb = (vectors: readonly number[][], coefficients: readonly number[]): number[] => {
    if (!instrumented) return vectors.map((vector) => sigmoid(dot(coefficients, vector)));
    const t0 = performance.now();
    const result = vectors.map((vector) => sigmoid(dot(coefficients, vector)));
    STATS.onProbMs += performance.now() - t0;
    STATS.onProbSigmoids += vectors.length;
    return result;
  };

  const apc = (
    design: Design,
    rows: readonly Row[],
    vectors: readonly number[][],
    coefficients: readonly number[],
    onProbabilities: readonly number[],
    column: string
  ): number => {
    const t0 = instrumented ? performance.now() : 0;
    if (instrumented) STATS.apcCalls += 1;
    let sum = 0;
    const columnIdx = design.columnIndex.get(column);
    if (columnIdx !== undefined && columnIdx !== 0) {
      for (let i = 0; i < rows.length; i++) {
        if (vectors[i]![columnIdx] === 0) continue;
        if (instrumented) STATS.apcActivePairs += 1;
        const on = onProbabilities[i]!;
        const offVector = vectors[i]!.slice();
        offVector[columnIdx] = 0;
        const off = sigmoid(dot(coefficients, offVector));
        sum += on - off;
      }
    }
    const result = rows.length === 0 ? 0 : sum / rows.length;
    if (instrumented) STATS.apcMs += performance.now() - t0;
    return result;
  };

  const pushValue = (map: Map<string, number[]>, key: string, value: number): void => {
    if (instrumented) STATS.drawsMapOps += 1;
    const list = map.get(key);
    if (list === undefined) map.set(key, [value]);
    else list.push(value);
  };

  return function fit(
    rows: readonly OfflineRow[],
    options?: { readonly maxIter?: number; readonly bootstrap?: number; readonly seed?: number }
  ): AttributionReport {
    const tTotal0 = instrumented ? performance.now() : 0;
    const maxIter = options?.maxIter ?? MAX_ITER_DEFAULT;
    const bootstrapDraws = options?.bootstrap ?? BOOTSTRAP_DEFAULT;
    const seed = options?.seed ?? SEED_DEFAULT;

    const tBase0 = instrumented ? performance.now() : 0;
    const baseRows: Row[] = rows.map((r) => ({
      scenarioId: r.scenarioId,
      modelVersion: r.modelVersion,
      projectId: r.projectId,
      y: r.y
    }));
    if (instrumented) STATS.baseCopyMs += performance.now() - tBase0;
    const effects: Array<{ name: string; point: number; lcb: number; ucb: number }> = [];

    if (baseRows.length === 0 || baseRows.every((r) => r.y === 0) || baseRows.every((r) => r.y === 1)) {
      return uncertainReport(baseRows.length, "INVALID_ESTIMATE: degenerate or empty design");
    }

    const design = buildDesign(baseRows);
    const vectors = baseRows.map((r) => design.build(r));
    const supports = computeSupports(vectors);
    const keys = canonicalRowKeys(baseRows);
    const fit = irlsImpl(design, baseRows, vectors, supports, keys, baseRows.length, maxIter);
    if (fit.coefficients === null) {
      return uncertainReport(baseRows.length, "INVALID_ESTIMATE: singular or non-finite Hessian");
    }

    const onProbabilities = onProb(vectors, fit.coefficients);
    const pointEffects = new Map<string, number>();
    for (const name of design.names) {
      if (name === "intercept") continue;
      pointEffects.set(
        name,
        apc(design, baseRows, vectors, fit.coefficients, onProbabilities, name)
      );
    }
    for (const ref of design.referenceLevels) {
      if (!pointEffects.has(`${ref.factor}:${ref.name}`)) {
        pointEffects.set(`${ref.factor}:${ref.name}`, 0);
      }
    }

    const random = rng(seed);
    const draws = new Map<string, number[]>();
    let successful = 0;
    for (let draw = 0; draw < bootstrapDraws; draw++) {
      const tSample0 = instrumented ? performance.now() : 0;
      const sample: Row[] = [];
      const sampleVectors: number[][] = [];
      const sampleSupports: number[][] = [];
      const sampleKeys: number[] = [];
      for (let i = 0; i < baseRows.length; i++) {
        const index = Math.floor(random() * baseRows.length);
        sample.push(baseRows[index]!);
        sampleVectors.push(vectors[index]!);
        sampleSupports.push(supports[index]!);
        sampleKeys.push(keys[index]!);
      }
      const collapsed = sample.every((r) => r.y === 0) || sample.every((r) => r.y === 1);
      if (instrumented) STATS.samplingMs += performance.now() - tSample0;
      if (collapsed) {
        if (instrumented) STATS.collapsedDraws += 1;
        continue;
      }
      const bootFit = irlsImpl(design, sample, sampleVectors, sampleSupports, sampleKeys, baseRows.length, maxIter);
      if (bootFit.coefficients === null) {
        if (instrumented) STATS.singularDraws += 1;
        continue;
      }
      successful += 1;
      if (instrumented) STATS.successfulDraws += 1;
      const sampleOnProbabilities = onProb(sampleVectors, bootFit.coefficients);
      for (const [name] of pointEffects.entries()) {
        const value = apc(
          design,
          sample,
          sampleVectors,
          bootFit.coefficients,
          sampleOnProbabilities,
          name
        );
        pushValue(draws, name, value);
      }
    }

    const tTail0 = instrumented ? performance.now() : 0;
    for (const [name, point] of pointEffects.entries()) {
      const values = draws.get(name) ?? [];
      if (successful < MIN_SUCCESSFUL_DRAWS || values.length < MIN_SUCCESSFUL_DRAWS) {
        return uncertainReport(baseRows.length, "INVALID_ESTIMATE: fewer than 20 successful bootstrap draws");
      }
      values.sort((a, b) => a - b);
      effects.push({
        name,
        point,
        lcb: percentile(values, 0.025),
        ucb: percentile(values, 0.975)
      });
    }

    const n = baseRows.length;
    const mean = baseRows.reduce((acc, r) => acc + r.y, 0) / n;
    const muPosterior = { alpha: 1 + n * mean, beta: 1 + n * (1 - mean) };
    const muLcb = betaQuantileLcbReplica(muPosterior, 0.05);
    const models = new Set(baseRows.map((r) => r.modelVersion)).size;
    const projects = new Set(baseRows.map((r) => r.projectId)).size;
    const ZERO_EPS = 0.005 * ATTRIBUTION_EFFECT;
    const containsZero = (e: { lcb: number; ucb: number }): boolean =>
      e.lcb <= ZERO_EPS && e.ucb >= -ZERO_EPS;
    const interactionsFor = (prefix: string): Array<{ name: string; lcb: number; ucb: number; point: number }> =>
      effects.filter((e) => e.name.startsWith(`w:${prefix}`));

    let diagnosis: AttributionReport["diagnosis"] = "uncertain";
    const scenarioHard = muLcb < QUALITY_FLOOR && models >= 2 && projects >= 3;
    if (
      effects.some(
        (e) => e.name.startsWith("u:") && e.lcb < -ATTRIBUTION_EFFECT && interactionsFor(modelOf(e.name)).every(containsZero)
      )
    ) {
      diagnosis = "model-problem";
    } else if (
      effects.some(
        (e) =>
          e.name.startsWith("v:") &&
          e.lcb < -ATTRIBUTION_EFFECT &&
          effects
            .filter((w) => w.name.startsWith("w:") && w.name.endsWith(`|${e.name.slice("v:".length)}`))
            .every(containsZero)
      )
    ) {
      diagnosis = "project-problem";
    } else if (effects.some((e) => e.name.startsWith("w:") && e.lcb < -ATTRIBUTION_EFFECT)) {
      diagnosis = "interaction-only";
    } else if (scenarioHard) {
      diagnosis = "scenario-hard";
    }

    const report: AttributionReport = {
      estimator: "logit-additive",
      rowsUsed: baseRows.length,
      effects,
      diagnosis,
      reason: diagnosis === "uncertain" ? "no effect beyond threshold or intervals too wide" : `${diagnosis} beyond the ${ATTRIBUTION_EFFECT} effect threshold`,
      writesActivePointer: false
    };
    if (instrumented) {
      STATS.tailMs += performance.now() - tTail0;
      STATS.totalMs += performance.now() - tTotal0;
    }
    return report;
  };
}

function compareReports(label: string, expected: AttributionReport, actual: AttributionReport): void {
  check(`${label}.estimator`, expected.estimator === actual.estimator);
  check(`${label}.rowsUsed`, expected.rowsUsed === actual.rowsUsed);

  check(`${label}.reason`, expected.reason === actual.reason);
  check(`${label}.writesActivePointer`, expected.writesActivePointer === actual.writesActivePointer);
  check(`${label}.effects.length`, expected.effects.length === actual.effects.length);
  if (expected.effects.length === actual.effects.length) {
    for (let i = 0; i < expected.effects.length; i++) {
      const e = expected.effects[i]!;
      const g = actual.effects[i]!;

      check(`${label}.effects[${i}].point`, Object.is(e.point, g.point));

      check(`${label}.effects[${i}].ucb`, Object.is(e.ucb, g.ucb));
    }
  }
}

function fixtureRng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function pick<T>(r: () => number, items: readonly T[]): T {
  return items[Math.floor(r() * items.length)]!;
}

function randomRows(
  r: () => number,
  options: {
    scenarios: number;
    models: number;
    projects: number;
    rows: number;
    passRate: number;
    pipeInModel?: boolean;
  }
): OfflineRow[] {
  const scenarios = Array.from({ length: options.scenarios }, (_, i) => `fam${i}|role${i % 2}`);
  const models = Array.from({ length: options.models }, (_, i) =>
    options.pipeInModel === true && i === 0 ? "weird|model-0" : `model-${i}`
  );
  const projects = Array.from({ length: options.projects }, (_, i) => `prj_${i}`);
  const rows: OfflineRow[] = [];
  for (let i = 0; i < options.rows; i++) {
    const modelVersion = pick(r, models);
    const rate = modelVersion === models[0] ? options.passRate * 0.4 : options.passRate;
    rows.push({
      scenarioId: pick(r, scenarios),
      modelVersion,
      projectId: pick(r, projects),
      y: r() < rate ? 1 : 0,
      occurredAtMs: 1_000 + i,
    });
  }
  return rows;
}

      y: r() < 0.6 ? 1 : 0,

  return rows;
}

/* ------------------------------- face B -------------------------------- *
 * Instrumented verbatim replicas of outcomes.ts / posterior.ts / r1.ts at
 * the 183df9b-landed state. The replica lane's ONLY additions are BSTAT
 * counter increments and performance.now() reads at call boundaries; no
 * float operation, operand, or ordering differs. The replica lane is
 * validated bitwise against the production prepareR1Observations /
 * estimateForKey / routeR1 imports AT EVERY DOSE before its numbers are
 * used. */

const BSTAT = {
  powCalls: 0,
  betacfCalls: 0,
  betacfIters: 0,
  ribCalls: 0,
  lnGammaCalls: 0,
  quantileCalls: 0,
  estimateComputes: 0,
  cacheHits: 0,
  prepareMs: 0,
  updMs: 0,
  lcbMs: 0,
  reset(): void {
    this.powCalls = 0; this.betacfCalls = 0; this.betacfIters = 0; this.ribCalls = 0;
    this.lnGammaCalls = 0; this.quantileCalls = 0; this.estimateComputes = 0;
    this.cacheHits = 0; this.prepareMs = 0; this.updMs = 0; this.lcbMs = 0;
  },
};

function outcomeKeyB(parts: {
  readonly taskFamily: string;
  readonly role: string;
  readonly modelVersion: string;
  readonly featureVersion: string;
}): string {
  return `${parts.taskFamily}|${parts.role}|${parts.modelVersion}|${parts.featureVersion}`;
}

function isInformativeOutcomeB(observation: OutcomeObservation): boolean {
  return observation.outcome === "PASS" || observation.outcome === "FAIL";
}

function observationsForR1B(observations: readonly OutcomeObservation[]): OutcomeObservation[] {
  return observations.filter((row) => {
    if (row.criterion !== "taskSuccess" || !isInformativeOutcomeB(row)) return false;
    if (row.source !== "deterministic-check") return false;
    if (row.outcome === "FAIL" && row.failureClass !== "model") return false;
    return true;
  });
}

function updatePosteriorB(
  config: PosteriorConfig,
  observations: readonly OutcomeObservation[],
  nowMs: number
): { alpha: number; beta: number } {
  const t0 = performance.now();
  let alpha = config.priorAlpha;
  let beta = config.priorBeta;
  for (const observation of observations) {
    if (!isInformativeOutcomeB(observation)) continue;
    const ageMs = Math.max(0, nowMs - observation.occurredAtMs);
    BSTAT.powCalls += 1;
    const weight = Math.pow(2, -ageMs / config.halfLifeMs);
    if (observation.outcome === "PASS") alpha += weight;
    else beta += weight;
  }
  BSTAT.updMs += performance.now() - t0;
  return { alpha, beta };
}

function posteriorMeanB(posterior: { alpha: number; beta: number }): number {
  const total = posterior.alpha + posterior.beta;
  return total > 0 ? posterior.alpha / total : 0;
}

function posteriorVarianceB(posterior: { alpha: number; beta: number }): number {
  const total = posterior.alpha + posterior.beta;
  if (total <= 0) return 0;
  return (posterior.alpha * posterior.beta) / (total * total * (total + 1));
}

function lnGammaB(z: number): number {
  BSTAT.lnGammaCalls += 1;
  const g = 7;
  const c = [
    0.99999999999980993, 676.5203681218851, -1259.1392167224028, 771.32342877765313,
    -176.61502916214059, 12.507343278686905, -0.13857109526572012, 9.9843695780195716e-6,
    1.5056327351493116e-7
  ];
  if (z < 0.5) {
    return Math.log(Math.PI / Math.sin(Math.PI * z)) - lnGammaB(1 - z);
  }
  const x = z - 1;
  let a = c[0]!;
  for (let i = 1; i < g + 2; i++) a += c[i]! / (x + i);
  const t = x + g + 0.5;
  return 0.5 * Math.log(2 * Math.PI) + (x + 0.5) * Math.log(t) - t + Math.log(a);
}

function lnBetaFunctionB(a: number, b: number): number {
  return lnGammaB(a) + lnGammaB(b) - lnGammaB(a + b);
}

function betacfB(a: number, b: number, x: number): number {
  BSTAT.betacfCalls += 1;
  const maxIt = 200;
  const eps = 3e-12;
  const qab = a + b;
  const qap = a + 1;
  const qam = a - 1;
  let c = 1;
  let d = 1 - (qab * x) / qap;
  if (Math.abs(d) < 1e-30) d = 1e-30;
  d = 1 / d;
  let h = d;
  for (let m = 1; m <= maxIt; m++) {
    BSTAT.betacfIters += 1;
    const m2 = 2 * m;
    let aa = (m * (b - m) * x) / ((qam + m2) * (a + m2));
    d = 1 + aa * d;
    if (Math.abs(d) < 1e-30) d = 1e-30;
    c = 1 + aa / c;
    if (Math.abs(c) < 1e-30) c = 1e-30;
    d = 1 / d;
    h *= d * c;
    aa = (-(a + m) * (qab + m) * x) / ((a + m2) * (qap + m2));
    d = 1 + aa * d;
    if (Math.abs(d) < 1e-30) d = 1e-30;
    c = 1 + aa / c;
    if (Math.abs(c) < 1e-30) c = 1e-30;
    d = 1 / d;
    const del = d * c;
    h *= del;
    if (Math.abs(del - 1) < eps) break;
  }
  return h;
}

function ribB(x: number, a: number, b: number, lnBeta: number): number {
  BSTAT.ribCalls += 1;
  if (x <= 0) return 0;
  if (x >= 1) return 1;
  const prefix = Math.exp(a * Math.log(x) + b * Math.log(1 - x) - lnBeta);
  if (x < (a + 1) / (a + b + 2)) {
    return (prefix * betacfB(a, b, x)) / a;
  }
  return 1 - (prefix * betacfB(b, a, 1 - x)) / b;
}

function invB(p: number, a: number, b: number): number {
  const lnBeta = lnBetaFunctionB(a, b);
  let lo = 0;
  let hi = 1;
  for (let i = 0; i < 80; i++) {
    const mid = (lo + hi) / 2;
    if (ribB(mid, a, b, lnBeta) > p) hi = mid;
    else lo = mid;
  }
  return (lo + hi) / 2;
}

function betaQuantileLcbB(posterior: { alpha: number; beta: number }, p = 0.05): number {
  BSTAT.quantileCalls += 1;
  if (!(p > 0 && p < 1)) return p <= 0 ? 0 : 1;
  if (posterior.alpha <= 0 || posterior.beta <= 0) return 0;
  return invB(p, posterior.alpha, posterior.beta);
}

function lowerConfidenceBoundB(
  config: PosteriorConfig,
  posterior: { alpha: number; beta: number },
  kind: "normal" | "beta-quantile" = config.lcbKind
): number {
  const t0 = performance.now();
  let result: number;
  if (kind === "beta-quantile") {
    result = betaQuantileLcbB(posterior, 0.05); // oneSidedTail(config.lcbZ) === 0.05 in production
  } else {
    const mean = posteriorMeanB(posterior);
    const sd = Math.sqrt(posteriorVarianceB(posterior));
    result = Math.max(0, mean - config.lcbZ * sd);
  }
  BSTAT.lcbMs += performance.now() - t0;
  return result;
}

function weightedSampleSizeB(config: PosteriorConfig, posterior: { alpha: number; beta: number }): number {
  return posterior.alpha + posterior.beta - config.priorAlpha - config.priorBeta;
}

function isWellSampledB(config: PosteriorConfig, posterior: { alpha: number; beta: number }): boolean {
  return weightedSampleSizeB(config, posterior) >= config.minSamples;
}

function groupObservationsByKeyB(
  observations: readonly OutcomeObservation[]
): ReadonlyMap<string, readonly OutcomeObservation[]> {
  const groups = new Map<string, OutcomeObservation[]>();
  for (const observation of observations) {
    const key = outcomeKeyB(observation);
    const group = groups.get(key);
    if (group === undefined) {
      groups.set(key, [observation]);
    } else {
      group.push(observation);
    }
  }
  return groups;
}

function prepareR1ObservationsB(observations: readonly OutcomeObservation[]): PreparedB {
  const t0 = performance.now();
  const prepared: PreparedB = {
    byKey: groupObservationsByKeyB(observationsForR1B(observations)),
    estimateCache: new Map(),
  };
  BSTAT.prepareMs += performance.now() - t0;

function estimateFingerprintB(config: PosteriorConfig, nowMs: number): string {
  return `${nowMs}|${config.priorAlpha}|${config.priorBeta}|${config.halfLifeMs}|${config.minSamples}|${config.lcbZ}|${config.lcbKind}`;
}

function estimateForKeyB(
  prepared: PreparedB,
  config: PosteriorConfig,
  nowMs: number,
  parts: { readonly taskFamily: string; readonly role: string; readonly modelVersion: string; readonly featureVersion: string }
): R1KeyEstimateB {
  const key = outcomeKeyB(parts);
  const fingerprint = estimateFingerprintB(config, nowMs);
  const hit = prepared.estimateCache.get(key);
  if (hit !== undefined && hit.fingerprint === fingerprint) {
    BSTAT.cacheHits += 1;
    return hit.estimate;
  }
  BSTAT.estimateComputes += 1;
  const posterior = updatePosteriorB(config, prepared.byKey.get(key) ?? [], nowMs);
  const estimate: R1KeyEstimateB = {
    key,
    alpha: posterior.alpha,
    beta: posterior.beta,
    mean: posteriorMeanB(posterior),
    lcb: lowerConfidenceBoundB(config, posterior),
    samples: weightedSampleSizeB(config, posterior),
    wellSampled: isWellSampledB(config, posterior),
  };
  prepared.estimateCache.set(key, { fingerprint, estimate });
  return estimate;
}

function costOfB(r0: R0Decision, modelId: string): number {
  return r0.candidates.find((row) => row.modelId === modelId)?.estimatedCostUsd ?? Number.POSITIVE_INFINITY;
}

function tierIndexB(r0: R0Decision, modelId: string): number {
  if (r0.selection === modelId) return 0;
  const fallback = r0.fallbacks.indexOf(modelId);
  return fallback === -1 ? Number.MAX_SAFE_INTEGER : fallback + 1;
}

function cheaperEstimateB(left: R1EstimateB, right: R1EstimateB, r0: R0Decision): R1EstimateB {
  const leftCost = costOfB(r0, left.modelId);
  const rightCost = costOfB(r0, right.modelId);
  if (rightCost < leftCost) return right;
  if (rightCost > leftCost) return left;
  const leftIndex = tierIndexB(r0, left.modelId);
  const rightIndex = tierIndexB(r0, right.modelId);
  return rightIndex < leftIndex ? right : left;
}

function routeR1B(input: R1InputB): R1DecisionB {
  const config: PosteriorConfig = { ...DEFAULT_POSTERIOR_CONFIG_B, ...input.config };
  const qualityFloor = input.qualityFloor ?? DEFAULT_QUALITY_FLOOR_B;
  const hysteresisMargin = input.hysteresisMargin ?? DEFAULT_HYSTERESIS_MARGIN_B;
  const request = input.r0.request;
  void request;

  if (input.r0.selection === undefined) {
    return {
      selection: undefined,
      estimates: [],
      reason: "no eligible model; R0 refused",
      exploratory: false,
      fallback: false,
    };
  }

  const prepared = input.observations;
  const tierIds = [input.r0.selection, ...input.r0.fallbacks];
  const estimates: R1EstimateB[] = [];
  const modelsById = new Map(input.models.map((m) => [m.modelId, m]));

  for (const modelId of tierIds) {
    const model = modelsById.get(modelId);
    const estimate = estimateForKeyB(prepared, config, input.nowMs, {
      taskFamily: input.r0.request.taskFamily,
      role: input.role,
      modelVersion: model?.version ?? modelId,
      featureVersion: input.featureVersion,
    });
    estimates.push({ modelId, ...estimate });
  }

  const sampled = estimates.filter((e) => e.wellSampled);
  if (sampled.length === 0) {
    return {
      selection: input.r0.selection,
      estimates,
      reason: "no well-sampled estimate; conservative R0 baseline selected",
      exploratory: false,
      fallback: true,
    };
  }

  const aboveFloor = sampled.filter((e) => e.lcb >= qualityFloor);
  if (aboveFloor.length === 0) {
    return {
      selection: input.r0.selection,
      estimates,
      reason: "no well-sampled model above quality floor; conservative R0 baseline selected",
      exploratory: false,
      fallback: true,
    };
  }

  const cheapest = aboveFloor.reduce((acc, current) => cheaperEstimateB(acc, current, input.r0));
  const previous = aboveFloor.find((e) => e.modelId === input.previousModelId);
  if (
    previous !== undefined &&
    previous.modelId !== cheapest.modelId &&
    cheapest.lcb < qualityFloor + hysteresisMargin
  ) {
    return {
      selection: previous.modelId,
      estimates,
      reason: `r1-hysteresis: previous selection still above floor (${previous.lcb.toFixed(4)})`,
      exploratory: false,
      fallback: false,
    };
  }

  return {
    selection: cheapest.modelId,
    estimates,
    reason: `cheapest above quality floor ${qualityFloor} (LCB ${cheapest.lcb.toFixed(4)})`,
    exploratory: false,
    fallback: false,
  };
}

function sameDecisionLists(
  a: readonly (R1Decision | R1DecisionB)[],
  b: readonly (R1Decision | R1DecisionB)[]
): boolean {
  if (a.length !== b.length) return false;
  for (let e = 0; e < a.length; e++) {
    const x = a[e]!;
    const y = b[e]!;
    if (
      x.selection !== y.selection ||
      x.reason !== y.reason ||
      x.fallback !== y.fallback ||
      x.exploratory !== y.exploratory ||
      x.estimates.length !== y.estimates.length
    ) {
      return false;
    }
    for (let k = 0; k < x.estimates.length; k++) {
      const p = x.estimates[k]!;
      const q = y.estimates[k]!;
      if (
        p.modelId !== q.modelId ||
        p.key !== q.key ||
        !Object.is(p.alpha, q.alpha) ||
        !Object.is(p.beta, q.beta) ||
        !Object.is(p.mean, q.mean) ||
        !Object.is(p.lcb, q.lcb) ||
        !Object.is(p.samples, q.samples) ||
        p.wellSampled !== q.wellSampled
      ) {
        return false;
      }
    }
  }
  return true;
}

function scenarioAgexFaceB(): void {
  out(`  face B (posterior/R1 shadow face — the slice's ONLY live time-coordinate consumers):`);
  out(
    `    frozen dims: keys=${B_KEYS} (${B_FAMILIES} fam x ${B_ROLES} roles x ${B_MODELS} model versions), ` +
      `obs/key=${B_OBS_PER_KEY} (20 PASS + 5 FAIL[model], frozen order), episodes=${B_EPISODES}, M=${B_MODELS}, ` +
      `nowMs=${NOW_MS}, config=DEFAULT (halfLife=24h, minSamples=5, beta-quantile LCB); only occurredAtMs moves`
  );
  const models = buildModelsB();
  const families = Array.from({ length: B_FAMILIES }, (_, f) => `family-${f}`);
  const r0ByFamily = new Map(families.map((f) => [f, r0ForFamilyB(models, f)]));
  const episodes = episodesB();

  // Steady-state warmup over every dose (COLDX stays closed; first-touch
  // effects do not enter readings).

    const observations = buildObservationsB(dose);
    runWorkloadProd(observations, models, r0ByFamily, episodes);
    runWorkloadReplica(observations, models, r0ByFamily, episodes);
  }
  BSTAT.reset();

  const decisionsByDose = new Map<string, R1Decision[]>();
  const doseLines: string[] = [];
  const doseTotals: Array<{ name: string; prodMs: number }> = [];

  for (const dose of AGE_DOSES_B) {
    const observations = buildObservationsB(dose);

    // (a) bitwise fidelity: replica lane vs production lane AT THIS DOSE.
    const prodOnce = runWorkloadProd(observations, models, r0ByFamily, episodes);
    const replicaOnce = runWorkloadReplica(observations, models, r0ByFamily, episodes);
    check(`R25C-agexB[${dose.name}].replica-vs-production-bitwise`, sameDecisionLists(prodOnce.decisions, replicaOnce.decisions));

    // (b) per-key estimate fidelity on a fresh index (all 160 keys, all fields).
    const preparedProd = prepareR1Observations(observations);
    const preparedRepl = prepareR1ObservationsB(observations);
    let keyFieldsSame = true;
    let wellSampledCount = 0;
    let sampleEstimate: R1KeyEstimateB | undefined;
    for (let f = 0; f < B_FAMILIES; f++) {

          if (

            !Object.is(pe.alpha, re.alpha) ||
            !Object.is(pe.beta, re.beta) ||
            !Object.is(pe.mean, re.mean) ||
            !Object.is(pe.lcb, re.lcb) ||
            !Object.is(pe.samples, re.samples) ||
            pe.wellSampled !== re.wellSampled

            keyFieldsSame = false;
          }
          if (re.wellSampled) wellSampledCount += 1;
          if (sampleEstimate === undefined) sampleEstimate = re;
        }
      }
    }
    check(`R25C-agexB[${dose.name}].per-key-estimates-bitwise (160 keys x 7 fields)`, keyFieldsSame);

    // (c) production self-determinism at this dose.
    const prodTwice = runWorkloadProd(observations, models, r0ByFamily, episodes);
    check(`R25C-agexB[${dose.name}].production-self-determinism`, sameDecisionLists(prodOnce.decisions, prodTwice.decisions));
    decisionsByDose.set(dose.name, prodOnce.decisions);

    // (d) counter determinism: 3 replica reps, counters bitwise-identical.
    const counterReps: BCounters[] = [];
    const replicaTimes: number[] = [];
    for (let rep = 0; rep < 3; rep++) {
      BSTAT.reset();
      const run = runWorkloadReplica(observations, models, r0ByFamily, episodes);
      replicaTimes.push(run.ms);
      counterReps.push(snapshotB());
    }

        c.powCalls === c0.powCalls &&
        c.betacfCalls === c0.betacfCalls &&
        c.betacfIters === c0.betacfIters &&
        c.ribCalls === c0.ribCalls &&

    check(`R25C-agexB[${dose.name}].count-determinism (3 replica reps)`, countsSame);

    check(`R25C-agexB[${dose.name}].powCalls==${B_KEYS * B_OBS_PER_KEY}`, c0.powCalls === B_KEYS * B_OBS_PER_KEY);
    check(`R25C-agexB[${dose.name}].quantileCalls==${B_KEYS}`, c0.quantileCalls === B_KEYS);

    );
    check(`R25C-agexB[${dose.name}].ribCalls==80*quantiles`, c0.ribCalls === 80 * c0.quantileCalls);
    check(`R25C-agexB[${dose.name}].betacfCalls==ribCalls`, c0.betacfCalls === c0.ribCalls);

    // (f) production-lane wall time: 5 timed reps, median.

    for (let rep = 0; rep < 5; rep++) {
      prodTimes.push(runWorkloadProd(observations, models, r0ByFamily, episodes).ms);
    }
    const prodMs = median(prodTimes);
    doseTotals.push({ name: dose.name, prodMs });

    // (g) decision-class histogram (deterministic witness).
    const classOf = (d: R1Decision): string =>
      d.reason.startsWith("cheapest above quality floor")
        ? "selected-cheapest"
        : d.reason.startsWith("no well-sampled estimate")
          ? "sparse-fallback"
          : d.reason.startsWith("no well-sampled model above quality floor")
            ? "below-floor-fallback"
            : `other(${d.reason})`;
    const hist = new Map<string, number>();
    for (const d of prodOnce.decisions) hist.set(classOf(d), (hist.get(classOf(d)) ?? 0) + 1);
    const histStr = [...hist.entries()].map(([k, v]) => `${k}=${v}`).join(" ");

    const s = sampleEstimate!;
    doseLines.push(
      `    dose age=${dose.name}: prod ${prodMs.toFixed(2)} ms/workload (replica ${median(replicaTimes).toFixed(2)}; ` +
        `prepare ${(c0.prepareMs / 1).toFixed(2)} / decay-scan ${c0.updMs.toFixed(2)} / lcb-quantile ${c0.lcbMs.toFixed(2)} ms); ` +
        `betacfIters=${c0.betacfIters} (${(c0.betacfIters / c0.quantileCalls).toFixed(1)}/quantile); ` +
        `sample key alpha=${s.alpha} beta=${s.beta} lcb=${s.lcb.toFixed(6)} nObsEff=${s.samples} wellSampled=${s.wellSampled}; ` +
        `wellSampledKeys=${wellSampledCount}/${B_KEYS}; decisions: ${histStr}`
    );
  }

  for (const line of doseLines) out(line);

  // (h) clamp witness: future (occurredAtMs = nowMs + 24h) must be bitwise
  // identical to fresh — Math.max(0, age) maps both to age 0.
  check(
    `R25C-agexB[future==fresh bitwise via Math.max clamp]`,
    sameDecisionLists(decisionsByDose.get("future")!, decisionsByDose.get("fresh")!)
  );

  // (i) dose arithmetic vs the 35 ms offline-logit band.
  const worst = doseTotals.reduce((a, b) => (b.prodMs > a.prodMs ? b : a));

      `(x${(35 / Math.max(delta, 1e-9)).toFixed(1)} below band); ` +

function scenarioApcFloor(): void {
  const rows = perfRows();
  const baseRows: Row[] = rows.map((r) => ({
    scenarioId: r.scenarioId,
    modelVersion: r.modelVersion,
    projectId: r.projectId,
    y: r.y
  }));
  const design = buildDesign(baseRows);
  const vectors = baseRows.map((r) => design.build(r));
  const supports = computeSupports(vectors);
  const keys = canonicalRowKeys(baseRows);
  const fit = irlsPlain(design, baseRows, vectors, supports, keys, baseRows.length, MAX_ITER_DEFAULT);
  const coefficients = fit.coefficients!;
  const onProbabilities = vectors.map((vector) => sigmoid(dot(coefficients, vector)));
  const names = design.names.filter((name) => name !== "intercept");
  let apcSink = 0;
  const sweepReps = 60;
  for (let rep = 0; rep < 5; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const on = onProbabilities[i]!;
          const offVector = vectors[i]!.slice();
          offVector[columnIdx] = 0;
          const off = sigmoid(dot(coefficients, offVector));
          sum += on - off;
        }
      }
      apcSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const tSweep0 = performance.now();
  for (let rep = 0; rep < sweepReps; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const on = onProbabilities[i]!;
          const offVector = vectors[i]!.slice();
          offVector[columnIdx] = 0;
          const off = sigmoid(dot(coefficients, offVector));
          sum += on - off;
        }
      }
      apcSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const perSweepMs = (performance.now() - tSweep0) / sweepReps;
  out(
    `probe APCX: standalone timer-free APC sweep (59 names x 400 rows, base-fit coefficients) ` +
      `${perSweepMs.toFixed(3)} ms/fit -> x201 fits = ${(perSweepMs * 201).toFixed(1)} ms/report ` +
      `(base-shape approximation; sink=${apcSink.toFixed(3)})`
  );

  let floorSink = 0;
  for (let rep = 0; rep < 5; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const active = supports[i]!;
          let value = 0;
          for (let ai = 0; ai < active.length; ai++) {
            const j = active[ai]!;
            if (j === columnIdx) continue;
            value += coefficients[j]!;
          }
          sum += onProbabilities[i]! - sigmoid(value);
        }
      }
      floorSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const tFloor0 = performance.now();
  for (let rep = 0; rep < sweepReps; rep++) {
    for (const name of names) {
      let sum = 0;
      const columnIdx = design.columnIndex.get(name);
      if (columnIdx !== undefined && columnIdx !== 0) {
        for (let i = 0; i < baseRows.length; i++) {
          if (vectors[i]![columnIdx] === 0) continue;
          const active = supports[i]!;
          let value = 0;
          for (let ai = 0; ai < active.length; ai++) {
            const j = active[ai]!;
            if (j === columnIdx) continue;
            value += coefficients[j]!;
          }
          sum += onProbabilities[i]! - sigmoid(value);
        }
      }
      floorSink += baseRows.length === 0 ? 0 : sum / baseRows.length;
    }
  }
  const perFloorMs = (performance.now() - tFloor0) / sweepReps;
  out(
    `probe FLOORX: cheapest bitwise-legal sweep ${perFloorMs.toFixed(3)} ms/fit -> x201 fits = ` +

      `${(perSweepMs * 201 - perFloorMs * 201).toFixed(1)} ms (floorSink=${floorSink.toFixed(3)})`
  );
}
