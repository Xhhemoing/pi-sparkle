## 2. 本轮实扫新轴 + 第二十五组新角度枚举：有理由的空枚举

### 2.0 新轴 CG：per-key 重述链面（本切片首次；链深 D × 链态增长类 G/P）

R1-E..R24-E 的十七个维度没有任何一维剂量过「同一任务键上的
TASK_RETRY 重述链」。该链是本切片唯一的**流内自耦合**结构：
`applyCascadeRetry`（`from-episode.ts:252-273`）对每次生效重试做
`current.eligibleModels.includes(nextModel)` O(L) 扫描 + （新模型时）
`[...eligibleModels, nextModel]` O(L) 拷贝 + 全载荷 spread +
`oneHotDistribution(eligible, next)` O(L) 重建——L 是**同键此前
重试的累积函数**，所以第 i 次重试的单价随 i 增长，R17-E 的「每类
常数单价 × 计数」可加账本对这一项**结构性失明**（其可加性重构恰在
浅链构成上成立）。S9-E-1 在浅链上定价并否决了该位点的两个改写形态，
但从未画出深度-增长曲线；首选候选「键基数/扇入面」在立项前被 KFAN
换名判定拦下（§0），CG 是它的合法残差：**键数钉死为 1，变量是每键
存储状态的增长本身**。双段构造与数字见结论 1–4（此处不重抄），
关键机制事实与裁决：

- **两条边际曲线的形状互证机制**：G 边际单调升（127.6→153.8，
  斜率 0.26–0.43 ns/深度 = includes+拷贝+oneHot 三个 O(L) 项的
  每元素价 ~0.3ns 之和的均值化）、P 边际单调降（105→72，纯摊薄）
  ——同深度差价 G−P = 66.9–80.0 ns/retry 是「状态增长」的直接价签，
  两类载荷字节形逐位同构（4ch 模型 id、同一批事件对象）。
- **超线性存在但极弱且被域封顶**：曲率比 47 深链只到 1.07–1.11；
  更重要的是生产 |eligible| ≤ M+1 ≤ ~11（级联只在目录 tier 内升级）
  且 D ≤ 2（maxAttempts 硬编码）——超线性项的两个操作数在生产里
  都被钉死，实测生产天花板 258–358 ns/任务。
- **collect 侧是结构性零面**：两遍扫描都没有 TASK_RETRY 分支，重试
  事件只付 ~4–10 ns 的分派落空税（六次 type 比较），G/P 不可分辨
  ——R17-E/R24-E 对 collect 的账本无需链项即完备（本轮实测补钉）。
- **穿越点与既有门槛收敛**：aD+bD²=10ms 解出 D≈4.7–6.0×10³，与
  S9-E-1 重开门槛「每 run 数千次重试」独立同阶——两条不同方法
  （改写形态定价 vs 剂量曲线外推）指向同一收口线，收口可信度加固。

**裁决**：该轴在频率（级联重试 ≤2 次/任务 × ≤10 任务/run）× 单价

 * R25-E new-axis suite: CG — per-key restatement-chain face
 * (chain depth D x state-growth class) at the two mint-level sites.
 *
 * Object: the TASK_RETRY cascade chain on a single task key.
 *   - outcomes side (from-episode applyCascadeRetry): the ONLY site in the
 *     slice where per-event cost depends on accumulated per-key in-stream
 *     state (eligibleModels growth + payload spread + oneHotDistribution
 *     O(|eligible|)) -> per-retry price is chain-position dependent
 *     (superlinear total), invisible to R17-E's constant per-class prices.
 *   - collect side (signals.ts): TASK_RETRY is structurally ignored (no
 *     handler in either pass) -> pure dispatch fall-through price, and the
 *     TASK_RESULT signal binds the PRE-retry model while the outcome binds
 *     the POST-retry model (cross-site binding divergence witness).
 *
 * Deterministic explicit construction, NO RNG. Payload identity pinned:
 * the same disc/route/result event objects are reused across every arm;
 * retry events are 4-char models on both classes (g001..g047 vs p001) so
 * fresh-vs-repeat differs ONLY in the chain-state effect, not in byte shape
 * or length (NVG / content-class confounds pinned).
 *
 * Fate pinned: every arm mints exactly 1 outcome (PASS, full binding) and
 * exactly 1 collect signal. The only variable is D and the growth class.
 *
 * argv[2] optional segment filter: A (collect) | B (outcomes).

import { performance } from "node:perf_hooks";
import { collectSignalsFromEvents } from "/workspace/src/learning/signals.js";
import { outcomesFromRoutedRun } from "/workspace/src/learning/from-episode.js";
import { oneHotDistribution } from "/workspace/src/routing/catalog-model.js";
import { isAgentRole } from "/workspace/src/domain/roles.js";
import type { Event } from "/workspace/src/run/events.js";
import type { IsoTimestamp } from "/workspace/src/domain/timestamp.js";

const SEG = process.argv[2] ?? "AB";
const NOW = "2026-08-25T17:30:00.000Z" as IsoTimestamp;
const RUN_ID = "run_cg000001";
const TASK_ID = "tsk_cg000000";

let failures = 0;
let checks = 0;
function check(name: string, ok: boolean): void {
  checks += 1;
  if (!ok) {
    failures += 1;
    console.error(`FAIL ${name}`);
  }
}
function bench(fn: () => void, reps: number): number {
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}

/* Domain pin: agentRole must be a real AgentRole (mirrors R24-E B0). */
check("domain pin: tester is an AgentRole", isAgentRole("tester"));

const BASE_ELIGIBLE = ["orig", "p001", "x001"];
const discEvent = {
  type: "PROJECT_DISCOVERED",
  runId: RUN_ID,
  occurredAt: NOW,
  payload: { project: { id: "prj_cg000001", rootPath: "/tmp/proj-cg" } }
} as unknown as Event;
const routeEvent = {
  type: "MODEL_ROUTED",
  runId: RUN_ID,
  occurredAt: NOW,
  payload: {
    taskId: TASK_ID,
    model: "orig",
    role: "tester",
    agentRole: "tester",
    eligibleModels: BASE_ELIGIBLE,
    behaviorDistribution: oneHotDistribution(BASE_ELIGIBLE, "orig"),
    family: "test",
    featureVersion: "fv1",
    modelVersion: "v1"
  }
} as unknown as Event;
const resultEvent = {
  type: "CHILD_MESSAGE",
  runId: RUN_ID,
  occurredAt: NOW,
  payload: {
    message: {
      type: "TASK_RESULT",
      taskId: TASK_ID,
      runId: RUN_ID,
      outcome: "SUCCESS",
      verification: { kind: "PASSED" },
      summary: "tests passed",
      evidenceIds: []
    }
  }
} as unknown as Event;

function retryEvent(nextModel: string, attempt: number, previousModel: string): Event {
  return {
    type: "TASK_RETRY",
    runId: RUN_ID,
    occurredAt: NOW,
    taskId: TASK_ID,
    payload: {
      childRunId: "run_cg000002",
      attempt,
      reason: "deterministic model FAIL",
      previousModel,
      nextModel
    }
  } as unknown as Event;
}

/** grow class: D fresh 4-char models g001..gNNN (eligible grows 3 -> 3+D). */
function growStream(depth: number): { events: Event[]; finalModel: string } {
  const events: Event[] = [discEvent, routeEvent];
  let prev = "orig";
  let finalModel = "orig";
  for (let i = 1; i <= depth; i += 1) {
    finalModel = `g${String(i).padStart(3, "0")}`;
    events.push(retryEvent(finalModel, i, prev));
    prev = finalModel;
  }
  events.push(resultEvent);
  return { events, finalModel };
}

/** pinned class: D retries all to p001 (already in eligible -> no growth). */
function pinnedStream(depth: number): { events: Event[]; finalModel: string } {
  const events: Event[] = [discEvent, routeEvent];
  let prev = "orig";
  for (let i = 1; i <= depth; i += 1) {
    events.push(retryEvent("p001", i, prev));
    prev = "p001";
  }
  events.push(resultEvent);
  return { events, finalModel: depth > 0 ? "p001" : "orig" };
}

const prelude = pinnedStream(0); // disc + route + result, D=0

  readonly events: Event[];

    collectSignalsFromEvents(cells.find((c) => c.name === "G47")!.events, {})[0]!.modelId === "orig"
);
