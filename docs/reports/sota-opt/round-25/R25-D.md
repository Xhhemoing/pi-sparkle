function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** Recursively sort object keys so JSON compare ignores key order. */
function canonicalJson(value: unknown): string {
  return JSON.stringify(value, (_key, val: unknown) => {
    if (val !== null && typeof val === "object" && !Array.isArray(val)) {
      const sorted: Record<string, unknown> = {};
      for (const k of Object.keys(val as Record<string, unknown>).sort()) {
        sorted[k] = (val as Record<string, unknown>)[k];
      }
      return sorted;
    }
    return val;
  });
}

  const eps = Array.from({ length: episodes }, (_, i) => ({
    episodeHash: `eh_${i}`,
    taskId: `tsk_e${i}`,
    role: pick(rng, ROLES),
    objective: pick(rng, OBJECTIVES),
    originalWorkspace: "/repos/alpha",
    ...(rng() < 0.7 ? { taskSuccess: rng() < 0.5 ? "PASS" : "FAIL" } : {}),
    ...(rng() < 0.5 ? { taskFamily: pick(rng, ["edit", "test", "review"]) } : {})
  }));
  const manifest = { datasetId: "ds-r25d", environmentVersion: "env-1", episodes: eps };
  await writeFile(join(dir, "manifest.json"), `${JSON.stringify(manifest, null, 2)}\n`, "utf8");
}

const SHARED_PROJECT = createProjectId(() => "r25dproj"); // "prj_r25dproj", scopeKey "project:prj_r25dproj" (20 chars)
const PROJECT_SCOPE: ResourceScope = { kind: "project", projectId: SHARED_PROJECT };

const KEY_TARGET = 48; // total resourceIdentityKey length equalized across duel arms

function paddedName(i: number, kind: ResourceKind, scope: ResourceScope): string {
  const scopeKeyLen = scope.kind === "project" ? 8 + String(scope.projectId).length : 11;
  const len = KEY_TARGET - (kind.length + 2 + scopeKeyLen);
  const base = `res-${String(i).padStart(4, "0")}`;
  return base + "x".repeat(len - base.length);
}

interface CoordRow {
  readonly identity: ResourceIdentity;
  readonly v1: ResourceVersion;
  readonly candidate: ImprovementCandidate;
  readonly input: PromoteInput;
}

interface CoordFleet {
  readonly registry: ResourceRegistry;
  readonly rows: readonly CoordRow[];
}

/** N identities at one lattice coordinate. Contents are byte-identical
 * across coordinates; names are padded so resourceIdentityKey length is
 * identical across duel arms; same tag => identical id streams. All
 * promotes are explicit-mode over the default profile (production mode;
 * the approval-authority face stays untouched, R24-D territory). */
function buildCoordFleet(tag: string, n: number, kind: ResourceKind, scope: ResourceScope): CoordFleet {
  const registry = new ResourceRegistry({ now: () => NOW, generateId: sequentialIds(tag) });
  const rows: CoordRow[] = [];
  for (let i = 0; i < n; i += 1) {
    const identity: ResourceIdentity = { kind, name: paddedName(i, kind, scope), scope };
    const body = `# resource row ${String(i).padStart(4, "0")}\nKeep the diff bounded to the stated scope and cite evidence for every claim.\n`;
    const v1 = registry.registerBaseline({ identity, content: body, author: HUMAN });
    const content = `${body}\n## amendment\nAdd an explicit check for the attributable failure (row ${String(i).padStart(4, "0")}).\n`;
    const candidate = registry.createCandidate({
      identity,
      content,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    const input: PromoteInput = {
      candidateId: candidate.candidateId,
      expectedCurrentVersionId: v1.versionId,
      content,
      approvedBy: HUMAN,
      review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}${i}`),
      changeNote: mkNote(`row:${String(i).padStart(4, "0")}`, v1.versionId),
      explicitApproval: true
    };
    rows.push({ identity, v1, candidate, input });
  }
  return { registry, rows };
}

function promoteFleet(fleet: CoordFleet): number {
  const t0 = performance.now();
  for (const row of fleet.rows) {
    promoteWithRegistry(fleet.registry, row.input);
  }
  return performance.now() - t0;
}

/** Mask exactly the coordinate-valued fields: identity objects and the
 * kind-derived autoPromotable bit. Everything else must be byte-equal. */
function maskedSnapshot(registry: ResourceRegistry): string {
  const snap = registry.snapshot();
  const masked = {
    ...snap,
    versions: snap.versions.map((v) => ({ ...v, identity: "MASKED" })),
    candidates: snap.candidates.map((c) => ({ ...c, identity: "MASKED", autoPromotable: false }))
  };
  return canonicalJson(masked);
}
