function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function pick<T>(rng: () => number, items: readonly T[]): T {
  return items[Math.floor(rng() * items.length)] as T;
}
function bench(fn: () => void, reps: number): number {
  fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}

function genObjective(rng: () => number): string {
  if (rng() < 0.5) return pick(rng, OBJECTIVE_TEMPLATES);

function genTasks(rng: () => number, n: number): AssignableTask[] {

    mode: "cr" | "ct" | "ot" | "all" | "mix" | "alt" | "huge" | "crInvert"

    const rng = mulberry32(0x525b02);
    return bare.map((task, i) => {
      const ctVal = Math.floor(rng() * 100000);
      const otVal = Math.floor(rng() * 4000);
      const mixDraws = { cr: rng() < 0.1, crVal: rng() < 0.5, ct: rng() < 0.3, ot: rng() < 0.3 };
      switch (mode) {
        case "cr":
          return { ...task, contractRisk: regexRisk[i]! };
        case "crInvert":
          return { ...task, contractRisk: !regexRisk[i]! };
        case "ct":
          return { ...task, contextTokens: ctVal };
        case "ot":
          return { ...task, outputTokens: otVal };
        case "all":
          return { ...task, contractRisk: regexRisk[i]!, contextTokens: ctVal, outputTokens: otVal };
        case "huge":
          return { ...task, contextTokens: 2 ** 28 + ctVal, outputTokens: 2 ** 20 + otVal };
        case "mix":
          return {
            ...task,
            ...(mixDraws.cr ? { contractRisk: mixDraws.crVal } : {}),
            ...(mixDraws.ct ? { contextTokens: ctVal } : {}),
            ...(mixDraws.ot ? { outputTokens: otVal } : {})
          };
        case "alt":
          return i % 2 === 0
            ? { ...task, contractRisk: regexRisk[i]!, contextTokens: ctVal, outputTokens: otVal }
            : { ...task };
      }
    });
  }
  const arms = {
    cr: inject("cr"),
    ct: inject("ct"),
    ot: inject("ot"),
    all: inject("all"),
    mix: inject("mix"),
    alt: inject("alt"),
    huge: inject("huge"),
    crInvert: inject("crInvert")
  };

  // W4: presence census -- the injectors actually produce the intended corners.
  {
    const allOwn = arms.all.every(
      (task) => task.contractRisk !== undefined && task.contextTokens !== undefined && task.outputTokens !== undefined
    );
    const bareOwn = bare.every(
      (task) => !("contractRisk" in task) && !("contextTokens" in task) && !("outputTokens" in task)
    );
    check("presence W4: ALL@100% tasks own all three fields; bare tasks own none", allOwn && bareOwn);
    const outAll = assignTasks({ catalog: catalog2, tasks: arms.all });
    const outBareCensus = assignTasks({ catalog: catalog2, tasks: bare });
    const analysisCensus =
      outAll.every((row) => row.analysis.contextTokens !== undefined && row.analysis.outputTokens !== undefined) &&
      outBareCensus.every((row) => !("contextTokens" in row.analysis) && !("outputTokens" in row.analysis));
    check("presence W4: ALL@100% analyses echo both token fields; bare analyses echo neither", analysisCensus);
  }

  // W2: contractRisk agreement-value presence is output-inert (the override
  // leg computes the same verdict the skipped regex would), with an inverted
  // control proving the witness discriminates.
  {
    const outBare = assignTasks({ catalog: catalog2, tasks: bare });
    const outCr = assignTasks({ catalog: catalog2, tasks: arms.cr });
    let inert = outCr.length === outBare.length;
    for (let i = 0; i < outBare.length; i += 1) {
      if (JSON.stringify(outCr[i]) !== JSON.stringify(outBare[i])) {
        inert = false;
        break;
      }
    }
    check("presence W2 [M=2]: CR@100% agreement values element-wise byte-identical to bare", inert);
    const outInv = assignTasks({ catalog: catalog2, tasks: arms.crInvert });
    let moved = 0;
    for (let i = 0; i < outBare.length; i += 1) {
      if (JSON.stringify(outInv[i]) !== JSON.stringify(outBare[i])) moved += 1;
    }
    check("presence W2 control [M=2]: inverted contractRisk moves >0 assignments", moved > 0, `moved=${moved}`);
    console.log(`presence W2 control [M=2]: inverted contractRisk moves ${moved}/${outBare.length} assignments`);
  }

  // W3: in-window token presence leaves every DECISION byte-identical on the
  // M=2 default catalog (no contextWindow/maxOutputTokens declared; default
  // limits are infinite) -- only the analysis gains the two echoed fields.
  // Discriminating control: on the M=7 configured catalog a 32001-token
  // output requirement produces max-output rejections.
  {
    const outBare = assignTasks({ catalog: catalog2, tasks: bare });
    const outAll = assignTasks({ catalog: catalog2, tasks: arms.all });
    let projected = outAll.length === outBare.length;
    for (let i = 0; i < outBare.length; i += 1) {
      const stripped = JSON.parse(JSON.stringify(outAll[i])) as {
        analysis: Record<string, unknown>;
      };
      delete stripped.analysis["contextTokens"];
      delete stripped.analysis["outputTokens"];
      if (JSON.stringify(stripped) !== JSON.stringify(outBare[i])) {
        projected = false;
        break;
      }
    }
    check(
      "presence W3 [M=2]: ALL@100% (agreement CR + in-window tokens) strip-projection byte-identical to bare",
      projected
    );
    const bigOut = bare.map((task) => ({ ...task, outputTokens: 32001 }));
    const outBig = assignTasks({ catalog: catalogCfg, tasks: bigOut });
    const rejectionRows = outBig.filter((row) =>
      row.decision.rejections.some((refusal) => refusal.constraint === "max-output")
    ).length;
    check(
      "presence W3 control [M=7 configured]: outputTokens=32001 produces max-output rejections",
      rejectionRows > 0,
      `rows=${rejectionRows}`
    );
    console.log(
      `presence W3 control [M=7 configured]: outputTokens=32001 -> max-output rejection rows ${rejectionRows}/${outBig.length}`
    );
  }

  // W1: double-run determinism at the most field-loaded corner.
  {
    const runA = JSON.stringify(assignTasks({ catalog: catalog2, tasks: arms.all }));
    const runB = JSON.stringify(assignTasks({ catalog: catalog2, tasks: arms.all }));
    check("presence W1 [M=2 ALL@100%]: double run byte-identical", runA === runB);
  }

  // Timing lattice: every arm on the SAME objective stream, same window, plus
  // the historical genTasks corpus for the gap attribution.
  const tBare = bench(() => assignTasks({ catalog: catalog2, tasks: bare }), 30);
  const tCr = bench(() => assignTasks({ catalog: catalog2, tasks: arms.cr }), 30);
  const tCt = bench(() => assignTasks({ catalog: catalog2, tasks: arms.ct }), 30);
  const tOt = bench(() => assignTasks({ catalog: catalog2, tasks: arms.ot }), 30);
  const tAll = bench(() => assignTasks({ catalog: catalog2, tasks: arms.all }), 30);
  const tMix = bench(() => assignTasks({ catalog: catalog2, tasks: arms.mix }), 30);
  const tAlt = bench(() => assignTasks({ catalog: catalog2, tasks: arms.alt }), 30);
  const tHuge = bench(() => assignTasks({ catalog: catalog2, tasks: arms.huge }), 30);
  const ceilingCorpus = genTasks(mulberry32(0xb44b01), N);
  const tCeiling = bench(() => assignTasks({ catalog: catalog2, tasks: ceilingCorpus }), 30);
  const line = (label: string, cost: number): string =>
    `presence [M=2 ${label}]: ${(cost * 1e3).toFixed(1)}us per call (${(cost * 2).toFixed(2)}ms per eval x2; delta vs bare ${((cost - tBare) * 2).toFixed(2)}ms)`;
  console.log(`presence [M=2 bare production corner]: ${(tBare * 1e3).toFixed(1)}us per call (${(tBare * 2).toFixed(2)}ms per eval x2)`);
  console.log(line("CR@100% agree", tCr));
  console.log(line("CT@100%", tCt));
  console.log(line("OT@100%", tOt));
  console.log(line("ALL@100%", tAll));
  console.log(line("mix 10/30/30 (ceiling presence state, same objectives)", tMix));
  console.log(line("ALT 50% (shape-dimorphism dose)", tAlt));
  console.log(line("HUGE token magnitudes", tHuge));
  console.log(
    `presence gap attribution [M=2, same window]: ceiling-corpus total=${(tCeiling * 2).toFixed(2)}ms | gap vs bare=${((tCeiling - tBare) * 2).toFixed(2)}ms | presence share (mix - bare)=${((tMix - tBare) * 2).toFixed(2)}ms | objective-stream share (ceiling - mix)=${((tCeiling - tMix) * 2).toFixed(2)}ms per eval x2`
  );
  conclusions.push(
    `presence lattice M=2 per-eval: bare=${(tBare * 2).toFixed(2)} CR=${(tCr * 2).toFixed(2)} CT=${(tCt * 2).toFixed(2)} OT=${(tOt * 2).toFixed(2)} ALL=${(tAll * 2).toFixed(2)} mix=${(tMix * 2).toFixed(2)} alt50=${(tAlt * 2).toFixed(2)} huge=${(tHuge * 2).toFixed(2)}ms | gap attribution: total=${((tCeiling - tBare) * 2).toFixed(2)} presence=${((tMix - tBare) * 2).toFixed(2)} stream=${((tCeiling - tMix) * 2).toFixed(2)}ms`
  );

  // Configured-state spot arms: the M=1 collapse catalog (the eval face's
  // configured state, R24-B) and the M=7 configured catalog at the ALL corner
  // -- presence tax must stay flat across catalog provenance states.
  {
    const catalogM1 = catalogFromPrimary({ primaryModelId: "acme/frontier" });
    const tM1Bare = bench(() => assignTasks({ catalog: catalogM1, tasks: bare }), 30);
    const tM1All = bench(() => assignTasks({ catalog: catalogM1, tasks: arms.all }), 30);
    const tCfgBare = bench(() => assignTasks({ catalog: catalogCfg, tasks: bare }), 30);
    const tCfgAll = bench(() => assignTasks({ catalog: catalogCfg, tasks: arms.all }), 30);
    console.log(
      `presence spot [M=1 collapse]: bare=${(tM1Bare * 2).toFixed(2)}ms ALL@100%=${(tM1All * 2).toFixed(2)}ms (delta ${((tM1All - tM1Bare) * 2).toFixed(2)}ms per eval x2)`
    );
    console.log(
      `presence spot [M=7 configured]: bare=${(tCfgBare * 2).toFixed(2)}ms ALL@100%=${(tCfgAll * 2).toFixed(2)}ms (delta ${((tCfgAll - tCfgBare) * 2).toFixed(2)}ms per eval x2)`
    );
    conclusions.push(
      `presence spot deltas per-eval: M=1 ${(tM1All * 2 - tM1Bare * 2).toFixed(2)}ms M=7 ${(tCfgAll * 2 - tCfgBare * 2).toFixed(2)}ms`
    );
  }

  // Real face N=30 (production callers pass the bare trio; the ALL corner is
  // therefore a synthetic-only surcharge even here) + 10x stress corner.
  {
    const bare30 = bare.slice(0, 30);
    const all30 = arms.all.slice(0, 30);
    const tBare30 = bench(() => assignTasks({ catalog: catalog2, tasks: bare30 }), 2000);
    const tAll30 = bench(() => assignTasks({ catalog: catalog2, tasks: all30 }), 2000);
    console.log(
      `presence REAL face N=30 [M=2]: bare=${(tBare30 * 1e3).toFixed(1)}us ALL@100%=${(tAll30 * 1e3).toFixed(1)}us per call (delta ${((tAll30 - tBare30) * 1e3).toFixed(1)}us)`
    );
    const bareStress = genReplayTasks(mulberry32(0xb44b01), 20000);
    const rngS = mulberry32(0x525b02);
    const riskStress = bareStress.map((task) => analyzeTask(task.objective, task.role).highRisk);
    const allStress = bareStress.map((task, i) => ({
      ...task,
      contractRisk: riskStress[i]!,
      contextTokens: Math.floor(rngS() * 100000),
      outputTokens: Math.floor(rngS() * 4000)
    }));
    const tBareStress = bench(() => assignTasks({ catalog: catalog2, tasks: bareStress }), 5);
    const tAllStress = bench(() => assignTasks({ catalog: catalog2, tasks: allStress }), 5);
    console.log(
      `presence 10x stress N=20000 [M=2]: bare=${(tBareStress * 2).toFixed(1)}ms ALL@100%=${(tAllStress * 2).toFixed(1)}ms per eval x2 (delta ${((tAllStress - tBareStress) * 2).toFixed(1)}ms)`
    );
    conclusions.push(
      `presence real-face N=30 delta=${((tAll30 - tBare30) * 1e3).toFixed(1)}us | stress N=20000 delta=${((tAllStress - tBareStress) * 2).toFixed(1)}ms per eval`
    );
  }
}

console.log(`\nCONCLUSIONS: ${conclusions.join(" | ")}`);
if (failures > 0) {

  process.exit(1);
}
