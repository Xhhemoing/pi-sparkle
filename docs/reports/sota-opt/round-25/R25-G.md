## §2 新轴：注入指令种类面（injection command-kind face，战役首创）

### 2.0 为什么这是 R1-G…R24-G 从未拉伸的完备格

R7-I 教训的本轮应用：14 格矩阵的 inject 格是**配置态盲区**——它固定
kind=fact，把三元种类枚举当成了单点。历史 G 报告全文核对：inject 只以
「一次性命令类」（R8-G）、「会话恢复死载荷」（R9-G S9-G-1）、「skip
未知节点拒绝格」（R17-G）、「fact × 决策策略剂量」（R24-G M5）四个
侧面出现过，**种类维从未被扫**。而该维不是死代码（§0 调用面核查：

run 的解锁语义差分——恰好构成一条新完备格：**kind × 运行态 ×
目标节点态**。按纪律不硬凑：先理论后仿真，达不了线就诚实关闭。

function normalizeFactory(stateRoot: string): (raw: string) => string {
  return (raw: string) =>
    raw.replaceAll(stateRoot, "<STATE>").replaceAll(projectRoot, "<PROJ>")

async function driveApprovalCell(spec: InjectSpec, index: number): Promise<CellOut> {
  const digests: string[] = [];
  let out: CellOut | undefined;
  for (let rep = 0; rep < REPS; rep++) {
    const stateRoot = mkdtempSync(join(tmpdir(), `r25g-injkind-a${index}-`));
    try {
      const deps = {
        stateRoot, router: router(), now: () => T0,
        generateId: sequenceGenerator(String(index + 1)),
        pause: createFilePauseController(stateRoot, () => T0)
      };
      const phases: PhaseOut[] = [];

      resetCounts();
      let started = performance.now();
      const startOut = await startFlowchartRun(deps, {
        projectRoot, flowchart: injFlowchart(), objective: "injkind drive",
        childResults: { a: ok(0.9, "evd_a") }
      });
      phases.push({ counts: countsLine(), wallMs: performance.now() - started });
      if (startOut.status !== "WAITING_FOR_USER") throw new Error(`${spec.name}: start status ${startOut.status}`);
      const planId = startOut.pendingApproval?.plan.id ?? "";

      if (spec.request !== undefined) {
        resetCounts();
        started = performance.now();
        const injectOut = await injectFlowchartRun(deps, startOut.runId, spec.request);
        phases.push({ counts: countsLine(), wallMs: performance.now() - started });
        if (injectOut.status !== spec.expectInjectStatus) {
          throw new Error(`${spec.name}: inject status ${injectOut.status} != ${spec.expectInjectStatus}`);
        }
      }

      resetCounts();
      started = performance.now();
      const resumeOut = await resumeFlowchartRun(deps, startOut.runId, {
        approvalReply: { approvalPlanId: planId, selectedActionIds: ["pA"] },
        childResults: { pA: ok(0.85, "evd_pA"), tail: ok(0.8, "evd_tail") }
      });
      phases.push({ counts: countsLine(), wallMs: performance.now() - started });
      if (resumeOut.status !== "COMPLETED") throw new Error(`${spec.name}: resume status ${resumeOut.status}`);

      const dir = join(stateRoot, "runtime", "runs", startOut.runId);
      const normalize = normalizeFactory(stateRoot);
      const eventsRaw = await readFile(join(dir, "events.jsonl"), "utf8");
      const checkpointRaw = await readFile(join(dir, "checkpoint.json"), "utf8");
      digests.push(createHash("sha256").update(normalize(eventsRaw)).update(normalize(checkpointRaw)).digest("hex"));

      const nodes = resumeOut.snapshot.nodes as unknown as Record<string, { state: string; confidence?: number }>;
      out = {
        name: spec.name,
        phases,
        eventTypes: eventsRaw.trim().split("\n").map((line) => (JSON.parse(line) as { type: string }).type).join(","),
        nodeCarry: ["a", "sel", "pA", "pB", "tail"]
          .map((id) => `${id}=${nodes[id]!.state}:${JSON.stringify(nodes[id]!.confidence)}`)
          .join(" "),
        digest: digests[digests.length - 1]!,
        bytes: checkpointRaw.length
      };
    } finally {
      rmSync(stateRoot, { recursive: true, force: true });
    }
  }
  if (!digests.every((d) => d === digests[0])) throw new Error(`${spec.name}: unstable artifacts`);
  return out!;
}
