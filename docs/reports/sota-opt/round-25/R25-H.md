/** Representation recipes. Every recipe returns a byte-identical string;
 * only the V8-internal representation class differs (by documented

  FF: (s) => s.split("").join(""),

  SLICED: (s) => ("\u0001" + s).slice(1),
  TB: (s) => ("\u4e2d" + s).slice(1)
