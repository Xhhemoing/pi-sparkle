function runInterleaved(plan: ExperimentPlan, s: readonly ExperimentOutcome[]): ShadowState {

    if (state.halted) {
      break;
    }
    state = runner.assign(state, outcome.episodeHash, 0);
    state = runner.recordOutcome(state, outcome, 0);
  }
  return state;
}
