## 2. 新轴：内容存储人口面

### 2.1 定义与盲区

受控变量：`contentsByHash` 存储的人口三分量——**别名重数 μ**
（逻辑引用数 refs / 唯一 blob 数 B；refs = versions ∪ candidates
的 contentHash 引用）、**分区**（固定唯一总字节 U 与固定 B 下的
blob 尺寸分布）、**引用状态**（活/孤儿）。钉死变量：逻辑总字节
（逐臂断言恰等）、引用结构（K 个 prompt×project 身份 ×
(registerBaseline + createCandidate)，掩码后快照结构逐臂断言相等
——contentHash 掩码、contents 人口除外，其余全等）、叶内容类
（扁平全小写 ASCII **非 JSON** 串，首字符非 `{`/`[`——R27-D 树
几何域与 R25-A/R27-A 值域全部外置）、注入臂（全程 arm A）、
数据集（E=200 同 R8-D…R27-D 口径）。

盲区成立的三个首手事实：(a) R14/R16-D P 轴的字节剂量在**固定
人口结构**下缩放单 blob 尺寸——μ、B、引用状态全程锁死；(b)
R21-D 状态增长方向轴的版本链增长使 contents 人口与版本数**同比**
增长（每版本新内容 → μ≈常数）——μ 作为独立自由度从未受控；(c)
putContent 的去重语义（L155-159 "Identical content is stored
once"）使 μ>1 的折扣**静默兑现**——历史夹具自身 μ=1.33（S1 普查，
21.6% 逻辑字节从不落盘）但该属性二十七遍从未被度量或定价。

### 2.2 与近邻轴的区分（换名自查核心）

- **vs R14/R16-D payload 形态/字节轴**：其剂量单 blob 字节于固定
  人口；本轴剂量人口于固定逻辑字节（S2 逐臂断言）。正交补。
- **vs R27-D 树几何面（上轮 D 轴）**：其剂量单 blob 内部 JSON
  树形于固定字节；本轴 blob 内容为扁平非 JSON 串（首字符非

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

async function benchAsync(fn: () => Promise<void>, reps: number): Promise<number> {
  await fn();
  const t0 = performance.now();

  return (performance.now() - t0) / reps;
}
/** Time-boxed sync bench, returns ms/call. */
function bench(fn: () => void, minMs = 60, maxReps = 50000): number {
  fn();
  fn();
  let reps = 0;
  const t0 = performance.now();
  let elapsed = 0;
  do {
    fn();
    reps += 1;
    elapsed = performance.now() - t0;
  } while (elapsed < minMs && reps < maxReps);
  return elapsed / reps;
}

const FILLER = "abcdefghijklmnopqrstuvwxyz abcdefghij lowercase ascii filler text ";
/** Flat lowercase-ASCII non-JSON blob of EXACTLY len bytes; distinct per (tag, j).
 * Never starts with '{' or '[' so the candidate boundary lint's JSON.parse arm
 * stays constant-off across every arm (leaf content class pinned). */
function poolContent(tag: string, j: number, len: number): string {
  const head = `pool ${tag} ${String(j).padStart(6, "0")} `;
  if (head.length > len) throw new Error("len too small");
  let s = head;
  while (s.length < len) s += FILLER;
  const out = s.slice(0, len);
  if (out.length !== len) throw new Error("pad mismatch");
  return out;
}

interface Fleet {
  registry: ResourceRegistry;
  pool: readonly string[];
  refs: readonly string[]; // logical reference contents in put order
  identities: readonly ResourceIdentity[];
  buildMs: number;
}
/** K prompt-x-project identities, each registerBaseline + createCandidate.
 * Logical refs = 2K contents of exactly `len` bytes each, drawn round-robin
 * from a pool of B distinct blobs => aliasing multiplicity mu = 2K / B. */
function buildFleet(tag: string, K: number, len: number, B: number): Fleet {
  const registry = new ResourceRegistry(armA(tag));
  const pool = Array.from({ length: B }, (_, j) => poolContent(tag, j, len));
  const refs: string[] = [];
  const identities: ResourceIdentity[] = [];
  const t0 = performance.now();
  for (let i = 0; i < K; i += 1) {
    const identity: ResourceIdentity = {
      kind: "prompt",
      name: `agent-prompt-${String(i).padStart(4, "0")}`,
      scope: { kind: "project", projectId: createProjectId(() => "popproj") }
    };
    identities.push(identity);
    const c1 = pool[(2 * i) % B] as string;
    const c2 = pool[(2 * i + 1) % B] as string;
    const v1 = registry.registerBaseline({ identity, content: c1, author: HUMAN });
    registry.createCandidate({
      identity,
      content: c2,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    refs.push(c1, c2);
  }
  const buildMs = performance.now() - t0;
  return { registry, pool, refs, identities, buildMs };
}

/** Cross-arm structure witness: snapshot canonical form with contentHash
 * values masked and the contents population excluded (the population IS the
 * dosed variable; everything else must be identical across arms). */
function maskedCanonical(registry: ResourceRegistry): string {
  const snap = registry.snapshot();

    versions: snap.versions.map((v) => ({ ...v, contentHash: "MASK" })),

    pending: snap.pending,
    autoPromotionsUsed: snap.autoPromotionsUsed,
    rollbackLedger: snap.rollbackLedger,
    retiredVersionIds: snap.retiredVersionIds
  });
}

interface PopulationCensus {
  refs: number;
  blobs: number;
  logicalKB: number;
  uniqueKB: number;
  mu: number;
}
function censusOf(snap: ResourceRegistrySnapshot): PopulationCensus {
  const contents = snap.contents ?? [];
  const byHash = new Map(contents.map((b) => [b.hash, b.content] as const));
  let logical = 0;
  let refs = 0;
  for (const v of snap.versions) {
    refs += 1;
    logical += (byHash.get(v.contentHash) ?? "").length;
  }
  for (const c of snap.candidates) {
    refs += 1;
    logical += (byHash.get(c.contentHash) ?? "").length;
  }
  const unique = contents.reduce((s, b) => s + b.content.length, 0);
  return {
    refs,
    blobs: contents.length,
    logicalKB: logical / 1024,
    uniqueKB: unique / 1024,
    mu: contents.length === 0 ? 0 : refs / contents.length
  };
}

async function runMuTier(
  tierTag: string,
  K: number,
  len: number,
  Bs: readonly number[],
  ioReps: number

  const logicalTarget = 2 * K * len;
  for (const B of Bs) {
    const fleet = buildFleet(`f${tierTag}b${B}x`, K, len, B);
    const { registry, refs, identities } = fleet;
    // W1: logical byte parity by construction
    const logical = refs.reduce((s, c) => s + c.length, 0);
    check(`W1 ${tierTag}/B=${B} logical byte parity`, logical === logicalTarget, `${logical} != ${logicalTarget}`);
    const snap = registry.snapshot();
    const census = censusOf(snap);
    check(`W1 ${tierTag}/B=${B} blob count`, census.blobs === B, `${census.blobs}`);
    check(`W1 ${tierTag}/B=${B} refs`, census.refs === 2 * K, `${census.refs}`);

  const fit = (pts: { x: number; y: number }[]): { slope: number; intercept: number } => {
    const n = pts.length;
    const sx = pts.reduce((s, p) => s + p.x, 0);
    const sy = pts.reduce((s, p) => s + p.y, 0);
    const sxx = pts.reduce((s, p) => s + p.x * p.x, 0);
    const sxy = pts.reduce((s, p) => s + p.x * p.y, 0);
    const slope = (n * sxy - sx * sy) / (n * sxx - sx * sx);
    return { slope, intercept: (sy - slope * sx) / n };
  };
