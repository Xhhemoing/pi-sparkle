## 结论

**无可落地的新更优解，本轮生产代码零改动，切片维持关闭
（MORE_OPTIMA=no）。** 第二十八遍为本切片**连续第十七次完全空枚举**
（R12-G…R27-G、本轮），并新增一条前二十七轮从未做过的真实新轴：
**决策门裁决面 / decision-gate resolution face（战役首创）**——
`human` / `router` 角色节点是**决策门**（isDecisionGate
flowchart-supervisor.ts:219-221）：它们不执行子工作而是选择后继，其
裁决机械是与普通 actor 租借结构不同的第二条流。被扫格 = **门角色**
（router / human / actor 对照）× **审批需求**（`approvalRequired=false`
租借内同步自动裁决 vs `=true` BRANCH 等待；加同族 ROUTE 类三臂）×
**后继计划几何**（扇出 F、边级 `defaultSelected` 混合、零后继自门退化
形）× **选择子集**（全选 / 部分 / 空选；ROUTE authorize / cancel /
empty）——外加多门单等待者序列化维。该格不是死代码而是运行时裁决
引擎本体：

- **自动裁决臂**（:707,:732-746）：门 `approvalRequired=false` 时在
  `leaseReadyNodes` 内同步完成——`buildBranchPlan`（:624-642）按
  `edge.defaultSelected !== false` 收集后继目标（:630），
  `resolveBranchGate`（:909-935）把门置 COMPLETED（success=true、
  confidence=routedConfidence）并 SKIP 全部未选后继——零用户交互、
  零子结果，租约状态 "COMPLETED"。
- **BRANCH 等待臂**（:709-730）：门 `approvalRequired=true` 时挂起
  监督者构建的后继计划为 pendingApproval；回复的选择子集决定每个后继
  的命运；空选**合法**跳过全部可选分支（:904-907 注释即契约）。
- **自门退化形**（:632-634,:915-925）：零出边门的计划退化为
  `[{id: node.id}]`；批准 → 门 COMPLETED success=true，拒绝 → SKIPPED
  success=false 而 run 终态仍 COMPLETED。
- **同族 ROUTE 类**（:845-855）：普通（非门）节点 `approvalRequired=

    raw.replaceAll(stateRoot, "<STATE>").replaceAll(projectRoot, "<PROJ>")

}

async function driveCell(spec: CellSpec, index: number): Promise<CellOut> {
  const digests: string[] = [];
  let out: CellOut | undefined;
  for (let rep = 0; rep < REPS; rep++) {
    const stateRoot = mkdtempSync(join(tmpdir(), `r28g-gateface-c${index}-`));
    try {
      const deps = {
        stateRoot, router: router(), now: () => T0,
        generateId: sequenceGenerator(String((index % 9) + 1)),
        pause: createFilePauseController(stateRoot, () => T0)
      };
      const phases: Phase[] = [];
      let planNote = "-";

      resetCounts();
      let started = performance.now();
      let outcome = await startFlowchartRun(deps, {
        projectRoot, flowchart: spec.flowchart,
        objective: "gateface drive", childResults: spec.childResults
      });
      phases.push({ label: "start", counts: countsLine(), wallMs: performance.now() - started });
      if (outcome.status !== spec.expectStartStatus) {
        throw new Error(`${spec.name}: start status ${outcome.status} != ${spec.expectStartStatus}`);
      }
      if (spec.probeReadyDuringWait !== undefined) {
        const state = (outcome.snapshot.nodes as Record<string, { state: string }>)[spec.probeReadyDuringWait]!.state;
        witness(`    WAITER ${spec.name} :: node ${spec.probeReadyDuringWait} is ${state} while the first gate waits (single-waiter invariant)`);
        if (state !== "READY") throw new Error(`${spec.name}: expected ${spec.probeReadyDuringWait} READY, got ${state}`);
      }

      for (const [stepIndex, step] of (spec.approvals ?? []).entries()) {
        const pending = outcome.pendingApproval;
        if (pending === undefined) throw new Error(`${spec.name}: expected a pending approval`);
        if (rep === 0 && stepIndex === 0) {
          planNote = `plan ${pending.plan.id} items=[${pending.plan.items
            .map((item) => `${item.id}(def=${item.defaultSelected})`)
            .join(", ")}]`;
        }
        const selected = step.select(pending.plan);
        resetCounts();
        started = performance.now();
        outcome = await resumeFlowchartRun(deps, outcome.runId, {
          approvalReply: { approvalPlanId: pending.plan.id, selectedActionIds: selected },
          ...(step.withResults !== undefined ? { childResults: step.withResults } : {})
        });
        phases.push({ label: `resume${stepIndex + 1}`, counts: countsLine(), wallMs: performance.now() - started });
        if (outcome.status !== step.expectStatus) {
          throw new Error(`${spec.name}: resume status ${outcome.status} != ${step.expectStatus}`);
        }
        if (rep === 0) witness(`    APPROVAL ${spec.name} :: step ${stepIndex + 1} [${selected.join(",") || "none"}] -> ${outcome.status} (${step.note})`);
      }

      const dir = join(stateRoot, "runtime", "runs", outcome.runId);
      const normalize = normalizeFactory(stateRoot);
      const eventsRaw = await readFile(join(dir, "events.jsonl"), "utf8");
      const checkpointRaw = await readFile(join(dir, "checkpoint.json"), "utf8");
      digests.push(createHash("sha256").update(normalize(eventsRaw)).update(normalize(checkpointRaw)).digest("hex"));

      const nodes = outcome.snapshot.nodes as Record<string, { state: string; success?: boolean; confidence?: number }>;
      for (const [id, fate] of Object.entries(spec.expectFates)) {
        if (nodes[id]!.state !== fate) {
          throw new Error(`${spec.name}: node ${id} fate ${nodes[id]!.state} != ${fate}`);
        }
      }
      for (const [id, carry] of Object.entries(spec.expectCarry ?? {})) {
        if (carry.success !== undefined && nodes[id]!.success !== carry.success) {
          throw new Error(`${spec.name}: node ${id} success ${nodes[id]!.success} != ${carry.success}`);
        }
        if (carry.confidence !== undefined && nodes[id]!.confidence !== carry.confidence) {
          throw new Error(`${spec.name}: node ${id} confidence ${nodes[id]!.confidence} != ${carry.confidence}`);
        }
      }
      out = {
        name: spec.name,
        phases,
        eventTypes: eventsRaw.trim().split("\n").map((line) => (JSON.parse(line) as { type: string }).type).join(","),
        carry: Object.keys(spec.expectFates)
          .map((id) => `${id}=${nodes[id]!.state}${nodes[id]!.success !== undefined ? `/succ:${nodes[id]!.success}` : ""}${nodes[id]!.confidence !== undefined ? `/conf:${nodes[id]!.confidence}` : ""}`)
          .join(" "),
        planNote,
        digest: digests[digests.length - 1]!,
        bytes: checkpointRaw.length
      };
    } finally {
      rmSync(stateRoot, { recursive: true, force: true });
    }
  }
  if (!digests.every((d) => d === digests[0])) throw new Error(`${spec.name}: unstable artifacts`);
  return out!;
}

/* SHAPE / PRUNE / TWIN comparisons */
{
  const events = (cell: CellOut): number => cell.eventTypes.split(",").length;
  const routed = (cell: CellOut): number => cell.eventTypes.split(",").filter((t) => t === "MODEL_ROUTED").length;

  const g1 = cOuts.get("G1")!;
  const g2 = cOuts.get("G2")!;
  const sameEvents = g1.eventTypes === g2.eventTypes;
  const sameCounts = JSON.stringify(g1.phases.map((p) => p.counts)) === JSON.stringify(g2.phases.map((p) => p.counts));
  witness(`  SHAPE G2 vs G1 (human vs router auto-gate): events-identical=${sameEvents} counts-identical=${sameCounts} bytes ${g2.bytes} vs ${g1.bytes}`);
  if (!sameEvents || !sameCounts) throw new Error("gate-role swap changed the run shape");

  const g0 = cOuts.get("G0")!;
  witness(
    `  AUTO vs ACTOR G1 vs G0: events ${events(g1)} vs ${events(g0)}; MODEL_ROUTED ${routed(g1)} vs ${routed(g0)}; ` +
    `start-counts {${g1.phases[0]!.counts}} vs {${g0.phases[0]!.counts}} ` +
    "(auto-gate completes without a child result; the gate still routes and costs one MODEL_ROUTED)"
  );

  for (const [miss, hit, label] of [
    ["G3", "G1", "defaultSelected:false deselection"],
    ["G4E", "G4", "empty vs full BRANCH selection"],
    ["G7X", "G7", "ROUTE cancel vs approve"]
  ] as const) {
    const missCell = cOuts.get(miss)!;
    const hitCell = cOuts.get(hit)!;
    witness(
      `  PRUNE ${miss} vs ${hit} (${label}): MODEL_ROUTED ${routed(missCell)} vs ${routed(hitCell)}; events ${events(missCell)} vs ${events(hitCell)}; ` +
      `checkpoint ${missCell.bytes} vs ${hitCell.bytes} B`
    );
    if (routed(missCell) >= routed(hitCell) || events(missCell) >= events(hitCell)) {
      throw new Error(`${miss}: deselection did not prune work`);
    }
  }

  const g7x = cOuts.get("G7X")!;
  const g7e = cOuts.get("G7E")!;
  const twinCounts = JSON.stringify(g7x.phases.map((p) => p.counts)) === JSON.stringify(g7e.phases.map((p) => p.counts));
  witness(
    `  TWIN G7E vs G7X (empty vs cancel): events-identical=${g7x.eventTypes === g7e.eventTypes} counts-identical=${twinCounts} ` +
    `digest-equal=${g7x.digest === g7e.digest} (fates identical; only the USER_ANSWER payload differs)`
  );
  if (!twinCounts) throw new Error("G7E/G7X count vectors diverged");
}

function preppedSupervisor(fan: number, gate: MNodeOpts): FlowchartSupervisor {
  const s = createFlowchartSupervisor({
    flowchart: diamond(`m1-${fan}`, fan, gate),
    router: router(),
    limits: { maxConcurrentNodes: 1, maxConsecutiveStalls: 3 },
    generateId: () => "00000000-0000-4000-8000-000000000000",
    now: () => T0
  });
  s.leaseReadyNodes();
  s.applyChildResult("a", OK);
  return s;
}
function leasePass(fan: number, gate: MNodeOpts, expectG: string, expectS0: string | undefined, pool: number): number {
  const supervisors = Array.from({ length: pool }, () => preppedSupervisor(fan, gate));
  const start = performance.now();
  for (const s of supervisors) s.leaseReadyNodes();
  const us = ((performance.now() - start) * 1000) / pool;
  const g = supervisors[0]!.nodeState("g");
  if (g !== expectG) throw new Error(`M1 fan=${fan}: g=${g} != ${expectG}`);
  if (expectS0 !== undefined) {
    const s0 = supervisors[0]!.nodeState("s0");
    if (s0 !== expectS0) throw new Error(`M1 fan=${fan}: s0=${s0} != ${expectS0}`);
  }
  return us;
}
