function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

}

/** Widen an object with n surplus own enumerable keys (fixed ASCII
 * texture names, SMI values -- content/value classes pinned). Declared
 * keys are inserted first (declaration order preserved). */
function widen<T extends object>(base: T, n: number): T {
  const t: Record<string, unknown> = { ...base };
  for (let i = 0; i < n; i += 1) t[`x_${String(i).padStart(5, "0")}`] = i;
  return t as unknown as T;
}

/** Slow-mode twin: same declared keys/values, property backing forced to
 * dictionary mode by deleting a non-last own property (V8 map
 * deprecation on non-trailing delete). Descriptor class stays data,
 * lookup base stays own, no membrane -- only the BACKING moves. */
function dictTwin<T extends object>(base: T): T {
  const t: Record<string, unknown> = { ...base };
  t.__f8_a = 1;
  t.__f8_b = 2;
  delete t.__f8_a;
  delete t.__f8_b;
  return t as unknown as T;
}

/** Insertion-order twin: same own keys/values, reversed insertion order. */
function reorderTwin<T extends object>(base: T): T {
  const t: Record<string, unknown> = {};
  const keys = Object.keys(base).reverse();
  for (const k of keys) t[k] = (base as Record<string, unknown>)[k];
  return t as unknown as T;
}

function widenRow(e: Event, n: number): Event {
  const p: Record<string, unknown> = { ...(e as unknown as { payload: Record<string, unknown> }).payload };
  for (let i = 0; i < n; i += 1) p[`px_${String(i).padStart(3, "0")}`] = i;
  const w: Record<string, unknown> = { ...(e as unknown as Record<string, unknown>), payload: p };
  for (let i = 0; i < n; i += 1) w[`ex_${String(i).padStart(3, "0")}`] = i;
  return w as unknown as Event;
}
