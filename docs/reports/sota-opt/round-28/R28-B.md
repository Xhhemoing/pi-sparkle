function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function pick<T>(rng: () => number, items: readonly T[]): T {
  return items[Math.floor(rng() * items.length)] as T;
}
function bench(fn: () => void, reps: number): number {
  fn();

  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}

  ] as const) {
    for (const [armLabel, order] of arrangements) {
      const catalog = permuteModels(baseCatalog, order);
      const replayCost = bench(() => assignTasks({ catalog, tasks: replayTasks, learned }), 30);
      const liveCost = bench(() => assignTasks({ catalog, tasks: cli, learned }), 2000);
      console.log(
        `arrangement [${cellLabel}] arm=${armLabel}: replay N=2000 ${(replayCost * 1e3).toFixed(1)}us per call | live face N=30 ${(liveCost * 1e3).toFixed(2)}us per call`
      );
    }
  }

  // Prelude micro dose: planAssignmentPolicy sort-input-order sensitivity.
  for (const [cellLabel, baseCatalog] of [["M=7", configuredCatalog()], ["M=10", tenModelCatalog()]] as const) {
    const parts: string[] = [];
    for (const [armLabel, order] of arrangements) {
      if (armLabel === "canonicalB" || armLabel.startsWith("shuffle")) continue;
      const catalog = permuteModels(baseCatalog, order);
      const models = createModelRouter(catalog).config.models;
      const catalogIds = catalog.models.map((m) => m.id);
      const cost = bench(() => planAssignmentPolicy(models, catalogIds), 100000);
      parts.push(`${armLabel}=${(cost * 1e6).toFixed(0)}ns`);
    }
    console.log(`arrangement prelude micro [${cellLabel}] planAssignmentPolicy: ${parts.join(" ")}`);
  }

  // W1: double-run byte identity at the most loaded arm (canonical M=7 +learned+prior).
  {
    const catalogCfg = configuredCatalog();
    const runA = JSON.stringify(assignTasks({ catalog: catalogCfg, tasks: cli, learned: configuredLearned(), prior }));
    const runB = JSON.stringify(assignTasks({ catalog: catalogCfg, tasks: cli, learned: configuredLearned(), prior }));
    check("arrangement W1 [M=7 learned+prior, N=30]: double run byte-identical", runA === runB);
  }

  // W2: selection invariance on the distinct-cost catalog (M=10; strict total
  // order). Per arm, per task: model/preferredModel/justification/scores and
  // the multisets of eligibleModels/rejections plus the behaviorDistribution
  // mapping match canonical byte-for-byte; allowedModels order equals the
  // arm's own catalog id order (the documented order-carrying surface).
  {
    const base = tenModelCatalog();
    const learned = tenModelLearned();
    const canonical = assignTasks({ catalog: base, tasks: cli, learned });
    for (const [armLabel, order] of arrangements) {
      if (armLabel === "canonicalA" || armLabel === "canonicalB") continue;
      const catalog = permuteModels(base, order);
      const armIds = catalog.models.map((m) => m.id);
      const out = assignTasks({ catalog, tasks: cli, learned });
      let invariant = out.length === canonical.length;
      let orderCarrying = true;
      for (let i = 0; i < out.length && invariant && orderCarrying; i += 1) {
        const a = canonical[i]!;
        const b = out[i]!;
        invariant =
          b.decision.model === a.decision.model &&
          b.preferredModel === a.preferredModel &&
          b.decision.justification === a.decision.justification &&
          b.decision.coldStartRoutingScore === a.decision.coldStartRoutingScore &&
          b.decision.family === a.decision.family &&
          JSON.stringify([...b.decision.eligibleModels].sort()) === JSON.stringify([...a.decision.eligibleModels].sort()) &&
          JSON.stringify([...b.decision.rejections].sort((l, r) => l.modelId.localeCompare(r.modelId))) ===
            JSON.stringify([...a.decision.rejections].sort((l, r) => l.modelId.localeCompare(r.modelId))) &&
          JSON.stringify(Object.entries(b.decision.behaviorDistribution).sort()) ===
            JSON.stringify(Object.entries(a.decision.behaviorDistribution).sort());
        orderCarrying =
          JSON.stringify(b.allowedModels) === JSON.stringify(armIds.filter((id) => b.allowedModels.includes(id)));
      }
      check(`arrangement W2 [M=10+learned] arm=${armLabel}: selection invariant + allowedModels carries arm order`, invariant && orderCarrying);
    }
  }

  // W3: tie-flip witness on the M=7 configured anchor catalog (exact cost
  // ties via alias rows). Plan level and end-to-end.
  {
    const canonical = configuredCatalog();
    const aliasFirst = permuteModels(canonical, (models) => {
      const byId = new Map(models.map((m) => [m.id, m]));
      const front = [byId.get("premium")!, byId.get("cheap")!];
      return [...front, ...models.filter((m) => m.id !== "premium" && m.id !== "cheap")];
    });
    const planOf = (catalog: ModelRouterConfig) =>
      planAssignmentPolicy(createModelRouter(catalog).config.models, catalog.models.map((m) => m.id));
    const planCanonical = planOf(canonical);
    const planAlias = planOf(aliasFirst);
    check(
      "arrangement W3a [M=7]: canonical plan keeps earliest tie rows (acme/frontier primary, orion/lite cheapest)",
      planCanonical.primaryPreferredId === "acme/frontier" && planCanonical.cheapestAssignableId === "orion/lite",
      `${planCanonical.primaryPreferredId}/${planCanonical.cheapestAssignableId}`
    );
    check(
      "arrangement W3b [M=7]: alias-first permutation flips both tie picks (premium primary, cheap cheapest)",
      planAlias.primaryPreferredId === "premium" && planAlias.cheapestAssignableId === "cheap",
      `${planAlias.primaryPreferredId}/${planAlias.cheapestAssignableId}`
    );
    const plannerTask: AssignableTask = {
      taskId: "tsk_w3" as AssignableTask["taskId"],
      role: "planner",
      objective: "Break down the ledger migration"
    };
    const outCanonical = assignTasks({ catalog: canonical, tasks: [plannerTask] })[0]!;
    const outAlias = assignTasks({ catalog: aliasFirst, tasks: [plannerTask] })[0]!;
    check(
      "arrangement W3c [M=7]: preferPrimary decision.model flips acme/frontier -> premium end-to-end",
      outCanonical.decision.model === "acme/frontier" && outAlias.decision.model === "premium",
      `${outCanonical.decision.model} vs ${outAlias.decision.model}`
    );
  }

  // W4: derived-artifact invariance across arms (each re-sorts with an id
  // tie-break, restoring a strict total order).
  {
    const base = tenModelCatalog();
    const editTask: AssignableTask = {
      taskId: "tsk_w4" as AssignableTask["taskId"],
      role: "implementer",
      objective: "Implement retry logic for the ledger sync"
    };
    const refTiers = JSON.stringify(
      cheapFirstTiers(
        assignTasks({ catalog: base, tasks: [editTask] })[0]!.decision.eligibleModels,
        base.models as never
      )
    );
    const refPick = JSON.stringify(
      pickFromPublicPrior(prior, "edit", planAssignmentPolicy(createModelRouter(base).config.models, base.models.map((m) => m.id)).assignableModels)
    );
    const r0Request = {
      taskFamily: "edit",
      privacyRequired: "cloud-general",
      requiredCapabilities: ["tool-use"],
      contextNeeded: 0,
      outputNeeded: 0,
      budgetUsd: Number.POSITIVE_INFINITY,
      deadlineMs: Number.MAX_SAFE_INTEGER,
      highRisk: false
    } as const;
    const refR0 = JSON.stringify(
      routeR0(
        { confidenceGate: 0.75, cascade: true, policyVersion: "sim-r28b-r0" },
        createModelRouter(base).config.models.map((m) => toModelDescriptor(m)),
        r0Request
      )
    );
    for (const [armLabel, order] of arrangements) {
      if (armLabel === "canonicalA" || armLabel === "canonicalB") continue;
      const catalog = permuteModels(base, order);
      const armTiers = JSON.stringify(
        cheapFirstTiers(
          assignTasks({ catalog, tasks: [editTask] })[0]!.decision.eligibleModels,
          catalog.models as never
        )
      );
      const armPick = JSON.stringify(
        pickFromPublicPrior(prior, "edit", planAssignmentPolicy(createModelRouter(catalog).config.models, catalog.models.map((m) => m.id)).assignableModels)
      );
      const armR0 = JSON.stringify(
        routeR0(
          { confidenceGate: 0.75, cascade: true, policyVersion: "sim-r28b-r0" },
          createModelRouter(catalog).config.models.map((m) => toModelDescriptor(m)),
          r0Request
        )
      );
      check(`arrangement W4 [M=10] arm=${armLabel}: cheapFirstTiers byte-identical`, armTiers === refTiers);
      check(`arrangement W4 [M=10] arm=${armLabel}: pickFromPublicPrior byte-identical`, armPick === refPick);
      check(`arrangement W4 [M=10] arm=${armLabel}: routeR0 decision byte-identical`, armR0 === refR0);
    }
  }

  // W5: order-carrying census at the M=10 corner. All-eligible MEDIUM task:
  // under the reversed catalog eligibleModels is the exact reverse and the
  // behaviorDistribution key sequence follows it; model/justification stay
  // identical.
  {
    const base = tenModelCatalog();
    const reversed = permuteModels(base, (models) => [...models].reverse());
    const task: AssignableTask = {
      taskId: "tsk_w5" as AssignableTask["taskId"],
      role: "implementer",
      objective: "Implement retry logic for the ledger sync"
    };
    const a = assignTasks({ catalog: base, tasks: [task] })[0]!;
    const b = assignTasks({ catalog: reversed, tasks: [task] })[0]!;
    check(
      "arrangement W5a [M=10]: reversed catalog keeps model+justification identical",
      a.decision.model === b.decision.model && a.decision.justification === b.decision.justification
    );
    check(
      "arrangement W5b [M=10]: eligibleModels under reversed catalog is the exact reverse (order-carrying)",
      JSON.stringify([...a.decision.eligibleModels].reverse()) === JSON.stringify(b.decision.eligibleModels) &&
        a.decision.eligibleModels.length === 10
    );
    check(
      "arrangement W5c [M=10]: behaviorDistribution key sequence follows the arm's eligible order",
      JSON.stringify(Object.keys(b.decision.behaviorDistribution)) === JSON.stringify([...b.decision.eligibleModels])
    );
  }

  conclusions.push("arrangement-face: perf grid + prelude micro measured; W1-W5 fidelity witnesses asserted (tie-flip proves catalog order is decision-carrying)");
}

console.log(`\nCONCLUSIONS: ${conclusions.join(" | ")}`);
if (failures > 0) {
  console.error(`\n${failures} check(s) FAILED`);
  process.exit(1);
}
console.log("ALL CHECKS PASSED (measurement harness + catalog-row-arrangement fidelity witnesses; corpora seeded + deterministic)");
