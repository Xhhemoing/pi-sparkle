/**

 * once per arm (setup, untimed); the per-call dose is exactly the dispatch
 * mechanism. The DIRECT arm keeps the per-call property load — that is the
 * 27-round fixture (and production) coordinate, so it stays the baseline.
 */
function buildArmOps(name: ArmName, runner: ShadowRunner): BoundaryOps {
  const assignFn: AssignFn = runner.assign;
  const recordFn: RecordFn = runner.recordOutcome;
  switch (name) {
    case "DIRECT":
      return {
        assign: (s, h, t) => runner.assign(s, h, t),
        record: (s, o, t) => runner.recordOutcome(s, o, t),

    case "LOCAL":
      return {
        assign: (s, h, t) => assignFn(s, h, t),
        record: (s, o, t) => recordFn(s, o, t),
      };
    case "CALL":
      return {
        assign: (s, h, t) => assignFn.call(undefined, s, h, t),
        record: (s, o, t) => recordFn.call(undefined, s, o, t),
      };

      return {
        assign: (s, h, t) => assignFn.apply(undefined, [s, h, t]),
        record: (s, o, t) => recordFn.apply(undefined, [s, o, t]),
      };
    case "REFLECT":

        assign: (s, h, t) => Reflect.apply(assignFn, undefined, [s, h, t]),
        record: (s, o, t) => Reflect.apply(recordFn, undefined, [s, o, t]),
      };

      const boundAssign = assignFn.bind(undefined);
      const boundRecord = recordFn.bind(undefined);

      }
      return {
        assign: (s, h, t) => boundAssign(s, h, t),
        record: (s, o, t) => boundRecord(s, o, t),
      };
    }
    case "SPREAD":
      return {
        assign: (s, h, t) => {
          const args: [ShadowState, string, number] = [s, h, t];
          return assignFn(...args);
        },
        record: (s, o, t) => {
          const args: [ShadowState, ExperimentOutcome, number] = [s, o, t];
          return recordFn(...args);
        },
      };
    case "WRAP": {
      const wrappedAssign = (...args: Parameters<AssignFn>): ShadowState => assignFn(...args);

      if ((wrappedAssign as unknown) === (assignFn as unknown)) {

        assign: (s, h, t) => wrappedAssign(s, h, t),
        record: (s, o, t) => wrappedRecord(s, o, t),
      };
    }
    default:
      throw new Error(`unknown arm: ${String(name)}`);

          for (let c = 0; c < CALLS; c++) call();
        })
      );
    }
    process.stdout.write(
      `CHILD unitprobe ${armName} ${((median(samples) * 1e6) / CALLS).toFixed(3)} ns/call sink=${sink > 0 ? "ok" : "zero"}\n`
    );
    process.exit(0);
  }

  throw new Error(`unknown cell: ${cell}`);
}

/* ----------------------- isolated child-cell driver ------------------------ */
function runChild(cell: string, arm: string): string[] {
  const child = spawnSync(
    process.execPath,
    ["--import", "tsx", process.argv[1] as string, `--cell=${cell}`, `--arm=${arm}`],
    { encoding: "utf8", timeout: 300_000 }
  );
  if (child.status !== 0) {
    return [`CHILD ${cell} ${arm} FAILED status=${String(child.status)} ${child.stderr.slice(0, 300)}`];
  }
  return child.stdout.trim().split("\n");
}

function runIsolatedCell(title: string, cell: string, arms: readonly string[]): Map<string, number> {
  process.stdout.write(`${title}\n`);
  const values = new Map<string, number>();
  for (const arm of arms) {
    for (const line of runChild(cell, arm)) {
      process.stdout.write(`  ${line}\n`);
      const match = line.match(/^CHILD \S+ \S+ ([0-9.]+)/);
      if (match !== null) {
        values.set(arm, Number(match[1]));
      }
    }
  }
  return values;
}
