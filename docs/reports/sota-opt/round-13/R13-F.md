```ts
/**
 * R13-F new-axis measurements (temporary; NOT candidate adjudications and NOT
 * committed — no winner this round). Two axes no earlier round measured
 * directly:
 *
 *   axis 1  validate Ω(P) body decomposition: iteration floor vs native-Set
 *           hash work. R8-F..R12-F carried "the remaining bulk is the native
 *           Set hash-work floor" on two negative witnesses (S6-F-4 user-space
 *           table 38–45% slower; S9-F-3 clear-reuse stably negative) without
 *           single-listing the hash fraction. This measures, per
 *           assertUniqueNonEmpty(population) call at P=2000:
 *             (a) landed mirror (head-guard + add + size-counter)
 *             (b) iteration floor: same loop shape, no Set operations
 *                 (informational probe only — NOT semantically equivalent,
 *                 it proves nothing about uniqueness; never a candidate)
 *           (a)−(b) = the native hash/table work that any uniqueness proof
 *           must pay (S6-F-4/S9-F-3 witnesses say cheaper substitutes lose),
 *           i.e. the non-eliminable fraction of the biggest ledger line.
 *
 *   axis 2  full-experiment anchor P-scaling curve. All rounds measured the
 *           historical cell P=2000/A=1000 (plus the small r5f C2 view at
 *           P=200/A=100). The ledger model predicts cost ≈ c·A·P (2A
 *           validates × Ω(P) each) ⟹ quadratic in the scale factor when
 *           A = P/2. Fitting P ∈ {500, 1000, 2000, 4000} checks there is no
 *           hidden superlinear pool the fixed cell could mask.
 *
 * Uses the production modules directly. Run: npx tsx tmp-r13f-axes.ts
 */

import { createCandidateId, createResourceVersionId } from "./src/domain/ids.js";
import { validateExperimentPlan, type ExperimentPlan } from "./src/experiments/plan.js";
import { createShadowRunner, type ExperimentOutcome } from "./src/experiments/shadow.js";

function timeMs(fn: () => void, rounds = 7): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

function makePopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

function makeShadowPlan(population: readonly string[], maxAssignments: number): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: "exp_r13f_bench",
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r13fbase"),
    candidateId: createCandidateId(() => "r13fcand"),
    population,
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments, maxWallClockMs: 1e12 },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

/* ------------------------------------------------------------------ */
/* axis 1 — validate body: landed mirror vs iteration floor            */
/* ------------------------------------------------------------------ */
const P = 2000;
const CALLS = 2000; // 2A calls at the historical cell
const population = makePopulation(P);

class MirrorError extends Error {}

/** Landed assertUniqueNonEmpty mirror (S5-F + S7-F-2 form, verbatim shape). */
function landedMirror(values: readonly string[]): void {
  if (!Array.isArray(values) || values.length === 0) {
    throw new MirrorError("must be a non-empty array");
  }
  const seen = new Set<string>();
  let unique = 0;
  for (const value of values) {
    if (typeof value !== "string") {
      throw new MirrorError("contains an empty entry");
    }
    const head = value.charCodeAt(0);
    if (!(head > 32 && head < 127) && value.trim() === "") {
      throw new MirrorError("contains an empty entry");
    }
    seen.add(value);
    unique += 1;
    if (seen.size !== unique) {
      throw new MirrorError(`contains a duplicate: ${value}`);
    }
  }
}

/** Iteration floor probe: identical loop shape, zero Set work. Proves nothing
 *  about uniqueness — informational lower bound only, never a candidate. */
let floorSink = 0;
function iterationFloor(values: readonly string[]): void {
  if (!Array.isArray(values) || values.length === 0) {
    throw new MirrorError("must be a non-empty array");
  }
  let unique = 0;
  for (const value of values) {
    if (typeof value !== "string") {
      throw new MirrorError("contains an empty entry");
    }
    const head = value.charCodeAt(0);
    if (!(head > 32 && head < 127) && value.trim() === "") {
      throw new MirrorError("contains an empty entry");
    }
    unique += 1;
    floorSink += head;
  }
  if (unique !== values.length) {
    throw new MirrorError("unreachable");
  }
}

landedMirror(population); // warm + sanity
iterationFloor(population);
const tLanded = timeMs(() => {
  for (let i = 0; i < CALLS; i++) landedMirror(population);
}, 5);
const tFloor = timeMs(() => {
  floorSink = 0;
  for (let i = 0; i < CALLS; i++) iterationFloor(population);
}, 5);
process.stdout.write(
  `axis 1 (P=${P}, x${CALLS} calls): landed mirror=${tLanded.toFixed(2)}ms ` +
    `iteration floor=${tFloor.toFixed(2)}ms ` +
    `hash/table work=${(tLanded - tFloor).toFixed(2)}ms ` +
    `(${(((tLanded - tFloor) / tLanded) * 100).toFixed(1)}% of the mirror)\n`
);

/* ------------------------------------------------------------------ */
/* axis 2 — full-experiment anchor P-scaling curve (production runner) */
/* ------------------------------------------------------------------ */
function fullExperimentMs(populationSize: number, assignments: number): number {
  const pop = makePopulation(populationSize);
  const plan = makeShadowPlan(pop, assignments);
  validateExperimentPlan(plan); // fixture sanity
  const runner = createShadowRunner(plan);
  return timeMs(() => {
    let state = runner.start(0);
    for (let k = 0; k < assignments; k++) {
      state = runner.assign(state, pop[k] as string, 0);
      const outcome: ExperimentOutcome = {
        episodeHash: pop[k] as string,
        utility: 0.5,
        costUsd: 0.01,
        guardrailBreached: false,
      };
      state = runner.recordOutcome(state, outcome, 0);
    }
    if (state.assignments.length !== assignments) {
      throw new Error("unexpected assignment count");
    }
  }, 3);
}

const cells: readonly (readonly [number, number])[] = [
  [500, 250],
  [1000, 500],
  [2000, 1000],
  [4000, 2000],
];
const results: number[] = [];
for (const [p, a] of cells) {
  const ms = fullExperimentMs(p, a);
  results.push(ms);
  process.stdout.write(`axis 2: P=${p} A=${a} full experiment=${ms.toFixed(2)}ms\n`);
}
// Quadratic-fit check: with A = P/2 the ledger model predicts cost ~ c*P^2.
for (let i = 1; i < results.length; i++) {
  const ratio = (results[i] as number) / (results[i - 1] as number);
  process.stdout.write(
    `axis 2: P x2 step ${cells[i - 1]?.[0]} -> ${cells[i]?.[0]}: cost ratio ${ratio.toFixed(2)} (quadratic predicts ~4)\n`
  );
}
process.stdout.write(`sink: ${floorSink}\n`);
```
