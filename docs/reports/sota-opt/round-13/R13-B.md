## 2d. 新测量格：learned 策略规模（矩阵最后一个未变维度闭合）

方法：同一 replay-faithful 语料（seed `0xb44b01`，N=2000），对照两种
learned——**small**（|avoid|=1 + |prefer|=1，矩阵历史格）与 **fat**
（生产 dedup 上界：avoid = 全 (family,model) 格网减每 family 两个幸存者
（primary 恒存活 + 轮转位），prefer = 每 family 一条；M=2/7/10 时
|avoid| = 4/41/64）。fat 策略经探针确认真实过滤（每 family 的 allow-list
收缩到 ≤2 行）。三次运行：

```text
learned-size cell N=2000 M=2 default: delta(fat − small) = −48.8 ~ −199.6us per call（−0.10~−0.40ms per eval x2）
learned-size cell N=2000 M=7 configured: delta = −2605.0 ~ −2714.6us per call（−5.21~−5.43ms per eval x2）
learned-size cell N=2000 M=10 stress: delta = −3850.3 ~ −4763.3us per call（−7.70~−9.53ms per eval x2）
learned-size fat-bound M=10 per-eval: 10.5–11.7ms（still-below-landing-line=true）
```

三条结论：

1. **learned 规模通道是负方向的**：|avoid| 推到上界反而整批变快（四次
   运行 12/12 格全负号稳定）——肥 avoid 把 allowedModels 收缩到 ≤2 行，
   route() 的 partition 主循环（切片成本主体）随之变薄，收缩省费
   （M=10 上 ~8 ms/eval）远大于 applyLearnedRouting 本体的 O(|avoid|)
   过滤增费（≤~1 ms/eval）。
2. 该格不存在隐藏悬崖：learned 无论多肥，整片天花板只降不升（fat M=10
   per-eval 10.5–11.7 ms < small 的 18.7–18.8 ms）。矩阵的 learned 维度
   就此**双端闭合**（空策略端 S6-B-4 已裁决、上界端本轮实测为负方向）。
3. 该发现同时构成 S13-B-1 的机制性否决旁证：learned 层被调方本体从不是
   成本主体，其记忆化没有收益池（§4.1）。

## 4. 关键裁决细节

### 4.1 S13-B-1：等价完备 + 真实流量在场，仍无收益池

S13-B-1 是本切片对 S12-B-2 判决书中最诱人假设的正面检验：S12-B-2 输在
「N=2000 调用方不传 prior」，而 learned 层恰好相反——eval-routing 真实
传 learned。若 learned 层被调方有 prior 层那样的每调用重校验成本
（19.5–19.9 µs/call），S13-B-1 将是全战役 B 切片第一个落地赢家。

实测证伪了收益池的存在：`applyLearnedRouting` 本体在现实 learned
（|avoid|=1）下仅约百 ns/任务（矩阵 learned 增量 0.1–0.6 ms/eval ÷ 4000
任务），其中还包含 memo 无法省掉的每任务防御拷贝；memo 换来的键构造
（`${family}\u0000${preferredModel}` 串分配）+ 新鲜串 Map 哈希本身即
数十 ns/任务——净差额坍缩进调度噪声，三个目录规模 15 轮基准全部符号
翻转。fat 格（|avoid|=64）把本体抬到 ~µs/任务后差额才显形（M=10 15/15
正），但 §2d 已证 fat learned 使**整批更快**——即 learned 层本体成本
的天花板本来就被 allow-list 收缩红利压死，记忆化在任何现实形态下都够
不到落地线。与 X4-6 / S12-B-2 的既有纪律一致（调用次数复杂度类下降 +
等价完备仍以真实规模量级裁决），淘汰。

落地设计成本一并记档（重开时即用）：与 S12-B-2 不同，S13-B-1 的落地
**不需要**改公开签名——memo 与 canonical allowed 副本都可作 `assignTasks`

首抛奇偶性）本报告已备。

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/**
 * Fat learned policy at the production dedup bound: policyFromOutcomes dedups
 * avoid by (modelId, family), so |avoid| <= M x 8. Build the full grid minus
 * two survivors per family — the primary (always kept: it is the only
 * high-risk-approved model in the default catalog, so high-risk tasks stay
 * routable) plus one rotating survivor — and one prefer per family.
 */
function fatLearned(catalog: ModelRouterConfig, primaryId: string): LearnedRoutingPolicy {
  const ids = catalog.models.map((m) => m.id);
  const avoid: { modelId: string; family: string; reason: string }[] = [];
  for (const [fi, family] of FAMILIES.entries()) {
    const rotating = ids[fi % ids.length]!;
    for (const id of ids) {
      if (id === rotating || id === primaryId) continue;
      avoid.push({ modelId: id, family, reason: "sim: fat grid FAIL" });
    }
  }
  const prefer = FAMILIES.map((family, fi) => ({ family, modelId: ids[fi % ids.length]! }));
  return { primaryModelId: primaryId, avoid, prefer };
}

 * Within one assignTasks batch, applyLearnedRouting's four inputs are
 * (family, allowed-contents, preferredModel, learned) where allowed-contents
 * and learned are batch-invariant, so the result is a pure function of
 * (family, preferredModel) — a domain of <= 8 x |preferred-values| per batch.
 * The memo stores the returned { allowedModels, preferredModel } object; the
 * call-site defensive copy [...applied.allowedModels] stays per task, so the
 * per-assignment array identity contract (S1-B-8) is untouched.
 * Replica chain: (1) replicaCurrent outcome-identical to real assignTasks
 * (byte-equal rows OR first-throw class+message parity); (2) replicaMemo
 * outcome-identical to replicaCurrent; (3) per-assignment allowedModels

 * ============================================================ */
const DEFAULT_LIMITS: RoutingLimits = { remainingTimeMs: Number.MAX_SAFE_INTEGER };
function assignTasksReplica(input: AssignTasksInput, memoized: boolean): readonly TaskAssignment[] {
  const router: ModelRouter = createModelRouter(input.catalog);
  const catalogIds = input.catalog.models.map((model) => model.id);
  const plan: AssignmentPolicyPlan = planAssignmentPolicy(router.config.models, catalogIds);
  const limits: RoutingLimits = {
    remainingTimeMs: input.remainingTimeMs ?? DEFAULT_LIMITS.remainingTimeMs,
    ...(input.remainingCostUsd !== undefined ? { remainingCostUsd: input.remainingCostUsd } : {})
  };
  const learned = input.learned;
  const memo =
    memoized && learned !== undefined
      ? new Map<string, { allowedModels: readonly string[]; preferredModel: string }>()
      : undefined;
  const canonicalAllowed: readonly string[] = memo !== undefined ? [...plan.allowedIds] : [];
  return input.tasks.map((task) => {
    const analysis = analyzeTask(task.objective, task.role, {
      ...(task.contractRisk !== undefined ? { contractRisk: task.contractRisk } : {}),
      ...(task.contextTokens !== undefined ? { contextTokens: task.contextTokens } : {}),
      ...(task.outputTokens !== undefined ? { outputTokens: task.outputTokens } : {})
    });
    let allowedModels: readonly string[];
    let preferredModel = pickPreferredModel(plan, analysis, input.prior);
    if (learned !== undefined) {
      if (memo !== undefined) {
        const key = `${analysis.family}\u0000${preferredModel}`;
        let applied = memo.get(key);
        if (applied === undefined) {
          applied = applyLearnedRouting(analysis.family, canonicalAllowed, preferredModel, learned);
          memo.set(key, applied);
        }
        allowedModels = [...applied.allowedModels];
        preferredModel = applied.preferredModel;
      } else {
        const perTaskAllowed: readonly string[] = [...plan.allowedIds];
        const applied = applyLearnedRouting(analysis.family, perTaskAllowed, preferredModel, learned);
        allowedModels = [...applied.allowedModels];
        preferredModel = applied.preferredModel;
      }
    } else {
      allowedModels = [...plan.allowedIds];
    }
    const decision = router.route({
      taskId: task.taskId,
      role: flowchartRoleForAgentRole(task.role),
      complexity: analysis.complexity,
      modelPolicy: { allowedModels, preferredModel },
      approvalRequired: analysis.highRisk,
      highRisk: analysis.highRisk,
      family: analysis.family,
      featureVersion: ASSIGN_FEATURE_VERSION,
      agentRole: task.role,
      requiredCapabilities: analysis.requiredCapabilities,
      ...(analysis.contextTokens !== undefined ? { contextNeeded: analysis.contextTokens } : {}),
      ...(analysis.outputTokens !== undefined ? { outputNeeded: analysis.outputTokens } : {}),
      limits
    });
    return {
      taskId: task.taskId,
      role: task.role,
      analysis,
      decision,
      allowedModels,
      preferredModel
    };
  });
}
