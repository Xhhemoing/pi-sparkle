function buildDesign(rows: readonly Row[]): Design {
  const scenarios = [...new Set(rows.map((r) => r.scenarioId))];
  const models = [...new Set(rows.map((r) => r.modelVersion))];
  const projects = [...new Set(rows.map((r) => r.projectId))];
  const dropLast = (levels: readonly string[]): string[] =>
    levels.slice(0, Math.max(0, levels.length - 1));
  const scenarioLevels = dropLast(scenarios);
  const modelLevels = dropLast(models);
  const projectLevels = dropLast(projects);

  const pairCounts = new Map<string, number>();
  for (const row of rows) {
    const key = `${row.modelVersion}|${row.projectId}`;
    pairCounts.set(key, (pairCounts.get(key) ?? 0) + 1);
  }
  const interactionPairs = [...pairCounts.entries()]
    .filter(([, n]) => n >= INTERACTION_MIN_N)
    .map(([key]) => key);

  const names = [
    "intercept",
    ...scenarioLevels.map((s) => `a:${s}`),
    ...modelLevels.map((m) => `u:${m}`),
    ...projectLevels.map((p) => `v:${p}`),
    ...interactionPairs.map((key) => `w:${key}`)
  ];
  const columnIndex = new Map(names.map((name, index) => [name, index] as const));
  const interactionPairSet = new Set(interactionPairs);

  const referenceLevels: Array<{ factor: "a" | "u" | "v"; name: string }> = [];
  const lastModel = models[models.length - 1];
  const lastProject = projects[projects.length - 1];
  if (lastModel !== undefined) referenceLevels.push({ factor: "u", name: lastModel });
  if (lastProject !== undefined) referenceLevels.push({ factor: "v", name: lastProject });

  return {
    names,
    columnIndex,
    referenceLevels,
    build(row: Row, skip?: string): number[] {
      const vec = new Array<number>(names.length).fill(0);
      vec[0] = 1;
      const set = (name: string): void => {
        if (name === skip) return;
        const index = columnIndex.get(name);
        if (index !== undefined && index > 0) vec[index] = 1;
      };
      if (row.scenarioId !== scenarios[scenarios.length - 1]) set(`a:${row.scenarioId}`);
      if (row.modelVersion !== models[models.length - 1]) set(`u:${row.modelVersion}`);
      if (row.projectId !== projects[projects.length - 1]) set(`v:${row.projectId}`);
      const pairKey = `${row.modelVersion}|${row.projectId}`;
      if (interactionPairSet.has(pairKey)) set(`w:${pairKey}`);
      return vec;
    }
  };
}

/* verbatim landed solveSymmetric (lin-alg.ts) */
function solveInst(a: readonly (readonly number[])[], b: readonly number[]): number[] | null {
  const n = b.length;
  if (a.length !== n || a.some((row) => row.length !== n)) {
    throw new DomainValidationError("solveSymmetric requires a square matrix matching b");
  }
  const m: number[][] = a.map((row) => [...row]);
  const x: number[] = [...b];
  const eps = 1e-12;

  for (let col = 0; col < n; col++) {
    let pivotRow = col;
    let pivotAbs = Math.abs(m[col]![col]!);
    for (let row = col + 1; row < n; row++) {
      const abs = Math.abs(m[row]![col]!);
      if (abs > pivotAbs) {
        pivotAbs = abs;
        pivotRow = row;
      }
    }
    if (pivotAbs < eps) return null;
    if (pivotRow !== col) {
      const tmp = m[col]!;
      m[col] = m[pivotRow]!;
      m[pivotRow] = tmp;
      const tb = x[col]!;
      x[col] = x[pivotRow]!;
      x[pivotRow] = tb;
    }
    const colArr = m[col]!;
    const pivot = colArr[col]!;
    const xCol = x[col]!;
    for (let row = col + 1; row < n; row++) {
      const rowArr = m[row]!;
      const factor = rowArr[col]! / pivot;
      if (factor === 0) continue;
      let k = col;
      const stop = n - 3;
      for (; k < stop; k += 4) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
        rowArr[k + 1] = rowArr[k + 1]! - factor * colArr[k + 1]!;
        rowArr[k + 2] = rowArr[k + 2]! - factor * colArr[k + 2]!;
        rowArr[k + 3] = rowArr[k + 3]! - factor * colArr[k + 3]!;
      }
      for (; k < n; k++) {
        rowArr[k] = rowArr[k]! - factor * colArr[k]!;
      }
      x[row] = x[row]! - factor * xCol;
    }
  }

  const solution = new Array<number>(n).fill(0);
  for (let row = n - 1; row >= 0; row--) {
    const rowArr = m[row]!;
    let sum = x[row]!;
    for (let k = row + 1; k < n; k++) sum -= rowArr[k]! * solution[k]!;
    const diag = rowArr[row]!;
    if (Math.abs(diag) < eps) return null;
    solution[row] = sum / diag;
  }
  for (const value of solution) {
    if (!Number.isFinite(value)) return null;
  }
  return solution;
}

    solution[row] = sum / diag;
  }
  for (const value of solution) {
    PROBE.finiteCheckVisits += 1;
    if (!Number.isFinite(value)) return null;
  }
  return solution;
}

/* verbatim landed irls (offline-logit.ts), timers/counters only added;
 * parameterized by the solve implementation for the counting run. */
function irlsInst(
  design: Design,
  rows: readonly Row[],
  vectors: readonly number[][],
  supports: readonly (readonly number[])[],
  keys: readonly number[],
  keySpace: number,
  maxIter: number,
  solve: SolveImpl
): FitResult {
  const tIrls0 = performance.now();
  STATS.fits += 1;
  const p = design.names.length;
  const n = rows.length;
  const eta = new Array<number>(n).fill(0);
  const mu = new Array<number>(n).fill(0);
  const xtwx: number[][] = Array.from({ length: p }, () => new Array<number>(p).fill(0));
  const xtwz: number[] = new Array<number>(p).fill(0);
  const stamp = new Int32Array(keySpace);
  const etaByKey = new Float64Array(keySpace);
  const muByKey = new Float64Array(keySpace);
  let beta = new Array<number>(p).fill(0);
  let result: FitResult | null = null;
  for (let iter = 0; iter < maxIter; iter++) {
    STATS.irlsIterations += 1;
    STATS.rowVisits += n;
    const mark = iter + 1;
    for (let i = 0; i < n; i++) {
      const key = keys[i]!;
      if (stamp[key] !== mark) {
        stamp[key] = mark;
        STATS.sigmoidIrls += 1;
        let value = 0;
        const active = supports[i]!;
        for (let ai = 0; ai < active.length; ai++) value += beta[active[ai]!]!;
        etaByKey[key] = value;
        muByKey[key] = sigmoid(value);
      }
      eta[i] = etaByKey[key]!;
      mu[i] = muByKey[key]!;
    }
    for (let d = 0; d < p; d++) xtwx[d]!.fill(0);
    xtwz.fill(0);
    for (let i = 0; i < n; i++) {
      const w = Math.max(mu[i]! * (1 - mu[i]!)!, 1e-10);
      const z = eta[i]! + ((rows[i]!.y - mu[i]!) / w);
      const wz = w * z;
      const active = supports[i]!;
      switch (active.length) {
        case 5: {
          const a0 = active[0]!;
          const a1 = active[1]!;
          const a2 = active[2]!;
          const a3 = active[3]!;
          const a4 = active[4]!;
          xtwz[a0] = xtwz[a0]! + wz;
          const r0 = xtwx[a0]!;
          r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w; r0[a2] = r0[a2]! + w; r0[a3] = r0[a3]! + w; r0[a4] = r0[a4]! + w;
          xtwz[a1] = xtwz[a1]! + wz;
          const r1 = xtwx[a1]!;
          r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w; r1[a2] = r1[a2]! + w; r1[a3] = r1[a3]! + w; r1[a4] = r1[a4]! + w;
          xtwz[a2] = xtwz[a2]! + wz;
          const r2 = xtwx[a2]!;
          r2[a0] = r2[a0]! + w; r2[a1] = r2[a1]! + w; r2[a2] = r2[a2]! + w; r2[a3] = r2[a3]! + w; r2[a4] = r2[a4]! + w;
          xtwz[a3] = xtwz[a3]! + wz;
          const r3 = xtwx[a3]!;
          r3[a0] = r3[a0]! + w; r3[a1] = r3[a1]! + w; r3[a2] = r3[a2]! + w; r3[a3] = r3[a3]! + w; r3[a4] = r3[a4]! + w;
          xtwz[a4] = xtwz[a4]! + wz;
          const r4 = xtwx[a4]!;
          r4[a0] = r4[a0]! + w; r4[a1] = r4[a1]! + w; r4[a2] = r4[a2]! + w; r4[a3] = r4[a3]! + w; r4[a4] = r4[a4]! + w;
          break;
        }
        case 4: {
          const a0 = active[0]!;
          const a1 = active[1]!;
          const a2 = active[2]!;
          const a3 = active[3]!;
          xtwz[a0] = xtwz[a0]! + wz;
          const r0 = xtwx[a0]!;
          r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w; r0[a2] = r0[a2]! + w; r0[a3] = r0[a3]! + w;
          xtwz[a1] = xtwz[a1]! + wz;
          const r1 = xtwx[a1]!;
          r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w; r1[a2] = r1[a2]! + w; r1[a3] = r1[a3]! + w;
          xtwz[a2] = xtwz[a2]! + wz;
          const r2 = xtwx[a2]!;
          r2[a0] = r2[a0]! + w; r2[a1] = r2[a1]! + w; r2[a2] = r2[a2]! + w; r2[a3] = r2[a3]! + w;
          xtwz[a3] = xtwz[a3]! + wz;
          const r3 = xtwx[a3]!;
          r3[a0] = r3[a0]! + w; r3[a1] = r3[a1]! + w; r3[a2] = r3[a2]! + w; r3[a3] = r3[a3]! + w;
          break;
        }
        case 3: {
          const a0 = active[0]!;
          const a1 = active[1]!;
          const a2 = active[2]!;
          xtwz[a0] = xtwz[a0]! + wz;
          const r0 = xtwx[a0]!;
          r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w; r0[a2] = r0[a2]! + w;
          xtwz[a1] = xtwz[a1]! + wz;
          const r1 = xtwx[a1]!;
          r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w; r1[a2] = r1[a2]! + w;
          xtwz[a2] = xtwz[a2]! + wz;
          const r2 = xtwx[a2]!;
          r2[a0] = r2[a0]! + w; r2[a1] = r2[a1]! + w; r2[a2] = r2[a2]! + w;
          break;
        }
        case 2: {
          const a0 = active[0]!;
          const a1 = active[1]!;
          xtwz[a0] = xtwz[a0]! + wz;
          const r0 = xtwx[a0]!;
          r0[a0] = r0[a0]! + w; r0[a1] = r0[a1]! + w;
          xtwz[a1] = xtwz[a1]! + wz;
          const r1 = xtwx[a1]!;
          r1[a0] = r1[a0]! + w; r1[a1] = r1[a1]! + w;
          break;
        }
        default: {
          for (let ai = 0; ai < active.length; ai++) {
            const a = active[ai]!;
            xtwz[a] = xtwz[a]! + wz;
            const rowA = xtwx[a]!;
            for (let bi = 0; bi < active.length; bi++) {
              const b = active[bi]!;
              rowA[b] = rowA[b]! + w;
            }
          }
        }
      }
    }
    for (let d = 0; d < p; d++) xtwx[d]![d] = xtwx[d]![d]! + 1e-6;
    const tSolve0 = performance.now();
    const next = solve(xtwx, xtwz);
    STATS.solveMs += performance.now() - tSolve0;
    STATS.solveCalls += 1;
    if (next === null) { result = { coefficients: null }; break; }
    const delta = next.map((value, index) => value - beta[index]!);
    const l2 = Math.sqrt(delta.reduce((acc, d) => acc + d * d, 0));
    beta = next;
    if (!beta.every(Number.isFinite)) { result = { coefficients: null }; break; }
    if (l2 < TOL) break;
  }
  STATS.irlsMs += performance.now() - tIrls0;
  return result ?? { coefficients: beta };
}

function dot(a: readonly number[], b: readonly number[]): number {
  let sum = 0;
  for (let i = 0; i < a.length; i++) sum += a[i]! * b[i]!;
  return sum;
}

function onProbInst(vectors: readonly number[][], coefficients: readonly number[]): number[] {
  const t0 = performance.now();
  const result = vectors.map((vector) => sigmoid(dot(coefficients, vector)));
  STATS.onProbMs += performance.now() - t0;
  STATS.onProbSigmoids += vectors.length;
  return result;
}

function apcInst(
  design: Design,
  rows: readonly Row[],
  vectors: readonly number[][],
  coefficients: readonly number[],
  onProbabilities: readonly number[],
  column: string
): number {
  const t0 = performance.now();
  STATS.apcCalls += 1;
  let sum = 0;
  const columnIdx = design.columnIndex.get(column);
  if (columnIdx !== undefined && columnIdx !== 0) {
    for (let i = 0; i < rows.length; i++) {
      if (vectors[i]![columnIdx] === 0) continue;
      STATS.apcActivePairs += 1;
      const on = onProbabilities[i]!;
      const offVector = vectors[i]!.slice();
      offVector[columnIdx] = 0;
      const off = sigmoid(dot(coefficients, offVector));
      sum += on - off;
    }
  } else {
    STATS.apcColumnlessCalls += 1;
  }
  const result = rows.length === 0 ? 0 : sum / rows.length;
  STATS.apcMs += performance.now() - t0;
  return result;
}

function percentile(sortedValues: readonly number[], q: number): number {
  if (sortedValues.length === 0) return Number.NaN;
  const index = (sortedValues.length - 1) * q;
  const lo = Math.floor(index);
  const hi = Math.ceil(index);
  if (lo === hi) return sortedValues[lo]!;
  return sortedValues[lo]! * (hi - index) + sortedValues[hi]! * (index - lo);
}

function modelOf(effectName: string): string {
  return effectName.slice("u:".length);
}

function pushValue(map: Map<string, number[]>, key: string, value: number): void {
  STATS.drawsMapOps += 1;
  const list = map.get(key);
  if (list === undefined) map.set(key, [value]);
  else list.push(value);
}

function uncertainReport(rowsUsed: number, reason: string): AttributionReport {
  return {
    estimator: "logit-additive",
    rowsUsed,
    effects: [],
    diagnosis: "uncertain",
    reason,
    writesActivePointer: false
  };
}

function fitInst(
  rows: readonly OfflineRow[],
  options?: { readonly maxIter?: number; readonly bootstrap?: number; readonly seed?: number },
  solve: SolveImpl = solveInst
): AttributionReport {
  const tTotal0 = performance.now();
  const maxIter = options?.maxIter ?? MAX_ITER_DEFAULT;
  const bootstrapDraws = options?.bootstrap ?? BOOTSTRAP_DEFAULT;
  const seed = options?.seed ?? SEED_DEFAULT;

  const tBase0 = performance.now();
  const baseRows: Row[] = rows.map((r) => ({
    scenarioId: r.scenarioId,
    modelVersion: r.modelVersion,
    projectId: r.projectId,
    y: r.y
  }));
  STATS.baseCopyMs += performance.now() - tBase0;
  const effects: Array<{ name: string; point: number; lcb: number; ucb: number }> = [];

  if (baseRows.length === 0 || baseRows.every((r) => r.y === 0) || baseRows.every((r) => r.y === 1)) {
    return uncertainReport(baseRows.length, "INVALID_ESTIMATE: degenerate or empty design");
  }

  const design = buildDesign(baseRows);
  const vectors = baseRows.map((r) => design.build(r));
  const supports = computeSupports(vectors);
  const keys = canonicalRowKeys(baseRows);
  const fit = irlsInst(design, baseRows, vectors, supports, keys, baseRows.length, maxIter, solve);
  if (fit.coefficients === null) {
    return uncertainReport(baseRows.length, "INVALID_ESTIMATE: singular or non-finite Hessian");
  }

  const onProbabilities = onProbInst(vectors, fit.coefficients);
  const pointEffects = new Map<string, number>();
  for (const name of design.names) {
    if (name === "intercept") continue;
    pointEffects.set(
      name,
      apcInst(design, baseRows, vectors, fit.coefficients, onProbabilities, name)
    );
  }
  for (const ref of design.referenceLevels) {
    if (!pointEffects.has(`${ref.factor}:${ref.name}`)) {
      pointEffects.set(`${ref.factor}:${ref.name}`, 0);
    }
  }

  const referenceNames = new Set(
    design.referenceLevels.map((ref) => `${ref.factor}:${ref.name}`)
  );

  const random = rng(seed);
  const draws = new Map<string, number[]>();
  let successful = 0;
  for (let draw = 0; draw < bootstrapDraws; draw++) {
    const tSample0 = performance.now();
    const sample: Row[] = [];
    const sampleVectors: number[][] = [];
    const sampleSupports: number[][] = [];
    const sampleKeys: number[] = [];
    for (let i = 0; i < baseRows.length; i++) {
      const index = Math.floor(random() * baseRows.length);
      sample.push(baseRows[index]!);
      sampleVectors.push(vectors[index]!);
      sampleSupports.push(supports[index]!);
      sampleKeys.push(keys[index]!);
    }
    const collapsed = sample.every((r) => r.y === 0) || sample.every((r) => r.y === 1);
    STATS.samplingMs += performance.now() - tSample0;
    if (collapsed) continue;
    const bootFit = irlsInst(design, sample, sampleVectors, sampleSupports, sampleKeys, baseRows.length, maxIter, solve);
    if (bootFit.coefficients === null) continue;
    successful += 1;
    const sampleOnProbabilities = onProbInst(sampleVectors, bootFit.coefficients);
    for (const [name] of pointEffects.entries()) {
      const value = apcInst(
        design,
        sample,
        sampleVectors,
        bootFit.coefficients,
        sampleOnProbabilities,
        name
      );
      if (referenceNames.has(name)) STATS.refDrawsMapOps += 1;
      pushValue(draws, name, value);
    }
  }

  const tTail0 = performance.now();
  for (const [name, point] of pointEffects.entries()) {
    const values = draws.get(name) ?? [];
    if (successful < MIN_SUCCESSFUL_DRAWS || values.length < MIN_SUCCESSFUL_DRAWS) {
      return uncertainReport(baseRows.length, "INVALID_ESTIMATE: fewer than 20 successful bootstrap draws");
    }
    values.sort((a, b) => a - b);
    effects.push({
      name,
      point,
      lcb: percentile(values, 0.025),
      ucb: percentile(values, 0.975)
    });
  }

  const n = baseRows.length;
  const mean = baseRows.reduce((acc, r) => acc + r.y, 0) / n;
  const muPosterior = { alpha: 1 + n * mean, beta: 1 + n * (1 - mean) };
  const muLcb = betaQuantileLcbLocal(muPosterior, 0.05);
  const models = new Set(baseRows.map((r) => r.modelVersion)).size;
  const projects = new Set(baseRows.map((r) => r.projectId)).size;
  const ZERO_EPS = 0.005 * ATTRIBUTION_EFFECT;
  const containsZero = (e: { lcb: number; ucb: number }): boolean =>
    e.lcb <= ZERO_EPS && e.ucb >= -ZERO_EPS;
  const interactionsFor = (prefix: string): Array<{ name: string; lcb: number; ucb: number; point: number }> =>
    effects.filter((e) => e.name.startsWith(`w:${prefix}`));

  let diagnosis: AttributionReport["diagnosis"] = "uncertain";
  const scenarioHard = muLcb < QUALITY_FLOOR && models >= 2 && projects >= 3;
  if (
    effects.some(
      (e) => e.name.startsWith("u:") && e.lcb < -ATTRIBUTION_EFFECT && interactionsFor(modelOf(e.name)).every(containsZero)
    )
  ) {
    diagnosis = "model-problem";
  } else if (
    effects.some(
      (e) =>
        e.name.startsWith("v:") &&
        e.lcb < -ATTRIBUTION_EFFECT &&
        effects
          .filter((w) => w.name.startsWith("w:") && w.name.endsWith(`|${e.name.slice("v:".length)}`))
          .every(containsZero)
    )
  ) {
    diagnosis = "project-problem";
  } else if (effects.some((e) => e.name.startsWith("w:") && e.lcb < -ATTRIBUTION_EFFECT)) {
    diagnosis = "interaction-only";
  } else if (scenarioHard) {
    diagnosis = "scenario-hard";
  }

  const report: AttributionReport = {
    estimator: "logit-additive",
    rowsUsed: baseRows.length,
    effects,
    diagnosis,
    reason: diagnosis === "uncertain" ? "no effect beyond threshold or intervals too wide" : `${diagnosis} beyond the ${ATTRIBUTION_EFFECT} effect threshold`,
    writesActivePointer: false
  };
  STATS.tailMs += performance.now() - tTail0;
  STATS.totalMs += performance.now() - tTotal0;
  return report;
}

  return betaQuantileLcb(posterior, quantile);
}

function compareReports(label: string, expected: AttributionReport, actual: AttributionReport): void {

    }
  }
}

function fixtureRng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function pick<T>(r: () => number, items: readonly T[]): T {
  return items[Math.floor(r() * items.length)]!;
}

function randomRows(
  r: () => number,
  options: {
    scenarios: number;
    models: number;
    projects: number;
    rows: number;
    passRate: number;
    pipeInModel?: boolean;
  }
): OfflineRow[] {
  const scenarios = Array.from({ length: options.scenarios }, (_, i) => `fam${i}|role${i % 2}`);
  const models = Array.from({ length: options.models }, (_, i) =>
    options.pipeInModel === true && i === 0 ? "weird|model-0" : `model-${i}`
  );
  const projects = Array.from({ length: options.projects }, (_, i) => `prj_${i}`);
  const rows: OfflineRow[] = [];
  for (let i = 0; i < options.rows; i++) {
    const modelVersion = pick(r, models);
    const rate = modelVersion === models[0] ? options.passRate * 0.4 : options.passRate;
    rows.push({
      scenarioId: pick(r, scenarios),
      modelVersion,
      projectId: pick(r, projects),
      y: r() < rate ? 1 : 0,
      occurredAtMs: 1_000 + i,
    });
  }
  return rows;
}

function allUniqueRows(r: () => number, count: number): OfflineRow[] {
  const rows: OfflineRow[] = [];
  for (let i = 0; i < count; i++) {
    rows.push({
      scenarioId: `s${i % 2}`,
      modelVersion: `m${i}`,
      projectId: `p${i}`,
      y: r() < 0.6 ? 1 : 0,
      occurredAtMs: 1_000 + i,
    });
  }
  return rows;
}

function scenarioProfile(): void {
  const shapes: Array<[string, OfflineRow[]]> = [
    ["small (3 models x 3 projects, 120 rows)",
      randomRows(fixtureRng(0x51a1), { scenarios: 3, models: 3, projects: 3, rows: 120, passRate: 0.6 })],
    ["perf (6 models x 8 projects, 400 rows)", perfRows()],
  ];
  for (const [label, rows] of shapes) {

    for (let rep = 0; rep < (label.startsWith("perf") ? 7 : 3); rep++) {
      const t0 = performance.now();
      fitLogitAdditive(rows, PERF_OPTIONS);
      times.push(performance.now() - t0);
    }
    out(`  shape ${label}: production median ${median(times).toFixed(1)} ms/report (${times.length} reps)`);
  }

  // Instrumented pool table on the perf shape (3 runs; medians reported).
  const rows = perfRows();
  const expected = fitLogitAdditive(rows, PERF_OPTIONS);
  const runs: Array<Record<string, number>> = [];
  for (let rep = 0; rep < 3; rep++) {
    STATS.reset();
    const actual = fitInst(rows, PERF_OPTIONS);
    if (rep === 0) compareReports("R13C-perf-fixture", expected, actual);
    runs.push({
      totalMs: STATS.totalMs, irlsMs: STATS.irlsMs, solveMs: STATS.solveMs,
      onProbMs: STATS.onProbMs, apcMs: STATS.apcMs, samplingMs: STATS.samplingMs,
      tailMs: STATS.tailMs, baseCopyMs: STATS.baseCopyMs,
      fits: STATS.fits, irlsIterations: STATS.irlsIterations, solveCalls: STATS.solveCalls,
      sigmoidIrls: STATS.sigmoidIrls, rowVisits: STATS.rowVisits,
      onProbSigmoids: STATS.onProbSigmoids, apcActivePairs: STATS.apcActivePairs,
      apcCalls: STATS.apcCalls, apcColumnlessCalls: STATS.apcColumnlessCalls,
      drawsMapOps: STATS.drawsMapOps, refDrawsMapOps: STATS.refDrawsMapOps,
    });
  }
  const med = (key: string): number => median(runs.map((r) => r[key]!));
  const total = med("totalMs");
  const solveMs = med("solveMs");
  const irlsOther = med("irlsMs") - solveMs;
  const rest = total - med("irlsMs") - med("onProbMs") - med("apcMs") - med("samplingMs") - med("tailMs");
  out(`instrumented pool table (perf shape, median of 3 instrumented runs; timers only, bitwise-checked):`);
  out(`  total                 ${total.toFixed(1)} ms`);
  out(`  solveSymmetric        ${solveMs.toFixed(1)} ms  (${(100 * solveMs / total).toFixed(1)}%)  [${med("solveCalls")} calls]`);
  out(`  irls minus solve      ${irlsOther.toFixed(1)} ms  (${(100 * irlsOther / total).toFixed(1)}%)  [${med("irlsIterations")} iterations, ${med("rowVisits")} row visits, ${med("sigmoidIrls")} irls sigmoids]`);
  out(`  onProbabilitiesFor    ${med("onProbMs").toFixed(1)} ms  (${(100 * med("onProbMs") / total).toFixed(1)}%)  [${med("onProbSigmoids")} sigmoid+dot]`);
  out(`  APC (all sweeps)      ${med("apcMs").toFixed(1)} ms  (${(100 * med("apcMs") / total).toFixed(1)}%)  [${med("apcActivePairs")} active pairs, ${med("apcCalls")} calls of which ${med("apcColumnlessCalls")} columnless]`);
  out(`  bootstrap sampling    ${med("samplingMs").toFixed(1)} ms  (${(100 * med("samplingMs") / total).toFixed(1)}%)`);
  out(`  sort/percentile tail  ${med("tailMs").toFixed(1)} ms  (${(100 * med("tailMs") / total).toFixed(1)}%)  [${med("drawsMapOps")} draws-map ops of which ${med("refDrawsMapOps")} reference-level]`);
  out(`  base-row projection   ${med("baseCopyMs").toFixed(3)} ms  (once/report)`);
  out(`  outside pools (build/supports/keys/base copies) ${rest.toFixed(1)} ms  (${(100 * rest / total).toFixed(1)}%)`);
  out(`  fits ${med("fits")}`);
}

/* --------- scenario 4: thirteenth-group closure probes (no lanes) --------- */

function scenarioProbes(): void {
  // PIVCSE + FINTAIL exact counts: counting run on the perf shape via
  // solveProbe. The counting run's full report is bitwise-checked against
  // the production import (also covered case-by-case in scenario 1) before
  // counts are read.
  {
    PROBE.reset();
    const rows = perfRows();
    const expected = fitLogitAdditive(rows, PERF_OPTIONS);
    const counted = fitInst(rows, PERF_OPTIONS, solveProbe);
    compareReports("R13C-counting-run", expected, counted);
    out(
      `probe counts: solves=${PROBE.solves} elimRowVisits=${PROBE.elimRowVisits} ` +
        `pivotSearchVisits=${PROBE.pivotSearchVisits} pivotSwaps=${PROBE.pivotSwaps} ` +
        `backSubRows=${PROBE.backSubRows} finiteCheckVisits=${PROBE.finiteCheckVisits}`
    );
    out(
      `  -> PIVCSE arithmetic: a raw-value stash during pivot search deletes exactly ` +
        `${PROBE.elimRowVisits} elimination-loop loads of rowArr[col] and adds exactly ` +
        `${PROBE.pivotSearchVisits} scratch stores during the search (equal counts; a store is ` +
        `not cheaper than an L1-hot load, and the swap needs fix-ups on top) -> zero-or-negative pool; ` +
        `whole row-loop control class already bounded ~16-48 ms sub-band by R12-C FZERO arithmetic`
    );
  }

  // FINTAIL: the final finiteness-check loop over solution (n values per
  // solve). Fusing it into back substitution or micro-restructuring it cannot
  // save more than the standalone cost of the loop itself.
  {
    const n = 60;
    const r = fixtureRng(0xf1a7);
    const solution = new Array<number>(n);
    for (let i = 0; i < n; i++) solution[i] = r() * 8 - 4;
    let sink = 0;
    const reps = 200000;
    const t0 = performance.now();
    for (let rep = 0; rep < reps; rep++) {
      let ok = true;
      for (const value of solution) {
        if (!Number.isFinite(value)) { ok = false; break; }
      }
      sink += ok ? 1 : 0;
    }
    const nsPerSolve = ((performance.now() - t0) / reps) * 1e6;
    out(
      `probe FINTAIL: ${nsPerSolve.toFixed(1)} ns per 60-value finiteness sweep -> x8966 solves = ` +
        `${((nsPerSolve * 8966) / 1e6).toFixed(2)} ms/report ceiling (sink=${sink})`
    );
  }

  // REFZ: reference-level bootstrap draws. Each successful draw calls the
  // columnless APC (Map miss + division; the row scan is skipped) and pushes
  // a constant +0 into the draws map, for each of the 2 reference names.
  // Skipping all of it cannot save more than this standalone pool. The exact
  // production count comes from the counting run above (refDrawsMapOps).
  {
    const rows = perfRows();
    const baseRows: Row[] = rows.map((r) => ({
      scenarioId: r.scenarioId,
      modelVersion: r.modelVersion,
      projectId: r.projectId,
      y: r.y
    }));
    const design = buildDesign(baseRows);
    const refNames = design.referenceLevels.map((ref) => `${ref.factor}:${ref.name}`);
    let sink = 0;
    const reps = 3000;
    const t0 = performance.now();
    for (let rep = 0; rep < reps; rep++) {
      const draws = new Map<string, number[]>();
      for (let draw = 0; draw < 195; draw++) {
        for (const name of refNames) {
          // verbatim columnless-APC body
          let sum = 0;
          const columnIdx = design.columnIndex.get(name);
          if (columnIdx !== undefined && columnIdx !== 0) {
            sum += 1; // unreachable for reference names
          }
          const value = baseRows.length === 0 ? 0 : sum / baseRows.length;
          const list = draws.get(name);
          if (list === undefined) draws.set(name, [value]);
          else list.push(value);
        }
      }
      sink += draws.size;
    }
    const msPerReport = (performance.now() - t0) / reps;
    out(
      `probe REFZ: ${(msPerReport * 1e3).toFixed(1)} us/report for ${refNames.length} reference names x 195 draws ` +
        `(columnless APC + draws push; sink=${sink})`
    );
  }

  // TSORT: the whole sort/percentile tail pool timed standalone (59 names x
  // ~195 values), plus the -0 ordering bitwise hazard of the typed-array
  // cheap form: Array.prototype.sort with (a,b)=>a-b is stable and keeps
  // insertion order for the -0/+0 tie, while %TypedArray%.sort orders -0
  // before +0.
  {
    const r = fixtureRng(0x7501);
    const names = Array.from({ length: 59 }, (_, i) => `name-${i}`);
    const source = new Map<string, number[]>();
    for (const name of names) {
      const values: number[] = [];
      for (let i = 0; i < 195; i++) values.push(r() * 0.2 - 0.1);
      source.set(name, values);
    }
    let sink = 0;
    const reps = 300;
    const t0 = performance.now();
    for (let rep = 0; rep < reps; rep++) {
      for (const name of names) {
        const values = [...source.get(name)!];
        values.sort((a, b) => a - b);
        sink += percentile(values, 0.025) + percentile(values, 0.975);
      }
    }
    const msPerReport = (performance.now() - t0) / reps;
    out(`probe TSORT: ${msPerReport.toFixed(2)} ms/report for 59x195 copy+sort+2 percentiles (whole tail ceiling; sink=${sink.toFixed(3)})`);

    // -0 ordering hazard demo (41 values so index = 40*0.025 = 1 exactly).
    const hazard: number[] = [0, -0];
    for (let i = 0; i < 39; i++) hazard.push(i + 1);
    const stable = [...hazard].sort((a, b) => a - b);
    const typed = Array.from(new Float64Array(hazard).sort());
    const stableLcb = percentile(stable, 0.025);
    const typedLcb = percentile(typed, 0.025);
    check("tsort.hazard-diverges", !Object.is(stableLcb, typedLcb));
    out(
      `  -> -0 hazard: stable comparator sort keeps [+0, -0] -> lcb=${Object.is(stableLcb, -0) ? "-0" : String(stableLcb)}; ` +
        `Float64Array sort orders [-0, +0] -> lcb=${Object.is(typedLcb, -0) ? "-0" : String(typedLcb)} (Object.is differs)`
    );
  }

  // DEG: the base degenerate double-`every` scan (once per report). Fusing
  // the two passes into one cannot save more than this standalone pool; the
  // per-draw collapse-check side is S2-C-5 (excluded) and lives inside the
  // SAMP pool closed by R12-C.
  {
    const rows = perfRows();
    const baseRows: Row[] = rows.map((r) => ({
      scenarioId: r.scenarioId,
      modelVersion: r.modelVersion,
      projectId: r.projectId,
      y: r.y
    }));
    let sink = 0;
    const reps = 100000;
    const t0 = performance.now();
    for (let rep = 0; rep < reps; rep++) {
      const degenerate =
        baseRows.length === 0 ||
        baseRows.every((r) => r.y === 0) ||
        baseRows.every((r) => r.y === 1);
      sink += degenerate ? 1 : 0;
    }
    const usPerReport = ((performance.now() - t0) / reps) * 1e3;
    out(`probe DEG: ${usPerReport.toFixed(2)} us per double-every degenerate scan, once/report (sink=${sink})`);
  }
}
