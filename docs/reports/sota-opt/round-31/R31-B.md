function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };

 * anchors). Full grid + CLI live face + per-child cascade probes.

const anchoredCells: readonly [string, ModelRouterConfig, LearnedRoutingPolicy | undefined][] = [

  // W1: double-run byte identity at production scale (N=2000, M=10, learned).
  {
    const catalog10 = tenModelCatalog();
    const learned10 = tenModelLearned();
    const first = assignTasks({ catalog: catalog10, tasks: replayTasks, learned: learned10 });
