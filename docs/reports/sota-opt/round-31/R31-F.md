## 3. 唯一残格的裁决：边界故障通道输出处置面（不铸、不计时、按分解归属收口）

**格的定义**：切片边界抛出故障对象后，异常离开边界帧、控制权回到调用方——调用方对该故障对象的处置集从未是受控变量：PROPAGATE（无 catch 原样上抛）/ SWALLOW / MESSAGE（读 `.message`）/ STACK（首读 `.stack`——V8 惰性栈物化，一次性符号化）/ RETAIN（保持可达跨后续实验）/ WRAP-RETHROW（读 message 包装新错误重抛）。R17-F 定价的是**制造侧**（抛出成本 + 拒绝路径通道单价）；R30-F/R29-F 定价的是**成功通道**输出的读取集与保留（快照）；处置格是故障通道的调用方侧——三十轮夹具全部把它钉在「断言类即刻消费」形态，从未剂量。

**铸轴前置证明（R31-A 三步，失败于第 1 步）**：

1. **独立物理对象 + 生产流量：失败。** (a) 分解归属——处置臂逐一落入既有属主变量类：MESSAGE/STACK 是对边界制造对象的**字段载入** = R30-F 载入集变量类（换了通道不换机制）；RETAIN 是**可达性存活期** = R29-F 保留变量类；每故障通道单价（构造 + 栈捕获点）= R17-F；处置格唯一特有机制——`.stack` 首读的惰性栈物化——是**引擎面**（V8 lazy stack trace；干预形态 `Error.stackTraceLimit`/`prepareStackTrace` 全是平台面拒绝类 + 切片外行为变更）。(b) 载荷普查（W2 见证）——规范故障对象 own 槽**全字符串**（name + message；stack 为引擎管理惰性槽）：零账本/计划指称 ⇒ RETAIN 腿无重指称物（R29-F 成功通道的 16 MB 账本闭包在故障通道**无类似物**），MESSAGE/STACK 腿载入集 = 两个字符串槽 ⇒ 构造上 µs–ns 域。(c) 生产流量——热通道零流量（W1：干净锚点尺度流 P=2000/A=1000 生产交错 2001 次边界调用**零故障**；热消费者无 catch，PROPAGATE 坐标由 W4 直接见证：中流重复指派故障穿出 `evaluateProposalShadow` 类与生产消息原样不变）；冷通道唯一 catch 位点每 eval 至多一次（once-per-run 拒绝类，派单明文 reject）。

const P = 2000;
const A = 1000;

function makePackedPopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

const basePopulation = makePackedPopulation(P);
const baseHashes = basePopulation.slice(0, A);

function makePlan(): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: "exp_r31fbase",
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r31fbase"),
    candidateId: createCandidateId(() => "r31fcand"),
    population: [...basePopulation],
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments: A, maxWallClockMs: 1e12 },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

function cleanStream(): ExperimentOutcome[] {
  return baseHashes.map((h) => ({
    episodeHash: h,
    utility: 0.5,
    costUsd: 0.01,
    guardrailBreached: false,
  }));
}

const sharedStream = cleanStream();

function ledgerDigest(state: ShadowState): string {
  return stableStringify({
    assignments: state.assignments,
    outcomes: state.outcomes,
    guardrailBreaches: state.guardrailBreaches,
    halted: state.halted,
    haltReason: state.haltReason,
  });
}

function assertBudgetExhausted(name: string, state: ShadowState): void {
  if (!state.halted || state.haltReason !== "budget-exhausted") {
    throw new Error(`${name}: expected budget-exhausted final state`);
  }
  if (state.assignments.length !== A || state.outcomes.length !== A) {
    throw new Error(`${name}: unexpected ledger sizes`);
  }
}

{

  let opCalls = 0;
  let state = runner.start(0);
  opCalls += 1;

    for (const outcome of sharedStream) {

  );
