### A.3 门定价探针（/tmp/r31i-bench/doorprobe.mjs，败者脚本仅入附录）

```javascript
// R31-I door-pricing probe (loser probe — appendix only, never committed).
// Prices the ONLY two doors found by the thirty-first enumeration audit that
// carry any production flow at all, to make their REJECT/decompose verdicts
// quantitative rather than merely structural:
//
//   Door A — outbound OS child-process boundary: the slice's complete OS
//   spawn surface is exactly two spawnSync("git", ...) sites in
//   src/cli/commits.ts (L79 rev-parse guard, L99 commit apply), reachable
//   only via `pi-sparkle commits apply` (main.ts:1441 dynamic import). This
//   probe prices (1) the exact L79 call form, (2) the bare OS fork/exec
//   floor (/bin/true), and (3) the exact L99 commit call form, so the
//   external-binary share (total - floor) and the slice-side share
//   (argument assembly, measured in-process) can be split.
//
//   Door B — collation site: the slice's complete locale-sensitive surface
//   is exactly one localeCompare in src/cli/doctor-overlay.ts L108
//   (agent-drift overlay summary sort), reachable only via
//   `pi-sparkle doctor`. This probe prices the exact sort against a
//   codepoint comparator at realistic (k=12) and stress (k=100) entry
//   counts.
//
// Nothing is dosed; both doors are priced observationally.
// usage: <node-binary> doorprobe.mjs
import { spawnSync } from "node:child_process";
import { mkdirSync, rmSync } from "node:fs";
import { join } from "node:path";
import { performance } from "node:perf_hooks";

const ROOT = "/tmp/r31i-bench";
const REPO = join(ROOT, "gitrepo");

function q(sorted, p) { const idx = (sorted.length - 1) * p, lo = Math.floor(idx), hi = Math.ceil(idx); return sorted[lo] + (sorted[hi] - sorted[lo]) * (idx - lo); }
function stats(times) { const t = [...times].sort((a, b) => a - b); return `median=${q(t, 0.5).toFixed(2)}ms min=${t[0].toFixed(2)} p90=${q(t, 0.9).toFixed(2)}`; }

// --- Door A setup: scratch git repo ---
rmSync(REPO, { recursive: true, force: true });
mkdirSync(REPO, { recursive: true });
for (const args of [["init", "-q"], ["config", "user.email", "bench@example.invalid"], ["config", "user.name", "Bench"], ["commit", "--allow-empty", "-q", "-m", "seed"]]) {
  const r = spawnSync("git", args, { cwd: REPO, encoding: "utf8", windowsHide: true });
  if (r.status !== 0) { console.error(`setup failed: git ${args.join(" ")}: ${r.stderr}`); process.exit(1); }
}

// A1: exact commits.ts L79 call form (isGitWorkTree guard).
{
  const WARM = 2, N = 20, times = [];
  for (let i = 0; i < WARM + N; i++) {
    const t0 = performance.now();
    const r = spawnSync("git", ["rev-parse", "--is-inside-work-tree"], { cwd: REPO, encoding: "utf8", windowsHide: true });
    const dt = performance.now() - t0;
    if (r.status !== 0 || r.stdout.trim() !== "true") { console.error("A1 unexpected"); process.exit(1); }
    if (i >= WARM) times.push(dt);
  }
  console.log(`A1 git rev-parse (commits.ts L79 form): ${stats(times)}`);
}

// A2: bare OS fork/exec floor.
{
  const WARM = 2, N = 20, times = [];
  for (let i = 0; i < WARM + N; i++) {
    const t0 = performance.now();
    const r = spawnSync("/bin/true", [], { cwd: REPO, encoding: "utf8", windowsHide: true });
    const dt = performance.now() - t0;
    if (r.status !== 0) { console.error("A2 unexpected"); process.exit(1); }
    if (i >= WARM) times.push(dt);
  }
  console.log(`A2 /bin/true OS fork/exec floor: ${stats(times)}`);
}

// A3: exact commits.ts L99 call form (applyProposal commit; message on stdin
// via -F - exactly like gitCommitArgs + input option).
{
  const WARM = 2, N = 10, times = [];
  for (let i = 0; i < WARM + N; i++) {
    const t0 = performance.now();
    const r = spawnSync("git", ["commit", "--allow-empty", "-q", "-F", "-"], { cwd: REPO, encoding: "utf8", input: `probe: commit ${i}\n\nbody line\n`, windowsHide: true });
    const dt = performance.now() - t0;
    if (r.status !== 0) { console.error(`A3 unexpected: ${r.stderr}`); process.exit(1); }
    if (i >= WARM) times.push(dt);
  }
  console.log(`A3 git commit --allow-empty (commits.ts L99 form): ${stats(times)}`);
}

// A4: slice-side JS share — argument-array assembly + message template,
// shaped like gitCommitArgs + formatCommitMessage input construction.
{
  const proposal = { type: "feat", scope: "routing", subject: "adopt learned table", nodeId: "n7", evidenceIds: ["evd_1", "evd_2"] };
  const N = 1e6;
  let sink = 0;
  const t0 = performance.now();
  for (let i = 0; i < N; i++) {
    const msg = `${proposal.type}(${proposal.scope}): ${proposal.subject}\n\nnode: ${proposal.nodeId}\nevidence: ${proposal.evidenceIds.join(", ")}\n`;
    const args = ["commit", "--allow-empty", "-q", "-F", "-"];
    sink += msg.length + args.length;
  }
  const perCall = ((performance.now() - t0) / N) * 1e6;
  console.log(`A4 slice-side arg+message assembly: ${perCall.toFixed(1)} ns/call (sink=${sink})`);
}

// --- Door B: doctor-overlay L108 collation sort ---
function priceSort(k, label) {
  const names = [];
  for (let i = 0; i < k; i++) names.push(`agent-${String.fromCharCode(97 + (i % 26))}${Math.floor(i / 26)}`);
  const entries = names.map((n, i) => [n, i + 1]);
  const SORTS = 1e4;
  let t0 = performance.now();
  let sink = 0;
  for (let s = 0; s < SORTS; s++) {
    const parts = [...entries].sort(([a], [b]) => a.localeCompare(b));
    sink += parts[0][1];
  }
  const localeUs = ((performance.now() - t0) / SORTS) * 1e3;
  t0 = performance.now();
  for (let s = 0; s < SORTS; s++) {
    const parts = [...entries].sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0));
    sink += parts[0][1];
  }
  const cpUs = ((performance.now() - t0) / SORTS) * 1e3;
  console.log(`B ${label} k=${k}: localeCompare=${localeUs.toFixed(2)}us/sort codepoint=${cpUs.toFixed(2)}us/sort delta=${(localeUs - cpUs).toFixed(2)}us (sink=${sink})`);
}

priceSort(100, "stress");
```
