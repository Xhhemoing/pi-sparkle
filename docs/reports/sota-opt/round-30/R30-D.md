## 2. 新轴：批准档案约束列表几何面

### 2.1 定义与盲区

`ApprovalProfile`（approval-profile.ts L7-15）携带两个
`readonly ResourceKind[]` 约束列表：`autoPromoteClasses`（n）与
`neverAutoPromote`（m）。每次 `beginPromotion`（registry.ts
L608-613）执行 `validateApprovalProfile`（L50-94）+
`canAutoPromote`（L101-131）。validate 的成本结构恰四段列表循环：

| 循环 | 行号 | 成本 |
|---|---|---|
| L1 枚举查验 + 禁类拒绝 | L65-74 | O(n·(10+3)) |
| L2 枚举查验 | L75-79 | O(m·10) |
| L3 required 扫描 | L80-86 | O(3·m) |
| L4 交叉成员校验 | L87-93 | **O(n·m) 双线性** |

`canAutoPromote`：L109 O(m) miss 扫 + L118 O(n) 位置敏感扫。

盲区：R24-D 批准权限面剂量的是**模式旗与标量计数器**
（explicitApproval × budget × autoPromotionsUsed），其划界原文
明载「profile 列表长度钉在内建形态 0/1/3 恒定……非列表维」；
R22-D 导出流普查给的 `canAutoPromote` per-op 价在两个内建
**恒定几何**上；R24-D S1a 的 validate 价是 n∈{0,1}/m=3 两个
**定点**。列表维（n / m 受控、重复重数方向、L4 双线性项）
三十遍从未定价。剂量合法性：`ResourceKind` 是封闭 10 值枚举
⇒ 互异值封顶（n≤7 且与 never 不交、m≤10 且含三禁类；合法互异
双线性格 n·m ≤ n·(10−n) ≤ **25 格**）；校验**无唯一性规则**
（L50-94 首手逐段核对）⇒ 重复重数是唯一无界剂量方向且合法通过。

function bench(fn: () => void, minMs = 60, maxReps = 50000): number {
  fn();
  fn();
  let reps = 0;

  let elapsed = 0;
  do {

    elapsed = performance.now() - t0;
  } while (elapsed < minMs && reps < maxReps);
  return elapsed / reps;
}

function configuredPolicy(variantTag: string, k = 1): string {
  const avoid = Array.from({ length: 40 * k }, (_, i) => ({
    modelId: i % 8 === 0 ? "cheap" : `m${i % 7}`,
    family: i % 3 === 0 ? pick(() => (i % 10) / 10, ["edit", "test", "review"]) : `fam${i % 12}`,
    reason: `deterministic FAIL pattern ${i} (${variantTag}): attributable replay regression observed across paired episodes with stable family-level attribution`
  }));
  const prefer = Array.from({ length: 40 * k }, (_, i) => ({
    family: i % 4 === 0 ? "test" : `fam${i % 12}`,
    modelId: i % 4 === 0 ? "cheap" : `m${(i + 1) % 7}`
  }));
  const assignments = Array.from({ length: 7 }, (_, i) => ({
    role: ["planner", "executor", "tester", "reviewer", "scout", "debugger", "worker"][i] as string,
    model: i % 2 === 0 ? "premium" : "cheap",
    family: ["edit", "test", "review"][i % 3] as string
  }));
  if (variantTag === "candidate") {
    avoid.push({ modelId: "cheap", family: "refactor", reason: "candidate-only: deterministic FAIL on refactor replay (new attribution)" });
  }
  return JSON.stringify({ primaryModelId: "premium", avoid, prefer, assignments });
}

function buildConfiguredRegistry(

  const built = buildRoutingRegistry(tag, CONFIGURED_BASELINE_POLICY, CONFIGURED_CANDIDATE_POLICY);

    };

    const v1 = registry.registerBaseline({ identity, content: body, author: HUMAN });

      identity,
      content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    const entry: (typeof versions)[number] = { identity, v1, candidate };
    if (i < 5) {
      const result = promoteWithRegistry(registry, {
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
      entry.v2 = result.newVersion as ResourceVersion;
    } else if (i === 5) {
      registry.beginPromotion({
        candidateId: candidate.candidateId,
        expectedCurrentVersionId: v1.versionId,
        content: `${body}\n## amendment ${i}\nAdd an explicit check for the attributable failure.\n`,
        approvedBy: HUMAN,
        review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}p${i}`),
        changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId),
        explicitApproval: true
      });
    }
    versions.push(entry);
  }
  const first = versions[0] as (typeof versions)[number];
  registry.rollback({
    identity: first.identity,
    expectedCurrentVersionId: (first.v2 as ResourceVersion).versionId,
    targetVersionId: first.v1.versionId,
    reason: "degradation",
    evidence: ["ev:configured-degradation"],

  });

    identity: second.identity,
    expectedCurrentVersionId: (second.v2 as ResourceVersion).versionId,
    targetVersionId: second.v1.versionId,
    reason: "user",
    evidence: ["ev:configured-user"],
    automatic: false
  });
  registry.retire((versions[2] as { v1: ResourceVersion }).v1.versionId);
  registry.retire((versions[3] as { v1: ResourceVersion }).v1.versionId);
  return built;
}

/* ===== geometry-dose profile builders (duplicate enum tokens; all validate-legal) ===== */
/** autoPromoteClasses: n duplicates of one auto-promotable kind; neverAutoPromote:
 * (m-2) duplicates of "permission" then one "security", one "credential" (all
 * three required kinds present; loop3 includes() hits at depth O(m)).
 * Cross-check loop4: every autoPromoteClasses entry misses -> full n*m scan. */
function dosedProfile(n: number, m: number, autoKind: ResourceKind = "prompt", budget = 0): ApprovalProfile {
  if (m < 3) throw new Error("m >= 3 required");
  const never: ResourceKind[] = Array.from({ length: m - 2 }, () => "permission" as ResourceKind);
  never.push("security", "credential");
  return {
    profileId: `apr_geom-n${n}-m${m}`,
    profileVersion: 1,
    defaultMode: "proposal-first",
    autoPromoteClasses: Array.from({ length: n }, () => autoKind),
    neverAutoPromote: never,
    budget: { maxAutoPromotions: budget }
  };
}
function dedup(profile: ApprovalProfile): ApprovalProfile {
  return {
    ...profile,
    autoPromoteClasses: [...new Set(profile.autoPromoteClasses)],
    neverAutoPromote: [...new Set(profile.neverAutoPromote)]
  };
}
function throwsValidation(fn: () => void): boolean {
  try {
    fn();
    return false;
  } catch (error) {
    return error instanceof DomainValidationError;
  }
