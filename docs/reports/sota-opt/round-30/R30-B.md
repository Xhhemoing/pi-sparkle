const MISS_TOKENS = ["zorv", "quel", "brint", "moxa", "felp", "durn", "sakle", "wemb"] as const;
function missTextOfLength(rng: () => number, targetLen: number): string {
  let out = "";
  while (out.length < targetLen) out += `${pick(rng, MISS_TOKENS)} `;
  let text = out.slice(0, targetLen);
  const lastSpace = text.lastIndexOf(" ");
  if (lastSpace < 0) {
    text = "x".repeat(text.length);
  } else if (lastSpace < text.length - 1) {
    text = text.slice(0, lastSpace + 1) + "x".repeat(text.length - lastSpace - 1);
  }
  return text;
}
