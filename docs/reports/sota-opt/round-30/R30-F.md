/** Historical fixture shape: `ep_` + zero-padded base36 index, L=11. */
function makePackedPopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

function cleanStream(): ExperimentOutcome[] {
  return baseHashes.map((h) => ({
    episodeHash: h,

const sharedStream = cleanStream();

const ARMS = ["NONE", "HALT", "SCALARS", "LENGTHS", "LAST", "FULL"] as const;

/* ------------------------- per-arm readback folds ------------------------- */
/* Shared verbatim between the e2e drivers and the unit-price probes so the
 * cell-E' sweep is the exact caller-side load multiset of one experiment.
 * Every fold lands in a primitive number: nothing retains a snapshot. */

function readScalars(state: ShadowState): number {
  return (
    (state.halted ? 1 : 0) +
    (state.haltReason === undefined ? 0 : 1) +
    state.guardrailBreaches +
    state.startedAtMs +
    state.elapsedMs
  );
}

function readLengths(state: ShadowState): number {
  return readScalars(state) + state.assignments.length + state.outcomes.length;
}

function readLast(state: ShadowState): number {
  let sink = readLengths(state);
  const assignments = state.assignments;
  const aLen = assignments.length;
  if (aLen > 0) {
    const last = assignments[aLen - 1] as ShadowState["assignments"][number];
    sink +=
      last.episodeHash.length +
      (last.liveAction === "baseline" ? 1 : 0) +
      (last.shadowDecision === "candidate" ? 1 : 0) +
      (last.changedLiveAction ? 1 : 0);
  }
  const outcomes = state.outcomes;
  const oLen = outcomes.length;
  if (oLen > 0) {
    const last = outcomes[oLen - 1] as ExperimentOutcome;
    sink +=
      last.episodeHash.length + last.utility + last.costUsd + (last.guardrailBreached ? 1 : 0);
  }
  return sink;
}

function readFull(state: ShadowState): number {
  let sink = readLengths(state);
  const assignments = state.assignments;
  for (let i = 0; i < assignments.length; i++) {
    const record = assignments[i] as ShadowState["assignments"][number];
    sink +=
      record.episodeHash.length +
      (record.liveAction === "baseline" ? 1 : 0) +
      (record.shadowDecision === "candidate" ? 1 : 0) +
      (record.changedLiveAction ? 1 : 0);
  }
  const outcomes = state.outcomes;
  for (let i = 0; i < outcomes.length; i++) {
    const record = outcomes[i] as ExperimentOutcome;
    sink +=
      record.episodeHash.length +
      record.utility +
      record.costUsd +
      (record.guardrailBreached ? 1 : 0);
  }
  return sink;
}

const READBACK: Record<Exclude<ArmName, "NONE" | "HALT">, (state: ShadowState) => number> = {
  SCALARS: readScalars,
  LENGTHS: readLengths,
  LAST: readLast,
  FULL: readFull,
};

interface RunResult {
  readonly finalState: ShadowState;
  readonly sink: number;
}

/**
 * Production interleave (the evaluateProposalShadow shape), clock pinned to
 * 0, DIRECT dispatch on one runner, latest-only threading in every arm. The
 * caller's LOAD SET over each returned snapshot is the arm; the operand fed
 * to every call is the freshest snapshot in every arm. The halted branch
 * never fires mid-stream (budget == stream length), so the boundary call
 * multiset and order are identical in all six arms — witnessed by cross-arm
 * ledger digest identity.
 */
function runExperiment(runner: ShadowRunner, arm: ArmName): RunResult {
  if (arm === "NONE") {
    let state = runner.start(0);
    for (const outcome of sharedStream) {
      state = runner.assign(state, outcome.episodeHash, 0);
      state = runner.recordOutcome(state, outcome, 0);
    }
    return { finalState: state, sink: 0 };
  }
  if (arm === "HALT") {
    // The production coordinate, verbatim shape (reflection.ts L216-223).
    let state = runner.start(0);
    for (const outcome of sharedStream) {
      if (state.halted) {
        break;
      }
      state = runner.assign(state, outcome.episodeHash, 0);
      state = runner.recordOutcome(state, outcome, 0);
    }
    return { finalState: state, sink: 0 };
  }
  const readback = READBACK[arm];
  let sink = 0;
  let state = runner.start(0);
  sink += readback(state);
  for (const outcome of sharedStream) {
    if (state.halted) {
      break;
    }
    state = runner.assign(state, outcome.episodeHash, 0);
    sink += readback(state);
    state = runner.recordOutcome(state, outcome, 0);
    sink += readback(state);
  }
  return { finalState: state, sink };
}

function ledgerDigest(state: ShadowState): string {
  return stableStringify({
    assignments: state.assignments,
    outcomes: state.outcomes,
    guardrailBreaches: state.guardrailBreaches,
    halted: state.halted,
    haltReason: state.haltReason,
  });
}

function verifyFinalState(name: string, state: ShadowState): void {
  if (!state.halted || state.haltReason !== "budget-exhausted") {
    throw new Error(`${name}: expected budget-exhausted final state`);
  }
  if (state.assignments.length !== A || state.outcomes.length !== A) {
    throw new Error(`${name}: unexpected ledger sizes`);
  }
  for (let i = 0; i < A; i++) {
    const assignment = state.assignments[i] as { episodeHash: string };
    if (assignment.episodeHash !== (basePopulation[i] as string)) {
      throw new Error(`${name}: assignment ${i} not the head-aligned prefix`);
    }
  }
  if (state.guardrailBreaches !== 0 || state.startedAtMs !== 0 || state.elapsedMs !== 0) {
    throw new Error(`${name}: unexpected counters/clock fields`);
  }
  if (Object.isFrozen(state.plan.population) || Object.isSealed(state.plan.population)) {
    throw new Error(`${name}: storage-state face moved (must stay extensible)`);
  }
}

function verifyArm(name: ArmName, result: RunResult): void {
  verifyFinalState(name, result.finalState);
  if (typeof result.sink !== "number" || !Number.isFinite(result.sink)) {
    throw new Error(`${name}: readback sink must be a finite primitive number`);
  }
  if ((name === "NONE" || name === "HALT") && result.sink !== 0) {
    throw new Error(`${name}: no-readback arm must have a zero sink`);
  }
  if (name !== "NONE" && name !== "HALT" && result.sink <= 0) {
    throw new Error(`${name}: readback arm produced an empty sink (loads optimized away?)`);
  }
}

/** Untimed instrument: all 2001 op-return snapshots of one experiment. */
function buildSnapshotSequence(runner: ShadowRunner): ShadowState[] {
  const sequence: ShadowState[] = [];
  let state = runner.start(0);
  sequence.push(state);
  for (const outcome of sharedStream) {
    state = runner.assign(state, outcome.episodeHash, 0);
    sequence.push(state);
    state = runner.recordOutcome(state, outcome, 0);
    sequence.push(state);
  }
  if (sequence.length !== OP_RETURNS) {
    throw new Error(`snapshot sequence holds ${sequence.length}, expected ${OP_RETURNS}`);
  }
  verifyFinalState("sequence-final", sequence[sequence.length - 1] as ShadowState);
  return sequence;
}

/* ----------------------- isolated child-cell driver ------------------------ */
function runChild(cell: string, arm: string): string[] {
  const child = spawnSync(
    process.execPath,
    ["--import", "tsx", process.argv[1] as string, `--cell=${cell}`, `--arm=${arm}`],
    { encoding: "utf8", timeout: 300_000 }
  );
  if (child.status !== 0) {
    return [`CHILD ${cell} ${arm} FAILED status=${String(child.status)} ${child.stderr.slice(0, 300)}`];
  }
  return child.stdout.trim().split("\n");
}

function runIsolatedCell(title: string, cell: string, arms: readonly string[]): Map<string, string> {
  process.stdout.write(`${title}\n`);
  const lines = new Map<string, string>();
  for (const arm of arms) {
    for (const line of runChild(cell, arm)) {
      process.stdout.write(`  ${line}\n`);
      if (line.startsWith("CHILD")) {
        lines.set(arm, line);
      }
    }
  }
  return lines;
}

  const match = line?.match(/ ([0-9.]+) ms gc=([0-9.]+)ms/);
