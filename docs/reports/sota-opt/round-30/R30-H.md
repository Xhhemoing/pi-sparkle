function mkContract(spec: FleetSpec): RequirementContract {
  const pathway = spec.pathway;
  const assumeIds = spec.assumeIds ?? ["a-defaults"];
  const assumptionIds_ = spec.assumptions ?? ["a-defaults", "a-padded0"];
  const prov = (): {
    sourceRefs?: { kind: "message"; ref: string }[];
    assumptionIds?: string[];
  } => {
    if (pathway === "refs") return { sourceRefs: [freshRef()] };
    if (pathway === "both") return { sourceRefs: [freshRef()], assumptionIds: [...assumeIds] };
    if (pathway === "assume") return { assumptionIds: [...assumeIds] };
    return {};
  };
  const constraints: unknown[] = [
    { id: "c-item-0001", description: "item-desc", enforceable: false, ...prov() },
    { id: "c-item-0002", description: "item-desc", enforceable: true, ...prov() }
  ];
  const extra = spec.extraUnsourcedConstraints ?? 0;
  for (let i = 0; i < extra; i += 1) {
    constraints.push({
      id: `c-uns-${String(i).padStart(4, "0")}`,
      description: "item-desc",
      enforceable: false
    });
  }
  return validateRequirementContract({
    schemaVersion: 1,
    objective: "provenance axis pinned objective",
    deliverables: [
      { id: "d-item-0001", description: "item-desc", artifactKind: "diff", ...prov() },
      { id: "d-item-0002", description: "item-desc", artifactKind: "file", ...prov() }
    ],
    constraints,
    nonGoals: [],
    acceptanceCriteria: [
      { id: "ac-item-0001", description: "item-desc", observableCheck: "run.status is COMPLETED", ...prov() },
      { id: "ac-item-0002", description: "item-desc", observableCheck: "verification is PASSED", ...prov() }
    ],
    assumptions: assumptionIds_.map((id) => ({ id, statement: "s", source: "heuristic-default" })),
    questions: [],
    authority: [],
    sourceRefs: [freshRef()]
  });
}

interface SlopePoint {
  x: number;
  ns: number;
}
function slopeOf(points: SlopePoint[]): number {
  const n = points.length;
  const sx = points.reduce((s, p) => s + p.x, 0);
  const sy = points.reduce((s, p) => s + p.ns, 0);
  const sxx = points.reduce((s, p) => s + p.x * p.x, 0);
  const sxy = points.reduce((s, p) => s + p.x * p.ns, 0);
  return (n * sxy - sx * sy) / (n * sxx - sx * sx);
}
