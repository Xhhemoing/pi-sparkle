/* member-dup leg: per-task list = k member edges (duplicates keep C=2,
 * T=5 pinned); foreign-pad leg: per-task list = 1 member + (k-1) DISTINCT
 * width-12 foreign ids. Slope = ns per declared edge of that class. */
function memberFleet(k: number): CoverageTaskRef[] {
  const lists: string[][] = [];
  for (let t = 0; t < T; t += 1) {
    const l: string[] = [];
    for (let j = 0; j < k; j += 1) l.push(j % 2 === 0 ? C1 : C2);
    lists.push(l);
  }
  return fleet(lists);
}
function foreignFleet(k: number): CoverageTaskRef[] {
  const lists: string[][] = [];

    const l: string[] = [t % 2 === 0 ? C1 : C2];
    for (let j = 1; j < k; j += 1) l.push(`zz-p${String(t)}-${String(j).padStart(6, "0")}`); /* width 12 */
    lists.push(l);
  }
  /* guarantee both columns covered */
  (lists[0] as string[])[0] = C1;
  (lists[1] as string[])[0] = C2;
  return fleet(lists);
}
interface SlopePoint {
  e: number;
  ns: number;
}
function slopeOf(points: SlopePoint[]): number {
  const n = points.length;
  const sx = points.reduce((s, p) => s + p.e, 0);
  const sy = points.reduce((s, p) => s + p.ns, 0);
  const sxx = points.reduce((s, p) => s + p.e * p.e, 0);
  const sxy = points.reduce((s, p) => s + p.e * p.ns, 0);
  return (n * sxy - sx * sy) / (n * sxx - sx * sx);
}
