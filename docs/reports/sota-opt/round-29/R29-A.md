function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

}

  const dag = decF.assessment;
  const twin = unsharedTwin(dag);

  // twin sanity: identical values, fully unshared object graph
  check("F9c twin JSON-equal to production DAG assessment", JSON.stringify(twin) === JSON.stringify(dag));
  check(
    "F9c twin is fully unshared (top arrays, gate, human, dimensions, per-dim refs all fresh)",
    twin.evidenceRefs !== dag.evidenceRefs &&
      twin.gate !== dag.gate &&
      twin.human !== dag.human &&
      twin.dimensions !== dag.dimensions &&
      twin.dimensions.every((d, i) => d !== dag.dimensions[i]) &&
      twin.dimensions.every(
        (d) => d.evidenceRefs === undefined || d.evidenceRefs !== dag.evidenceRefs
      )
  );
  const failDimCount = dag.dimensions.filter((d) => d.verdict === "FAIL").length;
  check(
    "F9c production DAG: failRefs shared at (1 + failDims) positions; twin: all distinct",
    failDimCount > 0 &&
      dag.dimensions.filter((d) => d.evidenceRefs === dag.evidenceRefs).length === failDimCount &&
      new Set(
        twin.dimensions.filter((d) => d.evidenceRefs !== undefined).map((d) => d.evidenceRefs)
      ).size === failDimCount
  );

  // (i) hash face identity-invisible
  const hashDag = hashAssessment(dag);
  check("F9c(i) DAG and unshared twin hash identically", hashAssessment(twin) === hashDag);

  // (ii) parse alias-transmission exactly on string arrays
  const parsed = parseTrackingAssessment(dag);
  check(
    "F9c(ii) parse output top-level evidenceRefs IS the caller array (asStringArray alias)",
    parsed.evidenceRefs === dag.evidenceRefs
  );
  check(
    "F9c(ii) parse output per-FAIL-dimension evidenceRefs IS the caller array (DAG transmitted)",
    parsed.dimensions
      .filter((d) => d.verdict === "FAIL")
      .every((d) => d.evidenceRefs === dag.evidenceRefs)
  );
  check(
    "F9c(ii) every other parse leg is alias-breaking (fresh dimensions/gate/human objects)",
    parsed.dimensions.every((d, i) => d !== dag.dimensions[i]) &&
      parsed.gate !== dag.gate &&
      parsed.human !== dag.human &&
      parsed.gate.codes !== dag.gate.codes
  );

  // (iii) whole-gate semantic identity-invariance (fresh apply + replay classes)
  {
    const seq = nextTrackingSeq(events41);
    let c1 = 300;
    let c2 = 300;
    const outDag = applyTrackingGate({
      events: events41,
      assessment: dag,
      assessmentHash: hashDag,
      expectedSeq: seq,
      policyVersion: "track-v1",
      nowIso: NOW_ISO,
      generateEventId: () => `evt_${String(c1++).padStart(8, "0")}` as EventId
    });
    const outTwin = applyTrackingGate({
      events: events41,
      assessment: twin,
      assessmentHash: hashDag,
      expectedSeq: seq,
      policyVersion: "track-v1",
      nowIso: NOW_ISO,
      generateEventId: () => `evt_${String(c2++).padStart(8, "0")}` as EventId
    });

      "F9c(iii) gate result + appended bytes bitwise identical for DAG vs unshared twin",
      JSON.stringify(outDag.result) === JSON.stringify(outTwin.result) &&
        JSON.stringify(outDag.events.slice(events41.length)) ===
          JSON.stringify(outTwin.events.slice(events41.length))

    check("F9c(iii) recomputed twin hash is value-equal", twinHash === hashDag);
    const replays = [
      { label: "aliased original", a: dag, h: hashDag },
      { label: "unshared object twin", a: twin, h: hashDag },
      { label: "fresh-string twin", a: twin, h: twinHash }

        events: outDag.events,
        assessment: r.a,
        assessmentHash: r.h,
        expectedSeq: seq,
        policyVersion: "track-v1",

          throw new Error("replay must not generate ids");

    let c4 = 300;

      events: dupTable,
      assessment: dag,
      assessmentHash: hashDag,
      expectedSeq: nextTrackingSeq(dupTable),
      policyVersion: "track-v1",
      nowIso: NOW_ISO,
      generateEventId: () => `evt_${String(c3++).padStart(8, "0")}` as EventId
    });
    const outClone = applyTrackingGate({
      events: cloneTable,
      assessment: dag,
      assessmentHash: hashDag,
      expectedSeq: nextTrackingSeq(cloneTable),
      policyVersion: "track-v1",
      nowIso: NOW_ISO,

      JSON.stringify(outDup.result) === JSON.stringify(outClone.result) &&
        JSON.stringify(outDup.events.slice(41)) === JSON.stringify(outClone.events.slice(41))

  }

  {

      valueCopy !== (summary as unknown) && executionMayNotReadSummary(valueCopy, summary) === false
    );
    check(
      "F9c(iv) clean packet passes",
      executionMayNotReadSummary({ facts: ["a"] }, summary) === true
    );
  }
  console.log(
    `F9c: contract corners -- (i) the hash face is identity-invisible by construction (fresh ` +
      `projection every call, no cache): production DAG and fully-unshared twin hash identically; ` +
      `(ii) parseTrackingAssessment is alias-TRANSMITTING exactly on string arrays (output evidenceRefs ` +
      `=== caller arrays, top-level and per FAIL dimension -- the validated output keeps live caller ` +
      `handles; R28-A alias-reachable-ledger family, parent decision face) while every other leg ` +
      `rebuilds fresh; (iii) the whole gate is semantically identity-INVARIANT: DAG vs unshared twins ` +
      `and dup-row vs clone-row tables give bitwise identical results, the replay equivalence class is ` +
      `value-keyed (aliased original / object twin / fresh-string twin all no-op), and the [...events] ` +
      `copy is shallow (row aliasing flows through); (iv) the isolation identity leg is fail-closed AND ` +
      `redundant on schema-complete summaries (name probes already reject the alias and every value ` +
      `copy) -- deleting it is defense-in-depth-deletion, rejected by rename check at sub-ns.`
  );
}
