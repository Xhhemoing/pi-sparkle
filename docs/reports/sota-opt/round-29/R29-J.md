### 2.4 裁决

六腿全部「无候选」，三重收口：

1. **生产已坐在最便宜拼写上，且外来拼写不是自由变量**：两个同号税
   （K 读税、pretty 水合税）都挂在**外来**拼写臂上，规范臂即最快臂
   ——`writeFeedbackRecords` / `appendJsonlLine` / `saveToDisk` /
   级联侧车写手铸的全是规范拼写；空白类外来拼写在首个重写动词处
   自动湮灭（W-2/W-4），键序拼写虽粘滞但仓内**无铸造动词**（只能
   外部编辑铸入）。「让读缝更快」没有生产侧剩余动作。
2. **全部代码侧改形撞已闭族**：读时规范化（发现外来拼写即重写）=
   读路径改写持久态（分析不改在途态纪律 + 数据面契约）且写放大
   负优化；侧车改铸 compact（唯一「规范拼写非最小拼写」的工件，
   363 vs 301 字节）= 磁盘数据面字节发散（S4-G-6 / S6-E-4 已闭族）
   且 Leg S 实测平坦（±5 µs 内）；BOM/畸形容忍读 = 接受包络放宽 =
   fail-open 族（W-7 三缝现皆 fail-closed 或既定吞取）；跨命令按
   mtime/拼写缓存解析产物 = X1-1（多进程外部改写）；split/扫描
   读器改形 = S3-J-4 已裁；水合再投影单态化（load 时重建规范字面量
   以消 K 粘滞与 PIC 税）= 对外部铸造文件的下次落盘字节发散 +
   收益上界即 K−C 水合差（±1 µs 平坦）。
3. **量级**：K 读税 +8–28 µs × 零调用方 API / pretty 水合税
   +5.7–7.1 µs × 每 pref 命令一次——once-per-command µs 拒收类，
   距落地线 3 个数量级；本面无自有规模坐标（钉死逻辑内容下字节
   膨胀被小常数因子界住：W +9.4% / I8 +33% / pretty +39%；把膨胀
   推到病理量级需外部对抗性铸造且量级坐标属 R14-J / SIDEC /
   DELINV 已闭域）。

四条语义普查事实随轴入档（proposal-first，仅观察不改产码；见结论
(a)–(d)）：键序拼写穿透两个生产重写动词永久粘滞（无归一动词）；
损坏 records 读缝跨缝不对称（readFeedback 抛 / 级联吞——吞取端点
本体 R18-J 已定价）；corrupt 快照 configure 半态武装下一写动词
静默替换；侧车畸形 JSON 逃出为裸 SyntaxError（卫只覆盖良构错形）。
按纪律不铸任何 S29-J-\* ID；CSPELL 只入本报告、不写 EXCLUSIONS 行

const N = 200;
const HITS_EVERY = 10;

const stamp = (i: number): IsoTimestamp =>
  `2026-08-26T00:${pad(Math.floor(i / 6000), 2)}:${pad(Math.floor(i / 100) % 60, 2)}.${pad(i % 100, 3)}Z` as IsoTimestamp;

function makeRecord(i: number): FeedbackRecord {
  return {
    id: `fbk_${pad(i, 8)}`,
    episodeId: i % HITS_EVERY === 0 ? EPS_HIT : EPS_OTHER,
    kind: "human",
    rubricVersion: "r1",
    score: 50,
    evidenceRefs: [] as readonly EvidenceId[],
    redacted: true,
    createdAt: stamp(i),
    body: "uniform-body-24-chars-xx"
  };
}

const RECORDS: readonly FeedbackRecord[] = Array.from({ length: N }, (_, i) => makeRecord(i));
const HIT_IDS_SORTED = RECORDS.filter((r) => r.episodeId === EPS_HIT)
  .map((r) => r.id)
  .sort();
/** Constant foreign tombstone stock (T=20 pinned; ids match no record). */
const FOREIGN_IDS: readonly string[] = Array.from({ length: 20 }, (_, i) => `fbk_zz${pad(i, 6)}`);

/* --------------------------- spelling writers ----------------------------- */

const reverseKeys = (value: Record<string, unknown>): Record<string, unknown> =>
  Object.fromEntries(Object.entries(value).reverse());

/** Single-line whitespace-padded spelling of one JSON value. */
const padSpelling = (value: unknown): string => JSON.stringify(value, null, 1).replace(/\n\s*/g, " ");

type RecordsArm = "C" | "W" | "K" | "R" | "B" | "T";
const RECORDS_ARMS: readonly RecordsArm[] = ["C", "W", "K", "R", "B", "T"];

function recordsSpelling(arm: RecordsArm, records: readonly FeedbackRecord[]): string {
  const canonical = records.map((r) => JSON.stringify(r));
  switch (arm) {
    case "C":
      return `${canonical.join("\n")}\n`;
    case "W":
      return `${records.map((r) => padSpelling(r)).join("\n")}\n`;
    case "K":
      return `${records.map((r) => JSON.stringify(reverseKeys({ ...r }))).join("\n")}\n`;
    case "R":
      return `${canonical.join("\r\n")}\r\n`;
    case "B":
      return `${canonical.join("\n\n")}\n`;
    case "T":
      return canonical.join("\n");
  }
}

type SidecarArm = "P2" | "M" | "I8";
const SIDECAR_ARMS: readonly SidecarArm[] = ["P2", "M", "I8"];

function sidecarSpelling(arm: SidecarArm, ids: readonly string[]): string {
  switch (arm) {
    case "P2":
      return `${JSON.stringify(ids, null, 2)}\n`;
    case "M":
      return JSON.stringify(ids);
    case "I8":
      return `${JSON.stringify(ids, null, 8)}\n`;
  }
}

interface Root {
  readonly dir: string;
  readonly records: string;
  readonly sidecar: string;
}

function makeRoot(): Root {
  const dir = mkdtempSync(join(tmpdir(), "r29j-cspell-"));
  const records = feedbackLogPath(dir);
  const sidecar = feedbackTombstonesPath(dir);
  mkdirSync(dirname(records), { recursive: true });
  return { dir, records, sidecar };
}

function seed(root: Root, recordsArm: RecordsArm, sidecarArm: SidecarArm): void {
  writeFileSync(root.records, recordsSpelling(recordsArm, RECORDS), "utf8");
  writeFileSync(root.sidecar, sidecarSpelling(sidecarArm, FOREIGN_IDS), "utf8");
}
