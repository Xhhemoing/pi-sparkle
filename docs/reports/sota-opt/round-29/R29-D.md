blob**（单串），六个列表槽在其全部夹具中恒为单例；(b) R21-D 状态

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);

/** Provenance/plan list item of EXACTLY w bytes; deterministic per (slot, j).
 * Falls back to a compact head for tiny widths (partition narrow arms). */
function listItem(slot: string, j: number, w: number): string {
  const long = `ev ${slot} ${String(j).padStart(6, "0")} `;
  const head = long.length <= w ? long : `e${j.toString(36)}`;
  if (head.length > w) throw new Error(`w too small for ${slot}`);
  let s = head;
  while (s.length < w) s += FILLER;
  const out = s.slice(0, w);
  if (out.length !== w) throw new Error("pad mismatch");
  return out;
}

interface FleetRow {

/** K prompt identities x (baseline + candidate + promote) with dosed
 * provenance lists. Contents plane pinned: 2K blobs x 600 B in every arm. */
function buildPromoteFleet(tag: string, K: number, n: number, w: number): { registry: ResourceRegistry; provBytes: number } {
  const registry = new ResourceRegistry(armA(tag));
  let provBytes = 0;
  for (let i = 0; i < K; i += 1) {
    const identity: ResourceIdentity = {
      kind: "prompt",
      name: `agent-prompt-${String(i).padStart(4, "0")}`,
      scope: { kind: "project", projectId: createProjectId(() => "provproj") }
    };
    const c1 = fixedContent(tag, 2 * i);
    const c2 = fixedContent(tag, 2 * i + 1);
    const v1 = registry.registerBaseline({ identity, content: c1, author: HUMAN });
    const cand = registry.createCandidate({
      identity,
      content: c2,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    const refs = listOf(`refs-${i}`, n, w);
    const evidence = listOf(`evid-${i}`, n, w);
    const guardrails = listOf(`guard-${i}`, n, w);
    provBytes += 3 * n * w;
    promoteWithRegistry(registry, {
      candidateId: cand.candidateId,
      expectedCurrentVersionId: v1.versionId,
      content: c2,
      approvedBy: HUMAN,
      review: mkReview(cand.candidateId, cand.contentHash, `${tag}-${i}`, refs),
      changeNote: mkNote(`prompt:${tag}${i}`, v1.versionId, evidence, guardrails),
      explicitApproval: true
    });
  }
  return { registry, provBytes };
}

function maskedCanonical(snap: ResourceRegistrySnapshot, tag: string): string {
  const maskList = (): readonly string[] => ["MASK"];
  return JSON.stringify({
    versions: snap.versions.map((v) => ({ ...v, contentHash: "MASK" })),
    activeVersionIds: snap.activeVersionIds,
    candidates: snap.candidates.map((c) => ({
      ...c,
      contentHash: "MASK",
      evaluationPlan: { ...c.evaluationPlan, stages: maskList(), metrics: maskList() }
    })),
    ledger: snap.ledger.map((e) => ({
      ...e,
      review: e.review === undefined ? undefined : { ...e.review, contentHash: "MASK", evidenceRefs: maskList() },
      changeNote: { ...e.changeNote, evidence: maskList(), guardrails: maskList() }
    })),
    pending: snap.pending,
    autoPromotionsUsed: snap.autoPromotionsUsed,
    rollbackLedger: (snap.rollbackLedger ?? []).map((e) => ({ ...e, evidence: maskList() })),
    retiredVersionIds: snap.retiredVersionIds,
    contentsCount: (snap.contents ?? []).length,
    contentsBytes: (snap.contents ?? []).reduce((s, b) => s + b.content.length, 0)
  }).replaceAll(tag, "TAG");
}

/* ============================================================
 * Section 6: fits and crossing math.
 * ============================================================ */
{
  const fit = (pts: { x: number; y: number }[]): { slope: number; intercept: number } => {
    const nn = pts.length;
    const sx = pts.reduce((s, p) => s + p.x, 0);
    const sy = pts.reduce((s, p) => s + p.y, 0);
    const sxx = pts.reduce((s, p) => s + p.x * p.x, 0);
    const sxy = pts.reduce((s, p) => s + p.x * p.y, 0);
    const slope = (nn * sxy - sx * sy) / (nn * sxx - sx * sx);
    return { slope, intercept: (sy - slope * sx) / nn };
  };
  const loadFit = fit(s4rows.map((r) => ({ x: r.provKB, y: r.loadMs })));
  const saveFit = fit(s4rows.map((r) => ({ x: r.provKB, y: r.saveMs })));
  const parseFit = fit(s4rows.map((r) => ({ x: r.provKB, y: r.parseMs })));
  // per-item cost from the S2 cardinality lattice (parse leg, delta per added item):
  const s2lo = s2rows[0] as FleetRow;
  const s2hi = s2rows[s2rows.length - 1] as FleetRow;
  const items = (n: number): number => S2_K * 3 * n * 2; // 3 slots, 2 materializations
  const perItemNs = ((s2hi.parseMs - s2lo.parseMs) / (items(512) - items(1))) * 1e6;
  console.log(
    `S6 fits vs provenance KB (S4, on-disk incl. 2x materialization): load=${(loadFit.slope * 1e3).toFixed(1)}us/KB save=${(saveFit.slope * 1e3).toFixed(1)}us/KB parse=${(parseFit.slope * 1e3).toFixed(1)}us/KB | S2 parse per-item=${perItemNs.toFixed(0)}ns/item(@48B)`
  );
  const crossingKB = 10 / loadFit.slope;
  const perPromoteProdBytes = 2 * (21 + 14 + 18 + 40); // 2x (evidence+guardrails+refs singleton production widths approx)
  console.log(
    `S6 crossings: 10ms of load needs ~${crossingKB.toFixed(0)}KB provenance (~${Math.round((crossingKB * 1024) / perPromoteProdBytes)} production-shaped promotes at ~${perPromoteProdBytes}B/promote incl. 2x materialization); per-item validation crossing: 10ms/load needs ~${Math.round(10e6 / perItemNs)} list items`
  );
}
