 *   cell A'  process-isolated end-to-end (median full-experiment ms/arm)
 *            WITH per-arm GC telemetry (cell G': perf_hooks 'gc' entries
 *            captured during the timed passes — total pause ms, counts by
 *            kind) and heap growth across the timed section.
 *   cell E'  bookkeeping unit price: ns/op of the retention write itself
 *            (push-to-2001 / slot write / ring write) — closes the
 *            arithmetic that any ms-level e2e delta must be collector work,
 *            not bookkeeping.
 *   cell d   arithmetic closures: e2e delta vs GC-time delta vs bookkeeping.
 *
 * Deterministic witnesses before any timing (every arm): budget-exhausted
 * halt, ledger sizes A/A, assignment sequence = head-aligned prefix, zero
 * clock fields, within-arm double-execution digest equality, and — the
 * decisive witness — the ledger digests of ALL arms byte-identical to the
 * THREAD anchor digest (snapshot liveness must not leak into a single
 * ledger byte). Arm-construction fidelity is asserted directly: RETAIN
 * holds exactly 2A+1 snapshots at pass end (first element the start state,
 * last element identical to the final state); WINDOW64's ring holds 64 with
 * the newest slot identical to the final state; PUSH1's holder holds the
 * final state; BALLAST's ballast array still holds 2A+1 states after
 * timing (liveness across the whole timed section).
 *
 * Uses the production modules directly. Run: npx tsx tmp-r29f-axes.ts
 */

function onceMs(fn: () => void): number {
  const t0 = process.hrtime.bigint();
  fn();
  const t1 = process.hrtime.bigint();
  return Number(t1 - t0) / 1e6;
}

function median(xs: readonly number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  return s[Math.floor(s.length / 2)] as number;
}

/** Historical fixture shape: `ep_` + zero-padded base36 index, L=11. */
function makePackedPopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

const P = 2000;
const A = 1000;
const PASSES = 5;
const SNAPSHOTS_PER_EXPERIMENT = 2 * A + 1; // 1 start + A assign + A record
const basePopulation = makePackedPopulation(P); // ONE shared string pool
const baseHashes = basePopulation.slice(0, A);

/** Plain-literal plan (own data / Object.prototype / no membrane / packed). */
function makePlan(): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: "exp_r29fbase",
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r29fbase"),
    candidateId: createCandidateId(() => "r29fcand"),
    population: [...basePopulation],
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments: A, maxWallClockMs: 1e12 },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

function cleanStream(): ExperimentOutcome[] {
  return baseHashes.map((h) => ({
    episodeHash: h,
    utility: 0.5,
    costUsd: 0.01,
    guardrailBreached: false,
  }));
}

const sharedStream = cleanStream();

const ARMS = ["THREAD", "PUSH1", "WINDOW64", "RETAIN", "BALLAST"] as const;
type ArmName = (typeof ARMS)[number];

const WINDOW = 64;

interface RunResult {
  readonly finalState: ShadowState;
  /** Arm-construction fidelity payload (checked by verifyArm). */
  readonly retainedCount: number;
  readonly retainedFirst: ShadowState | undefined;
  readonly retainedLast: ShadowState | undefined;
}

/**
 * Production interleave (the evaluateProposalShadow shape), clock pinned to
 * 0, DIRECT dispatch only. The retention SINK is the arm; the operand fed to
 * every call is the freshest snapshot in every arm (input residency pinned).
 * `keepAlive` receives the arm's retained container so the optimizer cannot
 * dead-store it; the driver holds it until after the timer stops.
 */
function runInterleaved(runner: ShadowRunner, arm: ArmName, sink: unknown[]): RunResult {
  let state = runner.start(0);
  if (arm === "RETAIN") {
    const retained: ShadowState[] = [state];
    for (const outcome of sharedStream) {
      if (state.halted) {
        break;
      }
      state = runner.assign(state, outcome.episodeHash, 0);
      retained.push(state);
      state = runner.recordOutcome(state, outcome, 0);
      retained.push(state);
    }
    sink.push(retained); // alive until the driver clears the sink
    return {
      finalState: state,
      retainedCount: retained.length,
      retainedFirst: retained[0],
      retainedLast: retained[retained.length - 1],
    };
  }
  if (arm === "WINDOW64") {
    const ring: ShadowState[] = new Array<ShadowState>(WINDOW);
    let writes = 0;
    ring[writes++ % WINDOW] = state;
    for (const outcome of sharedStream) {
      if (state.halted) {
        break;
      }
      state = runner.assign(state, outcome.episodeHash, 0);
      ring[writes++ % WINDOW] = state;
      state = runner.recordOutcome(state, outcome, 0);
      ring[writes++ % WINDOW] = state;
    }
    sink.push(ring);
    return {
      finalState: state,
      retainedCount: WINDOW,
      retainedFirst: undefined,
      retainedLast: ring[(writes - 1) % WINDOW],
    };
  }
  if (arm === "PUSH1") {
    const holder: ShadowState[] = [state];
    for (const outcome of sharedStream) {
      if (state.halted) {
        break;
      }
      state = runner.assign(state, outcome.episodeHash, 0);
      holder[0] = state;
      state = runner.recordOutcome(state, outcome, 0);
      holder[0] = state;
    }
    sink.push(holder);
    return {
      finalState: state,
      retainedCount: 1,
      retainedFirst: undefined,
      retainedLast: holder[0],
    };
  }
  // THREAD and BALLAST: latest-only threading — the production coordinate.
  for (const outcome of sharedStream) {
    if (state.halted) {
      break;
    }
    state = runner.assign(state, outcome.episodeHash, 0);
    state = runner.recordOutcome(state, outcome, 0);
  }
  return { finalState: state, retainedCount: 0, retainedFirst: undefined, retainedLast: undefined };
}

/** BALLAST arm: one untimed full-retention experiment whose snapshots stay alive. */
function buildBallast(runner: ShadowRunner): ShadowState[] {
  const sink: unknown[] = [];
  const result = runInterleaved(runner, "RETAIN", sink);
  const retained = sink[0] as ShadowState[];
  if (retained.length !== SNAPSHOTS_PER_EXPERIMENT || result.retainedCount !== SNAPSHOTS_PER_EXPERIMENT) {
    throw new Error("BALLAST: ballast construction did not retain 2A+1 snapshots");
  }
  return retained;
}

function ledgerDigest(state: ShadowState): string {
  return stableStringify({
    assignments: state.assignments,
    outcomes: state.outcomes,
    guardrailBreaches: state.guardrailBreaches,
    halted: state.halted,
    haltReason: state.haltReason,
  });
}

function verifyFinalState(name: string, state: ShadowState): void {
  if (!state.halted || state.haltReason !== "budget-exhausted") {
    throw new Error(`${name}: expected budget-exhausted final state`);
  }
  if (state.assignments.length !== A || state.outcomes.length !== A) {
    throw new Error(`${name}: unexpected ledger sizes`);
  }
  for (let i = 0; i < A; i++) {
    const assignment = state.assignments[i] as { episodeHash: string };
    if (assignment.episodeHash !== (basePopulation[i] as string)) {
      throw new Error(`${name}: assignment ${i} not the head-aligned prefix`);
    }
  }
  if (state.guardrailBreaches !== 0 || state.startedAtMs !== 0 || state.elapsedMs !== 0) {
    throw new Error(`${name}: unexpected counters/clock fields`);
  }
  if (Object.isFrozen(state.plan.population) || Object.isSealed(state.plan.population)) {
    throw new Error(`${name}: storage-state face moved (must stay extensible)`);
  }
}

function verifyArm(name: ArmName, result: RunResult): void {
  verifyFinalState(name, result.finalState);
  if (name === "RETAIN") {
    if (result.retainedCount !== SNAPSHOTS_PER_EXPERIMENT) {
      throw new Error(`RETAIN: retained ${result.retainedCount}, expected ${SNAPSHOTS_PER_EXPERIMENT}`);
    }
    const first = result.retainedFirst as ShadowState;
    if (first.assignments.length !== 0 || first.outcomes.length !== 0 || first.halted) {
      throw new Error("RETAIN: first retained snapshot is not the start state");
    }
    if (result.retainedLast !== result.finalState) {
      throw new Error("RETAIN: last retained snapshot is not the final state");
    }
  }
  if (name === "WINDOW64" && result.retainedLast !== result.finalState) {
    throw new Error("WINDOW64: newest ring slot is not the final state");
  }
  if (name === "PUSH1" && result.retainedLast !== result.finalState) {
    throw new Error("PUSH1: holder does not hold the final state");
  }
}

/* ----------------------------- child-cell modes ---------------------------- */
const cellFlag = process.argv.find((a) => a.startsWith("--cell="));
const armFlag = process.argv.find((a) => a.startsWith("--arm="));

if (cellFlag !== undefined) {
  const cell = cellFlag.slice("--cell=".length);
  const armName = (armFlag ?? "--arm=").slice("--arm=".length) as ArmName;

  if (cell === "e2e") {
    const runner = createShadowRunner(makePlan());
    // BALLAST: old-gen preload built (untimed) BEFORE warmup, held module-wide.
    const ballast: ShadowState[] | undefined = armName === "BALLAST" ? buildBallast(runner) : undefined;

    // Verification (untimed, twice; digests must match within arm).
    const digests: string[] = [];
    for (let round = 0; round < 2; round++) {
      const sink: unknown[] = [];
      const result = runInterleaved(runner, armName, sink);
      verifyArm(armName, result);
      digests.push(ledgerDigest(result.finalState));
      sink.length = 0;
    }
    if (digests[0] !== digests[1]) {
      throw new Error(`${armName}: nondeterministic ledger`);
    }

    // Warmup (one untimed pass with the arm's own retention shape).
    {
      const sink: unknown[] = [];
      runInterleaved(runner, armName, sink);
      sink.length = 0;
    }

    // Timed passes with GC telemetry (cell G').
    let gcMs = 0;
    let gcMinor = 0;
    let gcMajor = 0;
    let gcIncremental = 0;
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        gcMs += entry.duration;
        const kind = (entry.detail as { kind?: number } | undefined)?.kind ?? 0;
        if (kind === perfConstants.NODE_PERFORMANCE_GC_MINOR) gcMinor += 1;
        else if (kind === perfConstants.NODE_PERFORMANCE_GC_MAJOR) gcMajor += 1;
        else if (kind === perfConstants.NODE_PERFORMANCE_GC_INCREMENTAL) gcIncremental += 1;
      }
    });
    observer.observe({ entryTypes: ["gc"] });
    const heapBefore = process.memoryUsage().heapUsed;
    const samples: number[] = [];
    for (let pass = 0; pass < PASSES; pass++) {
      const sink: unknown[] = [];
      samples.push(
        onceMs(() => {
          runInterleaved(runner, armName, sink);
        })
      );
      // Retained snapshots stay alive across the WHOLE timed section of the
      // pass (the timer above) and are released only here, between passes.
      sink.length = 0;
    }
    const heapAfter = process.memoryUsage().heapUsed;
    await new Promise((resolve) => setTimeout(resolve, 100)); // flush gc entries
    observer.disconnect();

    // BALLAST liveness witness: the ballast survived the whole timed section.
    if (armName === "BALLAST") {
      if (ballast === undefined || ballast.length !== SNAPSHOTS_PER_EXPERIMENT) {
        throw new Error("BALLAST: ballast was not held alive across the timed section");
      }
      verifyFinalState("BALLAST-ballast-final", ballast[ballast.length - 1] as ShadowState);
    }

    process.stdout.write(
      `CHILD e2e ${armName} ${median(samples).toFixed(2)} ms gc=${gcMs.toFixed(1)}ms ` +
        `minor=${gcMinor} major=${gcMajor} incr=${gcIncremental} ` +
        `heapDelta=${((heapAfter - heapBefore) / 1048576).toFixed(1)}MB\n`
    );
    process.exit(0);
  }

  if (cell === "unitprobe") {
    // ns per retention write: the bookkeeping alone over inert snapshot-like
    // objects (the sink write is the only difference between arms).
    const fakeStates: Array<{ v: number }> = [];
    for (let i = 0; i < SNAPSHOTS_PER_EXPERIMENT; i++) fakeStates.push({ v: i });
    let sink = 0;
    let probe: () => void;
    if (armName === ("RETAIN" as ArmName)) {
      probe = () => {
        const retained: Array<{ v: number }> = [];
        for (let i = 0; i < SNAPSHOTS_PER_EXPERIMENT; i++) retained.push(fakeStates[i] as { v: number });
        sink += retained.length;
      };
    } else if (armName === ("WINDOW64" as ArmName)) {
      probe = () => {
        const ring: Array<{ v: number }> = new Array<{ v: number }>(WINDOW);
        for (let i = 0; i < SNAPSHOTS_PER_EXPERIMENT; i++) ring[i % WINDOW] = fakeStates[i] as { v: number };
        sink += ring.length;
      };
    } else {
      probe = () => {
        const holder: Array<{ v: number }> = [fakeStates[0] as { v: number }];
        for (let i = 0; i < SNAPSHOTS_PER_EXPERIMENT; i++) holder[0] = fakeStates[i] as { v: number };
        sink += holder.length;
      };
    }
    for (let c = 0; c < 200; c++) probe();
    const REPS = 2000;
    const samples: number[] = [];
    for (let pass = 0; pass < PASSES; pass++) {
      samples.push(
        onceMs(() => {
          for (let c = 0; c < REPS; c++) probe();
        })
      );
    }
    const nsPerWrite = (median(samples) * 1e6) / (REPS * SNAPSHOTS_PER_EXPERIMENT);
    process.stdout.write(
      `CHILD unitprobe ${armName} ${nsPerWrite.toFixed(3)} ns/write sink=${sink > 0 ? "ok" : "zero"}\n`
    );
    process.exit(0);
  }

  throw new Error(`unknown cell: ${cell}`);
}

/* ------------------- verification pass (untimed, all arms) ------------------ */
const verificationRunner = createShadowRunner(makePlan());
let anchorDigest = "";
for (const name of ARMS) {
  const digests: string[] = [];
  for (let round = 0; round < 2; round++) {
    const sink: unknown[] = [];
    const result = runInterleaved(verificationRunner, name, sink);
    verifyArm(name, result);
    digests.push(ledgerDigest(result.finalState));
    sink.length = 0;
  }
  if (digests[0] !== digests[1]) {
    throw new Error(`${name}: nondeterministic ledger`);
  }
  if (name === "THREAD") {
    anchorDigest = digests[0] as string;
  } else if (digests[0] !== anchorDigest) {
    throw new Error(`${name}: ledger diverged from anchor across retention arms`);
  }
}
process.stdout.write(
  "verification: all output-retention invariants hold (cross-arm ledger identity)\n"
);

/* ---------- cell a: in-process round-robin e2e (pollution view) ------------ */
for (const name of ARMS) {
  const sink: unknown[] = [];
  runInterleaved(verificationRunner, name, sink); // warmup
  sink.length = 0;
}
const ballastForRoundRobin = buildBallast(verificationRunner); // shared BALLAST preload
const samplesA = new Map<ArmName, number[]>(ARMS.map((name) => [name, []]));
for (let pass = 0; pass < PASSES; pass++) {
  for (const name of ARMS) {
    const sink: unknown[] = [];
    (samplesA.get(name) as number[]).push(
      onceMs(() => {
        runInterleaved(verificationRunner, name, sink);
      })
    );
    sink.length = 0;
  }
}
if (ballastForRoundRobin.length !== SNAPSHOTS_PER_EXPERIMENT) {
  throw new Error("round-robin BALLAST preload was not held alive");
}
const baseA = median(samplesA.get("THREAD") as number[]);
process.stdout.write(
  `cell a (IN-PROCESS round-robin e2e, P=${P}/A=${A}, median ms; delta vs THREAD):\n`
);
for (const name of ARMS) {
  const med = median(samplesA.get(name) as number[]);
  const label = name === "THREAD" ? "(baseline)" : `delta=${(med - baseA).toFixed(2)}`;
  process.stdout.write(`${name.padEnd(8)} median=${med.toFixed(2)} ${label}\n`);
}

/* ----------------------- isolated child-cell driver ------------------------ */
function runChild(cell: string, arm: string): string[] {
  const child = spawnSync(
    process.execPath,
    ["--import", "tsx", process.argv[1] as string, `--cell=${cell}`, `--arm=${arm}`],
    { encoding: "utf8", timeout: 300_000 }
  );
  if (child.status !== 0) {
    return [`CHILD ${cell} ${arm} FAILED status=${String(child.status)} ${child.stderr.slice(0, 300)}`];
  }
  return child.stdout.trim().split("\n");
}

function runIsolatedCell(title: string, cell: string, arms: readonly string[]): Map<string, string> {
  process.stdout.write(`${title}\n`);
  const lines = new Map<string, string>();
  for (const arm of arms) {
    for (const line of runChild(cell, arm)) {
      process.stdout.write(`  ${line}\n`);
      if (line.startsWith("CHILD")) {
        lines.set(arm, line);
      }
    }
  }
  return lines;
}

const e2eIso = runIsolatedCell(
  "cell A' (PROCESS-ISOLATED e2e + cell G' GC telemetry, one fresh child per arm, median full-experiment ms):",
  "e2e",
  ARMS
);
const unitIso = runIsolatedCell(
  "cell E' (retention bookkeeping unit price, ns/write, isolated children):",
  "unitprobe",
  ["PUSH1", "WINDOW64", "RETAIN"]
);

/* --------------------- cell d: arithmetic closures ------------------------ */
function parseMs(line: string | undefined): number {
  const match = line?.match(/ ([0-9.]+) ms gc=([0-9.]+)ms/);
  return match === null || match === undefined ? Number.NaN : Number(match[1]);
}
function parseGc(line: string | undefined): number {
  const match = line?.match(/gc=([0-9.]+)ms/);
  return match === null || match === undefined ? Number.NaN : Number(match[1]);
}
function parseNs(line: string | undefined): number {
  const match = line?.match(/ ([0-9.]+) ns\/write/);
  return match === null || match === undefined ? 0 : Number(match[1]);
}
const e2eBase = parseMs(e2eIso.get("THREAD"));
const gcBase = parseGc(e2eIso.get("THREAD"));
const lines: string[] = [];
for (const name of ARMS) {
  if (name === "THREAD") continue;
  const e2eDelta = parseMs(e2eIso.get(name)) - e2eBase;
  const gcDelta = parseGc(e2eIso.get(name)) - gcBase;
  const bookkeepingMs = (parseNs(unitIso.get(name)) * SNAPSHOTS_PER_EXPERIMENT) / 1e6;
  lines.push(
    `${name} e2e-delta=${e2eDelta.toFixed(2)}ms gc-delta(5 passes)=${gcDelta.toFixed(1)}ms ` +
      `bookkeeping=${bookkeepingMs.toFixed(4)}ms/experiment`
  );
}
process.stdout.write(`cell d closures (isolated readings): ${lines.join("; ")}\n`);
