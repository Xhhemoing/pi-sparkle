let failures = 0;
function check(name: string, ok: boolean, detail?: string): void {
  if (!ok) {
    failures += 1;
    console.error(`FAIL ${name}${detail === undefined ? "" : `: ${detail}`}`);
  }
}
function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function pick<T>(rng: () => number, items: readonly T[]): T {
  return items[Math.floor(rng() * items.length)] as T;
}
function bench(fn: () => void, reps: number): number {
  fn();
  const t0 = performance.now();
  for (let i = 0; i < reps; i += 1) fn();
  return (performance.now() - t0) / reps;
}

function genObjective(rng: () => number): string {
  if (rng() < 0.5) return pick(rng, OBJECTIVE_TEMPLATES);
  let text = "";
  const parts = 1 + Math.floor(rng() * 8);
  for (let i = 0; i < parts; i += 1) text += pick(rng, FRAGMENTS) + " ";
  return text;
}
function genTasks(rng: () => number, n: number): AssignableTask[] {
  return Array.from({ length: n }, (_, i) => ({
    taskId: `tsk_${i.toString().padStart(6, "0")}` as AssignableTask["taskId"],
    role: pick(rng, ROLES),
    objective: genObjective(rng),
    ...(rng() < 0.1 ? { contractRisk: rng() < 0.5 } : {}),
    ...(rng() < 0.3 ? { contextTokens: Math.floor(rng() * 100000) } : {}),
    ...(rng() < 0.3 ? { outputTokens: Math.floor(rng() * 4000) } : {})
  }));
}
/** Replay-faithful shape: every production caller maps only taskId/role/objective. */

    role: pick(rng, ROLES),
    objective: genObjective(rng)
  }));
}

    `matrix prior HYPOTHETICAL replay N=2000 [M=7 configured, prior]: ${(p7 * 1e3).toFixed(1)}us per call (${(p7 * 2).toFixed(2)}ms per eval x2)`
  );
  console.log(
    `matrix prior HYPOTHETICAL replay N=2000 [M=10 stress, prior]: ${(p10 * 1e3).toFixed(1)}us per call (${(p10 * 2).toFixed(2)}ms per eval x2)`
  );
  const cli2 = bench(() => assignTasks({ catalog: catalog2, tasks: cli, prior }), 2000);
  const cli7 = bench(() => assignTasks({ catalog: catalogCfg, tasks: cli, prior, learned: configuredLearned() }), 2000);
  console.log(
    `matrix prior REAL-SCALE live face N=30: M=2+prior ${(cli2 * 1e3).toFixed(1)}us | M=7+prior+learned ${(cli7 * 1e3).toFixed(1)}us per call`
  );
  const models7 = createModelRouter(catalogCfg).config.models;
  const plan7 = planAssignmentPolicy(models7, catalogCfg.models.map((m) => m.id));
  const perPick = bench(() => {
    pickFromPublicPrior(prior, "edit", plan7.assignableModels);
  }, 20000);
  console.log(`prior unit cost: pickFromPublicPrior(edit, M=7 pool, ${prior.scores.length} rows) = ${(perPick * 1e6).toFixed(0)}ns per call`);
  conclusions.push(
    `prior-cells real-scale-live-face M=7+prior+learned=${(cli7 * 1e3).toFixed(1)}us hypothetical-N2000 M=7+prior per-eval=${(p7 * 2).toFixed(1)}ms per-pick=${(perPick * 1e6).toFixed(0)}ns`
  );
}

/* ============================================================
 * §0e NEW measurement cell: the learned-policy container face.
 * The fourth and last assignTasks input container, never dosed at its
 * size-axis bottom (empty policy = the production-dominant shape), never
 * dosed on row arrangement, and never censused against the two production
 * writers' actual output shapes.
 * ============================================================ */
{
  const prior = realisticPrior();
  const replayTasks = genReplayTasks(mulberry32(0xb44b01), 2000);
  const cli = genTasks(mulberry32(0xb44b01), 30);
  const GHOST = "ghost/none";
  const FAMILIES = ["edit", "test", "review", "plan", "research", "refactor", "deploy", "unknown"] as const;
  const WRITER1_REASON = "deterministic-check taskSuccess FAIL (failureClass=model)";

  interface LearnedCell {
    readonly label: string;
    readonly base: ModelRouterConfig;
    readonly primary: string;
    readonly prodAvoidIds: readonly string[];
  }
  const cells: readonly LearnedCell[] = [
    {
      label: "M=7 configured",
      base: configuredCatalog(),
      primary: "acme/frontier",
      prodAvoidIds: ["orion/coder", "orion/lite", "acme/std", "acme/fast-mini"]
    },
    {
      label: "M=10 stress",
      base: tenModelCatalog(),
      primary: "m9",
      prodAvoidIds: ["m2", "m1", "m0", "m3"]
    }
  ];
  const armsFor = (
    cell: LearnedCell
  ): readonly [string, LearnedRoutingPolicy | undefined][] => {
    const empty: LearnedRoutingPolicy = { primaryModelId: cell.primary, avoid: [], prefer: [] };
    const prodCand: LearnedRoutingPolicy = {
      primaryModelId: cell.primary,
      avoid: cell.prodAvoidIds.map((modelId, i) => ({
        modelId,
        family: FAMILIES[i]!,
        reason: WRITER1_REASON
      })),
      prefer: []
    };
    const fixture = cell.label.startsWith("M=7") ? configuredLearned() : tenModelLearned();
    const prefer8Ghost: LearnedRoutingPolicy = {
      primaryModelId: cell.primary,
      avoid: [],
      prefer: FAMILIES.map((family) => ({ family, modelId: GHOST }))
    };
    const globalGhost: LearnedRoutingPolicy = {
      primaryModelId: cell.primary,
      avoid: Array.from({ length: 8 }, (_, i) => ({
        modelId: `ghost/g${i}`,
        reason: WRITER1_REASON
      })),
      prefer: []
    };
    const assignmentsRider: LearnedRoutingPolicy = {
      ...fixture,
      assignments: Array.from({ length: 30 }, (_, i) => ({
        role: ROLES[i % ROLES.length]!,
        model: cell.primary,
        family: FAMILIES[i % FAMILIES.length]!
      }))
    };
    return [
      ["none", undefined],
      ["empty", empty],
      ["prod-cand", prodCand],
      ["fixture", fixture],
      ["prefer8-ghost", prefer8Ghost],
      ["global-ghost", globalGhost],
      ["assignments-rider", assignmentsRider]
    ];
  };

  // Perf grid: replay N=2000 and live face N=30 per learned arm.
  for (const cell of cells) {
    for (const [armLabel, learned] of armsFor(cell)) {
      const catalog = cell.base;
      const replayCost = bench(
        () => assignTasks({ catalog, tasks: replayTasks, ...(learned !== undefined ? { learned } : {}) }),
        30
      );
      const liveCost = bench(
        () => assignTasks({ catalog, tasks: cli, ...(learned !== undefined ? { learned } : {}) }),
        2000
      );
      console.log(
        `learned-container [${cell.label}] arm=${armLabel}: replay N=2000 ${(replayCost * 1e3).toFixed(1)}us per call | live face N=30 ${(liveCost * 1e3).toFixed(2)}us per call`
      );
    }
  }

  // Eval-faithful pair: adaptation/eval-routing.ts:353-354 calls assignTasks
  // twice per eval with learned = parsed BASELINE (empty by construction via
  // ensureRoutingBaseline) and learned = parsed CANDIDATE (writer-1-faithful:
  // family-scoped avoid rows + prefer []). First-ever timed cell of the true
  // production eval learned configuration.
  for (const cell of cells) {
    const arms = new Map(armsFor(cell));
    const emptyArm = arms.get("empty")!;
    const prodArm = arms.get("prod-cand")!;
    const pairCost = bench(() => {
      assignTasks({ catalog: cell.base, tasks: replayTasks, learned: emptyArm });
      assignTasks({ catalog: cell.base, tasks: replayTasks, learned: prodArm });
    }, 15);
    console.log(
      `learned-container eval-faithful pair [${cell.label}] (baseline=empty + candidate=prod-cand): ${pairCost.toFixed(2)}ms per eval`
    );
    if (cell.label.startsWith("M=7")) {
      conclusions.push(
        `learned-container eval-faithful M=7 per-eval=${pairCost.toFixed(1)}ms (still-below-landing-line=${pairCost < 30})`
      );
    }
  }

  // W1: double-run byte identity at the heaviest container arm
  // (M=7 assignments-rider + prior, N=30).
  {
    const cell = cells[0]!;
    const rider = new Map(armsFor(cell)).get("assignments-rider")!;
    const runA = JSON.stringify(assignTasks({ catalog: cell.base, tasks: cli, learned: rider, prior }));
    const runB = JSON.stringify(assignTasks({ catalog: cell.base, tasks: cli, learned: rider, prior }));
    check("learned-container W1 [M=7 assignments-rider+prior, N=30]: double run byte-identical", runA === runB);
  }

  // W2: avoid-row arrangement inertness. A 6-row avoid list (mixed
  // family-scoped real ids, one global real id, ghost rows) in three orders
  // yields byte-identical output on both catalogs.
  {
    const shuffleRows = <T,>(rows: readonly T[], seed: number): T[] => {
      const rng = mulberry32(seed);
      const out = [...rows];
      for (let i = out.length - 1; i > 0; i -= 1) {
        const j = Math.floor(rng() * (i + 1));
        const tmp = out[i]!;
        out[i] = out[j]!;
        out[j] = tmp;
      }
      return out;
    };
    for (const cell of cells) {
      const ids = cell.base.models.map((m) => m.id);
      const avoidRows = [
        { modelId: ids[1]!, family: "edit", reason: WRITER1_REASON },
        { modelId: ids[2]!, family: "review", reason: WRITER1_REASON },
        { modelId: ids[0]!, reason: "auto-loop: meanScore 0.20 over 6 samples" },
        { modelId: "ghost/x", family: "plan", reason: WRITER1_REASON },
        { modelId: ids[3]!, family: "test", reason: WRITER1_REASON },
        { modelId: "ghost/y", reason: WRITER1_REASON }
      ];
      const orders: readonly [string, typeof avoidRows][] = [
        ["canonical", [...avoidRows]],
        ["reversed", [...avoidRows].reverse()],
        ["shuffle-0x51c1", shuffleRows(avoidRows, 0x51c1)]
      ];
      const outputs = orders.map(([, rows]) =>
        JSON.stringify(
          assignTasks({
            catalog: cell.base,
            tasks: cli,
            learned: { primaryModelId: cell.primary, avoid: rows, prefer: [] }
          })
        )
      );
      check(
        `learned-container W2 [${cell.label}]: avoid-row arrangement inert (3 orders byte-identical)`,
        outputs[0] === outputs[1] && outputs[0] === outputs[2]
      );
    }
  }

  // W3: prefer first-match flip. Duplicate-family prefer rows with DISTINCT
  // targets flip preferredModel AND decision.model end-to-end while
  // eligibleModels stay identical; the writer-2 duplicate shape (same target
  // twice) is order-inert.
  {
    const cell = cells[0]!;
    const editTask: AssignableTask = {
      taskId: "tsk_w3" as AssignableTask["taskId"],
      role: "worker",
      objective: "Implement retry logic for the ledger sync"
    };
    const mk = (prefer: readonly { family: string; modelId: string }[]): LearnedRoutingPolicy => ({
      primaryModelId: cell.primary,
      avoid: [],
      prefer
    });
    const outAB = assignTasks({
      catalog: cell.base,
      tasks: [editTask],
      learned: mk([
        { family: "edit", modelId: "acme/std" },
        { family: "edit", modelId: "orion/coder" }
      ])
    })[0]!;
    const outBA = assignTasks({
      catalog: cell.base,
      tasks: [editTask],
      learned: mk([
        { family: "edit", modelId: "orion/coder" },
        { family: "edit", modelId: "acme/std" }
      ])
    })[0]!;
    check(
      "learned-container W3a [M=7]: duplicate-family prefer order flips preferredModel (first-match-wins)",
      outAB.preferredModel === "acme/std" && outBA.preferredModel === "orion/coder",
      `${outAB.preferredModel} vs ${outBA.preferredModel}`
    );
    check(
      "learned-container W3b [M=7]: the flip carries into decision.model end-to-end",
      outAB.decision.model === "acme/std" && outBA.decision.model === "orion/coder",
      `${outAB.decision.model} vs ${outBA.decision.model}`
    );
    check(
      "learned-container W3c [M=7]: eligibleModels stay byte-identical across the flip",
      JSON.stringify(outAB.decision.eligibleModels) === JSON.stringify(outBA.decision.eligibleModels)
    );
    const dupSameA = JSON.stringify(
      assignTasks({
        catalog: cell.base,
        tasks: [editTask],
        learned: mk([
          { family: "edit", modelId: cell.primary },
          { family: "edit", modelId: cell.primary }
        ])
      })
    );
    const dupSameB = JSON.stringify(
      assignTasks({
        catalog: cell.base,
        tasks: [editTask],
        learned: mk([
          { family: "edit", modelId: cell.primary },
          { family: "edit", modelId: cell.primary }
        ])
      })
    );
    check("learned-container W3d [M=7]: writer-2 duplicate shape (same target) is order-inert", dupSameA === dupSameB);
  }

  // W4: dead-field / dose inertness. assignments-rider ≡ fixture;
  // prefer8-ghost ≡ empty ≡ global-ghost (both catalogs, N=30).
  for (const cell of cells) {
    const arms = new Map(armsFor(cell));
    const run = (learned: LearnedRoutingPolicy | undefined): string =>
      JSON.stringify(
        assignTasks({ catalog: cell.base, tasks: cli, ...(learned !== undefined ? { learned } : {}) })
      );
    check(
      `learned-container W4a [${cell.label}]: assignments rider is dead on the apply path (output ≡ fixture)`,
      run(arms.get("assignments-rider")!) === run(arms.get("fixture")!)
    );
    const emptyOut = run(arms.get("empty")!);
    check(
      `learned-container W4b [${cell.label}]: prefer8-ghost output ≡ empty (pure prefer-scan dose)`,
      run(arms.get("prefer8-ghost")!) === emptyOut
    );
    check(
      `learned-container W4c [${cell.label}]: global-ghost output ≡ empty (pure avoid-Set dose)`,
      run(arms.get("global-ghost")!) === emptyOut
    );
  }

  // W5: empty-container identity. The empty arm's output is byte-identical to
  // the none arm's on the live face (N=30) AND the replay corpus (N=2000).
  for (const cell of cells) {
    const arms = new Map(armsFor(cell));
    const empty = arms.get("empty")!;
    const liveNone = JSON.stringify(assignTasks({ catalog: cell.base, tasks: cli }));
    const liveEmpty = JSON.stringify(assignTasks({ catalog: cell.base, tasks: cli, learned: empty }));
    const replayNone = JSON.stringify(assignTasks({ catalog: cell.base, tasks: replayTasks }));
    const replayEmpty = JSON.stringify(assignTasks({ catalog: cell.base, tasks: replayTasks, learned: empty }));
    check(
      `learned-container W5 [${cell.label}]: empty policy ≡ no learned, byte-identical (N=30 and N=2000)`,
      liveNone === liveEmpty && replayNone === replayEmpty
    );
  }

  // W6: four-tier resolution census on the M=7 anchor (edit task), plus the
  // total-avoid kept==0 rescue. Shapes match the S13-B-1 equivalence-fuzz
  // inputs of R13-B; asserted here as explicit tier outcomes.
  {
    const cell = cells[0]!;
    const editTask: AssignableTask = {
      taskId: "tsk_w6" as AssignableTask["taskId"],
      role: "worker",
      objective: "Implement retry logic for the ledger sync"
    };
    const preferredOf = (learned: LearnedRoutingPolicy): string =>
      assignTasks({ catalog: cell.base, tasks: [editTask], learned })[0]!.preferredModel;
    check(
      "learned-container W6-tier1 [M=7]: prefer hit wins (edit -> acme/std)",
      preferredOf({
        primaryModelId: cell.primary,
        avoid: [],
        prefer: [{ family: "edit", modelId: "acme/std" }]
      }) === "acme/std"
    );
    check(
      "learned-container W6-tier2 [M=7]: empty policy keeps the plan preference (cheapest orion/lite)",
      preferredOf({ primaryModelId: cell.primary, avoid: [], prefer: [] }) === "orion/lite"
    );
    check(
      "learned-container W6-tier3 [M=7]: avoided plan preference falls back to learned.primaryModelId",
      preferredOf({
        primaryModelId: cell.primary,
        avoid: [{ modelId: "orion/lite", family: "edit", reason: WRITER1_REASON }],
        prefer: []
      }) === "acme/frontier"
    );
    check(
      "learned-container W6-tier4 [M=7]: ghost primary falls back to allowedModels[0] (acme/fast-mini)",
      preferredOf({
        primaryModelId: GHOST,
        avoid: [{ modelId: "orion/lite", family: "edit", reason: WRITER1_REASON }],
        prefer: []
      }) === "acme/fast-mini"
    );
    const allAvoided: LearnedRoutingPolicy = {
      primaryModelId: cell.primary,
      avoid: cell.base.models.map((m) => ({ modelId: m.id, family: "edit", reason: WRITER1_REASON })),
      prefer: []
    };
    const rescueOut = JSON.stringify(assignTasks({ catalog: cell.base, tasks: [editTask], learned: allAvoided }));
    const emptyOut = JSON.stringify(
      assignTasks({ catalog: cell.base, tasks: [editTask], learned: { primaryModelId: cell.primary, avoid: [], prefer: [] } })
    );
    check(
      "learned-container W6-rescue [M=7]: total avoid triggers the kept==0 fallback; output ≡ empty arm",
      rescueOut === emptyOut
    );
  }

  conclusions.push(
    "learned-container-face: size-bottom + arrangement + writer-shape census measured; W1-W6 witnesses asserted (prefer order is decision-carrying, avoid order and the assignments field are inert, empty ≡ none byte-identical)"
  );
}

console.log(`\nCONCLUSIONS: ${conclusions.join(" | ")}`);
if (failures > 0) {
  console.error(`\n${failures} check(s) FAILED`);
  process.exit(1);
}
console.log("ALL CHECKS PASSED (measurement harness + learned-policy-container fidelity witnesses; corpora seeded + deterministic)");
