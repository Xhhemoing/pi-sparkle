```javascript
// R14-I fat-configured-state axis: (a) pick one real model id from each of N
// distinct provider tables; (b) measure in-process linear per-provider import
// cost at N=5/10/20/39 distinct providers vs providers/all cold; (c) emit a
// providers.json "manyprov" state for the spawn matrix.
import { performance } from "node:perf_hooks";
import { writeFileSync, mkdirSync } from "node:fs";
import { join } from "node:path";

const which = process.argv[2];
const label = process.version;
const PROVIDERS = [
  "anthropic", "openai", "google", "mistral", "groq",
  "deepseek", "xai", "fireworks", "cerebras", "together",
  "baseten", "huggingface", "minimax", "moonshotai", "nvidia",
  "openrouter", "qwen-token-plan", "vercel-ai-gateway", "xiaomi", "zai"
];

async function firstModelId(providerId) {
  const ns = await import(`@earendil-works/pi-ai/providers/${providerId}.models`);
  const key = Object.keys(ns).find((k) => k.endsWith("_MODELS"));
  const ids = Object.keys(ns[key]);
  return ids[0];
}

if (which === "emit-state") {
  const n = Number(process.argv[3] ?? 10);
  const chosen = PROVIDERS.slice(0, n);
  const enabled = [];
  for (const p of chosen) enabled.push(`${p}/${await firstModelId(p)}`);
  const state = {
    version: 1,
    enabled,
    primary: enabled[0],
    fast: enabled[1] ?? enabled[0],
    customProviders: []
  };
  const dir = join("/tmp/r14i-bench/fixtures", `state-manyprov${n}`, "runtime");
  mkdirSync(dir, { recursive: true });
  writeFileSync(join(dir, "providers.json"), JSON.stringify(state, null, 2) + "\n");
  console.log(`emitted state-manyprov${n}: ${enabled.length} enabled across ${chosen.length} providers`);
} else if (which === "linear") {
  const n = Number(process.argv[3]);
  const { readdirSync } = await import("node:fs");
  const dir = "/workspace/node_modules/@earendil-works/pi-ai/dist/providers";
  const all = readdirSync(dir).filter((f) => f.endsWith(".models.js")).map((f) => f.replace(/\.models\.js$/, ""));
  const chosen = all.slice(0, n);
  const t0 = performance.now();
  for (const id of chosen) {
    await import(`@earendil-works/pi-ai/providers/${id}.models`);
  }
  const dt = performance.now() - t0;
  console.log(`${label} per-provider x${chosen.length}: ${dt.toFixed(1)}ms (${(dt / chosen.length).toFixed(2)}ms/table)`);
}
```

### A.4 manyprov10 spawn 格 harness（/tmp/r14i-bench/spot-fat.mjs；A.1 同纪律，仅状态源换 state-manyprov10）

```javascript
// R14-I fat-configured-state spawn cells: run-children/run-track @ manyprov10
// (10 enabled models across 10 distinct builtin providers). Same harness
// discipline as spot.mjs: fresh state-root copy per iteration, 2 warmup + 10.
import { spawnSync } from "node:child_process";
import { mkdirSync, rmSync, writeFileSync, cpSync, existsSync } from "node:fs";
import { join } from "node:path";
import { performance } from "node:perf_hooks";

const CLI = "/workspace/dist/cli/main.js";
const ROOT = "/tmp/r14i-bench";
const FIX = join(ROOT, "fixtures");
const WORK = join(ROOT, "work-fat");
const nodeBin = process.argv[2];
const WARMUP = 2, ITERS = 10;

if (!existsSync(join(FIX, "state-manyprov10"))) {
  console.error("state-manyprov10 missing; run r14i-fat.mjs emit-state 10 first");
  process.exit(1);
}
const PROJ = join(FIX, "project");
mkdirSync(join(PROJ, "src"), { recursive: true });
writeFileSync(join(PROJ, "package.json"), JSON.stringify({ name: "bench-project", version: "1.0.0" }) + "\n");
writeFileSync(join(PROJ, "src", "index.js"), "export const x = 1;\n");
const CHILDREN = join(FIX, "children.json");
writeFileSync(CHILDREN, JSON.stringify({ tasks: [
  { id: "tsk_a", role: "implementer", objective: "Do the work" },
  { id: "tsk_b", role: "reviewer", objective: "Review the work", dependsOn: ["tsk_a"] }
] }) + "\n");

const cells = ["children", "track"].map((mode) => ({
  id: `run-${mode}@manyprov10`,
  args: (s) => mode === "children"
    ? ["run", "--project", PROJ, "--objective", "bench objective", "--state-root", s, "--children", CHILDREN]
    : ["run", "--project", PROJ, "--objective", "bench objective", "--state-root", s, "--track", "--assume-defaults"]
}));
function freshState(cellId, iter) {
  const dir = join(WORK, `${cellId.replace(/[^a-z0-9-]/gi, "_")}-${iter}`);
  rmSync(dir, { recursive: true, force: true });
  cpSync(join(FIX, "state-manyprov10"), dir, { recursive: true });
  return dir;
}
function q(sorted, p) { const idx = (sorted.length - 1) * p, lo = Math.floor(idx), hi = Math.ceil(idx); return sorted[lo] + (sorted[hi] - sorted[lo]) * (idx - lo); }
rmSync(WORK, { recursive: true, force: true });
mkdirSync(WORK, { recursive: true });
for (const cell of cells) {
  const times = []; let exit;
  for (let i = 0; i < WARMUP + ITERS; i++) {
    const sd = freshState(cell.id, i);
    const env = { ...process.env };
    delete env.PI_PROVIDER; delete env.PI_MODEL; delete env.PI_API_KEY;
    const t0 = performance.now();
    const r = spawnSync(nodeBin, [CLI, ...cell.args(sd)], { env, encoding: "utf8", timeout: 120000 });
    const dt = performance.now() - t0;
    exit = r.status;
    if (i === 0 && r.status !== 0) console.error(`  FIRST-FAIL ${cell.id}: ${(r.stderr ?? "").slice(0, 300)}`);
    if (i >= WARMUP) times.push(dt);
  }
  times.sort((a, b) => a - b);
  console.log(`${cell.id}: median=${q(times, 0.5).toFixed(1)}ms min=${times[0].toFixed(1)} p90=${q(times, 0.9).toFixed(1)} exit=${exit}`);
}
```
