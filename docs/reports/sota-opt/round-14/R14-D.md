function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

  const first = versions[0] as { identity: ResourceIdentity; v1: ResourceVersion; v2?: ResourceVersion };
  registry.rollback({
    identity: first.identity,
    expectedCurrentVersionId: (first.v2 as ResourceVersion).versionId,
    targetVersionId: first.v1.versionId,

const FORBIDDEN_POLICY_FIELDS = new Set([
  "permission",
  "credential",
  "secret",

]);
function assertNoForbiddenFieldsReplica(value: unknown): void {
  if (Array.isArray(value)) {
    for (const entry of value) {
      assertNoForbiddenFieldsReplica(entry);
    }
    return;
  }
  if (!isRecord(value)) {
    return;
  }
  for (const [key, child] of Object.entries(value)) {
    if (FORBIDDEN_POLICY_FIELDS.has(key.toLowerCase())) {
      throw new DomainValidationError(`routing-policy contains forbidden field ${key}`);
    }
    assertNoForbiddenFieldsReplica(child);
  }
}
function parseEpisodeReplica(value: unknown, index: number): EpisodeShape {
  if (!isRecord(value)) {
    throw new DomainValidationError(`dataset episodes[${index}] must be an object`);
  }
  if (typeof value.episodeHash !== "string" || value.episodeHash.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires episodeHash`);
  }
  if (typeof value.taskId !== "string") {
    throw new DomainValidationError(`dataset episodes[${index}] requires taskId`);
  }
  parseTaskId(value.taskId);
  if (!isAgentRole(value.role)) {
    throw new DomainValidationError(`dataset episodes[${index}] has invalid role`);
  }
  if (typeof value.objective !== "string" || value.objective.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires objective`);
  }
  if (typeof value.originalWorkspace !== "string" || value.originalWorkspace.trim() === "") {
    throw new DomainValidationError(`dataset episodes[${index}] requires originalWorkspace`);
  }
  let taskSuccess: "PASS" | "FAIL" | undefined;
  if (value.taskSuccess !== undefined) {
    if (value.taskSuccess !== "PASS" && value.taskSuccess !== "FAIL") {
      throw new DomainValidationError(
        `dataset episodes[${index}] taskSuccess must be PASS or FAIL when present`
      );
    }
    taskSuccess = value.taskSuccess;
  }
  const taskFamily =
    typeof value.taskFamily === "string" && value.taskFamily.trim() !== ""
      ? value.taskFamily
      : undefined;
  return {
    episodeHash: value.episodeHash,
    taskId: value.taskId,
    role: value.role,
    objective: value.objective,
    originalWorkspace: value.originalWorkspace,
    ...(taskFamily !== undefined ? { taskFamily } : {}),
    ...(taskSuccess !== undefined ? { taskSuccess } : {})
  };
}

    environmentVersion: manifest.environmentVersion,
    evaluatorVersion: ROUTING_EVALUATOR_VERSION
  });

      datasetId: manifest.datasetId,
      cache: {
        candidateHash: candidate.contentHash,
        environmentVersion: manifest.environmentVersion,
        evaluatorVersion: ROUTING_EVALUATOR_VERSION

    })
  );

    candidateId,
    contentHash: candidate.contentHash,

  // (ms per episode), then E* = scale at which the component reaches the line.
  function slopePerEpisode(points: readonly ScalePoint[], value: (p: ScalePoint) => number): { slope: number; intercept: number } {
    const n = points.length;
    let sx = 0;
    let sy = 0;
    let sxx = 0;
    let sxy = 0;
    for (const p of points) {
      const x = p.scale;
      const y = value(p);
      sx += x;
      sy += y;
      sxx += x * x;
      sxy += x * y;
    }
    const slope = (n * sxy - sx * sy) / (n * sxx - sx * sx);
    const intercept = (sy - slope * sx) / n;
    return { slope, intercept };
  }
  function crossing(points: readonly ScalePoint[], value: (p: ScalePoint) => number, lineMs: number): number {
    const { slope, intercept } = slopePerEpisode(points, value);
    return (lineMs - intercept) / slope;

    [16, 10]
  ] as const) {
    const stateRoot = join(workRoot, `state-k${k}`);
    const { registry } = buildConfiguredRegistry(`k${k}s`, 8 * k);
    await saveAdaptationRegistry(stateRoot, registry);
    const path = adaptationRegistryPath(stateRoot);
    const raw = await readFile(path, "utf8");
    const parsed = JSON.parse(raw) as unknown;
    const snapshot = parseRegistrySnapshot(parsed);
    const contents = snapshot.contents ?? [];
    const blobKB = contents.reduce((sum, blob) => sum + blob.content.length, 0) / 1024;

    const loadMs = await benchAsync(async () => {
      await loadAdaptationRegistry(stateRoot);
    }, reps);
    const saveMs = await benchAsync(async () => {
      await saveAdaptationRegistry(stateRoot, registry);
    }, reps);
    const readMs = await benchAsync(async () => {
      await readFile(path, "utf8");
    }, reps);
    const jsonParseMs = bench(() => {
      JSON.parse(raw);
    }, reps * 2);
    const parseSnapshotMs = bench(() => {
      parseRegistrySnapshot(parsed);
    }, reps * 2);
    const fromSnapshotMs = bench(() => {
      ResourceRegistry.fromSnapshot(snapshot, { now: () => NOW });
    }, reps * 2);
    const rehashOnlyMs = bench(() => {
      for (const blob of contents) {
        if (hashCandidateContent(blob.content) !== blob.hash) {
          throw new DomainValidationError("unexpected hash mismatch");
        }
      }
    }, reps * 2);
    statePoints.push({
      k,
      rawKB: raw.length / 1024,
      blobKB,
      loadMs,
      saveMs,
      readMs,
      jsonParseMs,
      parseSnapshotMs,
      fromSnapshotMs,
      rehashOnlyMs
    });
    console.log(
      `S3 k=${String(k).padStart(2)} (registry.json=${(raw.length / 1024).toFixed(1)}KB blobs=${blobKB.toFixed(1)}KB V=${snapshot.versions.length} B=${contents.length}): load=${loadMs.toFixed(2)}ms save(+fsync)=${saveMs.toFixed(2)}ms | read=${readMs.toFixed(2)} JSON.parse=${jsonParseMs.toFixed(2)} parseSnapshot=${parseSnapshotMs.toFixed(2)} fromSnapshot(incl S8-D-1)=${fromSnapshotMs.toFixed(2)} rehash-only=${rehashOnlyMs.toFixed(2)}ms (${((rehashOnlyMs / loadMs) * 100).toFixed(1)}% of load)`
    );
  }
  // Crossing solve on raw registry KB (load is ~linear in bytes).
  function fit(points: readonly StatePoint[], value: (p: StatePoint) => number): { slope: number; intercept: number } {
    const n = points.length;
    let sx = 0;
    let sy = 0;
    let sxx = 0;
    let sxy = 0;
    for (const p of points) {
      const x = p.rawKB;
      const y = value(p);
      sx += x;
      sy += y;
      sxx += x * x;
      sxy += x * y;
    }
    const slope = (n * sxy - sx * sy) / (n * sxx - sx * sx);
    const intercept = (sy - slope * sx) / n;
    return { slope, intercept };
  }
  const base = statePoints[0]!;
  for (const [label, value] of [
    ["load (incl. mandatory S8-D-1 rehash)", (p: StatePoint) => p.loadMs],
    ["save(+fsync)", (p: StatePoint) => p.saveMs],
    ["S8-D-1 rehash-only", (p: StatePoint) => p.rehashOnlyMs]
  ] as const) {
    const { slope, intercept } = fit(statePoints, value);
    const kb10 = (10 - intercept) / slope;
    console.log(
      `S3 crossing [${label}]: slope=${(slope * 1e3).toFixed(2)}us/KB intercept=${intercept.toFixed(2)}ms -> crosses 10ms at ~${kb10.toFixed(0)}KB registry.json = ${(kb10 / base.rawKB).toFixed(1)}x the configured production anchor`
    );
  }
}
