 * R14-J new-axis harness: scale-crossing calibration + phase decomposition.
 * J-slice counterpart of the R14-A/D/E scale-crossing cells: at what data
 * scale would each I/O-floor path cross the tens-of-ms landing line, and
 * which phase (CPU vs serialize vs write) owns the cost at that scale?
 * Adjudicates the standing reopen condition ("store grows >=2 orders of
 * magnitude") numerically for the first time. Temporary script: not
 * committed (no production winner -> report only).
 */
import { mkdtempSync, rmSync, mkdirSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join, dirname } from "node:path";

import {
  configurePreferencePersistence,
  recordPreference,
  listObservations,
  listTombstones,
  resetPreferenceStore
} from "./src/preferences/store.js";
import { readJsonlObjects } from "./src/persist/jsonl.js";

import type { EpisodeId } from "./src/domain/ids.js";

const hr = (): bigint => process.hrtime.bigint();
const usPer = (a: bigint, b: bigint, n: number): number => Number(b - a) / 1000 / n;
const ms = (x: number): string => (x / 1000).toFixed(3);

// ---------------------------------------------------------------- cell A
// saveToDisk scale ladder: recordPreference end-to-end at store size N, then
// decompose the same snapshot into stringify-only and write-only phases.
// CPU remainder (applyObservation filter + rebuildViews) = end-to-end minus
// (stringify + write) approximately.
async function cellSaveLadder(): Promise<void> {
  for (const N of [50, 500, 5000, 20000]) {
    const dir = mkdtempSync(join(tmpdir(), "r14j-sl-"));
    resetPreferenceStore();
    configurePreferencePersistence(join(dir, "prefs.json"));
    for (let i = 0; i < N; i++) {
      recordPreference(
        (["user", "project", "role"] as const)[i % 3]!,
        `key-${i % 7}`,
        `pref-${i % 11}`,
        i % 2 === 0 ? `value-${i}` : i,
        `eps_seed${i.toString(36)}` as EpisodeId,
        1.0,
        i % 4 === 0
      );
    }
    const W = N >= 5000 ? 30 : 200;
    const t0 = hr();
    for (let i = 0; i < W; i++) {
      recordPreference(
        "project",
        `key-${i % 7}`,
        `pref-${i % 11}`,
        `value-w${i}`,
        `eps_w${i.toString(36)}` as EpisodeId,
        1.0,
        false
      );
    }
    const t1 = hr();
    const endToEnd = usPer(t0, t1, W);

    // phase decomposition on the equivalent snapshot object
    const snapshot = { observations: listObservations(), tombstones: listTombstones() };
    const R = N >= 5000 ? 30 : 200;
    let sink = 0;
    let t2 = hr();
    for (let i = 0; i < R; i++) sink += JSON.stringify(snapshot).length;
    let t3 = hr();
    const stringifyOnly = usPer(t2, t3, R);
    const bytes = JSON.stringify(snapshot);
    const phaseFile = join(dir, "phase.json");
    t2 = hr();
    for (let i = 0; i < R; i++) writeFileSync(phaseFile, bytes);
    t3 = hr();
    const writeOnly = usPer(t2, t3, R);
    const cpuRemainder = endToEnd - stringifyOnly - writeOnly;
    console.log(
      `saveToDisk N=${N}: end-to-end=${endToEnd.toFixed(1)}us/write ` +
        `(stringify=${stringifyOnly.toFixed(1)} write=${writeOnly.toFixed(1)} ` +
        `cpu-remainder~=${cpuRemainder.toFixed(1)}) bytes=${bytes.length} [sink=${sink}]`
    );
    resetPreferenceStore();
    configurePreferencePersistence(undefined);
    rmSync(dir, { recursive: true, force: true });
  }
}

// ---------------------------------------------------------------- cell B
// deletion-cascade scale ladder: N records with bodies, 10% hits. Decompose
// into read (readFeedbackRecordsRaw) and rewrite (writeFeedbackRecords).
function seedFeedbackN(stateRoot: string, N: number): void {
  const log = feedbackLogPath(stateRoot);
  mkdirSync(dirname(log), { recursive: true });
  const lines: string[] = [];
  for (let i = 0; i < N; i++) {
    const hit = i % 10 === 0;
    lines.push(
      JSON.stringify({
        id: `fb-${i}`,
        episodeId: hit ? "eps_hit" : `eps_other${i % 7}`,
        kind: "human",
        rubricVersion: "r1",
        score: 50 + (i % 40),
        evidenceRefs: [],
        redacted: false,

        body: `free text feedback payload number ${i} `.repeat(3),
        summary: `summary ${i}`
      })

  }
  writeFileSync(log, lines.join("\n") + "\n", "utf8");
  writeFileSync(
    feedbackTombstonesPath(stateRoot),
    JSON.stringify(["fb-pre-a", "fb-pre-b"], null, 2) + "\n",
    "utf8"
  );
}

async function cellCascadeLadder(): Promise<void> {
  for (const N of [200, 2000, 20000]) {
    const stateRoot = mkdtempSync(join(tmpdir(), "r14j-cl-"));
    seedFeedbackN(stateRoot, N);
    await cascadeFeedbackTombstones(stateRoot, "eps_warm" as EpisodeId);
    const iters = N >= 20000 ? 8 : 20;
    let total = 0n;
    for (let i = 0; i < iters; i++) {
      seedFeedbackN(stateRoot, N); // untimed re-seed
      const t0 = hr();
      const out = await cascadeFeedbackTombstones(stateRoot, "eps_hit" as EpisodeId);
      const t1 = hr();
      if (out.length !== N / 10) throw new Error(`expected ${N / 10} cascaded, got ${out.length}`);
      total += t1 - t0;
    }
    const matchUs = Number(total) / 1000 / iters;

    // phase decomposition: read side and rewrite side, same data
    seedFeedbackN(stateRoot, N);
    const R = N >= 20000 ? 8 : 20;
    let t2 = hr();
    let records: Awaited<ReturnType<typeof readFeedbackRecordsRaw>> = [];
    for (let i = 0; i < R; i++) records = await readFeedbackRecordsRaw(stateRoot);
    let t3 = hr();
    const readUs = usPer(t2, t3, R);
    const updated = records.map((record) =>
      record.episodeId === ("eps_hit" as EpisodeId) ? { ...record, body: undefined } : record
    );
    t2 = hr();
    for (let i = 0; i < R; i++) await writeFeedbackRecords(stateRoot, updated);
    t3 = hr();
    const rewriteUs = usPer(t2, t3, R);
    console.log(
      `cascade N=${N} (10% hits): match=${ms(matchUs)}ms/cascade ` +
        `(read=${ms(readUs)}ms rewrite=${ms(rewriteUs)}ms)`
    );
    rmSync(stateRoot, { recursive: true, force: true });
  }
}

// jsonl readAll scale ladder: L lines; decompose readFile-equivalent vs
// split+JSON.parse by timing a parse-only replica on the raw string.
async function cellReadLadder(): Promise<void> {
  const { readFile } = await import("node:fs/promises");
  for (const L of [1000, 10000, 100000]) {
    const dir = mkdtempSync(join(tmpdir(), "r14j-rl-"));

    for (let i = 0; i < L; i++) {

    }
    writeFileSync(file, lines.join("\n") + "\n", "utf8");
    const iters = L >= 100000 ? 5 : 20;
    let count = 0;
    let t0 = hr();
    for (let i = 0; i < iters; i++) {
      const { values } = await readJsonlObjects(file, (line) => new Error(`bad ${line}`));
      count = values.length;
    }
    let t1 = hr();
    const totalUs = usPer(t0, t1, iters);
    if (count !== L) throw new Error(`expected ${L} values, got ${count}`);
    t0 = hr();
    let raw = "";
    for (let i = 0; i < iters; i++) raw = await readFile(file, "utf8");
    t1 = hr();
    const readUs = usPer(t0, t1, iters);
    t0 = hr();
    let parsed = 0;
    for (let i = 0; i < iters; i++) {
      const segments = raw.split("\n");
      for (const segment of segments) {
        if (segment === "") continue;
        parsed += (JSON.parse(segment) as { n?: number } & object) !== null ? 1 : 0;
      }
    }
    t1 = hr();
    const parseUs = usPer(t0, t1, iters);
    console.log(
      `jsonl readAll L=${L}: total=${ms(totalUs)}ms/read (readFile=${ms(readUs)}ms split+parse=${ms(parseUs)}ms) [parsed=${parsed}]`
    );
    rmSync(dir, { recursive: true, force: true });
  }
}

// J1 fold CPU crossing: evaluatePreferenceLoop at N observations (zero

function cellLoopEvalLadder(): void {
  for (const N of [6000, 60000, 240000]) {
    const observations: PreferenceObservation[] = [];

    for (let i = 0; i < N; i++) {

        explicit: i % 5 === 0,
        recurrenceCount: 1
      });
    }
    const iters = N >= 240000 ? 3 : 10;
    let sink = 0;
    // warm
    sink += evaluatePreferenceLoop(observations, tombstoned).forgettingEvents;

    }
    const t1 = hr();
    console.log(`evaluatePreferenceLoop N=${N}: ${ms(usPer(t0, t1, iters))}ms/eval [sink=${sink}]`);
  }
}
