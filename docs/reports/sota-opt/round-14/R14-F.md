- 合计 **2 次 deopt 事件**：`normalizeString`（node/tsx 内部，切片外）1 次；`restoreShadowState` 1 次——**预热期一次性 eager deopt**（"insufficient type feedback for generic named access"，首次优化过早的标准分层行为），重优化后**零复发**。

```ts
/**
 * R14-F new-axis measurements (temporary; NOT candidate adjudications and NOT
 * committed — no winner this round). Three axes no earlier F round measured
 * directly:
 *
 *   axis 1  GC-time fraction of the full-experiment anchor. Every
 *           allocation-elimination candidate class in this slice is already
 *           individually excluded (defensive copies = hard invariant, append
 *           copies = X4-2/S1-F-3, pending-Set = the S6-F-1 mechanism itself,
 *           spreads = pool D <=0.2 ms, scratch-Set reuse = S9-F-3 stable
 *           negative). This measures the single global ceiling those
 *           exclusions share: total GC time attributable to the anchor run,
 *           via PerformanceObserver('gc'). Even a hypothetical zero-allocation
 *           rewrite could not recover more than this.
 *
 *   axis 2  string-provenance cell (R7-I fixture blind spot). The historical
 *           anchor fixture feeds outcome hashes that are pointer-equal to the
 *           population entries, so the S7-F-1 aligned compare and the
 *           assign-site Array.includes scan resolve mostly by pointer
 *           equality. The production caller (evaluateProposalShadow) receives
 *           outcome hashes from its own caller — typically parsed from
 *           disk/events, i.e. pointer-DISTINCT but content-equal strings.
 *           This cell re-prices the anchor with JSON-round-tripped caller
 *           hashes (population objects unchanged — the plan is shared by
 *           reference in production) and checks the final states are
 *           field-identical. Component views isolate where the delta lives.
 *
 *   axis 3  (run separately in the shell) deopt audit: node --trace-deopt
 *           over this same script, grepping for src/experiments frames.
 *
 * Uses the production modules directly. Run: npx tsx tmp-r14f-axes.ts
 */

import { PerformanceObserver } from "node:perf_hooks";
import { createCandidateId, createResourceVersionId } from "./src/domain/ids.js";
import { validateExperimentPlan, type ExperimentPlan } from "./src/experiments/plan.js";
import { createShadowRunner, type ExperimentOutcome, type ShadowState } from "./src/experiments/shadow.js";

function timeMs(fn: () => void, rounds = 7): number {
  let best = Infinity;
  for (let r = 0; r < rounds; r++) {
    const t0 = process.hrtime.bigint();
    fn();
    const t1 = process.hrtime.bigint();
    best = Math.min(best, Number(t1 - t0) / 1e6);
  }
  return best;
}

function makePopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

function makeShadowPlan(population: readonly string[], maxAssignments: number): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: "exp_r14f_bench",
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r14fbase"),
    candidateId: createCandidateId(() => "r14fcand"),
    population,
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments, maxWallClockMs: 1e12 },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

const P = 2000;
const A = 1000;
const population = makePopulation(P);
const plan = makeShadowPlan(population, A);
validateExperimentPlan(plan);

/** Interned caller hashes: pointer-equal to population entries (historic fixture). */
const internedHashes: string[] = population.slice(0, A);
/** De-interned caller hashes: content-equal, pointer-distinct (production provenance). */
const deinternedHashes: string[] = JSON.parse(JSON.stringify(internedHashes)) as string[];
for (let i = 0; i < A; i++) {
  if (deinternedHashes[i] !== internedHashes[i]) throw new Error("round-trip mismatch");
}

function fullExperiment(hashes: readonly string[]): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (let k = 0; k < A; k++) {
    const h = hashes[k] as string;
    state = runner.assign(state, h, 0);
    const outcome: ExperimentOutcome = {
      episodeHash: h,
      utility: 0.5,
      costUsd: 0.01,
      guardrailBreached: false,
    };
    state = runner.recordOutcome(state, outcome, 0);
  }
  if (state.assignments.length !== A) throw new Error("unexpected assignment count");
  return state;
}

/* determinism check: interned vs de-interned final states field-identical */
const sInterned = fullExperiment(internedHashes);
const sDeinterned = fullExperiment(deinternedHashes);
if (
  sInterned.halted !== sDeinterned.halted ||
  sInterned.haltReason !== sDeinterned.haltReason ||
  sInterned.guardrailBreaches !== sDeinterned.guardrailBreaches ||
  sInterned.assignments.length !== sDeinterned.assignments.length ||
  sInterned.outcomes.length !== sDeinterned.outcomes.length
) {
  throw new Error("state divergence between provenance variants");
}
for (let i = 0; i < A; i++) {
  const a = sInterned.assignments[i];
  const b = sDeinterned.assignments[i];
  if (a?.episodeHash !== b?.episodeHash || a?.shadowDecision !== b?.shadowDecision) {
    throw new Error(`assignment divergence at ${i}`);
  }
}
process.stdout.write("provenance determinism: interned and de-interned runs field-identical\n");

/* ------------------------------------------------------------------ */
/* axis 1 — GC-time fraction of the anchor (interned fixture)          */
/* ------------------------------------------------------------------ */
let gcMs = 0;
let gcCount = 0;
const obs = new PerformanceObserver((list) => {
  for (const entry of list.getEntries()) {
    gcMs += entry.duration;
    gcCount += 1;
  }
});
obs.observe({ entryTypes: ["gc"] });

const GC_ROUNDS = 10;
// warm
fullExperiment(internedHashes);
await new Promise((resolve) => setTimeout(resolve, 50));
gcMs = 0;
gcCount = 0;
const t0 = process.hrtime.bigint();
for (let r = 0; r < GC_ROUNDS; r++) {
  fullExperiment(internedHashes);
}
const t1 = process.hrtime.bigint();
// let trailing GC entries flush
await new Promise((resolve) => setTimeout(resolve, 100));
obs.disconnect();
const wallMs = Number(t1 - t0) / 1e6;
process.stdout.write(
  `axis 1 (GC fraction): ${GC_ROUNDS} anchor runs wall=${wallMs.toFixed(2)}ms ` +
    `gc=${gcMs.toFixed(2)}ms across ${gcCount} pauses ` +
    `(${((gcMs / wallMs) * 100).toFixed(1)}% of wall; ` +
    `${(gcMs / GC_ROUNDS).toFixed(2)}ms per experiment)\n`
);

/* ------------------------------------------------------------------ */
/* axis 2 — string-provenance cell: anchor + component views           */
/* ------------------------------------------------------------------ */
const tInterned = timeMs(() => {
  fullExperiment(internedHashes);
}, 5);
const tDeinterned = timeMs(() => {
  fullExperiment(deinternedHashes);
}, 5);
process.stdout.write(
  `axis 2 (anchor): interned=${tInterned.toFixed(2)}ms de-interned=${tDeinterned.toFixed(2)}ms ` +
    `delta=${(tDeinterned - tInterned).toFixed(2)}ms\n`
);

/* component view A: assign-site includes scan (requirePopulationEpisode shape) */
let includesSink = 0;
function includesScan(hashes: readonly string[]): void {
  for (let k = 0; k < A; k++) {
    if (population.includes(hashes[k] as string)) includesSink += 1;
  }
}
const tIncInterned = timeMs(() => includesScan(internedHashes), 5);
const tIncDeinterned = timeMs(() => includesScan(deinternedHashes), 5);

);

/* component view C: validate is provenance-invariant (population objects shared) */
const tValidate = timeMs(() => {
```
