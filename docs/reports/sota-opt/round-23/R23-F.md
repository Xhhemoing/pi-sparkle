function median(xs: readonly number[]): number {

/** Production interleave shape (evaluateProposalShadow), clock pinned to 0. */
function runInterleaved(plan: ExperimentPlan, stream: readonly ExperimentOutcome[]): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (const outcome of stream) {
    if (state.halted) {
      break;
    }
    state = runner.assign(state, outcome.episodeHash, 0);
    state = runner.recordOutcome(state, outcome, 0);
  }
  return state;
}

function ledgerDigest(state: ShadowState): string {
  return stableStringify({
    assignments: state.assignments,
    outcomes: state.outcomes,
    guardrailBreaches: state.guardrailBreaches,
    halted: state.halted,
    haltReason: state.haltReason,
  });
}

/* ------------------- least-squares surface identification ----------------- */
// Basis: x1 = (2A+1)·P (validate obligation count × population size),
// x2 = A², x3 = A, x4 = 1. Solve the 4×4 normal equations by Gaussian
// elimination with partial pivoting (measurement-side arithmetic only).
function solve4(matrix: number[][], rhs: number[]): number[] {
  const n = 4;
  const m = matrix.map((row, i) => [...row, rhs[i] as number]);
  for (let col = 0; col < n; col++) {
    let pivot = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs((m[row] as number[])[col] as number) > Math.abs((m[pivot] as number[])[col] as number)) {
        pivot = row;
      }
    }
    const tmp = m[col] as number[];
    m[col] = m[pivot] as number[];
    m[pivot] = tmp;
    const lead = (m[col] as number[])[col] as number;
    if (lead === 0) throw new Error("singular normal equations");
    for (let row = col + 1; row < n; row++) {
      const factor = ((m[row] as number[])[col] as number) / lead;
      for (let k = col; k <= n; k++) {
        (m[row] as number[])[k] = ((m[row] as number[])[k] as number) - factor * ((m[col] as number[])[k] as number);
      }
    }
  }
  const theta = new Array<number>(n).fill(0);
  for (let row = n - 1; row >= 0; row--) {
    let sum = (m[row] as number[])[n] as number;
    for (let k = row + 1; k < n; k++) {
      sum -= ((m[row] as number[])[k] as number) * (theta[k] as number);
    }
    theta[row] = sum / ((m[row] as number[])[row] as number);
  }
  return theta;
}

function basisOf(A: number, P: number): number[] {
  return [(2 * A + 1) * P, A * A, A, 1];
}

function fitCells(cells: readonly Arm[]): number[] {
  const xtx: number[][] = Array.from({ length: 4 }, () => [0, 0, 0, 0]);
  const xty: number[] = [0, 0, 0, 0];
  for (const cell of cells) {
    const x = basisOf(cell.A, cell.P);
    const y = medians.get(cell.name) as number;
    for (let i = 0; i < 4; i++) {
      xty[i] = (xty[i] as number) + (x[i] as number) * y;
      for (let j = 0; j < 4; j++) {
        (xtx[i] as number[])[j] = ((xtx[i] as number[])[j] as number) + (x[i] as number) * (x[j] as number);
      }
    }
  }
  return solve4(xtx, xty);
}

function predict(theta: readonly number[], A: number, P: number): number {
  const x = basisOf(A, P);
  return x.reduce((sum, v, i) => sum + v * (theta[i] as number), 0);
}
