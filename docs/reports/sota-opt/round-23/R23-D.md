save/load/恢复语义）都没有实测锚。一个未来接上两相 API 的调用方
（例如批量提案审查队列）会在第一天就踩进这两个未定价面。本轴分
四段关死：(2a) 理论——台账拷贝总量的调度不变性证明，使调度差分
成为 pending 扫描项的精确隔离器；(2b) 调度对格网 + 深度直测斜率；
(2c) 风暴中程持久化剂量 + 恢复见证；(2d) 同身份风暴 fail-closed
见证与 10 ms 越线求解。判读标尺沿用历轮验收线（数十–数百 ms
稳态；µs/ns 与 once-per-run 是拒绝类）。

### 2a. 理论：台账拷贝总量是调度不变量

begin 与 commit 各返回一次 `ledger()` 全量拷贝（S3-D-1 属地）。
交织调度第 i 对的台账长度为 2i（begin）与 2i+1（commit），总拷贝
Σ(2i)+Σ(2i+1) ≈ 2N²；风暴调度 begin_i 见长度 i、commit_i 见长度
N+i，总拷贝 Σ(i)+Σ(N+i) ≈ 2N²——**逐项相消**。版本 id 分配次序
（只在 begin 相分配）与时间戳（固定 NOW）在两调度下逐位相同。
因此风暴−交织残差**只**含 pending 面成本：preparePromotion 的
`Array.from(pendingByIntent.values())` 分配 + 线性 candidateId
比较 + Map 尺寸效应。

function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** Recursively sort object keys so JSON compare ignores key order. */
function canonicalJson(value: unknown): string {
  return JSON.stringify(value, (_key, val: unknown) => {
    if (val !== null && typeof val === "object" && !Array.isArray(val)) {
      const sorted: Record<string, unknown> = {};
      for (const k of Object.keys(val as Record<string, unknown>).sort()) {
        sorted[k] = (val as Record<string, unknown>)[k];
      }
      return sorted;
    }
    return val;
  });
}

interface FleetRow {
  readonly identity: ResourceIdentity;
  readonly v1: ResourceVersion;
  readonly candidate: ImprovementCandidate;
  readonly input: PromoteInput;

interface Fleet {
  readonly registry: ResourceRegistry;
  readonly rows: readonly FleetRow[];
}

/** N distinct prompt identities, each with a baseline and one candidate.
 * All PromoteInput objects are prebuilt so timed regions contain only
 * registry calls. Same tag => identical deterministic id streams. */
function buildPromptFleet(tag: string, n: number): Fleet {
  const registry = new ResourceRegistry({ now: () => NOW, generateId: sequentialIds(tag) });
  const rows: FleetRow[] = [];
  for (let i = 0; i < n; i += 1) {
    const identity: ResourceIdentity = {
      kind: "prompt",
      name: `fleet-prompt-${i}`,
      scope: { kind: "project", projectId: createProjectId(() => `${tag}proj`) }
    };
    const body = `# fleet prompt ${i}\nKeep the diff bounded to the stated scope and cite evidence for every claim (row ${i}).\n`;
    const v1 = registry.registerBaseline({ identity, content: body, author: HUMAN });
    const content = `${body}\n## amendment\nAdd an explicit check for the attributable failure (row ${i}).\n`;
    const candidate = registry.createCandidate({
      identity,
      content,
      parentVersionId: v1.versionId,
      author: HUMAN,
      evaluationPlan: EVAL_PLAN
    });
    const input: PromoteInput = {
      candidateId: candidate.candidateId,
      expectedCurrentVersionId: v1.versionId,
      content,
      approvedBy: HUMAN,
      review: mkReview(candidate.candidateId, candidate.contentHash, `${tag}${i}`),
      changeNote: mkNote(`fleet:${tag}${i}`, v1.versionId),
      explicitApproval: true
    };
    rows.push({ identity, v1, candidate, input });
  }
  return { registry, rows };
}

/** Production-equivalent trajectory: begin;commit pairs, P <= 1 throughout. */
function runInterleaved(fleet: Fleet): number {
  const t0 = performance.now();
  for (const row of fleet.rows) {
    const began = fleet.registry.beginPromotion(row.input);
    fleet.registry.commitPromotion(began.intentId);
  }
  return performance.now() - t0;
}

/** Begin storm then commit drain: P grows 0 -> N -> 0. Optionally records
 * bucketed begin timings for the per-begin-at-depth slope. */
function runStorm(
  fleet: Fleet,
  bucketSize?: number
): { totalMs: number; beginMs: number; commitMs: number; buckets: { midP: number; usPerBegin: number }[] } {
  const intentIds: string[] = [];
  const buckets: { midP: number; usPerBegin: number }[] = [];
  const t0 = performance.now();
  if (bucketSize === undefined) {
    for (const row of fleet.rows) {
      intentIds.push(fleet.registry.beginPromotion(row.input).intentId);
    }
  } else {
    for (let start = 0; start < fleet.rows.length; start += bucketSize) {
      const end = Math.min(start + bucketSize, fleet.rows.length);
      const b0 = performance.now();
      for (let i = start; i < end; i += 1) {
        intentIds.push(fleet.registry.beginPromotion((fleet.rows[i] as FleetRow).input).intentId);
      }
      const usPerBegin = ((performance.now() - b0) / (end - start)) * 1e3;
      buckets.push({ midP: (start + end - 1) / 2, usPerBegin });
    }
  }
  const t1 = performance.now();
  for (const intentId of intentIds) {
    fleet.registry.commitPromotion(intentId);
  }
  const t2 = performance.now();
  return { totalMs: t2 - t0, beginMs: t1 - t0, commitMs: t2 - t1, buckets };
}

function leastSquares(xs: readonly number[], ys: readonly number[]): { slope: number; intercept: number } {
  const n = xs.length;
  const mx = xs.reduce((a, b) => a + b, 0) / n;
  const my = ys.reduce((a, b) => a + b, 0) / n;
  let num = 0;
  let den = 0;
  for (let i = 0; i < n; i += 1) {
    num += ((xs[i] as number) - mx) * ((ys[i] as number) - my);
    den += ((xs[i] as number) - mx) ** 2;
  }
  const slope = num / den;
  return { slope, intercept: my - slope * mx };
}
