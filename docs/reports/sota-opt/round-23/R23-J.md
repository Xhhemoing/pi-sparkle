```ts
/**
 * R23-J new-axis harness: CALLB — call-boundary schedule face.
 *
 * Question: at a FIXED record multiset (fixed N, byte totals, class mix,
 * record order, aggregation identities, and object shapes), how does total
 * session cost respond when only the CALL-BOUNDARY STRUCTURE of the schedule
 * changes — (i) granularity: how many API calls deliver/consume the multiset,
 * and (ii) placement: where consuming calls sit relative to state growth?
 *
 * Every previous J fixture pinned the call schedule to one phase structure
 * (mailbox: all-enqueue-then-one-claim; preferences: per-write cost at pinned
 * store size or a single hydration; cascade: one call; host: register-all-
 * then-operate). R19-J permuted record order WITHIN pinned call boundaries;
 * this axis pins record order and moves the boundaries. On stateful surfaces
 * total session cost is a trajectory integral sum(f(state_t)), which the
 * schedule moves even at a fixed multiset — unlike R23-B's stateless batch
 * arrangement (flat by construction), non-flat responses are EXPECTED here;
 * the question is whether any lawful candidate can harvest them.
 *
 *  P1  mailbox claim-granularity dose D in {1,10,60,600} (M=600, one role,
 *      one claimer, zero self-mail, FIFO order pinned)
 *  P2  preferences write-schedule trajectory: HYDRATE (1 load) vs
 *      INCREMENTAL (N recordObservation calls) vs MIXED (hydrate N-g,
 *      record g), N=500 — the trajectory-integral cell
 *  P3  deletion-cascade call granularity: total hits H=40 fixed, spread over
 *      E in {1,2,4,8} target episodes = E sequential per-episode calls
 *  P4  cluster host register x send placement: P=300 registrations, S=300
 *      single-target role-cast sends placed EARLY / UNIFORM / LATE in the
 *      registration sequence (directory-scan trajectory)
 *
 * Fidelity gates (asserted inline):
 *  - semantics: delivered-id sequences / final store observations + views /
 *    cascaded-id totals + final file bytes / inbox contents bitwise equal or
 *    exactly conserved across all schedule doses (the schedule must be
 *    semantically invisible in the final state)
 *  - byte account: payload chars / line lengths / final log bytes equal
 *    across doses
 *  - class-mix, identity fan-out, record order, and object shapes pinned by
 *    construction (R20-J / R21-J / R19-J / R22-J dimensions held constant)
 */
import { mkdtempSync, mkdirSync, rmSync, statSync, readFileSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join, dirname } from "node:path";

import { createMailbox, type ClusterMail } from "./src/cluster/mailbox.js";
import { createClusterHost } from "./src/cluster/host.js";
import { createAgentProfileRegistry, defaultAgentProfiles } from "./src/agents/registry.js";
import { createTaskId } from "./src/domain/ids.js";
import {
  configurePreferencePersistence,
  recordObservation,
  listObservations,
  getView,
  resetPreferenceStore
} from "./src/preferences/store.js";
import type { PreferenceObservation } from "./src/preferences/types.js";
import { cascadeFeedbackTombstones } from "./src/privacy/deletion.js";
import { feedbackLogPath, feedbackTombstonesPath } from "./src/feedback/store.js";
import type { EpisodeId, AgentInstanceId, MessageId } from "./src/domain/ids.js";
import type { AgentRole } from "./src/domain/roles.js";
import type { IsoTimestamp } from "./src/domain/timestamp.js";

const hr = (): bigint => process.hrtime.bigint();
const us = (a: bigint, b: bigint, n: number): number => Number(b - a) / 1000 / n;
const fmt = (x: number): string => x.toFixed(x < 10 ? 2 : 1);
const pad = (n: number, w: number): string => n.toString().padStart(w, "0");

function assertEqual(label: string, a: unknown, b: unknown): void {
  const sa = JSON.stringify(a);
  const sb = JSON.stringify(b);
  if (sa !== sb) throw new Error(`FIDELITY FAIL ${label}: ${sa?.slice(0, 200)} !== ${sb?.slice(0, 200)}`);
}

/* ========== P1: mailbox claim-granularity dose (M=600, D claims) ========== */
/* One role, one claimer, zero self-mail, enqueue order pinned. Dose D =
 * number of claimRole calls: one claim after every 600/D enqueues. The mail
 * multiset, payload chars, identities, and FIFO order are D-invariant; only
 * the call-boundary count moves. */
{
  const M = 600;
  const DOSES = [1, 10, 60, 600];
  const REPEATS = 300;
  const role = "worker" as AgentRole;
  const claimer = "agt_claimer0000" as AgentInstanceId;

  const mails: ClusterMail[] = [];
  let chars = 0;
  for (let i = 0; i < M; i++) {
    const mail: ClusterMail = {
      id: `msg_${pad(i, 6)}` as MessageId,
      from: "agt_sender00000" as AgentInstanceId,
      addressRole: role,
      body: `payload ${pad(i, 6)}`,
      occurredAt: "2026-08-25T00:00:00.000Z" as IsoTimestamp
    };
    chars += mail.body.length;
    mails.push(mail);
  }

  // Untimed warm-up over every dose so all claim/enqueue call sites reach a
  // common JIT/IC state before any dose is measured.
  for (const D of DOSES) {
    const g = M / D;
    for (let r = 0; r < 50; r++) {
      const mb = createMailbox();
      let delivered = 0;
      for (let i = 0; i < M; i++) {
        mb.enqueue(mails[i]!);
        if ((i + 1) % g === 0) delivered += mb.claimRole(role, claimer).length;
      }
      assertEqual(`P1 warmup D=${D}`, delivered, M);
    }
  }

  let idSeqRef: string | undefined;
  for (const D of DOSES) {
    const g = M / D;
    let best = Infinity;
    let sum = 0;
    let idSeq = "";
    for (let r = 0; r < REPEATS; r++) {
      const mb = createMailbox();
      const deliveredIds: string[] = [];
      const t0 = hr();
      for (let i = 0; i < M; i++) {
        mb.enqueue(mails[i]!);
        if ((i + 1) % g === 0) {
          for (const mail of mb.claimRole(role, claimer)) deliveredIds.push(mail.id);
        }
      }
      const t1 = hr();
      assertEqual(`P1 D=${D} delivery conservation`, deliveredIds.length, M);
      assertEqual(`P1 D=${D} final inbox`, mb.inbox(claimer).length, M);
      idSeq = deliveredIds.join(",");
      const dt = us(t0, t1, 1);
      best = Math.min(best, dt);
      sum += dt;
    }
    if (idSeqRef === undefined) idSeqRef = idSeq;
    if (idSeq !== idSeqRef) throw new Error(`FIDELITY FAIL P1 D=${D}: delivered id sequence diverged`);
    console.log(
      `P1 mailbox claim-granularity D=${pad(D, 3)} (M=${M}): min=${fmt(best)} mean=${fmt(sum / REPEATS)} us/cycle`
    );
  }
  console.log(`P1 fidelity: payload chars ${chars}, exactly-once delivery, and delivered id order equal across D doses`);
}

/* ===== P2: preferences write-schedule trajectory (HYDRATE/MIXED/INCR) ===== */
/* Fixed multiset of N=500 observations (S=20 subjects, 30% explicit, 3-value
 * pool, monotonic createdAt, fixed-width fields, recurrenceCount carried =
 * what applyObservation computes). Schedules differ ONLY in how many write
 * calls deliver it: HYDRATE = 1 configure/load call (1 rebuild, 0 saves);
 * MIXED g = hydrate N-g then g recordObservation calls; INCREMENTAL = N
 * recordObservation calls (each pays filter O(i) + rebuild O(i) + stringify
 * O(i) + write at store size i — the trajectory integral). */
{
  const N = 500;
  const S = 20;

  const counts = new Map<string, number>();
  const multiset: PreferenceObservation[] = [];
  for (let i = 0; i < N; i++) {
    const explicit = i % 10 < 3;
    const scopeKey = `sk${pad(i % S, 2)}`;
    const ck = `${scopeKey}|${explicit}`;
    const rc = (counts.get(ck) ?? 0) + 1;
    counts.set(ck, rc);
    multiset.push({
      id: `evd_${pad(i, 8)}`,
      scope: "project",
      scopeKey,
      key: "k0",
      value: `v${i % 3}`,
      evidenceEpisodeId: "ep_fixed0000000" as EpisodeId,
      weight: 1,
      createdAt: `2026-08-25T00:00:${pad(Math.floor(i / 100), 2)}.${pad(i % 100, 3)}Z` as IsoTimestamp,
      explicit,
      recurrenceCount: rc
    });
  }
  const probes: readonly string[] = ["sk00", "sk07", "sk19"];

  const finalObs = (): string => JSON.stringify(listObservations());
  const finalViews = (): string =>
    JSON.stringify(
      probes.map((sk) => {
        const view = getView("project", sk);
        return view === undefined
          ? null
          : { aggregates: view.aggregates, confidence: view.confidence, sourceCount: view.sourceCount };
      })
    );

  function runSchedule(kind: "HYDRATE" | "MIXED" | "INCREMENTAL", g: number): {
    ms: number;
    obs: string;
    views: string;
  } {
    const root = mkdtempSync(join(tmpdir(), "r23j-callb-p2-"));
    const file = join(root, "preferences.json");
    const seedCount = N - g;
    resetPreferenceStore();
    configurePreferencePersistence(undefined);
    if (seedCount > 0) {
      writeFileSync(file, JSON.stringify({ observations: multiset.slice(0, seedCount), tombstones: [] }), "utf8");
    }
    const t0 = hr();
    configurePreferencePersistence(file);
    for (let i = seedCount; i < N; i++) recordObservation(multiset[i]!);
    const t1 = hr();
    const out = { ms: us(t0, t1, 1) / 1000, obs: finalObs(), views: finalViews() };
    resetPreferenceStore();
    configurePreferencePersistence(undefined);
    rmSync(root, { recursive: true, force: true });
    void kind;
    return out;
  }

  const SCHEDULES: readonly { kind: "HYDRATE" | "MIXED" | "INCREMENTAL"; g: number; reps: number }[] = [
    { kind: "HYDRATE", g: 0, reps: 30 },
    { kind: "MIXED", g: 50, reps: 10 },
    { kind: "MIXED", g: 250, reps: 5 },
    { kind: "INCREMENTAL", g: N, reps: 5 }
  ];
  let obsRef: string | undefined;
  let viewsRef: string | undefined;
  for (const sched of SCHEDULES) {
    let best = Infinity;
    let sum = 0;
    let obs = "";
    let views = "";
    for (let r = 0; r < sched.reps; r++) {
      const out = runSchedule(sched.kind, sched.g);
      obs = out.obs;
      views = out.views;
      best = Math.min(best, out.ms);
      sum += out.ms;
    }
    if (obsRef === undefined) {
      obsRef = obs;
      viewsRef = views;
    }
    if (obs !== obsRef) throw new Error(`FIDELITY FAIL P2 ${sched.kind} g=${sched.g}: final observations diverged`);
    if (views !== viewsRef) throw new Error(`FIDELITY FAIL P2 ${sched.kind} g=${sched.g}: final views diverged`);
    console.log(
      `P2 store write-schedule ${sched.kind.padEnd(11)} g=${pad(sched.g, 3)} (N=${N}, calls=${sched.g === 0 ? 1 : sched.g}): ` +
        `min=${fmt(best)} mean=${fmt(sum / sched.reps)} ms/session`
    );
  }
  // Per-write trajectory samples inside one incremental session (linear-in-i
  // per-write growth = the quadratic-integral signature).
  {
    const root = mkdtempSync(join(tmpdir(), "r23j-callb-p2t-"));
    resetPreferenceStore();
    configurePreferencePersistence(join(root, "preferences.json"));
    const milestones = new Set([1, 100, 250, 500]);
    const samples: string[] = [];
    for (let i = 0; i < N; i++) {
      const t0 = hr();
      recordObservation(multiset[i]!);
      const t1 = hr();
      if (milestones.has(i + 1)) samples.push(`i=${i + 1}: ${fmt(us(t0, t1, 1))}us`);
    }
    console.log(`P2 per-write trajectory samples (one incremental session): ${samples.join("  ")}`);
    resetPreferenceStore();
    configurePreferencePersistence(undefined);
    rmSync(root, { recursive: true, force: true });
  }
  console.log(`P2 fidelity: final observations (${N}) and probe views bitwise equal across all schedules`);
}

/* ==== P3: deletion-cascade call granularity (H=40 hits over E targets) ==== */
/* N=200 records, total hits H=40 pinned at positions i%5===0, spread over E
 * target episodes (hit j -> target j%E). The API is per-episode, so deleting
 * the E targets costs E sequential whole-log read+rewrite cycles. Line
 * lengths, total bytes, and the cascaded-id total are E-invariant. */
{
  const N = 200;
  const H = 40;
  const DOSES = [1, 2, 4, 8];
  const ITERS = 20;

  const targetOf = (t: number): EpisodeId => `ep_tg${t}${"0".repeat(9)}` as EpisodeId;

  function lineOf(i: number, E: number): string {
    const hit = i % 5 === 0;
    const hitIndex = Math.floor(i / 5);
    const episodeId = hit ? targetOf(hitIndex % E) : `ep_bg${pad(i % 7, 10)}`;
    return JSON.stringify({
      id: `fb_${pad(i, 6)}`,
      episodeId,
      kind: "human",
      rubricVersion: "r1",
      score: 50 + (i % 40),
      evidenceRefs: [],
      redacted: false,
      createdAt: "2026-08-25T00:00:00.000Z",
      body: `free text feedback payload number ${pad(i, 4)} `.repeat(3),
      summary: `summary ${pad(i, 4)}`
    });
  }

  function seed(stateRoot: string, E: number): { bytes: number; lineLens: string } {
    const log = feedbackLogPath(stateRoot);
    mkdirSync(dirname(log), { recursive: true });
    const lines: string[] = [];
    const lens: number[] = [];
    for (let i = 0; i < N; i++) {
      const line = lineOf(i, E);
      lines.push(line);
      lens.push(line.length);
    }
    writeFileSync(log, lines.join("\n") + "\n", "utf8");
    writeFileSync(
      feedbackTombstonesPath(stateRoot),
      JSON.stringify(["fb-pre-a", "fb-pre-b", "fb-pre-c", "fb-pre-d", "fb-pre-e"], null, 2) + "\n",
      "utf8"
    );
    return { bytes: statSync(log).size, lineLens: lens.join(",") };
  }

  let lineLensRef: string | undefined;
  let finalBytesRef: number | undefined;
  let sidecarRef: string | undefined;
  for (const E of DOSES) {
    const stateRoot = mkdtempSync(join(tmpdir(), "r23j-callb-p3-"));
    const { lineLens } = seed(stateRoot, E);
    if (lineLensRef === undefined) lineLensRef = lineLens;
    if (lineLens !== lineLensRef) throw new Error(`FIDELITY FAIL P3 E=${E}: per-line lengths diverged`);
    await cascadeFeedbackTombstones(stateRoot, "ep_warm000000000" as EpisodeId); // warm no-match
    let best = Infinity;
    let sum = 0;
    for (let it = 0; it < ITERS; it++) {
      seed(stateRoot, E); // untimed re-seed (also resets the sidecar)
      let cascadedTotal = 0;
      const t0 = hr();
      for (let t = 0; t < E; t++) {
        const cascaded = await cascadeFeedbackTombstones(stateRoot, targetOf(t));
        cascadedTotal += cascaded.length;
      }
      const t1 = hr();
      assertEqual(`P3 E=${E} cascaded total`, cascadedTotal, H);
      const dt = us(t0, t1, 1);
      best = Math.min(best, dt);
      sum += dt;
    }
    const finalBytes = statSync(feedbackLogPath(stateRoot)).size;
    const sidecar = JSON.stringify(
      (JSON.parse(readFileSync(feedbackTombstonesPath(stateRoot), "utf8")) as string[]).map((id) =>
        id.startsWith("fb_") ? id : id
      ).length
    );
    if (finalBytesRef === undefined) finalBytesRef = finalBytes;
    if (sidecarRef === undefined) sidecarRef = sidecar;
    assertEqual(`P3 E=${E} final log bytes`, finalBytes, finalBytesRef);
    assertEqual(`P3 E=${E} sidecar id count`, sidecar, sidecarRef);
    rmSync(stateRoot, { recursive: true, force: true });
    console.log(
      `P3 cascade call-granularity E=${E} (N=${N}, hits=${H}, ${H / E}/call): ` +
        `min=${fmt(best)} mean=${fmt(sum / ITERS)} us/delete-set  per-call min=${fmt(best / E)} us`
    );
  }
  console.log(
    `P3 fidelity: per-line lengths, final log bytes ${finalBytesRef} B, sidecar id count ${sidecarRef}, and cascaded total ${H} equal across E doses`
  );
}

/* ===== P4: host register x send placement (directory-scan trajectory) ===== */
/* P=300 registrations (1 reviewer target first, then 299 worker fillers),
 * S=300 role-cast sends to "reviewer" from an unregistered sender — each
 * send delivers exactly one mail but scans the whole directory. Placements
 * move the sends relative to directory growth: EARLY (all sends at P_t=1),
 * UNIFORM (interleaved), LATE (all sends at P_t=300). Delivered multiset and
 * id sequence are placement-invariant (deterministic id generator). */
{
  const FILLERS = 299;
  const SENDS = 300;
  const REPEATS = 200;
  const registry = createAgentProfileRegistry(defaultAgentProfiles());
  const target = "agt_target00000" as AgentInstanceId;
  const sender = "agt_sender00000" as AgentInstanceId;

  function makeHost(): ReturnType<typeof createClusterHost> {
    let n = 0;
    return createClusterHost({
      registry,
      generateId: () => `c${pad(n++, 10)}`,
      onSpawn: () => undefined
    });
  }

  type Placement = "EARLY" | "UNIFORM" | "LATE";
  const send = (host: ReturnType<typeof createClusterHost>): string =>
    host.send({ from: sender, body: `ping ${pad(0, 4)}`, addressRole: "reviewer" as AgentRole }).id;

  function runPlacement(placement: Placement): { dt: number; ids: string; inboxLen: number } {
    const host = makeHost();
    const ids: string[] = [];
    const t0 = hr();
    host.register(target, "reviewer" as AgentRole, createTaskId(() => `t${pad(0, 10)}`));
    if (placement === "EARLY") {
      for (let s = 0; s < SENDS; s++) ids.push(send(host));
      for (let f = 0; f < FILLERS; f++)
        host.register(`agt_fill${pad(f, 6)}` as AgentInstanceId, "worker" as AgentRole, createTaskId(() => `t${pad(f + 1, 10)}`));
    } else if (placement === "LATE") {
      for (let f = 0; f < FILLERS; f++)
        host.register(`agt_fill${pad(f, 6)}` as AgentInstanceId, "worker" as AgentRole, createTaskId(() => `t${pad(f + 1, 10)}`));
      for (let s = 0; s < SENDS; s++) ids.push(send(host));
    } else {
      for (let f = 0; f < FILLERS; f++) {
        host.register(`agt_fill${pad(f, 6)}` as AgentInstanceId, "worker" as AgentRole, createTaskId(() => `t${pad(f + 1, 10)}`));
        ids.push(send(host));
      }
      ids.push(send(host));
    }
    const t1 = hr();
    return { dt: us(t0, t1, 1), ids: ids.join(","), inboxLen: host.inbox(target).length };
  }

  let idsRef: string | undefined;
  for (const placement of ["EARLY", "UNIFORM", "LATE"] as const) {
    let best = Infinity;
    let sum = 0;
    let ids = "";
    for (let r = 0; r < REPEATS; r++) {
      const out = runPlacement(placement);
      assertEqual(`P4 ${placement} inbox`, out.inboxLen, SENDS);
      ids = out.ids;
      best = Math.min(best, out.dt);
      sum += out.dt;
    }
    if (idsRef === undefined) idsRef = ids;
    if (ids !== idsRef) throw new Error(`FIDELITY FAIL P4 ${placement}: delivered id sequence diverged`);
    console.log(
      `P4 host send-placement ${placement.padEnd(7)} (P=${FILLERS + 1}, S=${SENDS}): min=${fmt(best)} mean=${fmt(sum / REPEATS)} us/schedule`
    );
  }
  console.log(`P4 fidelity: delivered id sequence (deterministic gen) and inbox count ${SENDS} equal across placements`);
}
```
