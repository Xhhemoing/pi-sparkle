### A.4 进程内分解（/tmp/r23i-bench/inproc.mjs，双引擎各执行一次）

```javascript
// R23-I in-process decomposition of the .pi observability-tree plane.
// Run-family arm (E/J machines, replicating ingestSubagentDirectory's exact
// stage sequence — the function itself is module-private in auto-loop.ts):
//   readdir -> readFile x F -> JSON.parse x F -> collectSignalsFromSubagentRun x F (E)
//   -> appendFeedback x F (J: redact + stringify + mkdir + appendFile, fsync=false)
//   -> diagnoseModelProjectIssues (E; consumer face priced by R23-E, one witness here)
// Doctor arm (I-slice readers, called directly — both are exported):
//   unknownAgentDriftCheck(projectRoot) at F doses
//   skillRouteLogCheck(projectRoot, undefined) at L doses (marker-file enabled)
//   listPiAgentProfilesFromDirs at A doses
import { readdir, readFile, rm, mkdir } from "node:fs/promises";
import { join } from "node:path";
import { performance } from "node:perf_hooks";
import { collectSignalsFromSubagentRun } from "/workspace/dist/learning/signals.js";
import { diagnoseModelProjectIssues } from "/workspace/dist/learning/diagnostics.js";
import { appendFeedback } from "/workspace/dist/feedback/store.js";
import { skillRouteLogCheck, unknownAgentDriftCheck } from "/workspace/dist/cli/doctor-overlay.js";
import { listPiAgentProfilesFromDirs } from "/workspace/dist/agents/dispatch-preflight.js";

const FIX = "/tmp/r23i-bench/fixtures";
const SCRATCH = "/tmp/r23i-bench/inproc-scratch";
const context = { projectId: "prj_00000000", episodeId: "ep_11111111-2222-4333-8444-555555555555" };

function median(xs) { const s = [...xs].sort((a, b) => a - b); return s[Math.floor(s.length / 2)]; }
async function timeAsync(reps, warm, fn) {
  const ts = [];
  for (let r = 0; r < reps + warm; r++) {
    const t0 = performance.now();
    await fn(r);
    if (r >= warm) ts.push(performance.now() - t0);
  }
  return median(ts);
}

for (const F of [100, 500, 2000]) {
  const dir = join(FIX, `project-pi-f${F}`, ".pi", "subagents", "runs");
  const names = (await readdir(dir)).filter((n) => n.endsWith(".json"));
  const tReaddir = await timeAsync(20, 3, () => readdir(dir));
  // Sequential readFile (production order).
  const tRead = await timeAsync(7, 2, async () => {
    for (const n of names) await readFile(join(dir, n), "utf8");
  });
  const raws = [];
  for (const n of names) raws.push(await readFile(join(dir, n), "utf8"));
  const tParse = await timeAsync(9, 3, () => { for (const raw of raws) JSON.parse(raw); });
  const parsed = raws.map((raw) => JSON.parse(raw));
  const tCollect = await timeAsync(9, 3, () => {
    for (const p of parsed) collectSignalsFromSubagentRun(p, context);
  });
  const signals = parsed.flatMap((p) => collectSignalsFromSubagentRun(p, context));
  // Feedback append replica: fresh scratch state-root per rep.
  const tPersist = await timeAsync(5, 1, async (r) => {
    const root = join(SCRATCH, `persist-${F}-${r}`);
    for (const signal of signals) {
      await appendFeedback(root, {
        id: "fbk_00000000",
        episodeId: signal.episodeId,
        kind: signal.kind,
        rubricVersion: "auto-loop-v1",
        score: signal.score,
        evidenceRefs: ["evd_00000000"],
        redacted: false,
        createdAt: signal.createdAt,
        notes: signal.summary
      });
    }
  });
  const tDiagnose = await timeAsync(9, 3, () => diagnoseModelProjectIssues(signals));
  // Doctor drift reader (I-slice, sync, whole check).
  const tDrift = await timeAsync(9, 3, () => unknownAgentDriftCheck(join(FIX, `project-pi-f${F}`)));
  console.log(
    `F=${F} (signals=${signals.length}): readdir=${tReaddir.toFixed(2)}ms read=${tRead.toFixed(2)} parse=${tParse.toFixed(2)} ` +
    `collect(E)=${tCollect.toFixed(2)} persist(J)=${tPersist.toFixed(2)} diagnose(E)=${tDiagnose.toFixed(3)} ` +
    `| ingest-sum=${(tReaddir + tRead + tParse + tCollect).toFixed(2)} loop-sum=${(tReaddir + tRead + tParse + tCollect + tPersist + tDiagnose).toFixed(2)} ` +
    `| per-file: read=${((tRead * 1000) / F).toFixed(1)}us persist=${((tPersist * 1000) / F).toFixed(1)}us drift(I)=${((tDrift * 1000) / F).toFixed(1)}us driftTotal=${tDrift.toFixed(2)}ms`
  );
}
await rm(SCRATCH, { recursive: true, force: true });
await mkdir(SCRATCH, { recursive: true });

for (const L of [1000, 10000]) {
  const proj = join(FIX, `project-log-l${L}`);
  const tSkill = await timeAsync(9, 3, () => skillRouteLogCheck(proj, undefined));
  console.log(`L=${L}: skillRouteLogCheck(I)=${tSkill.toFixed(2)}ms per-line=${((tSkill * 1e6) / L).toFixed(0)}ns`);
}
for (const A of [8, 64]) {
  const dir = join(FIX, `agents-a${A}`);
  const tList = await timeAsync(15, 3, () => listPiAgentProfilesFromDirs([dir]));
  console.log(`A=${A}: listPiAgentProfilesFromDirs=${(tList * 1000).toFixed(1)}us`);
}
```
