/** Pre-built rubric pool so createRubric (Date.toISOString) never sits inside a timed region. */
function buildPool(count: number, scopeOf: (i: number) => RubricScope, prefix: string): Rubric[] {
  return Array.from({ length: count }, (_, i) => createRubric(`${prefix}-${i}`, scopeOf(i), PROBE_CRITERIA));
}
function fillRegistry(pool: readonly Rubric[]): void {
  resetRubricRegistry();
  for (const rubric of pool) registerRubric(rubric);
}
function assertRegistryEmpty(where: string): void {

/* Leg RP: membership population dose R=0/8/64/512 (uniform 4-scope mix).
 * Per-dose prices measured in interleaved rounds (7 rounds, per-dose median)
 * so allocation/GC drift loads equally on every dose. The register probe is
 * an OVERWRITE at fixed membership (same probe id re-registered), so the
 * spread size is held at exactly R+1 keys for every timed call. */
{
  const doses = [0, 8, 64, 512] as const;
  const pools = new Map<number, Rubric[]>(
    doses.map((r) => [r, buildPool(r, (i) => SCOPES[i % 4] ?? "task", `rp${r}`)])

  const probe = createRubric("rp-probe", "task", PROBE_CRITERIA);
  const reg = new Map<number, number[]>(doses.map((d) => [d, []]));
  const get = new Map<number, number[]>(doses.map((d) => [d, []]));
  const list = new Map<number, number[]>(doses.map((d) => [d, []]));
  for (let round = 0; round < 7; round += 1) {
    for (const dose of doses) {
      fillRegistry(pools.get(dose) ?? []);
      registerRubric(probe); // membership now R+1; timed calls below overwrite at fixed size
      const regReps = dose >= 512 ? 500 : 5000;
      reg.get(dose)?.push(bench(() => registerRubric(probe), regReps));
      get.get(dose)?.push(bench(() => void getActiveRubric("task"), 100000));
      list.get(dose)?.push(bench(() => void listRubrics(), dose >= 512 ? 2000 : 20000));
    }
  }
  const regM: Record<number, number> = {};
  const getM: Record<number, number> = {};
  const listM: Record<number, number> = {};
  for (const dose of doses) {
    regM[dose] = median(reg.get(dose) ?? []);
    getM[dose] = median(get.get(dose) ?? []);
    listM[dose] = median(list.get(dose) ?? []);
    log(
      `RP R=${dose}: registerRubric(overwrite @ membership ${dose + 1})=${((regM[dose] ?? 0) * 1e6).toFixed(0)}ns  getActiveRubric=${((getM[dose] ?? 0) * 1e6).toFixed(1)}ns  listRubrics=${((listM[dose] ?? 0) * 1e6).toFixed(0)}ns (7 interleaved rounds)`

  }

  );
  check(

    (regM[512] ?? Number.NaN) < 0.1,
