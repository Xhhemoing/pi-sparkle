## 1. 第二十三遍搜索方法：批排布几何透镜（同一多重集 × 六种排布）

configured+learned N=30 base vs family-grouped。四次干净运行 + run 0 冒烟
+ 两次 gate 后确认运行：


六条结论：

1. **排布轴是平的**：同一多重集的六种排布在同窗内的全距 M=2 仅 0.14–0.58
   ms/eval、M=10 仅 0.81–1.56 ms/eval——全部在 R15-B A/A 地板（±2.3 ms/窗）
   之内；跨运行的窗级电平移动（run2 M=2 全臂抬到 9.28–9.86）同步平移全部
   臂，臂间差不随之放大——排布税与 A/A 噪声不可分离。
2. **无稳定臂序**：最优臂身份在四次运行里遍历 role-grouped /
   family-grouped / cxrisk-grouped / max-alternate 四种，最差臂遍历 base /
   reversed / cxrisk-grouped / role-grouped / max-alternate——「对抗性
   交错」臂既当过最差也当过**最优**（run3 M=10、run4 M=2）＝分支切换税
   在本切片的逐任务循环里小于窗噪声。机制上合理：逐任务循环的主导站点
   （regex 链、目录扫描、对象构造）是数据依赖而非分支依赖，V8 的 IC 在
   7 种 role / 8 种 family 的小形态集上早已饱和为 megamorphic 稳态，
   相邻任务次序不再改变 IC 状态迁移计数。
3. **oracle 分组红利符号翻转**：max-alternate − 最优分组臂 M=2 −0.06~
   +0.20、M=10 −0.05~+1.22 ms/eval——上界一切可实现分组策略的红利池在
   两格都不稳定为正。即便取四次运行最大值 1.22 ms/eval，也低于落地线
   一个量级以上。
4. **候选自败**：「预分组+还原」候选必付的全管线开销实测 +0.53~+5.32
   ms/eval **七次运行全部为正**（analyzeTask 预扫描 ≈1.1 ms/call ×2 +
   decorate-sort ≈2000·log 项 + 还原数组）——开销稳定超过其所追逐的
   符号还不稳的红利。真实面 N=30 分组差 −2.01~+4.96 µs 亦符号翻转。
   该候选类（含 role 键零成本变体——role-grouped 臂已内含其收益上界）
   整体关闭，不铸 ID。
5. **置换等变性从推导升级为实验证明**：6 臂 × 2 格 × N=2000 逐元素逐
   字节 + 双运行确定性 ×7 运行零失败——「assignTasks 无跨任务状态」
   此前是源码推导（S3-B-1/S3-B-6 裁决的前提），现在有直接证据。任何
   未来的批内重排/并行化提案可直接引用本格的等变断言设计。
6. **批级输入三自由度清点完毕**：N（每轮 10× 压力）、组成（R17-B 类
   集中度）、排布（本格）全部有显式记录——批容器本身再无未测自由度，
   后续轮次对批级改写候选可直接对表拒绝。

```ts
function mulberry32(seed: number): () => number {

/* ============================================================
 * §0d NEW measurement cell: batch arrangement (order) geometry.
 * Same multiset of tasks (shared task object references -- exactly what a
 * pre-sorting candidate would pass), different arrangements:
 *   base          -- the seeded 0xb44b01 arrangement every round measured
 *   reversed      -- base reversed
 *   role-grouped  -- stable-grouped by task.role (realizable at zero
 *                    analysis cost by any candidate)
 *   family-grouped-- stable-grouped by ORACLE analysis family (upper bound
 *                    of every realizable grouping: perfect familyOf branch
 *                    homogeneity in the per-task loop)
 *   cxrisk-grouped-- stable-grouped by ORACLE (complexity, highRisk,
 *                    preferPrimary) triple (perfect policy/selection branch
 *                    homogeneity)
 *   max-alternate -- round-robin across the family groups (adversarial
 *                    floor: maximal adjacent-task branch alternation)
 * Cells: {M=2 default+learned, M=10 stress+learned} x arms at N=2000
 * replay-faithful, plus the M=7 configured+learned live face N=30 spot arm.
 * Fidelity per arm: permutation equivariance out_arm[i] == out_base[p[i]]
 * byte-for-byte for all i (assignPlanned is per-task pure; the plan is
 * catalog-only), plus a double-run determinism check at max-alternate.
 * Candidate pricing: pool ceiling = worst arm - best arm (per eval x2);
 * the decorate-sort-restore overhead a real "pre-group then un-permute"
 * candidate must pay is measured separately (oracle grouping includes the
 * analyzeTask pre-pass it would need).
 * ============================================================ */
{
  const rng = mulberry32(0xb44b01);
  const replayTasks = genReplayTasks(rng, 2000);
  const armDefs: readonly { readonly label: string; readonly make: (tasks: readonly AssignableTask[]) => AssignableTask[] }[] = [
    { label: "base", make: (tasks) => [...tasks] },
    { label: "reversed", make: (tasks) => [...tasks].reverse() },
    {
      label: "role-grouped",
      make: (tasks) => [...tasks].sort((a, b) => (a.role < b.role ? -1 : a.role > b.role ? 1 : 0))
    },
    {
      label: "family-grouped",
      make: (tasks) => {
        const keyed = tasks.map((task, i) => ({ task, i, key: analyzeTask(task.objective, task.role).family as string }));
        keyed.sort((a, b) => (a.key < b.key ? -1 : a.key > b.key ? 1 : a.i - b.i));
        return keyed.map((row) => row.task);
      }
    },
    {
      label: "cxrisk-grouped",
      make: (tasks) => {
        const keyed = tasks.map((task, i) => {
          const a = analyzeTask(task.objective, task.role);
          return { task, i, key: `${a.complexity}|${a.highRisk ? 1 : 0}|${a.preferPrimary ? 1 : 0}` };
        });
        keyed.sort((a, b) => (a.key < b.key ? -1 : a.key > b.key ? 1 : a.i - b.i));
        return keyed.map((row) => row.task);
      }
    },
    {
      label: "max-alternate",
      make: (tasks) => {
        const groups = new Map<string, AssignableTask[]>();
        for (const task of tasks) {
          const family = analyzeTask(task.objective, task.role).family as string;
          const bucket = groups.get(family);
          if (bucket === undefined) groups.set(family, [task]);
          else bucket.push(task);
        }
        const buckets = [...groups.keys()].sort().map((key) => groups.get(key)!);
        const out: AssignableTask[] = [];
        let remaining = tasks.length;
        let cursor = 0;
        const idx = buckets.map(() => 0);
        while (remaining > 0) {
          const b = cursor % buckets.length;
          const at = idx[b]!;
          if (at < buckets[b]!.length) {
            out.push(buckets[b]![at]!);
            idx[b] = at + 1;
            remaining -= 1;
          }
          cursor += 1;
        }
        return out;
      }
    }
  ];
  const orderCells: readonly [string, ModelRouterConfig, LearnedRoutingPolicy][] = [
    ["M=2 default+learned", catalogFromPrimary({ primaryModelId: "premium" }), learnedM2()],
    ["M=10 stress+learned", tenModelCatalog(), tenModelLearned()]
  ];
  for (const [cellLabel, catalog, learned] of orderCells) {
    const baseOut = assignTasks({ catalog, tasks: replayTasks, learned });
    const baseBytes = baseOut.map((row) => JSON.stringify(row));
    const indexOfTask = new Map<AssignableTask, number>();
    replayTasks.forEach((task, i) => indexOfTask.set(task, i));
    const costs = new Map<string, number>();
    for (const arm of armDefs) {
      const arranged = arm.make(replayTasks);
      check(`order-geometry [${cellLabel}] arm=${arm.label}: same multiset`, arranged.length === replayTasks.length);
      // Fidelity: permutation equivariance, element-by-element byte equality.
      const armOut = assignTasks({ catalog, tasks: arranged, learned });
      let equivariant = true;
      for (let i = 0; i < arranged.length; i += 1) {
        const baseIndex = indexOfTask.get(arranged[i]!)!;
        if (JSON.stringify(armOut[i]) !== baseBytes[baseIndex]) {
          equivariant = false;
          break;
        }
      }
      check(`order-geometry fidelity [${cellLabel}] arm=${arm.label}: permutation-equivariant byte-for-byte`, equivariant);
      const cost = bench(() => assignTasks({ catalog, tasks: arranged, learned }), 30);
      costs.set(arm.label, cost);
      console.log(
        `order-geometry unit [${cellLabel}] arm=${arm.label}: replay N=2000 ${(cost * 1e3).toFixed(1)}us per call (${(cost * 2).toFixed(2)}ms per eval x2)`
      );
    }
    // Determinism double run at the most degenerate arm.
    {
      const arranged = armDefs[armDefs.length - 1]!.make(replayTasks);
      const runA = JSON.stringify(assignTasks({ catalog, tasks: arranged, learned }));
      const runB = JSON.stringify(assignTasks({ catalog, tasks: arranged, learned }));
      check(`order-geometry fidelity [${cellLabel}]: double run at max-alternate byte-identical`, runA === runB);
    }
    const entries = [...costs.entries()];
    const best = entries.reduce((a, b) => (b[1] < a[1] ? b : a));
    const worst = entries.reduce((a, b) => (b[1] > a[1] ? b : a));
    const spreadPerEval = (worst[1] - best[1]) * 2;
    const groupedDividend = (costs.get("max-alternate")! - Math.min(costs.get("family-grouped")!, costs.get("cxrisk-grouped")!, costs.get("role-grouped")!)) * 2;
    console.log(
      `order-geometry summary [${cellLabel}]: best=${best[0]} ${(best[1] * 2).toFixed(2)}ms | worst=${worst[0]} ${(worst[1] * 2).toFixed(2)}ms | spread=${spreadPerEval.toFixed(2)}ms per eval | grouping dividend (max-alternate - best grouped)=${groupedDividend.toFixed(2)}ms per eval`
    );
    conclusions.push(
      `order-geometry ${cellLabel}: spread=${spreadPerEval.toFixed(2)}ms per eval (best=${best[0]}, worst=${worst[0]}); grouping-dividend=${groupedDividend.toFixed(2)}ms per eval`
    );
  }
  // The overhead a real "pre-group then un-permute" candidate must pay:
  // oracle family key pre-pass (analyzeTask x N) + decorate-sort + result
  // restore (out[origIndex] = res[j]) at N=2000.
  {
    const catalog = tenModelCatalog();
    const learned = tenModelLearned();
    const overhead = bench(() => {
      const keyed = replayTasks.map((task, i) => ({ task, i, key: analyzeTask(task.objective, task.role).family as string }));
      keyed.sort((a, b) => (a.key < b.key ? -1 : a.key > b.key ? 1 : a.i - b.i));
      const arranged = keyed.map((row) => row.task);
      const res = assignTasks({ catalog, tasks: arranged, learned });
      const restored = new Array(res.length);
      for (let j = 0; j < res.length; j += 1) restored[keyed[j]!.i] = res[j];
      if (restored.length !== res.length) throw new Error("unreachable");
    }, 30);
    const plainCost = bench(() => assignTasks({ catalog, tasks: replayTasks, learned }), 30);
    console.log(
      `order-geometry candidate overhead [M=10+learned]: pre-group+route+restore=${(overhead * 2).toFixed(2)}ms vs plain=${(plainCost * 2).toFixed(2)}ms per eval x2 (net=${((overhead - plainCost) * 2).toFixed(2)}ms)`
    );
    conclusions.push(
      `order-geometry candidate net [M=10+learned]: pre-group+restore costs ${((overhead - plainCost) * 2).toFixed(2)}ms per eval MORE than plain (the pool it chases is the grouping dividend above)`
    );
  }
  // Real-scale spot arm: M=7 configured+learned live face N=30.
  {
    const catalogCfg = configuredCatalog();
    const learned = configuredLearned();
    const cli = genTasks(mulberry32(0xb44b01), 30);
    const baseOut = assignTasks({ catalog: catalogCfg, tasks: cli, learned });
    const baseBytes = baseOut.map((row) => JSON.stringify(row));
    const keyed = cli.map((task, i) => ({ task, i, key: analyzeTask(task.objective, task.role).family as string }));
    keyed.sort((a, b) => (a.key < b.key ? -1 : a.key > b.key ? 1 : a.i - b.i));
    const grouped = keyed.map((row) => row.task);
    const groupedOut = assignTasks({ catalog: catalogCfg, tasks: grouped, learned });
    let equivariant = true;
    for (let j = 0; j < grouped.length; j += 1) {
      if (JSON.stringify(groupedOut[j]) !== baseBytes[keyed[j]!.i]) {
        equivariant = false;
        break;
      }
    }
    check("order-geometry fidelity [M=7 live face N=30]: family-grouped permutation-equivariant", equivariant);
    const baseCost = bench(() => assignTasks({ catalog: catalogCfg, tasks: cli, learned }), 2000);
    const groupedCost = bench(() => assignTasks({ catalog: catalogCfg, tasks: grouped, learned }), 2000);
    console.log(
      `order-geometry REAL face [M=7 configured+learned N=30]: base=${(baseCost * 1e3).toFixed(1)}us vs family-grouped=${(groupedCost * 1e3).toFixed(1)}us per call (delta=${((baseCost - groupedCost) * 1e3).toFixed(2)}us)`
    );
    conclusions.push(
      `order-geometry real face N=30: base=${(baseCost * 1e3).toFixed(1)}us grouped=${(groupedCost * 1e3).toFixed(1)}us (delta=${((baseCost - groupedCost) * 1e3).toFixed(2)}us per batch)`
    );
  }
}
```
