/**
 * R22-F new-axis measurements (temporary; NOT candidate adjudications and NOT
 * committed — no winner this round). One axis no earlier F round measured
 * directly: the MEMBERSHIP-TOPOLOGY COMPOSITION dimension — WHERE inside the
 * frozen population the assigned episode set sits.
 *
 * Twenty-one rounds of absolute anchors pinned the assigned set to the
 * population HEAD PREFIX (hashes = population.slice(0, A), aligned), and the
 * order/dose instruments that did travel through other membership positions
 * (r7f part E "reversed" = tail block reversed, "scattered" = full-range
 * sample; R18-F axis 1 alignment-fraction dose) always measured
 * LANDED-vs-FROZEN-REFERENCE DELTAS — both sides run the same S6-F-1
 * pending-Set fallback, so the fallback's population-scan depth cancels out
 * of every published number. Yet that scan depth is a genuine input
 * dimension of the landed form: `resolvePendingMembership` walks the
 * population from index 0 and early-exits at the HIGHEST population position
 * occupied by a pending assignment hash, so per restore the scan pays
 * (1 + max position) probes. Production `evaluateProposalShadow` feeds
 * caller-chosen episode hashes with no head-block guarantee, so the anchor's
 * aligned-head fixture sits at the cheapest corner of a never-priced grid.
 *
 * Arithmetic predictions (interleaved production schedule, P=2000, A=1000;
 * ~2 restores per ledger size m, scan depth per restore = 1 + max pending
 * position; the pending-Set BUILD cost is identical across all fallback
 * arms, so deltas vs HS isolate pure scan depth):
 *
 *   HS   head block, seeded shuffle       Σdepth ≈ Σ 2·(999·m/(m+1)) ≈ 1.99e6
 *   MID  positions 500..1499, natural     Σ ≈ Σ 2·(500+m)            ≈ 2.00e6   Δ≈0
 *   TAIL positions 1000..1999, natural    Σ ≈ Σ 2·(1000+m)           ≈ 3.00e6   Δ≈+1.0e6
 *   TREV tail block, reversed             Σ ≈ 2000·2000              ≈ 4.00e6   Δ≈+2.0e6
 *   FS   full-range sample, shuffled      Σ ≈ Σ 2·(1999·m/(m+1))     ≈ 3.97e6   Δ≈+2.0e6
 *   Z    head prefix, aligned (anchor)    no fallback at all (fast path)
 *
 *   cell a  end-to-end topology grid: the six arms above through the exact
 *           production interleave; deltas vs HS isolate scan depth, Z vs HS
 *           re-shows the aligned fast path in-process.
 *
 *   cell b  per-restore micro at a FULL non-halted ledger (maxAssignments =
 *           2A so nothing halts): runner.restore() × N on the captured final
 *           states. Full-ledger scan depths: Z none, HS 1000, MID 1500,
 *           TAIL 2000, TREV 2000 (same pending SET as TAIL at full ledger),
 *           FS ~2000. The depth-linear slope (ns/probe) must cross-predict
 *           cell a's end-to-end deltas (Δprobes × slope).
 *
 * Every arm is verified deterministically before timing (final ledger sizes,
 * halt reason, exact assignment sequence identity vs the intended order,
 * outcome pass-through identity, zero clock fields, per-arm digest equality
 * across two executions), then all arms are timed ROUND-ROBIN (1 warmup +
 * 5 passes) so process drift spreads evenly across arms.
 *
 * No candidate can come out of this axis: the scan IS the landed fail-closed
 * S6-F-1 mechanism (skipping the head misses needs a position index —
 * per-restore it costs the same O(P), cross-restore it is X1-1 hidden
 * state); scan-direction/start specialization is input-order-sensitive
 * coin-flipping (S6-F-2 family, S6-F-5 adjacency); and production A « 1000.
 * These are informational price cells and a methodological caveat on anchor
 * transfer, not pools.
 *

import { createCandidateId, createResourceVersionId } from "./src/domain/ids.js";
import {
  validateExperimentPlan,
  type ExperimentPlan,
} from "./src/experiments/plan.js";
import {
  createShadowRunner,
  type ExperimentOutcome,
  type ShadowState,
} from "./src/experiments/shadow.js";
import { stableStringify } from "./src/experiments/manifest.js";

function onceMs(fn: () => void): number {
  const t0 = process.hrtime.bigint();
  fn();
  const t1 = process.hrtime.bigint();
  return Number(t1 - t0) / 1e6;
}

function median(xs: readonly number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  return s[Math.floor(s.length / 2)] as number;
}

/** Historical fixture shape: `ep_` + zero-padded base36 index, L=11. */
function makePopulation(size: number): string[] {
  const population: string[] = [];
  for (let i = 0; i < size; i++) {
    population.push(`ep_${i.toString(36).padStart(8, "0")}`);
  }
  return population;
}

function makeShadowPlan(
  population: readonly string[],
  maxAssignments: number,
  id: string
): ExperimentPlan {
  return {
    planVersion: 1,
    experimentId: id,
    mode: "shadow",
    baselineVersionId: createResourceVersionId(() => "r22fbase"),
    candidateId: createCandidateId(() => "r22fcand"),
    population,
    metrics: ["utility"],
    thresholds: { maxGuardrailBreaches: 1_000_000, maxCostUsd: 1e12 },
    budget: { maxAssignments, maxWallClockMs: 1e12 },
    randomization: { seed: 42 },
    stopPolicy: { onGuardrail: "halt", onBudgetExhausted: "halt" },
    missingOutcomePolicy: "exclude",
    canary: undefined,
  };
}

/** Deterministic fixture rng (mulberry32 clone, measurement-side only). */
function fixtureRng(seed: number): () => number {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function seededShuffle<T>(items: readonly T[], seed: number): T[] {
  const out = [...items];
  const rng = fixtureRng(seed);
  for (let i = out.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1));
    const t = out[i] as T;
    out[i] = out[j] as T;
    out[j] = t;
  }
  return out;
}

const P = 2000;
const A = 1000;
const population = makePopulation(P);

/** The six membership topologies (assignment hash sequences). */
const orders: readonly (readonly [string, readonly string[]])[] = [
  ["Z-headAligned", population.slice(0, A)],
  ["HS-headScattered", seededShuffle(population.slice(0, A), 0x22f1)],
  ["MID-midBlock", population.slice(500, 500 + A)],
  ["TAIL-tailBlock", population.slice(P - A)],
  ["TREV-tailReversed", [...population.slice(P - A)].reverse()],
  ["FS-fullScattered", seededShuffle(population, 0x22f2).slice(0, A)],
];

function cleanStreamFor(hashes: readonly string[]): ExperimentOutcome[] {
  return hashes.map((h) => ({
    episodeHash: h,
    utility: 0.5,
    costUsd: 0.01,
    guardrailBreached: false,
  }));
}

/** Production interleave shape (evaluateProposalShadow), clock pinned to 0. */
function runInterleaved(plan: ExperimentPlan, stream: readonly ExperimentOutcome[]): ShadowState {
  const runner = createShadowRunner(plan);
  let state = runner.start(0);
  for (const outcome of stream) {
    if (state.halted) {
      break;
    }
    state = runner.assign(state, outcome.episodeHash, 0);
    state = runner.recordOutcome(state, outcome, 0);
  }
  return state;
}

function ledgerDigest(state: ShadowState): string {
  return stableStringify({
    assignments: state.assignments,
    outcomes: state.outcomes,
    guardrailBreaches: state.guardrailBreaches,
    halted: state.halted,
    haltReason: state.haltReason,
  });
}

interface Arm {
  readonly name: string;
  readonly stream: readonly ExperimentOutcome[];
  readonly exec: () => ShadowState;
}

const arms: Arm[] = orders.map(([name, hashes], i) => {
  const plan = makeShadowPlan(population, A, `exp_r22f_a${i}`);
  validateExperimentPlan(plan);
  const stream = cleanStreamFor(hashes);
  return { name, stream, exec: () => runInterleaved(plan, stream) };
});

/* ------------------- verification pass (untimed, twice) ------------------- */
for (const arm of arms) {
  const digests: string[] = [];
  for (let round = 0; round < 2; round++) {
    const state = arm.exec();
    if (!state.halted || state.haltReason !== "budget-exhausted") {
      throw new Error(`${arm.name}: expected budget-exhausted final state`);
    }
    if (state.assignments.length !== A || state.outcomes.length !== A) {
      throw new Error(`${arm.name}: unexpected ledger sizes`);
    }
    for (let i = 0; i < A; i++) {
      if ((state.assignments[i] as { episodeHash: string }).episodeHash !== arm.stream[i]?.episodeHash) {
        throw new Error(`${arm.name}: assignment sequence diverged at ${i}`);
      }
    }
    for (let i = 0; i < A; i += 97) {
      if (state.outcomes[i] !== arm.stream[i]) {
        throw new Error(`${arm.name}: outcome not pass-through at ${i}`);
      }
    }
    if (state.guardrailBreaches !== 0 || state.startedAtMs !== 0 || state.elapsedMs !== 0) {
      throw new Error(`${arm.name}: unexpected counters/clock fields`);
    }
    digests.push(ledgerDigest(state));
  }
  if (digests[0] !== digests[1]) {
    throw new Error(`${arm.name}: nondeterministic ledger`);
  }
}
process.stdout.write("verification: all topology-arm invariants hold\n");

/* --------------------- cell a: round-robin end-to-end --------------------- */
for (const arm of arms) {
  arm.exec(); // one warmup pass per arm
}
const PASSES = 5;
const samples = new Map<string, number[]>(arms.map((arm) => [arm.name, []]));
for (let pass = 0; pass < PASSES; pass++) {
  for (const arm of arms) {
    (samples.get(arm.name) as number[]).push(onceMs(() => {
      arm.exec();
    }));
  }
}
const baseHS = median(samples.get("HS-headScattered") as number[]);
process.stdout.write(
  `cell a (P=${P}/A=${A}, production interleave, round-robin ${PASSES} passes, ` +
    `min/median ms; delta vs HS-headScattered):\n`
);
for (const arm of arms) {
  const xs = samples.get(arm.name) as number[];
  const med = median(xs);
  const label = arm.name === "HS-headScattered" ? "(baseline)" : `delta=${(med - baseHS).toFixed(2)}`;
  process.stdout.write(
    `${arm.name.padEnd(24)} min=${Math.min(...xs).toFixed(2)} median=${med.toFixed(2)} ${label}\n`
  );
}

/* ------------- cell b: per-restore micro at a full 2A ledger -------------- */
// maxAssignments = 2A so the captured final states are NOT halted; every
// fallback arm builds the same A-entry pending Set, so deltas vs HS are pure
// scan depth (full-ledger depths: Z none, HS 1000, MID 1500, TAIL 2000,
// TREV 2000, FS ~2000).
const microStates: (readonly [string, ShadowRunnerLike, ShadowState])[] = [];
interface ShadowRunnerLike {
  restore(serialized: ShadowState): ShadowState;
}
for (const [i, [name, hashes]] of orders.entries()) {
  const plan = makeShadowPlan(population, 2 * A, `exp_r22f_b${i}`);
  validateExperimentPlan(plan);
  const runner = createShadowRunner(plan);
  const stream = cleanStreamFor(hashes);
  let state = runner.start(0);
  for (const outcome of stream) {
    state = runner.assign(state, outcome.episodeHash, 0);
    state = runner.recordOutcome(state, outcome, 0);
  }
  if (state.halted || state.assignments.length !== A) {
    throw new Error(`${name}: micro state must stay active with a full ledger`);
  }
  // restore() must be idempotent-equivalent on the captured state.
  if (ledgerDigest(runner.restore(state)) !== ledgerDigest(state)) {
    throw new Error(`${name}: restore altered the ledger`);
  }
  microStates.push([name, runner, state]);
}
const RESTORES = 200;
const microSamples = new Map<string, number[]>(microStates.map(([name]) => [name, []]));
for (const [, runner, state] of microStates) {
  for (let r = 0; r < 20; r++) runner.restore(state); // warmup
}
let sink = 0;
for (let pass = 0; pass < PASSES; pass++) {
  for (const [name, runner, state] of microStates) {
    (microSamples.get(name) as number[]).push(onceMs(() => {
      for (let r = 0; r < RESTORES; r++) {
        sink += runner.restore(state).assignments.length;
      }
    }));
  }
}
const microBase = median(microSamples.get("HS-headScattered") as number[]);
process.stdout.write(
  `cell b (full ${A}-entry active ledger, ${RESTORES} restores/pass, µs/restore medians; ` +
    `delta vs HS-headScattered):\n`
);
for (const [name] of microStates) {
  const xs = microSamples.get(name) as number[];
  const med = median(xs);
  const perOp = (med * 1000) / RESTORES;
  const delta = ((med - microBase) * 1000) / RESTORES;
  const label = name === "HS-headScattered" ? "(baseline)" : `delta=${delta.toFixed(2)}µs`;
  process.stdout.write(
    `${name.padEnd(24)} ${perOp.toFixed(2)} µs/restore ${label}\n`
  );
}
if (!Number.isFinite(sink)) throw new Error("sink overflow");
