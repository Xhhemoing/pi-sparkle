    (f) => ({ createdAt: f.createdAt, id: f.id, explicit: f.explicit, value: f.value, scope: f.scope, key: f.key, scopeKey: f.scopeKey, recurrenceCount: f.recurrenceCount, evidenceEpisodeId: f.evidenceEpisodeId, weight: f.weight }),

  function buildFixture(K: number): {

/* ====== P2: redaction optional-field shape mix (byte-compensated) ====== */
/* Two FeedbackRecord shapes: BASE (no runId/taskId, padded summary) and
 * EXT (runId+taskId, short summary). Pad width chosen so the serialized
 * INPUT length is identical; redaction output preserves the compensation
 * (copyFeedback's conditional spreads mirror input shape). Bodies are clean
 * (no needles) and < maxBodyChars, so classes = [pii] for every record at
 * every dose: redaction class mix is pinned. */
{
  const POOL = 1000;
  const DOSES = [0, 50, 100];
  const CPU_ITERS = 100000;
  const APPENDS = 2000;

  const PRODUCTION_POLICY = {
    redactPII: true,
    maxBodyChars: 400,
    forbiddenSubstrings: ["sk-", "api_key", "API_KEY", "BEGIN PRIVATE"]
  } as const;

  const BODY =
    "ordinary prose feedback body with no forbidden needles present; " +
    "fixed width padding text keeps the record byte account level.";

  function extRecord(i: number): FeedbackRecord {
    return {
      id: `fb_${pad(i, 6)}`,
      episodeId: "eps_shape00" as EpisodeId,
      kind: "human",
      rubricVersion: "r1",
      score: 60,
      evidenceRefs: [],
      redacted: false,
      createdAt: "2026-08-25T00:00:00.000Z" as FeedbackRecord["createdAt"],
      runId: `run_${pad(i, 8)}` as FeedbackRecord["runId"],
      taskId: `tsk_${pad(i, 8)}` as FeedbackRecord["taskId"],
      body: BODY,
      summary: "s"
    };
  }

  function baseRecord(i: number, padWidth: number): FeedbackRecord {
    return {
      id: `fb_${pad(i, 6)}`,
      episodeId: "eps_shape00" as EpisodeId,
      kind: "human",
      rubricVersion: "r1",
      score: 60,
      evidenceRefs: [],
      redacted: false,
      createdAt: "2026-08-25T00:00:00.000Z" as FeedbackRecord["createdAt"],
      body: BODY,
      summary: "s".repeat(padWidth)
    };
  }

  // Length compensation: pad the base summary until serialized lengths match.
  const extLen = JSON.stringify(extRecord(0)).length;
  let padWidth = 1;
  while (JSON.stringify(baseRecord(0, padWidth)).length < extLen) padWidth += 1;
  if (JSON.stringify(baseRecord(0, padWidth)).length !== extLen) {
    throw new Error("FIDELITY FAIL P2: cannot byte-compensate shapes");
  }

  function buildPool(x: number): { records: FeedbackRecord[]; inBytes: number; outBytes: number; classes: string } {
    const records: FeedbackRecord[] = [];
    let inBytes = 0;
    let outBytes = 0;
    const classCounts = new Map<string, number>();
    for (let i = 0; i < POOL; i++) {
      const record = i % 100 < x ? extRecord(i) : baseRecord(i, padWidth);
      records.push(record);
      inBytes += JSON.stringify(record).length;
      const out = redactFeedback(record, PRODUCTION_POLICY);
      outBytes += JSON.stringify(out.feedback).length;
      const key = out.decision.classes.join(",");
      classCounts.set(key, (classCounts.get(key) ?? 0) + 1);
    }
    return {
      records,
      inBytes,
      outBytes,
      classes: [...classCounts.entries()].map(([k, v]) => `${k}:${v}`).join("|")
    };
  }

  const inAccounts: number[] = [];
  const outAccounts: number[] = [];
  const classAccounts: string[] = [];
  for (const x of DOSES) {
    const { records, inBytes, outBytes, classes } = buildPool(x);
    inAccounts.push(inBytes);
    outAccounts.push(outBytes);
    classAccounts.push(classes);
    // (a) pure CPU
    const cpuResults: number[] = [];
    let sink = 0;
    for (let run = 0; run < 3; run++) {
      for (let i = 0; i < 20000; i++) sink += redactFeedback(records[i % POOL]!, PRODUCTION_POLICY).decision.classes.length;
      const t0 = hr();
      for (let i = 0; i < CPU_ITERS; i++) sink += redactFeedback(records[i % POOL]!, PRODUCTION_POLICY).decision.classes.length;
      const t1 = hr();
      cpuResults.push(us(t0, t1, CPU_ITERS) * 1000);
    }
    console.log(
      `P2 redact CPU shape-mix x=${pad(x, 3)}%: ${cpuResults.map((v) => v.toFixed(0)).join(" / ")} ns/record  [sink=${sink}]`
    );
    // (b) e2e appendFeedback
    const stateRoot = mkdtempSync(join(tmpdir(), "r22j-shape-"));
    const e2eResults: number[] = [];
    for (let run = 0; run < 3; run++) {
      for (let i = 0; i < 100; i++) await appendFeedback(stateRoot, records[(run * 100 + i) % POOL]!);
      const t0 = hr();
      for (let i = 0; i < APPENDS; i++) await appendFeedback(stateRoot, records[i % POOL]!);
      const t1 = hr();
      e2eResults.push(us(t0, t1, APPENDS));
    }
    console.log(`P2 e2e appendFeedback shape-mix x=${pad(x, 3)}%: ${e2eResults.map(fmt).join(" / ")} us/append`);
    rmSync(stateRoot, { recursive: true, force: true });
  }
  for (let i = 1; i < inAccounts.length; i++) {
    assertEqual("P2 input byte account", inAccounts[i], inAccounts[0]);
    assertEqual("P2 redacted-output byte account", outAccounts[i], outAccounts[0]);
    assertEqual("P2 redaction-class account", classAccounts[i], classAccounts[0]);
  }
  console.log(
    `P2 fidelity: input ${inAccounts[0]} B, redacted output ${outAccounts[0]} B, and class account ${classAccounts[0]} equal across x doses`
  );
}

/* ========== P3: mailbox optional-topic shape mix (byte parity) ========== */
/* M=600 role-cast mails on ONE role; only the fraction x with a topic field
 * varies. Body padded on no-topic mails so per-mail string payload length is
 * level. Delivery conserved exactly once per cycle. */
{
  const M = 600;
  const DOSES = [0, 50, 100];
  const REPEATS = 300;
  const role = "worker" as AgentRole;

  function buildMails(x: number): { mails: ClusterMail[]; chars: number } {
    const mails: ClusterMail[] = [];
    let chars = 0;
    const topic = "t0000";
    for (let i = 0; i < M; i++) {
      const withTopic = i % 100 < x;
      const mail: ClusterMail = withTopic
        ? {
            id: `msg_${pad(i, 6)}` as MessageId,
            from: "agt_sender0000" as AgentInstanceId,

            topic,
            body: `payload ${pad(i, 6)}`,
            occurredAt: "2026-08-25T00:00:00.000Z" as ClusterMail["occurredAt"]

            id: `msg_${pad(i, 6)}` as MessageId,
            from: "agt_sender0000" as AgentInstanceId,

            body: `payload ${pad(i, 6)}${"x".repeat(topic.length)}`,

          };
      chars += mail.body.length + (mail.topic?.length ?? 0);
      mails.push(mail);
    }
    return { mails, chars };
  }

  const charAccounts: number[] = [];
  for (const x of DOSES) {

    let best = Infinity;

      const dt = us(t0, t1, 1);
      best = Math.min(best, dt);
      sum += dt;
    }
    console.log(
      `P3 mailbox topic-mix x=${pad(x, 3)}% (M=${M}): min=${fmt(best)} mean=${fmt(sum / REPEATS)} us/enqueue+claim`
    );
  }
  for (let i = 1; i < charAccounts.length; i++) {
    assertEqual("P3 payload char account", charAccounts[i], charAccounts[0]);
  }
  console.log(`P3 fidelity: payload chars ${charAccounts[0]} and exactly-once delivery equal across x doses`);
}

/* ===== P4: deletion cascade JSONL key-order shape dose (I/O coupled) ===== */
/* N=200 records, 20 hits at fixed positions. Each line carries the same
 * fields; the JSON KEY ORDER rotates over K permutations (per-line byte
 * length invariant under permutation — asserted). JSON.parse materializes a
 * distinct hidden class per order, so the map/filter/stringify pipeline in
 * the cascade runs K-shape polymorphic. */
{
  const N = 200;
  const HITS = 20;
  const DOSES = [1, 2, 4];
  const ITERS = 30;
  const target = "ep_target000000" as EpisodeId;

  function lineOf(i: number, order: number): string {
    const hit = i % 10 === 0;
    const fields: Record<string, unknown> = {
      id: `fb_${pad(i, 6)}`,
      episodeId: hit ? target : `ep_bg${pad(i % 7, 10)}`,
      kind: "human",
      rubricVersion: "r1",
      score: 50 + (i % 40),
      evidenceRefs: [],
      redacted: false,
      createdAt: "2026-08-25T00:00:00.000Z",
      body: `free text feedback payload number ${pad(i, 4)} `.repeat(3),
      summary: `summary ${pad(i, 4)}`
    };
    const keys = Object.keys(fields);
    const permuted =
      order === 0
        ? keys
        : order === 1
          ? [...keys].reverse()
          : order === 2
            ? [...keys.filter((_, idx) => idx % 2 === 0), ...keys.filter((_, idx) => idx % 2 === 1)]
            : [...keys.filter((_, idx) => idx % 2 === 1), ...keys.filter((_, idx) => idx % 2 === 0)];
    const obj: Record<string, unknown> = {};
    for (const key of permuted) obj[key] = fields[key];
    return JSON.stringify(obj);
  }

  function seedFeedback(stateRoot: string, K: number): number {
    const log = feedbackLogPath(stateRoot);
    mkdirSync(dirname(log), { recursive: true });
    const lines: string[] = [];
    let bytes = 0;
    for (let i = 0; i < N; i++) {
      const line = lineOf(i, i % K);
      const reference = lineOf(i, 0);
      if (line.length !== reference.length) {
        throw new Error(`FIDELITY FAIL P4: line ${i} length differs across key orders`);
      }
      bytes += line.length + 1;
      lines.push(line);
    }
    writeFileSync(log, lines.join("\n") + "\n", "utf8");
    writeFileSync(
      feedbackTombstonesPath(stateRoot),
      JSON.stringify(["fb-pre-a", "fb-pre-b", "fb-pre-c", "fb-pre-d", "fb-pre-e"], null, 2) + "\n",
      "utf8"
    );
    return bytes;
  }

  const inputAccounts: number[] = [];
  const rewrittenAccounts: number[] = [];
  const cascadedAccounts: string[] = [];
  for (const K of DOSES) {
    const stateRoot = mkdtempSync(join(tmpdir(), "r22j-shape-casc-"));
    inputAccounts.push(seedFeedback(stateRoot, K));
    await cascadeFeedbackTombstones(stateRoot, "ep_warm00000000" as EpisodeId); // warm no-match
    let best = Infinity;
    let sum = 0;
    let cascadedRef: string[] = [];
    for (let it = 0; it < ITERS; it++) {
      seedFeedback(stateRoot, K); // untimed re-seed
      const t0 = hr();
      const cascaded = await cascadeFeedbackTombstones(stateRoot, target);
      const t1 = hr();
      assertEqual(`P4 K=${K} cascade count`, cascaded.length, HITS);
      cascadedRef = cascaded;
      const dt = us(t0, t1, 1);
      best = Math.min(best, dt);
      sum += dt;
    }
    cascadedAccounts.push(JSON.stringify(cascadedRef));
    rewrittenAccounts.push(statSync(feedbackLogPath(stateRoot)).size);
    const tsBytes = Buffer.byteLength(readFileSync(feedbackTombstonesPath(stateRoot), "utf8"));
    rmSync(stateRoot, { recursive: true, force: true });
    console.log(
      `P4 cascade key-order dose K=${K} (N=${N}, hits=${HITS}): min=${fmt(best)} mean=${fmt(sum / ITERS)} us/cascade  sidecar=${tsBytes} B`
    );
  }
  for (let i = 1; i < inputAccounts.length; i++) {
    assertEqual("P4 input byte-length account", inputAccounts[i], inputAccounts[0]);
    assertEqual("P4 rewritten byte-length account", rewrittenAccounts[i], rewrittenAccounts[0]);
    assertEqual("P4 cascaded id account", cascadedAccounts[i], cascadedAccounts[0]);
  }
  console.log(
    `P4 fidelity: input ${inputAccounts[0]} B, rewritten ${rewrittenAccounts[0]} B, and cascaded id set equal across K doses`
  );
}
