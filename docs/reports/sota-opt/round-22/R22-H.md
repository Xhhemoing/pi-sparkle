/* ================================================================
 * NEW AXIS, leg SD: synchronous stack-depth dose. Invoke the chain from
 * the innermost frame of a D-deep non-tail recursion, so the D frames are
 * live for the whole pre-checkpoint synchronous segment (where ~90% of
 * the chain's compute runs, per R21-H leg Y). Matched empty control at
 * each depth isolates the recursion's own wind/unwind cost from any tax

function atDepth<T>(depth: number, fn: () => T): T {
  if (depth === 0) return fn();
  const r = atDepth(depth - 1, fn);
  return r;
}
/**
 * Interleaved dose measurement: per round, measure every dose's dosed and
 * control price back-to-back; report per-dose MEDIAN net across rounds.
 * Residual call-sequence drift (R19-H axis — a confounder here, not the
 * subject) then loads equally on every dose instead of on the later ones.
 */
async function interleavedNets(
  doses: readonly number[],
  dosed: (dose: number) => Promise<void>,
  control: (dose: number) => Promise<void>,
  reps: number,
  rounds: number
): Promise<Map<number, number>> {
  const perDose = new Map<number, number[]>(doses.map((d) => [d, []]));
  for (let r = 0; r < rounds; r += 1) {
    for (const dose of doses) {
      const d = await benchAsync(() => dosed(dose), reps);
      const c = await benchAsync(() => control(dose), reps);
      perDose.get(dose)?.push(d - c);
    }
  }
  return new Map(doses.map((d) => [d, median(perDose.get(d) ?? [])]));
}

/* ================================================================

    // non-throwing path: handler must stay free
  }

async function viaAsyncDepth<T>(k: number, fn: () => Promise<T>): Promise<T> {
  if (k === 0) return await fn();
  return await viaAsyncDepth(k - 1, fn);
}
